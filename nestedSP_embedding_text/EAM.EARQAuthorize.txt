CREATE PROCEDURE [EAM].[EARQAuthorize]
(
 @RequestNo Int,
 @IsCallingForMultiLevelAuthorization tinyint = 0, 
 @OutputStatusNo tinyint = null output,
 @OutputStatusRemarks varchar(4000)  = null output
)
AS

BEGIN
	SET NOCOUNT ON;
	
	BEGIN TRY
		declare @GSTDNo int
				,@DocumentDate smalldatetime
				
		Select	-- @GSTDNo=GSTDNo,
				@DocumentDate=DocumentDate
			 ,@RequestNo=RequestNo
		from EAM.MaintenanceRequestMain where RequestNo=@RequestNo

       if @IsCallingForMultiLevelAuthorization = 1   ---------- When 1 then  Notification for Multi Level Authorization will call
	   Begin 
	   
		 Declare @MAPValuationNo smallint,@AuthorizedBy int,@CompanyNo tinyint ,@DivisionNo int ,@DocumentStatusNo tinyint
			,@AuthDetailNo int ,@CreatedBy smallint
	
 		Select @AuthorizedBy = AuthorizedBy
			  ,@CompanyNo =CompanyNo  
			  ,@DivisionNo = DivisionNo 
			  ,@DocumentStatusNo = DocumentStatusNo
			  ,@CreatedBy = sam.CreatedBy  
		From EAM.MaintenanceRequestMain sam where RequestNo = @RequestNo
		
	Declare  @UserCount int 
	Declare @NextUsers table(SNo smallint,AuthDetailNo int,LevelNo int,LoginNo int) 
	Declare @UserNO int,@LevelNo smallint,@UserCounter int =1
	   
	   If Exists (Select 1  From Notifications.MultiAuthorizationSetting where CompanyNo =@CompanyNo and ISNULL(DivisionNo,0) = ISNULL(@DivisionNo,0)
 					and FormCode ='EARQ') and @DocumentStatusNo = 20
 		BEGIN
 		
 			 if Exists (Select 1 From Notifications.TATAuthorizationDetail where FormCode ='EARQ' and InternalNo = @RequestNo  
 							and IsReady = 1 and IsEmailSent=0)
				BEGIN
						-- Send Email
					
					Insert into @NextUsers
					select ROW_NUMBER () over (order by LoginNo) SlNo,AuthDetailNo, LevelNo, LoginNo  from (
					Select LevelNo,LoginNo,AuthDetailNo
					 From Notifications.TATAuthorizationDetail where FormCode ='EARQ' and InternalNo = @RequestNo  
						and IsReady = 1 and IsEmailSent =0	and LoginNo is not null
					union 
					Select LevelNo,lrd.LoginNo,AuthDetailNo
					 From Notifications.TATAuthorizationDetail tad
					 inner join Security.mRoleMasterFormDetail rfd on rfd.RoleMasterNo = tad.RoleNo and rfd.FormNo in( 151,167,464)
					 inner join Security.LoginRoleDetail lrd on rfd.RoleMasterNo = lrd.RoleMasterNo 
					 where FormCode ='EARQ' and InternalNo = @RequestNo 
						and IsReady = 1 and IsEmailSent =0	and tad.RoleNo  is not null	
					union 
					Select LevelNo,lrd.LoginNo,AuthDetailNo
					 From Notifications.TATAuthorizationDetail tad
					 inner join Security.mRoleMasterFormDetail rfd on rfd.RoleMasterNo = tad.RoleNo and rfd.FormNo in( 151,167,464)
					 inner join Security.LoginTempRoleDetail lrd on rfd.RoleMasterNo = lrd.RoleMasterNo 
					 where FormCode ='EARQ' and InternalNo = @RequestNo and CONVERT(smalldatetime,convert(varchar(15), GETDATE(),106))
						 between lrd.FromDate and ISNULL( lrd.ToDate,CONVERT(smalldatetime,convert(varchar(15), GETDATE(),106)))
						and IsReady = 1 and IsEmailSent =0	and tad.RoleNo  is not null	) tt
						group by LevelNo, LoginNo 	,AuthDetailNo							
					
					
					Set @UserCount = @@ROWCOUNT	
					
					While @UserCounter <= @UserCount
					 Begin
					
						 Select @UserNO=LoginNo , @LevelNo = LevelNo,@AuthDetailNo=AuthDetailNo From @NextUsers Where SNo = @UserCounter
								
						 Exec [EAM].[EARQSendAuthorizationMail] @RequestNo,@UserNo,@LevelNo ,@AuthDetailNo
						 
						Update Notifications.TATAuthorizationDetail  set IsEmailSent =1 where FormCode = 'EARQ' and InternalNo = @RequestNo 
								and LevelNo = @LevelNo and IsReady =1
						 Set @UserCounter=@UserCounter+1 
					 End
		 
				END							   
				
				 
			 					  	
 		 END
 	  else If Exists (Select 1  From Notifications.MultiAuthorizationSetting where CompanyNo =@CompanyNo and ISNULL(DivisionNo,0) = ISNULL(@DivisionNo,0)
 					and FormCode ='EARQ') and Exists (Select 1 From Notifications.TATAuthorizationDetail where FormCode ='EARQ' 
 						and InternalNo = @RequestNo and IsRejectedMailSend =1) -- rejection mail

 			 if Exists (Select 1 From Notifications.TATAuthorizationDetail where FormCode ='EARQ' and InternalNo = @RequestNo 
 							and IsRejectedMailSend =1)
				BEGIN
						-- Send Email

					select ROW_NUMBER () over (order by LoginNo) SlNo,AuthDetailNo, LevelNo, LoginNo  from (

					 From Notifications.TATAuthorizationDetail where FormCode ='EARQ' and InternalNo = @RequestNo 
						and IsRejectedMailSend =1 and LoginNo is not null
					union 
					Select LevelNo,lrd.LoginNo,AuthDetailNo
					 From Notifications.TATAuthorizationDetail tad
					 inner join Security.mRoleMasterFormDetail rfd on rfd.RoleMasterNo = tad.RoleNo and rfd.FormNo in(151,167,464)
					 inner join Security.LoginRoleDetail lrd on rfd.RoleMasterNo = lrd.RoleMasterNo 
					 where FormCode ='EARQ' and InternalNo = @RequestNo 
						and IsRejectedMailSend =1 and tad.RoleNo  is not null	
					union 
					Select LevelNo,lrd.LoginNo,AuthDetailNo
					 From Notifications.TATAuthorizationDetail tad
					 inner join Security.mRoleMasterFormDetail rfd on rfd.RoleMasterNo = tad.RoleNo and rfd.FormNo in(151,167,464)
					 inner join Security.LoginTempRoleDetail lrd on rfd.RoleMasterNo = lrd.RoleMasterNo 
					 where FormCode ='EARQ' and InternalNo = @RequestNo and CONVERT(smalldatetime,convert(varchar(15), GETDATE(),106))
						 between lrd.FromDate and ISNULL( lrd.ToDate,CONVERT(smalldatetime,convert(varchar(15), GETDATE(),106)))
						and IsRejectedMailSend =1 and tad.RoleNo  is not null	) tt
						group by LevelNo, LoginNo 	,AuthDetailNo							
					
					
					Set @UserCount = @@ROWCOUNT	
					
					
					While @UserCounter <= @UserCount
					 Begin
					
						 Select @UserNO=LoginNo , @LevelNo = LevelNo,@AuthDetailNo=AuthDetailNo From @NextUsers Where SNo = @UserCounter
								
						 Exec [EAM].[EARQSendAuthorizationMail] @RequestNo,@UserNo,@LevelNo ,@AuthDetailNo ,1
						 
						Update Notifications.TATAuthorizationDetail  set IsRejectedMailSend =0 where FormCode = 'EARQ' and InternalNo = @RequestNo 
								and LevelNo = @LevelNo and IsRejectedMailSend =1
						 Set @UserCounter=@UserCounter+1 
					 End
				
				
				
				END							   
				
				 
			 					  	
 		 END 	 

		--MultiAuthorization Query Ends here 
		End 
		
		Declare @RequestDetail as table(RequestNo int,FaultNo int,RequestDescriptionNo int )
		DECLARE	@OutputStatus bit,@Result varchar(100)
				
		Insert into @RequestDetail(RequestNo ,FaultNo ,RequestDescriptionNo )
		Select	 MORFD.RequestNo
				,MORFD.FaultNo
				,MORFD.RequestDescriptionNo
		From EAM.MaintenanceRequestMain  MRM
		Inner Join  EAM.RequestFaultDetails MORFD On MRM.RequestNo = MORFD.RequestNo
		Where MRM.SourceDocumentNo = 2  And MRM.RequestNo = @RequestNo
				
		
		Declare @Count int = (Select COUNT(RequestNo) From @RequestDetail)
		Declare @i int =0
		While @i<@Count
		Begin
			Set @i=@i+1
			DECLARE @FaultNo int,@RequestDescriptionNo int
			
			Select @RequestNo =RequestNo
					,@FaultNo=FaultNo
					,@RequestDescriptionNo=RequestDescriptionNo 
			From (Select ROW_NUMBER() over (order by RequestNo, FaultNo, RequestDescriptionNo ) Rno
						,RequestNo
						,FaultNo
						,RequestDescriptionNo
				  From @RequestDetail 
				  ) tt where Rno=@i
			
			EXEC EAM.ChangeMRStatus
			@RequestNo = @RequestNo,
			@FaultNo =@FaultNo,
			@RequestDescriptionNo = @RequestDescriptionNo,
			@StatusNo=2,
			@OutputStatus =@OutputStatus  output,
			@Result =@Result output

			IF @OutputStatus = 0
				raiserror(@Result,16,-1)
		End
		
		--Else if @IsCallingForMultiLevelAuthorization = 0 -- Code for Order Account Assignment
		--Begin 
		--update SOTD set TaxCodeMiscChargesDetailNo = TCCD.TaxCodeMiscChargesDetailNo
		--FROM EAM.SalesOrderTaxDetail SOTD
		--	inner join masterdata.mMiscCharges MiscC on MiscC.MiscChargeNo=SOTD.MiscChargeNo
		--	left join masterdata.mTaxGroupSettingDetails gts on gts.TaxNo = miscC.MiscChargeNo
		--													and gts.GSTDNo=@GSTDNo
		--	left join (select TCMD.TaxCodeMiscChargesDetailNo ,TCMD.TaxCodeNo ,TCMD.MiscChargeNo
		--				From masterdata.mTaxCode TC
		--					inner join masterdata.mTaxCodeMiscChargesDetail TCMD on tcmd.TaxCodeNo = TC.TaxCodeNo 
		--				where TC.Inactive =0 and 
		--					TC.EffectiveFrom = (select MAX(EffectiveFrom) from   masterdata.mTaxCode PTC
		--										inner join masterdata.mTaxCodeMiscChargesDetail PTCMD on PTCMD.TaxCodeNo = PTC.TaxCodeNo 
		--										where PTCMD.MiscChargeNo= TCMD.MiscChargeNo
		--											and PTC.EffectiveFrom <= @DocumentDate 
		--											and PTCMD.TaxCodeNo=TCMD.TaxCodeNo )) TCCD on TCCD.MiscChargeNo = gts.TaxNo 
		--																					and TCCD.TaxCodeNo = gts.TaxCodeNo 
		--where SOTD.RequestNo=@RequestNo
	    
	 --   update SOATD set TaxCodeMiscChargesDetailNo = TCCD.TaxCodeMiscChargesDetailNo
		--FROM EAM.SalesOrderAmendmentTaxDetail SOATD
		--	inner join masterdata.mMiscCharges MiscC on MiscC.MiscChargeNo=SOATD.MiscChargeNo
		--	left join masterdata.mTaxGroupSettingDetails gts on gts.TaxNo = miscC.MiscChargeNo
		--													and gts.GSTDNo=@GSTDNo
		--	left join (select TCMD.TaxCodeMiscChargesDetailNo ,TCMD.TaxCodeNo ,TCMD.MiscChargeNo
		--				From masterdata.mTaxCode TC
		--					inner join masterdata.mTaxCodeMiscChargesDetail TCMD on tcmd.TaxCodeNo = TC.TaxCodeNo 
		--				where TC.Inactive =0 and 
		--					TC.EffectiveFrom = (select MAX(EffectiveFrom) from   masterdata.mTaxCode PTC
		--										inner join masterdata.mTaxCodeMiscChargesDetail PTCMD on PTCMD.TaxCodeNo = PTC.TaxCodeNo 
		--										where PTCMD.MiscChargeNo= TCMD.MiscChargeNo
		--											and PTC.EffectiveFrom <= @DocumentDate 
		--											and PTCMD.TaxCodeNo=TCMD.TaxCodeNo )) TCCD on TCCD.MiscChargeNo = gts.TaxNo 
		--																					and TCCD.TaxCodeNo = gts.TaxCodeNo 
		--where SOATD.RequestNo=@RequestNo
		
		--if exists (select 1 from EAM.SalesOrderTaxDetail where RequestNo=@RequestNo 
		--				and TaxCodeMiscChargesDetailNo is null and MiscChargeNo is not null and IncludeInBasic=0)
		--begin
		--	raiserror('Tax Rule is not set against charges.',16,-1)
		--end
		
		--declare @Return table (AccountType varchar(50),Particular varchar(500),AccountNo Int
		--						,AccountTypeNo tinyint,Id int,Flag varchar(10))
								
		--insert into @Return				
		--exec [Finance].[FOAMFillValuesOrderNo]
		--	@OrderTypeNo = 2
		--	,@OrderNo = @RequestNo
	
		--delete from EAM.SalesOrderAccountAssignmentDetail where RequestNo = @RequestNo
		--INSERT INTO  EAM.SalesOrderAccountAssignmentDetail
		--   (
		--    RequestNo
		--   ,Particular
		--   ,AccountNo
		--   ,AccountTypeNo
		--   ,Id
		--   ,Flag
		--   )
		--select
		--    @RequestNo
		--   ,Particular
		--   ,AccountNo
		--   ,AccountTypeNo
		--   ,Id
		--   ,Flag
		--from @Return
		--where AccountNo is not null

	 --   End 
			  
		  Select @OutputStatusNo = 1,@OutputStatusRemarks = 'Success'				  	  
	    Select 1,'Success'
	END TRY
	
	BEGIN CATCH
		Select 0,Error_Message()
		Select @OutputStatusNo = 0,@OutputStatusRemarks = ERROR_MESSAGE()				  
	END CATCH
END
