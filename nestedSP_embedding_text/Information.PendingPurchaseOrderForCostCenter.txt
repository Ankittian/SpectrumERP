CREATE PROCEDURE [Information].[PendingPurchaseOrderForCostCenter] 
	-- Add the parameters for the stored procedure here
	@CompanyNo tinyint,
	@DocumentDate smalldatetime,
	@CostCenterNo int,
	@ItemNo Int
AS
BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;

    -- Insert statements for procedure here
	
    
	declare @str varchar(max)
    select @str=masterdata.JSForInformation('MMPR')
    set @str=@str+masterdata.CSSForInformation('MMPR')
	set @str=@str+'<table class="altrowstable" id="alternatecolorPPOInfoCost">'
	set @str=@str+'<Caption><Font color="Maroon"><h3 align="Left"><b>Pending Purchase Order</b></h3></Font></Caption>'
    set @str=@str+'<tr><th>Sl. No.</th><th>Division</th><th>Department</th><th>Document No.</th><th>Document Date</th><th>Quantity</th><th>Received Quantity</th><th>Created By</th><th>Authorized By</th></tr>'
	set @str=@str
				 +(
				    REPLACE(
				    REPLACE(
				    REPLACE(
				    replace(
				    replace(
					replace(
					replace(
					replace(
					replace(
					    (    
							   Select    ROW_NUMBER() over ( Order by PO.DocumentDate desc) as td1
							             ,isnull(DM.DivisionName,'') td2
										 ,isnull(DT.DeptName,'') td3
										 ,PO.DocumentNoYearly td4
										 ,convert(varchar(20),PO.DocumentDate,103) td5
										 ,POI.POQty td6
										 ,isnull(cast(tt.RecvQty as decimal(12,3)),0) td7
										 ,L1.PrintingName td8
										 ,isnull(L2.PrintingName,'') td9
							From Purchase.POMain PO 
							Inner Join Purchase.POIndentDetail POI ON POI.PONo=PO.PONo
							inner Join Inventory.IndentMain PR ON PR.IndentNo=POI.IndentNo 
															   and PO.PORefDocumentTypeNo=3 and PODocumentTypeNo=2
							Inner Join Inventory.IndentItemDetail ID ON ID.IndentNo=PR.IndentNo
																	 And ID.ItemNo=POI.ItemNo
																	 And isnull(ID.MakeNo,0)=isnull(POI.MakeNo,0)
																	 And isnull(ID.TechnicalGradeNo,0)=isnull(POI.TechnicalGradeNo,0)                                   
							Inner Join Inventory.IndentRequisitionMain MR ON MR.IndentRequisitionNo=PR.RefDocumentNo and PR.IndentAgainstNo=3
							Left Join( Select  PO.PONo,GRI.ItemNo,GRI.MakeNo,GRI.TechnicalGradeNo,Sum(GRI.ReceivedQty) RecvQty		
							                    From Purchase.POAmendmentMain PO 
												Inner Join Purchase.POAmendmentItemDetail POI ON POI.POAmendmentNo=PO.POAmendmentNo
												Inner Join Inventory.GRNItemDetail GRI On GRI.POAmendmentNo=PO.POAmendmentNo
																					  And GRI.ItemNo=POI.ItemNo
																					  And ISNULL(GRI.MakeNo,0)=ISNULL(POI.MakeNo,0)
																					  And ISNULL(GRI.TechnicalGradeNo,0)=ISNULL(POI.TechnicalGradeNo,0)

												Group by  PO.PONo,GRI.ItemNo,GRI.MakeNo,GRI.TechnicalGradeNo
									 ) tt ON tt.ItemNo=POI.ItemNo 
									      And tt.PONo=PO.PONo
										  And isnull(tt.MakeNo,0)=isnull(POI.MakeNo,0)
										  And isnull(tt.TechnicalGradeNo,0)=isnull(POI.TechnicalGradeNo,0)

							Left Join payroll.tblDivisionMaster DM ON DM.DivisionNo=PO.DivisionNo
							Left Join payroll.tblDeptMaster DT ON DT.DeptNo=Po.DeptNo
							Inner Join Security.Login L1 ON L1.LoginNo=PO.CreatedBy
							Left Join Security.Login L2 ON L2.LoginNo=PO.AuthorizedBy
							where PO.StatusNo in(1,8,5,6)  ---- For Pending 
							      and ID.CostCenterNo=@CostCenterNo 
							      and PO.CompanyNo=@CompanyNo 
							      and PO.DocumentDate<=@DocumentDate
							      and POI.ItemNo=@ItemNo
						
						 for xml path('tr')
						 ),'<td1>','<td style="text-align: center;">'),'<td2>','<td>')
											,'<td3>','<td>')
											,'<td4>','<td>')
											,'<td5>','<td style="text-align: center;">')
											,'<td6>','<td  style="text-align: Right;">')
											,'<td7>','<td style="text-align: Right;">')
											,'<td8>','<td>')
											,'<td9>','<td>')
											)
								
				   
	set @str=@str+'</table>'
	
	select @str
END
