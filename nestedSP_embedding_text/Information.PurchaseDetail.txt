Reads: DiscreteRM.SINGateEntryItemDetail, DiscreteRM.SINOrderDetail, GateEntry.InvGateEntryItemDetail, GateEntry.InvGateEntryOrderDetail, GlobalData.PORefDocumentType, GlobalData.TaxCategory, Inventory.GRNItemDetail, Inventory.InvLotBillDetail, Inventory.InvLotSummary, masterdata.mItemMaster, masterdata.mMakeMaster, masterdata.mMiscCharges, masterdata.mTaxGroup, Purchase.POAmendmentItemDetail, Purchase.POAmendmentMain, Purchase.POItemDetail, Purchase.POItemLRate, Purchase.POItemTaxDetail, Purchase.POMain

CREATE PROCEDURE [Information].[PurchaseDetail] 
	-- Add the parameters for the stored procedure here
	@CompanyNo tinyint,
	@DocumentDate smalldatetime,
	@PONo int
	
AS
BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;

    -- Insert statements for procedure here
	

	declare @str varchar(max)
    select @str=masterdata.JSForInformation('PMPO')
    set @str=@str+masterdata.CSSForInformation('PMPO')
	set @str=@str+'<table class="altrowstable" id="alternatecolorPurchaseDetail">'
	set @str=@str+'<Caption><Font color="Maroon"><h3 align="Left"><b>Purchase Detail</b></h3></Font></Caption>'
    set @str=@str+'<tr><th>Sl.No.</th><th>PO Against</th><th>Route Applicable</th><th>Item Description</th><th>Make</th><th>PO Qty</th><th>Rate</th><th>LRate</th><th>Basic Amount</th><th>Discount</th><th>Excise</th><th>VAT/CST </th><th>Other Charges</th><th>Gate Entry Qty.</th> <th>GRN Qty.</th><th>Shortage Qty.</th><th>Excess Qty.</th><th>Purchase Bill Qty.</th><th>Freight Bill Qty.</th></tr>'
	set @str=@str
				 +( 
					replace( 
					replace(
					replace(
					replace( 
					replace(
					replace(
					replace( 
					replace(
					replace(
					replace( 
					replace(
					replace(
					replace( 
					replace(
					replace(
					replace( 
					replace(
					replace(
					replace( 
							(
								Select	ROW_NUMBER() over (Order By POM.PoNo) As  td1
										,PRF.RefDocumentType as td2,
									   RA.RouteApplicable as td3,
									   IM.ItemDescription as td4,
									   Isnull( MM.MakeName,'') as td5,
									   POID.Quantity as td6,
									   CONVERT(Decimal(12,2), POID.Rate) td7,
									   CONVERT(Decimal(12,2),t.LRate+ POID.Rate) td8,
									   CONVERT(Decimal(12,2), POID.BasicAmount )as td9
									   ,ISNULL( CONVERT(Decimal(12,2),t1.Discount),0) as td10
									   ,ISNULL( CONVERT(Decimal(12,2),t2.Excise ),0)td11
									   ,ISNULL( CONVERT(Decimal(12,2), t3.[Service Tax]),0) td12
									   ,ISNULL(CONVERT(Decimal(12,2),t4.[Other Tax]),0) td13
									   ,ISNULL( CONVERT(Decimal(12,3),GE.[Gate Entry Qty.]),0) td14
									   ,CONVERT(Decimal(12,3),ISNULL( ISNULL(PAM.AcceptedQty,PAM1.AcceptedQty),0)) td15
									   ,Case When (isnull(GE.[Gate Entry Qty.],0)-isnull(ISNULL(PAM.AcceptedQty,PAM1.AcceptedQty),0)) >=0 then (isnull(GE.[Gate Entry Qty.],0)-isnull(ISNULL(PAM.AcceptedQty,PAM1.AcceptedQty),0)) else 0 end as td16
									   ,CONVERT(Decimal(12,3),ISNULL( ISNULL(PAM.ExcessQty ,PAM1.ExcessQty),0)) td17
									   ,Isnull( Bill.PBPB,0) as td18
									   ,Isnull( Bill.FIFI,0) as td19
								From Purchase.POMain POM
								Inner Join (Select 0 Id, 'No' RouteApplicable
											Union all
											Select 1 Id, 'Yes' RouteApplicable
											 ) RA On POM.IsTransportationRouteApplicable = RA.Id
								Inner Join GlobalData.PORefDocumentType PRF on PRF.PORefDocumentTypeNo=POM.PORefDocumentTypeNo
								Inner Join Purchase.POItemDetail POID on POID.PONo=POM.PONo
								-----Calculate LRate
								Inner Join (		Select  SUM(POLR.LRate-POID.Rate ) LRate
																,POLR.PONo
																,POLR.ItemNo
																,POLR.MakeNo
																,POLR.TechnicalGradeNo
																,POID.Rate 
													From Purchase.POItemLRate POLR
													Inner Join Purchase.POItemDetail POID On POLR.PONo = POID.PONo
																								And POLR.ItemNo = POID.ItemNo
																								And ISNULL(POLR.MakeNo,0) = ISNULL(POID.MakeNo,0)
																								And ISNULL(POLR.TechnicalGradeNo,0) = ISNULL(POID.TechnicalGradeNo,0)

													Group by POLR.PONO,POLR.ItemNo,POLR.MakeNo,POLR.TechnicalGradeNo,POID.Rate 
												
											)t on  t.PONo=POM.PONo
												  And t.ItemNo=POID.ItemNo
												 And isNull(t.MakeNo,0)=isNull(POID.MakeNo,0)
												 And isNull(t.TechnicalGradeNo,0)=isNull(POID.TechnicalGradeNo,0)
								Inner Join masterdata.mItemMaster IM on IM.ItemNo=POID.ItemNo
								Left Join masterdata.mMakeMaster MM on MM.MakeNo=POID.MakeNo
								----Gate Entry Qty.
								Left Join (Select poam.PONo,
												  IGEID.ItemNo,IGEID.MakeNo,IGEID.TechnicalGradeNo,sum(IGEID.ChallanQty) [Gate Entry Qty.]
											From GateEntry.InvGateEntryItemDetail IGEID
											Inner Join Purchase.POAmendmentMain POAM on POAM.POAmendmentNo=IGEID.POAmendmentNo
											Group by poam.PONo,IGEID.ItemNo,IGEID.MakeNo,IGEID.TechnicalGradeNo
											union
											select poam.pono , gd.itemno ,gd.makeno ,null, sum(gd.ChallanQty)
											from GateEntry.InvGateEntryItemDetail gd
											inner join gateentry.InvGateEntryOrderDetail god on god.gateentryno = gd.gateentryno
											inner join purchase.poamendmentmain poam on poam.poamendmentno = god.poamendmentno
											group by poam.pono , gd.itemno ,gd.makeno 
											)GE on GE.PONo=POM.PONo
													And GE.ItemNo=POID.ItemNo
													And (GE.MakeNo= POID.MakeNo or poid.makeno is null)
													And isNull(GE.TechnicalGradeNo,0)=isNUll(POID.TechnicalGradeNo,0)
													
								---GRN Qty.
								--for Without Route Case										
								Left Join (
										  Select SUM(ReceivedQty) ReceivedQty,PAM.PONo,PAMI.ItemNo,PAMI.MakeNo,PAMI.TechnicalGradeNo
										   ,SUM(AcceptedQty) AcceptedQty,SUM(RejectedQty) RejectedQty,SUM(ChallanQty) ChallanQty,SUM(GRNI.ExcessQty) ExcessQty
										 from Inventory.GRNItemDetail GRNI
										  INNER JOIN Purchase.POAmendmentItemDetail PAMI ON PAMI.POAmendmentNo=GRNI.POAmendmentNo  
																							AND GRNI.ItemNo= PAMI.ItemNo
																							AND ISNULL(GRNI.MakeNo,0)= ISNULL(PAMI.MakeNo,0)
										  INNER JOIN  Purchase.POAmendmentMain PAM ON PAM.POAmendmentNo =PAMI.POAmendmentNo
										 
										  Where PAM.IsTransportationRouteApplicable=0
										  Group By PAM.PONo,PAMI.ItemNo,PAMI.MakeNo,PAMI.TechnicalGradeNo
										  
										  Union All
										  
										  Select SUM(ReceivedQty) ReceivedQty,PAM.PONo,PAMI.ItemNo,PAMI.MakeNo,PAMI.TechnicalGradeNo
										   ,SUM(AcceptedQty) AcceptedQty,SUM(RejectedQty) RejectedQty,0.00 ChallanQty,0.00 ExcessQty
										 from DiscreteRM.SINGateEntryItemDetail	GRNI
										 Inner Join DiscreteRM.SINOrderDetail OD on OD.SINNo=GRNI.SINNo	
										  INNER JOIN Purchase.POAmendmentItemDetail PAMI ON PAMI.POAmendmentNo=OD.POAmendmentNo  
																							AND GRNI.ItemNo= PAMI.ItemNo
																							AND (ISNULL(GRNI.MakeNo,0)= ISNULL(PAMI.MakeNo,0) or  pami.makeno is null)
										  INNER JOIN  Purchase.POAmendmentMain PAM ON PAM.POAmendmentNo =PAMI.POAmendmentNo
										 
										  Where PAM.IsTransportationRouteApplicable=0
										  Group By PAM.PONo,PAMI.ItemNo,PAMI.MakeNo,PAMI.TechnicalGradeNo
										  )	PAM on 	PAM.PONo =POM.PONo								
													And PAM.ItemNo = POID.ITEMNO 
													AND ISNULL(PAM.MakeNo,0)= ISNULL(POID.MakeNo,0)
								--For Route Case					
								Left Join (Select SUM(ReceivedQty) ReceivedQty,PAM.PONo,PAMI.ItemNo,PAMI.MakeNo,PAMI.TechnicalGradeNo
										  ,SUM(AcceptedQty) AcceptedQty,SUM(RejectedQty) RejectedQty,SUM(ChallanQty) ChallanQty, SUM(GRNI.ExcessQty) ExcessQty 
										 from Inventory.GRNItemDetail GRNI
										  INNER JOIN Purchase.POAmendmentItemDetail PAMI ON PAMI.POAmendmentNo=GRNI.POAmendmentNo  
																							AND GRNI.ItemNo= PAMI.ItemNo
																							AND ISNULL(GRNI.MakeNo,0)= ISNULL(PAMI.MakeNo,0)
										  INNER JOIN  Purchase.POAmendmentMain PAM ON PAM.POAmendmentNo =PAMI.POAmendmentNo
										  
										  Where PAM.IsTransportationRouteApplicable=1
										  And GRNI.RouteIdentity% 0.1 = 0
										  Group By PAM.PONo,PAMI.ItemNo,PAMI.MakeNo,PAMI.TechnicalGradeNo
										  )	PAM1 on 	PAM1.PONo =POM.PONo								
													And PAM1.ItemNo = POID.ITEMNO 
													AND ISNULL(PAM1.MakeNo,0)= ISNULL(POID.MakeNo,0)
								---Calculate Bill Qty. (Service Invoice Left)
								Left Join (Select PONo,ItemNo,MakeNo,PBPB,FIFI
											From(
													Select POM.PONo,InBD.ItemNo,InBD.MakeNo,InBD.BillQty
														  ,InBD.TransactionFlag
													From Purchase.POMain POM 
													Inner Join Purchase.POAmendmentMain POAM on POAM.PONo=POM.PONo
													Inner Join Inventory.InvLotSummary InVS on InVS.POAmendmentNo=POAM.POAmendmentNo
													Inner Join Inventory.InvLotBillDetail InBD on InVS.InvLotSummaryNo=InBD.InvLotSummaryNo
																							   And InBD.TransactionFlag In ('FIFI','PBPB'))t
													Pivot(sum(BillQty)
													For TransactionFlag in ([PBPB],[FIFI])
												) As [Purchase Bill Qty.]
											)Bill on Bill.PONo=POM.PONo
													  And Bill.ItemNo=POID.ItemNo
													  And isNull(Bill.MakeNo,0)=isNull(POID.MakeNo,0)
								
								--Calculate Discount                                                 
								Outer Apply (Select POITD.PONo,POITD.ItemNo,POITD.MakeNo,POITD.TechnicalGradeNo
												   ,POITD.TotalAmount [Discount]
												   ,MIS.MiscChargeName
												From Purchase.POItemTaxDetail POITD 
												Inner Join masterdata.mMiscCharges MIS on MIS.MiscChargeNo=POITD.MiscChargeNo 
												Inner Join masterdata.mTaxGroup TG on MIS.TaxGroupNo=TG.TaxGroupNo
												Inner Join GlobalData.TaxCategory TC on TG.TaxCategoryNo=TC.TaxCategoryNo and TC.TaxCategoryNo=11
																	   And POITD.PONo=POM.PONo
																	   And POITD.ItemNo=POID.ItemNo
																	   And isNull(POITD.MakeNo,0)=isNUll(POID.MakeNo,0)
																	   And isNull(POITD.TechnicalGradeNo,0)=isNull(POID.TechnicalGradeNo,0)
											)t1
								---Calculate Excise
								Outer Apply(Select POITD.PONo,POITD.ItemNo,POITD.MakeNo,POITD.TechnicalGradeNo
												   ,sum(POITD.TotalAmount) [Excise]
											From Purchase.POItemTaxDetail POITD 
											Inner Join masterdata.mMiscCharges MIS on MIS.MiscChargeNo=POITD.MiscChargeNo 
											Inner Join masterdata.mTaxGroup TG on MIS.TaxGroupNo=TG.TaxGroupNo
											Inner Join GlobalData.TaxCategory TC 
																	   on TG.TaxCategoryNo=TC.TaxCategoryNo and TC.TaxCategoryNo=1
																	   And POITD.PONo=POM.PONo
																	   And POITD.ItemNo=POID.ItemNo
																	   And isNull(POITD.MakeNo,0)=isNUll(POID.MakeNo,0)
																	   And isNull(POITD.TechnicalGradeNo,0)=isNull(POID.TechnicalGradeNo,0)
								Group BY POITD.PONo,POITD.ItemNo,POITD.MakeNo,POITD.TechnicalGradeNo)t2
								---Calculate Service Tax
								Outer Apply(	Select POITD.PONo,POITD.ItemNo,POITD.MakeNo,POITD.TechnicalGradeNo
													   ,sum(POITD.TotalAmount) [Service Tax]
												From Purchase.POItemTaxDetail POITD 
												Inner Join masterdata.mMiscCharges MIS on MIS.MiscChargeNo=POITD.MiscChargeNo 
												Inner Join masterdata.mTaxGroup TG on MIS.TaxGroupNo=TG.TaxGroupNo
												Inner Join GlobalData.TaxCategory TC 
																		   on TG.TaxCategoryNo=TC.TaxCategoryNo 
																		   And TC.TaxCategoryNo in (3,4)
																		   And POITD.PONo=POM.PONo
																		   And POITD.ItemNo=POID.ItemNo
																		   And isNull(POITD.MakeNo,0)=isNUll(POID.MakeNo,0)
																		   And isNull(POITD.TechnicalGradeNo,0)=isNull(POID.TechnicalGradeNo,0)
												Group BY POITD.PONo,POITD.ItemNo,POITD.MakeNo,POITD.TechnicalGradeNo
											)t3
								--Calculate Other Taxes
								Outer Apply(	Select POITD.PONo,POITD.ItemNo,POITD.MakeNo,POITD.TechnicalGradeNo
													   ,sum(POITD.TotalAmount) [Other Tax]
												From Purchase.POItemTaxDetail POITD 
												Inner Join masterdata.mMiscCharges MIS on MIS.MiscChargeNo=POITD.MiscChargeNo 
												Inner Join masterdata.mTaxGroup TG on MIS.TaxGroupNo=TG.TaxGroupNo
												Inner Join GlobalData.TaxCategory TC 
																		   On TG.TaxCategoryNo=TC.TaxCategoryNo 
																		   And TC.TaxCategoryNo not in (1,3,4,11)
																		   And POITD.PONo=POM.PONo
																		   And POITD.ItemNo=POID.ItemNo
																		   And isNull(POITD.MakeNo,0)=isNUll(POID.MakeNo,0)
																		   And isNull(POITD.TechnicalGradeNo,0)=isNull(POID.TechnicalGradeNo,0)
												Group BY POITD.PONo,POITD.ItemNo,POITD.MakeNo,POITD.TechnicalGradeNo
											)t4

								Where POM.PONo = @PONo
										And POM.DocumentDate = @DocumentDate
										And POM.CompanyNo = @CompanyNo
							for xml path('tr')),'<td1>','<td style="text-align: Center;">')
		   									   ,'<td2>','<td>')				
											   ,'<td3>','<td>')
											   ,'<td4>','<td style="text-align: Center;">')
		   									   ,'<td5>','<td>')				
											   ,'<td6>','<td style="text-align: Center;">')
											    ,'<td7>','<td>')
											   ,'<td8>','<td>')
											   ,'<td9>','<td>')
											    ,'<td10>','<td>')
											   ,'<td11>','<td>')												   									 			
											   ,'<td12>','<td>')
											    ,'<td13>','<td>')
											   ,'<td14>','<td style="text-align: Right;">')
											   ,'<td15>','<td style="text-align: Right;">')
											   ,'<td16>','<td style="text-align: Right;">')
											   ,'<td17>','<td style="text-align: Right;">')
											   ,'<td18>','<td>')
											   ,'<td19>','<td>')
									    	)
								
				   
	set @str=@str+'</table>'
	
	select @str
END
