CREATE PROCEDURE [Purchase].[PMPODelete] 

 @PONo int,
 @POAmendmentNo int,
 @Formtag char(4)
 ,@OutputStatusNo bit = null output
 ,@OutputStatusRemarks varchar(4000) = null output
 ,@IsCopy bit = 0 -- Used for Multi Level Authorization 
 AS
BEGIN
	SET NOCOUNT ON;
	BEGIN TRY
	
	Begin Transaction
		--Transaction Begins from here
		
		if @FormTag ='PMPO'
		begin
			if exists (select 1 from Purchase.POAmendmentMain where PONo = @PoNo and AmendmentNo>0)
				RaisError('PO Amendment is done againt this PO,so you can''t delete this entry.',16,-1)		
		end	
		
		if exists (select 1 from Purchase.POAmendmentMain where RefDocumentNo = @POAmendmentNo
					and PORefDocumentTypeNo= 4 )
		BEGIN				
			RaisError('PO is Prepared against this LOI,so you can''t delete this entry.',16,-1)			
		end

		if exists (select 1 from Finance.AdvanceRequestMain where SourceDocumentNo = @POAmendmentNo
					and AdvanceSourceDocumentNo = 1 )
		BEGIN				
			RaisError('Advance request is prepared against this PO,so you can''t delete this entry.',16,-1)			
		end

		if exists (select 1 from Finance.BGReceiptDetail rd
						inner join Purchase.POAmendmentMain poam on poam.PONo = @PoNo and AmendmentNo=0
					where rd.SourceDocumentNo = @PoNo
					and rd.AdvanceSourceDocumentNo = 1 )
		BEGIN				
			RaisError('BG is prepared against this PO,so you can''t delete this entry.',16,-1)			
		end
	 
	 declare @TempRefDocumentNo int,@PORefDocumentTypeNo tinyint
	 
declare @IsCS bit ,@tempPOAMendmentNo int,@QuotationNo int
		select @TempRefDocumentNo  = RefDocumentNo ,@QuotationNo= QuotationNo , @IsCS= IsCSAgainstAgreement ,@tempPOAMendmentNo = POAmendmentNo from 
			Purchase.POAmendmentMain pm 
			left join Purchase.RevisedQuotationMain qm on qm.RevisedQuotationNo = pm.RefDocumentNo and pm.IsCSAgainstAgreement =0
			where PONo = @PONo and AmendmentNo = (select MAX(AmendmentNo ) from Purchase.POAmendmentMain where PONo = pm.PONo)
			and PORefDocumentTypeNo  = 8 
	
	-- Agreement Reset	 
	if exists(select 1 from purchase.POMain where (PORefDocumentTypeNo=2 or (PORefDocumentTypeNo=8  and IsCSAgainstAgreement =1)) and PONO = @PONo)
	begin
		select @TempRefDocumentNo  = RefDocumentNo,@PORefDocumentTypeNo = PORefDocumentTypeNo
		from Purchase.POMain where PONo= @PONo

		
		update ad set StatusNo = 1
		from Purchase.POAmendmentMain pm
		inner join Purchase.POAmendmentItemDetail pd on 
					pm.POAmendmentNo = pd.POAmendmentNo
			inner join Purchase.AgreementItemDetail ad  on pm.refDocumentNo = ad.AgreementNo
					and ad.ItemNo = pd.itemno and isnull(ad.MakeNo,0) = isnull(pd.MakeNo,0)
			where PORefDocumentTypeNo=2 and pm.POAmendmentNo = @POAmendmentNo and ad.statusNo = 2
		
		update am set StatusNo = 1
		from Purchase.POAmendmentMain pm 
		inner join Purchase.AgreementMain am on pm.refDocumentNo = am.AgreementNo
			where PORefDocumentTypeNo=2 and pm.POAmendmentNo = @POAmendmentNo		
			and am.statusNo = 2
	end
	 
	 select @TempRefDocumentNo  = RefDocumentNo from Purchase.POAmendmentMain where POAmendmentNo=@POAmendmentNo
	 -- LOI reset
	 if exists(select 1 from purchase.POAmendmentMain where PORefDocumentTypeNo=4 and POAmendmentNo=@POAmendmentNo)
	begin
		
		
		
		update PM set StatusNo = 1
		from Purchase.POAmendmentMain PM 
			where  PONo in (select PONo from Purchase.POAmendmentMain where POAmendmentNo = @TempRefDocumentNo)
			
		update pd set StatusNo = 1
		from Purchase.POAmendmentMain PM 
			inner join Purchase.POAmendmentItemDetail pd on pm.POAmendmentNo = pd.PoAmendmentNo
			where  PONo in (select PONo from Purchase.POAmendmentMain where POAmendmentNo = @TempRefDocumentNo)
			
		update PM set StatusNo = 1
		from Purchase.POMain PM 
			where  PONo in (select PONo from Purchase.POAmendmentMain where POAmendmentNo = @TempRefDocumentNo)
			
		update pd set StatusNo = 1
		from Purchase.POItemDetail pd
			where  PONo in (select PONo from Purchase.POAmendmentMain where POAmendmentNo = @TempRefDocumentNo)
	end
	
	
	-- quotatation reset
	Declare @RevisedQuotationNo int = 0,@CompanyNo tinyint = 0
	Declare @RFQNo int = 0,@iRow int = 0 ,@iCount int = 0
	IF exists(
	
	          select 1 
			  from purchase.POMain 
			  where PONo = @PONo 
				  and (PORefDocumentTypeNo = 1 or (PORefDocumentTypeNo=8 and IsCSAgainstAgreement =0)) 
				  and PODocumentTypeNo<>1
			)
	BEGIN
		Select @RFQNo = RQM.RFQNo
		,@CompanyNo = PAM.CompanyNo
		,@RevisedQuotationNo =  RQM.RevisedQuotationNo
		From Purchase.POAmendmentMain PAM
		Inner Join Purchase.RevisedQuotationMain RQM on RQM.RevisedQuotationNo = PAM.RefDocumentNo
		Where  POAmendmentNo = @POAmendmentNo
			
		Declare @RevisedQuotationDetail as table(id int,RevisedQuotationNo int )	
		Insert into @RevisedQuotationDetail
		Select 1,@RevisedQuotationNo 
		Union 
		Select row_number() over (Order By RevisedQuotationNO) + 1,RevisedQuotationNO 
		From Purchase.RevisedQuotationMain Where RFQNo = @RFQNo and RevisedQuotationNO <> @RevisedQuotationNo
		
		set @iRow = @@rowcount
		set @iCount = 0
		while @iRow >@iCount
		Begin
		      Select @RevisedQuotationNo = RevisedQuotationNo
		      From @RevisedQuotationDetail Where id = @iCount + 1
		      
			  Exec [Purchase].[QuotationChangeStatus]  @RevisedQuotationNo,@POAmendmentNo,1,@CompanyNo,0,0
			  
			  set @iCOunt = @iCOunt + 1
		End		       
	END
	
	
		        DECLARE @prevPOAmendmentNo int
				--- GET THE PREVIOUS AMENDMENT NO TO INSERT IN MAIN TABLE
				
				SELECT @prevPOAmendmentNo = POAmendmentNo 
				FROM  [Purchase].[POAmendmentMain]
				WHERE AmendmentNo =( SELECT AmendmentNo -1
				                     FROM [Purchase].[POAmendmentMain] 
				                     WHERE POAmendmentNo = @POAmendmentNo ) 
					
				AND PONo =(SELECT PONo 
				           FROM [Purchase].[POAmendmentMain] 
				           WHERE POAmendmentNo = @POAmendmentNo)
		
		
		       IF ISNULL(@prevPOAmendmentNo ,0) = 0 OR EXISTS ( SELECT 1 
		                                                        FROM [Purchase].[POAmendmentMain]
		                                                        WHERE POAmendmentNo=@POAmendmentNo
		                                                        AND ISNULL([AuthorizedBy],0)<>0
		                                                       )
	          BEGIN
	          
	          	if @IsCopy = 1
	          	BEGIN
					set @prevPOAmendmentNo = @POAmendmentNo
				END	
	  
			  -- Delete PO Details Table
			  -- not in case of LOI type of Purchase Order
			Declare @IndentDetail as table (IDNo tinyint ,IndentNo int, ItemNo int, MakeNo smallint,TechnicalGradeNo tinyint)
			Set  @iRow = 0 
			Set @iCount  =0 
			if not exists(select 1 from Purchase.POMain where PODocumentTypeno=1 and PONO=@PONo)
			begin
				Insert into @IndentDetail select row_Number() over (order by IndentNo), IndentNo ,
					ItemNo ,IndentMakeNo ,IndentTechnicalGradeNo from 
						Purchase.POIndentDetail where PONO = @PoNo
				set @iRow= @@rowcount		
			end		
					Delete from Purchase.POAdvanceDetail where PONo = @PONo
					Delete from Purchase.POExpenseHeadOutputDetail where POExpenseDetailNo
							in (select POExpenseDetailNo from Purchase.POExpenseHead where PONo = @PONo)
					
					Delete From Purchase.POItemTechSpecificationDetail Where PONo = @PONo
					Delete from Purchase.POItemLRate  where PONo = @PONo		
					Delete from Purchase.POExpenseHead  where PONo = @PONo
					Delete from Purchase.POTransportationRoute   where PONo = @PONo
					
					Delete from Purchase.POItemConsigneeDetail where POItemDetailNo in(
					select POItemDetailNo  FROM Purchase.POItemDetail WHERE PONo = @PONo)
					
					Delete from Purchase.POConsigneeDetail where PONo = @PONo
					Delete from Purchase.POPaymentTermsDetail where PONo = @PONo
					DELETE FROM Purchase.POItemTaxDetail WHERE PONo = @PONo
					DELETE FROM Purchase.POItemSizeDetail WHERE PONo = @PONo
					DELETE FROM Purchase.POItemScheduleDetail WHERE PONo = @PONo
					
					DELETE FROM Purchase.POItemQCParameterDetail WHERE POItemDetailNo in(
					select POItemDetailNo  FROM Purchase.POItemDetail WHERE PONo = @PONo)
					
					DELETE FROM Purchase.POCustomFieldDetail WHERE PONo = @PONo
					DELETE FROM Purchase.POTaxConcessionFormDetail WHERE PONo = @PONo
					
					Delete from Purchase.POIndentReserveDetail where POIndentDetailNo in (
					select POIndentDetailNo from  Purchase.POIndentDetail WHERE PONo = @PONo)
					
					DELETE FROM Purchase.POIndentDetail WHERE PONo = @PONo
					
					DELETE FROM Purchase.POItemDetail WHERE PONo = @PONo
					DELETE FROM Purchase.POTaxDetail WHERE PONo = @PONo
					DELETE FROM Purchase.POTermsNConditionDetail WHERE PONo = @PONo
					Delete from [Purchase].[PONotesDetail] where PONo = @PONo
					
					Delete A From Purchase.FreightOrderOrderDetail  A 
					where PONo = @PONo  and @IsCopy = 0
					and (select count(1) from Purchase.FreightOrderOrderDetail FOOD where FOOD .PONo = A.PONo 
					and isnull(FOOD.RouteIdentity,0) = isnull(A.RouteIdentity,0))=1
					
			
			Declare @IndentNo int,@ItemNo int,@MakeNo smallint,@OutputStatus bit, @result varchar(100)
			,@TechnicalGradeNo tinyint
			
			select @iCOunt = 0
			while @iRow > @iCount
			begin
				select @IndentNo = IndentNo , @ItemNo = ItemNo , @MakeNo = MakeNo,
					@TechnicalGradeNo =TechnicalGradeNo 
					from @IndentDetail where 
					IDNo = @iCount +1
				
				exec Inventory.IndentStatus @IndentNo,@ItemNo,@MakeNo ,	2,@OutputStatus output,@Result output
						,@TechnicalGradeNo 
				
				if @OutputStatus =0
				   raiserror (@Result,16,-1)
				   
				set @iCount = @iCount + 1			
			end
			
			  END -- DELETE MAIN TABLE IF	
		      
		      
	 IF Exists(SELECT 1 FROM [Purchase].[POAmendmentMain] WHERE POAmendmentNo = @prevPOAmendmentNo 
		               AND ISNULL([AuthorizedBy],0)<>0 ) AND						 
		        Exists(SELECT 1 FROM [Purchase].[POAmendmentMain] WHERE POAmendmentNo = @POAmendmentNo  
		               AND ISNULL([AuthorizedBy],0)<>0 ) 
		     BEGIN
				
				Insert into Purchase.POAdvanceDetail(PONo, ReferenceNo,AdvanceAmount)
				select @PONo ,ReferenceNo,AdvanceAmount
				from Purchase.POAmendmentAdvanceDetail
				where POAmendmentNo = @prevPOAmendmentNo

				
				Insert into Purchase.poitemlrate	(PONo, ItemNo,MakeNo,TechnicalGradeNo ,RouteIdentity ,LRate)
				select @PONo, ItemNo,MakeNo,TechnicalGradeNo ,RouteIdentity ,LRate
				from Purchase.POAmendmentItemLRate
				where POAmendmentNo = @prevPOAmendmentNo
				
				Insert into Purchase.POConsigneeDetail	(PONo, ConsigneeLocationNo)
				select @PONo ,ConsigneeLocationNo
				from Purchase.POAmendmentConsigneeDetail
				where POAmendmentNo = @prevPOAmendmentNo	
				
				INSERT INTO Purchase.POPaymentTermsDetail(PONo, PaymentTypeNo, BaseDateTypeNo, PaymentPayOnNo,
				TermsValue, PayValue, NoOfDays, Date, AdditionalMonth, Discount, IsBGApplicable, BGAmount, 
				IsProgressBased, Progress, Tolerance, Remark,SequenceNo)
			select @PONo, PaymentTypeNo, BaseDateTypeNo, PaymentPayOnNo,
				TermsValue, PayValue, NoOfDays, Date, AdditionalMonth, Discount, IsBGApplicable, BGAmount, 
				IsProgressBased, Progress, Tolerance, Remark,SequenceNo
				from Purchase.POAmendmentPaymentTermsDetail where POAmendmentNo = @prevPOAmendmentNo
				
				 Insert into Purchase.POItemTaxDetail(PONo, Selected, ItemNo, MakeNo, MiscChargeNo, ChargeType, 
					Nature, ChargeOn, ChargeValue, TotalAmount,TechnicalGradeNo)
				 select @PONo, Selected, ItemNo, MakeNo, MiscChargeNo, ChargeType, 
					Nature, ChargeOn, ChargeValue, TotalAmount,TechnicalGradeNo 
					from Purchase.POAmendmentItemTaxDetail
					where POAmendmentNo = @prevPOAmendmentNo
				 
				 insert into Purchase.POItemSizeDetail(PONo,ItemNo,MakeNo,Length,Width,Pcs,UnitWeight,TechnicalGradeNo,TotalWeight)
				 select @PONo, ItemNo, MakeNo, Length, Width, Pcs, UnitWeight,TechnicalGradeNo,TotalWeight from
				 Purchase.POAmendmentItemSizeDetail where POAmendmentNo = @prevPOAmendmentNo
				 
				 Insert into Purchase.POItemScheduleDetail(PONo, ItemNo, MakeNo, ScheduleDate, ScheduleQty,TechnicalGradeNo)
				 select @PONo, ItemNo, MakeNo, ScheduleDate, ScheduleQty,TechnicalGradeNo FROM
				 Purchase.POAmendmentItemScheduleDetail where POAmendmentNo = @prevPOAmendmentNo
				 
				 Insert into Purchase.POCustomFieldDetail(PONo, CustomFieldNo, FieldValue)
				  select @PONo, CustomFieldNo, FieldValue from
				  Purchase.POAmendmentCustomFieldDetail  where POAmendmentNo = @prevPOAmendmentNo
				  
				  Insert into Purchase.POTaxConcessionFormDetail(PONo, IssueReceiveNo, TaxConcessionFormNo, PartyNo, PartyLocationNo)
					SELECT     @PONo , IssueReceiveNo, TaxConcessionFormNo, PartyNo, PartyLocationNo
					FROM         Purchase.POAmendmentTaxConcessionFormDetail
					WHERE     (POAmendmentNo = @prevPOAmendmentNo)
				  
				  Insert into Purchase.POIndentDetail(PONo, IndentNo, ItemNo, IndentMakeNo, MakeNo, FirstCF, 
					POUnitNo, SecondCF, POQty, PORate, ScheduleDate,TaskNo,UID ,PRSAmendmentNo,SalesOrderAmendmentNo
					,TechnicalGradeNo ,IndentTechnicalGradeNo)
					select @PONo, IndentNo, ItemNo, IndentMakeNo, MakeNo, FirstCF, 
					POUnitNo, SecondCF, POQty, PORate, ScheduleDate,TaskNo,UID ,PRSAmendmentNo , 
					SalesOrderAmendmentNo,TechnicalGradeNo ,IndentTechnicalGradeNo from
					Purchase.POAmendmentIndentDetail where POAmendmentNo = @prevPOAmendmentNo
			
				Insert into Purchase.POIndentReserveDetail (POIndentDetailNo ,IndentItemReserveNo ,POQty )  
				  select pd.POIndentDetailNo  , pid.IndentItemReserveNo ,pid.POQty 
				from Purchase.POAmendmentMain pam
				Inner join Purchase.POAmendmentIndentDetail piid on piid.POAmendmentNo = pam.POAmendmentNo 
				inner join purchase.POAmendmentIndentReserveDetail pid on pid.POAmendmentIndentDetailNo = piid.POAmendmentIndentDetailNo 
				inner join Purchase.POIndentDetail pd on pd.PONo = pam.PONo and pd.IndentNo = piid.IndentNo 
					and pd.ItemNo = piid.ItemNo and ISNULL(pd.IndentMakeNo,0)= ISNULL(piid.IndentMakeNo ,0)
					and ISNULL(pd.IndentTechnicalGradeNo ,0) = ISNULL(piid.IndentTechnicalGradeNo,0)
				where pam.POAmendmentNo = @prevPOAmendmentNo
				
				  
				 Insert into Purchase.POItemDetail(PONo, ItemNo, MakeNo, TechSpecification, Quantity, UnitNo, 
				 Rate, BasicAmount, NetAmount, Remark, StatusNo,TechnicalGradeNo ,ToleranceBasisNo,ToleranceMinus
				  ,TolerancePlus ,ChapterHeadingNo,AgreementItemScheduleDetailNo,HSNSACCodeNo  
				  ,CustomItemBasedPriceNo,CustomBaseRate,CustomDiscountPercentage,CustomDiscountAmount)
				 select @PONo, ItemNo, MakeNo, TechSpecification, Quantity, UnitNo, Rate, BasicAmount,
				  NetAmount, Remark, StatusNo,TechnicalGradeNo,ToleranceBasisNo,ToleranceMinus 
				  ,TolerancePlus ,ChapterHeadingNo ,AgreementItemScheduleDetailNo,HSNSACCodeNo 
				 ,CustomItemBasedPriceNo,CustomBaseRate,CustomDiscountPercentage,CustomDiscountAmount
				  from Purchase.POAmendmentItemDetail  where POAmendmentNo = @prevPOAmendmentNo
				 

				Insert into Purchase.POItemTechSpecificationDetail   ([PONo] ,[POItemDetailNo] ,[TechSpecificationHeadNo] ,[Description])
				Select @PONo,PD.POItemDetailNo,TD.TechSpecificationHeadNo,TD.[Description]
				FROM  purchase.POAmendmentItemDetail pid  
				Inner Join Purchase.POAmendmentItemTechSpecificationDetail TD on TD.POAmendmentNo=PID.POAmendmentNo
				inner join Purchase.POItemDetail pd On pd.PONo =@PONo and pid.ItemNo = pd.ItemNo  
													And isnull(pid.MakeNo,0) = isnull(pd.MakeNo,0) 
													And  isnull(pid.TechnicalGradeNo,0) = isnull(pd.TechnicalGradeNo ,0)
				where pid.POAmendmentNo = @prevPOAmendmentNo	
				 
				Insert into Purchase.POItemConsigneeDetail(POItemDetailNo, ConsigneeLocationNo, Qty)
				select pd.POItemDetailNo  , pcd.ConsigneeLocationNo, pcd.Qty
				from Purchase.POAmendmentMain pam
				inner join purchase.POAmendmentItemDetail pid on pam.POAmendmentNo = pid.POAmendmentNo 
				inner join Purchase.POAmendmentItemConsigneeDetail pcd on 
				pid.POAmendmentItemDetailNo = pcd.POAmendmentItemDetailNo
				inner join Purchase.POItemDetail pd on pd.PONo = pam.PONo and pid.ItemNo = pd.ItemNo 
				and isnull(pid.MakeNo,0) = isnull(pd.MakeNo,0) and 
				isnull(pid.TechnicalGradeNo,0) = isnull(pd.TechnicalGradeNo ,0)
				where pam.POAmendmentNo = @prevPOAmendmentNo	
				 
				 Insert into Purchase.POItemQCParameterDetail(POItemDetailNo, QCParameterNo, StandardValue,
				  ToleranceValue, DeductionRate, DeductionType, DoubleDeductionFrom,StandardValue2)
					select pd.POItemDetailNo , QCParameterNo, StandardValue,
					ToleranceValue, DeductionRate, DeductionType, DoubleDeductionFrom,StandardValue2 
					FROM  Purchase.POAmendmentMain pam
					inner join purchase.POAmendmentItemDetail pid on pid.POAmendmentNo = pam.POAmendmentNo
					inner join Purchase.POAmendmentItemQCParameterDetail pcd on pid.POAmendmentItemDetailNo = pcd.POAmendmentItemDetailNo
					inner join Purchase.POItemDetail pd on pd.PONo = pam.PONo and pid.ItemNo = pd.ItemNo 
					and isnull(pid.MakeNo,0) = isnull(pd.MakeNo,0) and 
					isnull(pid.TechnicalGradeNo,0) = isnull(pd.TechnicalGradeNo ,0)
					where pam.POAmendmentNo = @prevPOAmendmentNo	
				
				 
				 Insert into Purchase.POTaxDetail(PONo, MiscChargeNo, ChargeType, Nature, ChargeOn, ChargeValue,
				  TotalValue,TaxCodeMiscChargesDetailNo) 
				  select @PONo, MiscChargeNo, ChargeType, Nature, ChargeOn, ChargeValue,
				  TotalValue ,TaxCodeMiscChargesDetailNo 
				  from Purchase.POAmendmentTaxDetail where POAmendmentNo = @prevPOAmendmentNo
				  
				  Insert into Purchase.POTermsNConditionDetail(PONo, TermsNConditionHeadNo, TermsNCondition)
				  select @PONo, TermsNConditionHeadNo, TermsNCondition from
				  Purchase.POAmendmentTermsNConditionDetail where POAmendmentNo = @prevPOAmendmentNo
				  
				  INSERT INTO Purchase.PONotesDetail(PONo, PrintAreaNo, Notes, IsDefault)
					select @PONo, PrintAreaNo, Notes, IsDefault from 
					[Purchase].[POAmendmentNotesDetail] where POAmendmentNo = @prevPOAmendmentNo
				
				 INSERT INTO Purchase.POTransportationRoute
                      (PONo, RouteId, ItemNo, MakeNo, TechnicalGradeNo, DeliveryModeNo, FreightTypeNo,
                      FreightRateTypeNo, FreightAmount, FromLocationNo, ToLocationNo, WarehouseNo, 
                      ToleranceBasisNo, ToleranceMinus, TolerancePlus, Receipient,IsGateEntry,ReceipientPartyNo
                      ,RouteIdentity ,SenderPartyNo ,NoOfTrips ,ParentRouteId )
				 select @PONo, RouteId, ItemNo, MakeNo, TechnicalGradeNo, DeliveryModeNo, FreightTypeNo,
                      FreightRateTypeNo, FreightAmount, FromLocationNo, ToLocationNo, WarehouseNo, 
                      ToleranceBasisNo, ToleranceMinus, TolerancePlus, Receipient,IsGateEntry,ReceipientPartyNo
                      ,RouteIdentity ,SenderPartyNo ,NoOfTrips,ParentRouteId
                  from   Purchase.POAmendmentTransportationRoute where POAmendmentNo = @prevPOAmendmentNo
                     	      
				
				INSERT INTO Purchase.POExpenseHead
                      (PONo, ItemNo, MakeNo, TechnicalGradeNo, ExpenseheadNo, BillingLocationNo, BillingQtyNo,
                       FreightRateTypeNo , Rate, PartyNo, InputItemQty, Yield, Rejection, Loss, IncludeInValuation
                       ,RouteIdentity,SCAgreementNo)
                 select @PONo, ItemNo, MakeNo, TechnicalGradeNo, ExpenseheadNo, BillingLocationNo, BillingQtyNo,
                      FreightRateTypeNo, Rate, PartyNo, InputItemQty, Yield, Rejection, Loss, IncludeInValuation
                      ,RouteIdentity,SCAgreementNo
                   from    Purchase.POAmendmentExpenseHead where POAmendmentNo = @prevPOAmendmentNo
				
				INSERT INTO Purchase.POExpenseHeadOutputDetail(POExpenseDetailNo, ItemNo, Quantity)
				select pd.POExpenseDetailNo  , pcd.ItemNo , pcd.Quantity 
				from Purchase.POAmendmentMain pam
				inner join purchase.POAmendmentExpenseHead pid on pam.POAmendmentNo = pid.POAmendmentNo 
			inner join Purchase.POAmendmentExpenseHeadOutputDetail pcd on 
						pid.POAmendmentExpenseDetailNo = pcd.POAmendmentExpenseDetailNo
			inner join Purchase.POExpenseHead pd on pd.PONo = pam.PONo and pid.ItemNo = pd.ItemNo 
					and isnull(pid.MakeNo,0) = isnull(pd.MakeNo,0) and 
						isnull(pid.TechnicalGradeNo,0) = isnull(pd.TechnicalGradeNo ,0)
				where pam.POAmendmentNo = @prevPOAmendmentNo	
				
				 
				  
				  
			if not exists(select 1 from Purchase.POMain where PODocumentTypeno=1 and PONO=@PONo)
			begin
				Insert into @IndentDetail select row_Number() over (order by IndentNo), IndentNo ,
					ItemNo ,IndentMakeNo ,IndentTechnicalGradeNo from 
						Purchase.POIndentDetail where PONO = @PoNo
				set @iRow= @@rowcount		
			end		
			
			 
			select @iCOunt = 0
			while @iRow > @iCount
			begin
				select @IndentNo = IndentNo , @ItemNo = ItemNo , @MakeNo = MakeNo,@TechnicalGradeNo =TechnicalGradeNo
				  from @IndentDetail where  IDNo = @iCount +1
				
				exec Inventory.IndentStatus @IndentNo,@ItemNo,@MakeNo ,	2,@OutputStatus output,@Result output
						,@TechnicalGradeNo 
				
				if @OutputStatus =0
				   raiserror (@Result,16,-1)
				   
				set @iCOunt = @iCount + 1			
			end
			
		     END
			
                      
					UPDATE    POM SET
					CompanyNo = pam.CompanyNo, 
					--YearNo = pam.yearNo,  commented on 12 04 2019
					AmendmentNo = pam.AmendmentNo, 
					DivisionNo = pam.DivisionNo, 
					DocumentNoYearly = pam.DocumentNoYearly, 
					DocumentDate = pam.DocumentDate , 
                    PODocumentTypeNo = pam.PODocumentTypeNo , 
                    IndentTypeNo = pam.IndentTypeNo  , 
                    POPurchaseCategoryNo = pam.POPurchaseCategoryNo , 
                    PORefDocumentTypeNo =pam.PORefDocumentTypeNo , 
                    RefDocumentNo = pam.RefDocumentNo , 
                    VendorNo = pam.VendorNo , 
                    VendorLocationNo = pam.VendorLocationNo , 
                    VendorContactNo = pam.VendorContactNo , 
                    ValidityDate = pam.ValidityDate , 
                    CurrencyConversionNo =pam.CurrencyConversionNo , 
					DivisibilityRate =pam.DivisibilityRate ,
					MultiplicativeRate=pam.MultiplicativeRate, 
                    WayBillNo = pam.WayBillNo , 
                    WaybillDate = pam.WaybillDate , 
                    ConsigneeNo = pam.ConsigneeNo , 
                    ConsigneeLocationNo = pam.ConsigneeLocationNo , 
                    FreightTypeNo = pam.FreightTypeNo , 
                    FreightRateTypeNo  = pam.FreightRateTypeNo  , 
                    FreightAmount = pam.FreightAmount ,
