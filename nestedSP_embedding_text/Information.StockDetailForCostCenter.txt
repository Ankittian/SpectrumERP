CREATE PROCEDURE [Information].[StockDetailForCostCenter] 
	-- Add the parameters for the stored procedure here
	@DocumentDate smalldatetime,
	@ItemNo int,
	@CostCenterNo int
AS
BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;

    -- Insert statements for procedure here
	declare @str varchar(max)
    select @str=masterdata.JSForInformation('MMPR')
    set @str=@str+masterdata.CSSForInformation('MMPR')
	set @str=@str+'<table class="altrowstable" id="alternatecolorStockInfoCost">'
	set @str=@str+'<Caption><Font color="Maroon"><h3 align="Left"><b>Stock Detail</b></h3></Font></Caption>'
    set @str=@str+'<tr><th>Sl. No.</th><th>Company Name</th><th>Division</th><th>Warehouse</th><th>Make</th><th>Stock Quantity</th></tr>'
	set @str=@str
				 + (replace(
					replace(
					replace(
					replace(
					replace(
					replace((    
								 Select Row_number() over (order by CM.CompanyName) td1,
								        CM.CompanyName td2,
									    isNull(DV.DivisionName,'') as [td3],
									    WH.WareHouseName as [td4],
									    isNull(MM.MakeName,'') as td5,
									    t.[Opening Stock] td6 
							      From masterdata.mCompanyMaster CM
							     Cross Apply  Inventory.funcItemOpeningStock(CM.CompanyNo,null,@ItemNo,null
							                            ,@DocumentDate,null,null,@CostCenterNo,1,null,1) t
							     Left Join payroll.tblDivisionMaster DV on t.DivisionNo=DV.DivisionNo
							     Inner Join masterdata.mWarehouse WH on WH.WareHouseNo=t.WarehouseNo
							     Left Join masterdata.mMakeMaster MM on MM.MakeNo=t.MakeNo
							     for xml path('tr')),'<td1>','<td style="text-align: Center;">')
							                   ,'<td2>','<td>')
											   ,'<td3>','<td>')
											   ,'<td4>','<td>')
											   ,'<td5>','<td>')
											   ,'<td6>','<td style="text-align: right;">'))
				   
	set @str=@str+'</table>'
	
	select @str
END
