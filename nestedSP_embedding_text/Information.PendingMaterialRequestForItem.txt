CREATE PROCEDURE [Information].[PendingMaterialRequestForItem] 
	-- Add the parameters for the stored procedure here
	@CompanyNo tinyint,
	@DocumentDate smalldatetime,
	@ItemNo int
AS
BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;

    -- Insert statements for procedure here
	

	declare @str varchar(max)
    select @str=masterdata.JSForInformation('PMPO')
    set @str=@str+masterdata.CSSForInformation('PMPO')
	set @str=@str+'<table class="altrowstable" id="alternatecolorPendingMaterialRequestInfo">'
	set @str=@str+'<Caption><Font color="Maroon"><h3 align="Left"><b>Pending Material Request</b></h3></Font></Caption>'
    set @str=@str+'<tr><th>Sl. No.</th><th>Division</th><th>Department</th><th>Document No.</th><th>Document Date</th><th>Make</th><th>MR Quantity</th><th>PR Quantity</th><th>PO Quantity</th><th>Balance Quantity</th><th>Created By </th><th>Authorized By</th></tr>'
	set @str=@str
				 +( REPLACE(
					REPLACE(
				    REPLACE(
				    REPLACE(
				    REPLACE(
				    REPLACE(
				    replace(
					replace(
					replace(
					replace(
					replace(
					replace((    
							Select  ROW_NUMBER() over ( Order by IRM.DocumentDate desc) as td1
							        ,Isnull(DM.DivisionName,'')td2
									,Isnull( TDM.DeptName ,'') td3
								    ,IRM.DocumentNoYearly td4
								    ,convert(varchar(20),IRM.DocumentDate,103) td5
								    ,Isnull(MM.MakeName ,'') td6
								    ,sum( Isnull(IRID.Quantity,0)) td7
								    ,sum(Isnull(IID.IndentQty,0)) td8
								    ,sum(Isnull(IID.POQty,0)) td9
								    ,sum(IRID.Quantity - Isnull(IID.POQty,0)) td10
								    ,CRT.PrintingName td11
								    ,Isnull(AUT.PrintingName,'') td12
							From Inventory.IndentRequisitionMain IRM
							Inner Join Inventory.IndentRequisitionItemDetail IRID ON IRID.IndentRequisitionNo=IRM.IndentRequisitionNo
							Left Join Inventory.IndentMain IM ON IM.RefDocumentNo=IRM.IndentRequisitionNo and IndentAgainstNo=3
							Left Join Inventory.IndentItemDetail IID ON IID.IndentNo=IM.IndentNo
																	  And IID.ItemNo=IRID.ItemNo
																	  And isnull(IID.MakeNo,0)=isnull(IRID.MakeNo,0)
																	  And isnull(IID.TechnicalGradeNo,0)=isnull(IRID.TechnicalGradeNo,0)
							INNER JOIN payroll.tblDeptMaster TDM ON TDM.DeptNo=IRM.DeptNo
							Left JOIN payroll.tblDivisionMaster DM ON IRM.DivisionNo = DM.DivisionNo
							LEFT JOIN masterdata.mCostCenterMaster CCM ON CCM.CostCenterNo=IRID.CostCenterNo
							LEFT JOIN Masterdata.mMakeMaster MM ON MM.MakeNo=IRID.MakeNo
							Left join Security.Login CRT On IRM.CreatedBy = CRT.LoginNo
							Left join Security.Login AUT On IRM.AuthorizedBy = AUT.LoginNo
							Where Isnull(IID.POQty,0) < IRID.Quantity -----Pending Request
							      And IRID.ItemNo = @ItemNo
							      And IRM.DocumentDate<=@DocumentDate
							      And IRM.CompanyNo=@CompanyNo
							Group By DM.DivisionName
									,TDM.DeptName 
									,CCM.CostCenterName
									,IRM.DocumentNoYearly 
									,IRM.DocumentDate 
									,MM.MakeName
									,CRT.PrintingName 
									,AUT.PrintingName 
								 
								  
						 for xml path('tr')),'<td1>','<td style="text-align: center;">')
						                    ,'<td1>','<td>')
										    ,'<td2>','<td>')
											,'<td3>','<td>')
										    ,'<td4>','<td style="text-align: center;">')
											,'<td5>','<td>')
											,'<td6>','<td>')
											,'<td7>','<td style="text-align: right;">>')
											,'<td8>','<td style="text-align: right;">')
											,'<td9>','<td style="text-align: right;">>')
											,'<td10>','<td>')
											,'<td11>','<td>')
											)
								
				   
	set @str=@str+'</table>'
	
	select @str
END
