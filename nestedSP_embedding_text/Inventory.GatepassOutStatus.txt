-- =============================================
-- Author:		Akhil
-- Create date: 12 march 2012
-- Description:	Update Status of gatePass
-- =============================================
CREATE PROCEDURE [Inventory].[GatepassOutStatus] 
	-- Add the parameters for the stored procedure here
	@gatePassOutNo int  , 
	@ItemNo int ,
	@MakeNo smallint,
	@GRNNo int = null,
	@StatusNo tinyint,
	@OutputStatus bit output,
	@Result varchar(100) output

AS
BEGIN
begin try
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;
	declare @gatePassInQty as decimal(12,3)

if exists(select 1 from Inventory.GatePassOutMain where GatePassOutNo = @GatePassOutNo and GatePassTypeNo 
		in(2,5,6,8,9,10,11,12) 
					and DocumentStatusNo=30)
  update Inventory.GatePassOutItemDetail set Statusno = 2 where GatepassOutNo= @gatepassoutNo
				and itemno = @ItemNo and isnull(MakeNo,0) = isnull(@MakeNo,0) 
			and isnull(grnno,0) = isnull(@GRNNo,0) 

else
begin					      
	select @gatePassInQty =isnull(sum(receivedQty),0) from Inventory.GatePassInItemDetail  where GatepassOutNo= @GatePassOutNo
		and ItemNo = @ItemNo and isnull(MakeNo,0) = isnull(@MakeNo,0) 
			and Isnull(GRNNo ,0) = isnull(@GRnNo,0)
	select @gatePassInQty = @gatePassInQty + isnull(SUM(gid.AcceptedQty),0)   from Inventory.GRNMain grn
					inner join Inventory.GRNItemDetail gid on gid.GRNNo = grn.GRNNo
					inner join Inventory.GatePassOutItemDetail god on god.GatePassOutNo = @GatePassOutNo and gid.ItemNo = god.ItemNo
					inner join Inventory.GatePassOutMain gom on gom.GatePassOutNo = god.GatePassOutNo 
					inner join Inventory.GRNSCOrderRequisitionDetail grd on grd.GRNNo = grn.GRNNo 
						and grd.SCReqNo = gom.SCReqNo 
					inner join Inventory.GRNSCOrderDetail  grsd on grsd.GRNSCOrderDetailNo = grd.GRNSCOrderDetailNo 
						inner join Inventory.SCReqItemDetail srd on srd.SCReqNo = grd.SCReqNo and srd.itemno = god.ItemNo 
							and (grsd.ServiceItemNo = srd.ServiceItemNo or srd.ServiceItemNo is null) -- sometime srd.ServiceItemNo is null repairing case
					where grn.GatepassOutNo = @GatePassOutNo and god.ItemNo = @ItemNo and isnull(god.MakeNo ,0 ) =  ISNULL(@MakeNo,0)
					

	update Inventory.GatePassOutItemDetail set Statusno = 2 where GatepassOutNo= @gatepassoutNo
				and itemno = @ItemNo and isnull(MakeNo,0) = isnull(@MakeNo,0) 
			and isnull(grnno,0) = isnull(@GRNNo,0) and Quantity <= isnull(@gatePassInQty,0)

	update Inventory.GatePassOutItemDetail set Statusno = 1 where GatepassOutNo= @gatepassoutNo
				and itemno = @ItemNo and isnull(MakeNo,0) = isnull(@MakeNo,0) 
			and isnull(grnno,0) = isnull(@GRNNo,0) and Quantity > isnull(@gatePassInQty,0)
				and Statusno = 2
	
	update Inventory.GatePassOutItemDetail set Statusno = 8 where GatepassOutNo= @gatepassoutNo
				and itemno = @ItemNo and isnull(MakeNo,0) = isnull(@MakeNo,0) 
			and isnull(grnno,0) = isnull(@GRNNo,0) and  isnull(@gatePassInQty,0)>0
				and Statusno <>2
end
   
	if exists(select 1 from Inventory.GatePassOutItemDetail where  gatepassOutNo = @GatePassOutno and statusNo in(1,8))
	 begin
	   if exists (select 1 from Inventory.GatePassInItemDetail where GatePassOutNo = @GatePassOutNo)
		or exists(select 1 from Inventory.GRNMain where GatepassOutNo = @gatePassOutNo )
		 begin
		  update Inventory.GatePassOutMain set StatusNo = 8 where gatepassOutNo = @GatePassOutNo
		 end
	   else
	    begin
			update Inventory.GatePassOutMain set StatusNo = 1 where gatepassOutNo = @GatePassOutNo
		end
	 end
	else if exists(select 1 from Inventory.GatePassOutItemDetail 
				where  gatepassOutNo = @GatePassOutNo and statusNo =2)--completed
		begin
			update Inventory.GatePassOutMain set StatusNo = 2 where  gatepassOutNo = @GatePassOutNo
		end
				
 

set @OutputStatus =1
set @Result='Success'
end try

begin catch
	set @OutputStatus =0
	set @Result=error_message()

	select 0, error_message()
end Catch


END
