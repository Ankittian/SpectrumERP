CREATE PROCEDURE  [Inventory].[InvLotMapCalculation] 
(@Documentno int ,@TransactionFlag varchar(4)
 , @OutputStatusNo bit output
 ,@OutputStatusRemarks varchar(1000) output
)
as
begin

Begin try

declare @qty decimal(12,3) =0 , @MAPAmount decimal(18,6) =0,@itemNo int =0

if not exists  (select 1 from Inventory.InvLotBillDetail a,masterdata.mItemOtherInformation  im
			 where BillNo =@Documentno and TransactionFlag =@TransactionFlag and im.ItemNo = a.ItemNo and im.StockValueMethodNo  = 4)
begin
	select @OutputStatusNo = 1 ,@OutputStatusRemarks = 'Success'
	return
end
if not exists  (select 1 from Inventory.InvLotDetail a,masterdata.mItemOtherInformation  im
			 where DocumentNo  =@Documentno and TransactionFlag =@TransactionFlag and im.ItemNo = a.ItemNo and im.StockValueMethodNo  = 4)
begin
	select @OutputStatusNo = 1 ,@OutputStatusRemarks = 'Success'
	return
end	
	 
update bd set  
			 @qty = case when @itemNo=0 or @itemNo = ls.ItemNo then @qty + ls.BalanceQty else ls.BalanceQty  end  
			 ,@MAPAmount = case when @itemNo=0 or @itemNo = ls.ItemNo then 
					@MAPAmount +   bd.BillQty * bd.VRate    
				else   bd.BillQty * bd.VRate   
				end  
		 , ClosingQty = dt.ClosingQty + @qty 
		 , @Itemno = ls.ItemNo 
		, MAPAmount =  (dt.ClosingQty * dt.MAPRate) + @MAPAmount 
		,Status ='A'
		--,MAPDiffRate =   (( (dt.ClosingQty * dt.MAPRate) + (case when bd.TransactionFlag = ls.TransactionFlag then bd.BillQty * bd.VRate else 
		--				(ls.BalanceQty * ls.MAPRate ) + (bd.BillQty * bd.DiffRate) end) )
		--			/(dt.ClosingQty  + ls.BalanceQty  ))
		--					 - dt.MAPRate  
		,MAPDiffRate =   (((dt.ClosingQty * dt.MAPRate) + @MAPAmount )/(dt.ClosingQty + @qty) ) - dt.MAPRate 
	from Inventory.InvLotBillDetail bd With (Index = invLotBillDetail_Createddate_IX)
	inner join Inventory.InvLotSummary ls on ls.InvLotSummaryNo = bd.InvLotSummaryNo 
	inner join masterdata.mItemOtherInformation ioi on ioi.ItemNo = ls.ItemNo 
	outer apply 
	(select isnull(sum(BalanceQty),0)ClosingQty ,isnull(Max(MAPRate),0) MAPRate from
	Inventory.InvLotSummary  cls
	where cls.itemno = ls.ItemNo  and cls.warehouseno = ls.warehouseno 
	and cls.companyno = ls.CompanyNo and isnull(cls.Divisionno,0) = isnull(ls.DivisionNo,0)
	and isnull(cls.MakeNo,0)=ISNULL(ls.MakeNo,0)
	and cls.isCapital=0 and balanceQty>0      
	and not exists (select 1 where isnull(cls.DocumentNo,cls.InvLotSummaryNo)  =  isnull(ls.DocumentNo,ls.InvLotSummaryNo) and cls.TransactionFlag =ls.TransactionFlag) 
	) dt
	where isnull(bd.BillNo,bd.InvLotSummaryNo) = @Documentno and bd.TransactionFlag = ls.TransactionFlag 
	and bd.TransactionFlag = @TransactionFlag  
	and ls.IsCapital =0 and ioi.StockValueMethodNo = 4 
 
--update bd set 
--MAPDiffRate = case when bd.InvLotBillDetailNo = tt.InvLotBillDetailNo then
--					bd.MAPDiffRate
--					else 0 end

--from  Inventory.InvLotSummary pls  
--inner join Inventory.Setting s on s.CompanyNo = pls.CompanyNo 
--inner join inventory.InvLotBillDetail bd on bd.InvLotSummaryNo = pls.InvLotSummaryNo 
--outer apply(
--select ROW_NUMBER() over (order by nrbd.ItemNo,  InvLotBillDetailNo desc )SLNo, MAPAmount / ClosingQty NewRate  ,InvLotBillDetailNo
--			from inventory.invLotBillDetail nrbd 
--			inner join Inventory.InvLotSummary nrls on  nrbd.InvLotSummaryNo = nrls.InvLotSummaryNo 
--	where nrbd.BillNo = @Documentno   and nrbd.TransactionFlag = @TransactionFlag 
--	and nrls.IsCapital =0  --and nrls.DocumentNo = pls.DocumentNo and nrls.TransactionFlag = pls.TransactionFlag
--			and nrls.ItemNo = pls.ItemNo and ISNULL(nrls.MakeNo,0) = ISNULL(pls.MakeNo,0) 
--			and nrls.CompanyNo = pls.CompanyNo and ISNULL(nrls.DivisionNo ,0) = ISNULL(pls.DivisionNo ,0)
--			and nrls.WarehouseNo = pls.WarehouseNo and ClosingQty>0 
-- ) tt
-- where bd.BillNo = @Documentno  and bd.TransactionFlag = @TransactionFlag and SLNo=1 and 1=2
   
update ls set MAPRate   = NewRate 
from Inventory.InvLotSummary ls
inner join Inventory.InvLotSummary pls on pls.CompanyNo = ls.CompanyNo  
and ISNULL(pls.DivisionNo,0) = ISNULL(ls.DivisionNo,0)
and pls.ItemNo = ls.ItemNo and ISNULL(ls.MakeNo,0) = ISNULL(pls.MakeNo,0)
and pls.WarehouseNo = ls.WarehouseNo and ls.IsCapital = pls.IsCapital and pls.IsCapital =0
inner join inventory.InvLotBillDetail bd on bd.InvLotSummaryNo = pls.InvLotSummaryNo 
outer apply(
select ROW_NUMBER() over (order by InvLotBillDetailNo desc )SLNo, MAPAmount / ClosingQty NewRate  
			from inventory.invLotBillDetail nrbd 
			inner join Inventory.InvLotSummary nrls on  nrbd.InvLotSummaryNo = nrls.InvLotSummaryNo 
	where isnull(nrbd.BillNo,nrbd.InvLotSummaryNo ) = @Documentno   and nrbd.TransactionFlag = @TransactionFlag 
	and nrls.IsCapital =0 and isnull(nrls.DocumentNo,nrls.InvLotSummaryNo) =ISNULL( pls.DocumentNo,pls.InvLotSummaryNo)
	 and nrls.TransactionFlag = pls.TransactionFlag
			and nrls.ItemNo = pls.ItemNo and ISNULL(nrls.MakeNo,0) = ISNULL(pls.MakeNo,0) 
			and nrls.CompanyNo = pls.CompanyNo and ISNULL(nrls.DivisionNo ,0) = ISNULL(pls.DivisionNo ,0)
			and nrls.WarehouseNo = pls.WarehouseNo  
 ) tt
 where isnull(bd.BillNo,bd.InvLotSummaryNo) = @Documentno  and bd.TransactionFlag = @TransactionFlag and SLNo=1  and NewRate is not null
 
 update bd set ActualMAPRate = ls.MAPRate
	from Inventory.InvLotBillDetail bd
		inner join Inventory.InvLotSummary ls on ls.InvLotSummaryNo = bd.InvLotSummaryNo 
		where bd.BillNo = @Documentno and bd.TransactionFlag = @TransactionFlag 

 update ld set MAPRate  = ls.MAPRate
	from Inventory.InvLotBillDetail bd
		inner join Inventory.InvLotDetail ld on ld.InvLotDetailNo = bd.InvLotDetailNo 
		inner join Inventory.InvLotSummary ls on ls.InvLotSummaryNo = bd.InvLotSummaryNo 
		where bd.BillNo = @Documentno and bd.TransactionFlag = @TransactionFlag  
 
 
 select @OutputStatusNo  = 1,@OutputStatusRemarks = 'Success'
end try

begin catch
	select @OutputStatusNo  = 0,@OutputStatusRemarks = ERROR_MESSAGE ()
end catch
		
end
