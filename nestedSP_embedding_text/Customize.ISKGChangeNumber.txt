CREATE PROCEDURE  [Customize].[ISKGChangeNumber] 
	 @GRNNo int,
	 @LoginNo smallint
	 ,@OutputStatusNo tinyint = null output
	,@OutputStatusRemarks varchar(4000)  = null output
AS
 
BEGIN

begin try
 declare @DocumentDate smalldatetime, @DocumentSettingNo smallint ,@Status bit, @NewDocumentNo varchar(1000),@CompanyNo tinyint , @DivisionNo int
 ,@DocumentNoYearly varchar(30)  
 
  select @DocumentDate =DocumentDate , @DocumentSettingNo = DocumentSettingNo ,@DocumentNoYearly = DocumentNoYearly ,@CompanyNo = CompanyNo 
	,@DivisionNo = DivisionNo 
    from  Inventory.GRNMain where GRNNo = @GRNNo 
    
if exists(select 1 from Inventory.Setting where CompanyNo = @CompanyNo and GRNNoRearrange =1)
begin

		if @DocumentSettingNo is not null
			begin
				update Inventory.GRNMain set DocumentNoYearly ='' where GRNNo = @GRNNo 

				EXEC	[Customize].[ISKGNumberGeneration]
						@DocumentDate = @DocumentDate,
						@DocumentSettingNo = @DocumentSettingNo,
						@Status = @Status OUTPUT,
						@Remarks = @NewDocumentNo OUTPUT

						if @Status = 0
							raiserror(@NewDocumentNo,16,-1)
						 
			end
		

		exec [Utility].[UCYNSaveMain] @ChangeYearlyNumberNo  = null ,@FormNo = 65,@DocumentId  = @GRNNo ,@DocumentNo = @DocumentNoYearly
		,@DocumentSettingNo  = @DocumentSettingNo  ,@NewDocumentNo  = @NewDocumentNo ,@LoginNo = @LoginNo ,@DocumentDate = @DocumentDate 
		, @OutputStatusNo = @OutputStatusNo  output,@OutputStatusRemarks = @OutputStatusRemarks  output

			
	declare @Detail as table (IDNo smallint identity(1,1) not null, GRNNo int)
	declare @iROW smallint , @iCount smallint
	insert into @Detail 
	select GRNNo from Inventory.GRNMain where GRNNo <>@GRNNo and AuthorizedBy is null and DocumentDate  =@DocumentDate 
			and CompanyNo = @CompanyNo and ISNULL(DivisionNo,0) = ISNULL(@DivisionNo,0)
	set @iROW = @@ROWCOUNT 
	set @iCount = 0
	
	update grn set DocumentNoyearly =''
	from Inventory.GRNMain grn
	inner join @Detail dd on dd.GRNNo = grn.GRNNo 
	
	while @iROW > @iCount 
	begin
		select @GRNNo = GRNNO from @Detail where IDNo = @iCount + 1
					if @DocumentSettingNo is not null
					begin
						

						EXEC	[masterdata].[DocumentNumberGeneration]
								@DocumentDate = @DocumentDate,
								@DocumentSettingNo = @DocumentSettingNo,
								@Status = @Status OUTPUT,
								@Remarks = @NewDocumentNo OUTPUT

								if @Status = 0
									raiserror(@NewDocumentNo,16,-1)
								 
					end
			 	
			update Inventory.GRNMain set DocumentNoYearly = @NewDocumentNo where GRNNo = @GRNNo 
		
	   set @iCount = @iCount + 1
	end
end

select 1 ,'Success' 
set @OutputStatusNo = 1
set @OutputStatusRemarks ='Success'

end try
begin catch
	select 0, ERROR_MESSAGE ()
	set @OutputStatusNo = 0
set @OutputStatusRemarks = ERROR_MESSAGE ()
end catch


END
