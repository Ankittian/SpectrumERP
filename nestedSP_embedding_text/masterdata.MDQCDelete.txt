-- Description:	Delete QC Parameters
-- =============================================
CREATE PROCEDURE [masterdata].[MDQCDelete] 	
	@QCParameterNo smallint
	,@OutputStatus bit = Null output
	,@OutputRemarks varchar(100) = Null output	
AS
BEGIN
	SET NOCOUNT ON;

	BEGIN TRY
		begin tran	
			Delete From [masterdata].[mQCParametersCompanyDetail] Where QCParameterNo=@QCParameterNo
			Delete From [masterdata].[mQCParameters] Where QCParameterNo=@QCParameterNo

			Select 1
			Select @OutputStatus = 1,@OutputRemarks = 'Success'

		commit tran
	END TRY
	BEGIN CATCH
			rollback tran
			Select 0,Error_message()
			Select @OutputStatus = 0,@OutputRemarks = Error_message()
	END CATCH    
END
