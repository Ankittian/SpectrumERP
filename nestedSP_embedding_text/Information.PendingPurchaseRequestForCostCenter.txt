CREATE PROCEDURE [Information].[PendingPurchaseRequestForCostCenter] 
	-- Add the parameters for the stored procedure here
	@CompanyNo tinyint,
	@DocumentDate smalldatetime,
	@CostCenterNo int,
	@ItemNo Int 
AS
BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;

    -- Insert statements for procedure here
	
    
	declare @str varchar(max)
    select @str=masterdata.JSForInformation('MMPR')
    set @str=@str+masterdata.CSSForInformation('MMPR')
	set @str=@str+'<table class="altrowstable" id="alternatecolorPUPRInfoCost">'
	set @str=@str+'<Caption><Font color="Maroon"><h3 align="Left"><b>Pending Purchase Request</b></h3></Font></Caption>'
    set @str=@str+'<tr><th>Sl. No.</th><th>Division</th><th>Department</th><th>Document No.</th><th>Document Date</th><th>Make</th><th>Quantity</th><th>Balance Quantity</th><th>Created By</th><th>Authorized By</th></tr>'
	set @str=@str
				 +(REPLACE( 
				    REPLACE(
				    REPLACE(
				    REPLACE(
				    REPLACE(
				    replace(
					replace(
					replace(
					replace(
					replace(
					    (    
							 Select ROW_NUMBER() over ( Order by IM.DocumentDate desc) as td1
					                ,isnull(DM.DivisionName,'') td2
					                ,Dt.DeptName td3
					                , IM.DocumentNoYearly td4
									,convert(varchar(20),IM.DocumentDate,103)  td5
									,isnull(MM.MakeName,'') td6
									,cast(Sum(ID.IndentQty) as decimal(12,3)) td7
									,isnull(cast( Sum(ID.IndentQty -isnull(ID.POQty,0)) as decimal(12,3)),0) td8
									,L.PrintingName td9
									,isnull(L1.PrintingName,'')  td10		    
							From Inventory.IndentMain IM
							Inner Join Inventory.IndentItemDetail ID ON ID.IndentNo=IM.IndentNo
							Inner Join Inventory.IndentRequisitionMain IRD On IM.RefDocumentNo=IRD.IndentRequisitionNo 
							                                               and IM.IndentAgainstNo=3
							Inner Join GlobalData.IndentType IT ON IT.IndentTypeNo=IM.IndentTypeNo
							Inner Join payroll.tblDeptMaster DT ON DT.DeptNo=IM.DeptNo
							Left Join masterdata.mMakeMaster MM ON MM.MakeNo=ID.MakeNo
							Left Join payroll.tblDivisionMaster DM ON DM.DivisionNo=DT.DivisionNo
							Inner Join Security.Login L ON L.LoginNo=IM.CreatedBy
							Left Join Security.Login L1 On L1.LoginNo=IM.AuthorizedBy
							Where  (ID.IndentQty-isnull(ID.POQty,0)) > 0 -----For Pending 
							       And ID.CostCenterNo=@CostCenterNo 
							       And IM.CompanyNo=@CompanyNo 
							       And IM.DocumentDate<=@DocumentDate 
							       And ID.ItemNo=@ItemNo
							Group By DM.DivisionName,Dt.DeptName, IM.DocumentNoYearly
									,IM.DocumentDate,MM.MakeName,L.PrintingName,L1.PrintingName
							
														       
					 for xml path('tr')
			        ),'<td1>','<td style="text-align: center;">')
			         ,'<td2>','<td>')
					 ,'<td3>','<td>')
					 ,'<td4>','<td>')
					 ,'<td5>','<td  style="text-align: center;">')
					 ,'<td6>','<td>')
					 ,'<td7>','<td  style="text-align: Right;">')
					 ,'<td8>','<td  style="text-align: Right;">')
					 ,'<td9>','<td>')
					 ,'<td10>','<td>'))
								
				   
	set @str=@str+'</table>'
	
	select @str
END
