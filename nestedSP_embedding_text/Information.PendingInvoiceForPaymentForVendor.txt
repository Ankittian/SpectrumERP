Reads: Finance.accVoucherDetail, Finance.ARDVendorLocationRule, Finance.CashCounterMain, Finance.CashJournalMain, Finance.PurchaseBillItemDetail, Finance.PurchaseBillMain, Finance.ReferenceMain, Finance.ServiceBillItemDetail, Finance.ServiceBillMain, Inventory.GRNMain, masterdata.mAccountMaster, masterdata.mVendorLocationDetail, payroll.tblDeptMaster, payroll.tblDivisionMaster, Security.Login

CREATE PROCEDURE [Information].[PendingInvoiceForPaymentForVendor]  
	-- Add the parameters for the stored procedure here
	@CompanyNo tinyint,
	@DocumentDate smalldatetime,
	@VendorLocationNo int
AS
BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;

    -- Insert statements for procedure here
	declare @str varchar(max)
    select @str=masterdata.JSForInformation('PMPO')
    set @str=@str+masterdata.CSSForInformation('PMPO')
	set @str=@str+'<table class="altrowstable" id="alternatecolorMRNPendingInvoiceForPaymentInfo">'
	set @str=@str+'<Caption><Font color="Maroon"><h3 align="Left"><b>Pending Invoice For Payment</b></h3></Font></Caption>'
    set @str=@str+'<tr><th>Sl. No.</th><th>Division</th><th>Department</th><th>Document No.</th><th>Document Date</th><th>Invoice Amount</th><th>Balance Amount</th><th>Due Date</th><th>Created By</th><th>Authorized By</th></tr>'
	set @str=@str
				 +( replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					(Select ROW_NUMBER() over ( Order by td5 desc) as td1,
					        td2,
							td3,
							td4,
							td5,
							td6,
							td7,
							td8,
							td9,
							td10
					 From
					(Select   ------ for Purchase Bill
							 Isnull(DM.DivisionName,'') td2
							,Isnull(DE.DeptName,'') td3
							,PBM.Documentnoyearly td4
							,convert(varchar(20),PBM.DocumentDate,103) td5
							,Cast(PBM.NetAmount as Decimal(15,2)) td6
							,Cast(tt.BalanceAmount as Decimal(15,2)) td7
							,convert(varchar(20),tt.DueDate,103) td8
							,L1.PrintingName td9
							,Isnull(L2.PrintingName,'') td10
					  From Finance.PurchaseBillMain as PBM
					  Inner Join Finance.PurchaseBillItemDetail PBI ON PBM.PurchaseBillNo=PBI.PurchaseBillNo-----Group by Clause For this
					  Inner Join Inventory.GRNMain GRNM ON GRNM.GRNNo=PBI.GRNNo
					  LEFT Join payroll.tblDivisionMaster DM ON DM.DivisionNo=GRNM.DivisionNo
					  LEFT join payroll.tblDeptMaster DE ON DE.DeptNo=GRNM.DeptNo
					  Inner Join Security.Login L1 On L1.LoginNo=PBM.CreatedBy
					  Left Join Security.Login L2 On L2.LoginNo=PBM.AuthorizedBy
					  Inner Join
								  (Select AVD.VoucherNo
								          ,Abs(RM.BalanceAmount) BalanceAmount
								          ,ALD.VendorLocationNo
								          ,RM.DueDate  
									From Finance.accVoucherDetail AVD
									Inner Join Finance.ARDVendorLocationRule ALD ON ALD.AccountNo=AVD.AccountNo 
									Inner Join masterdata.mAccountMaster AM ON ALD.AccountNo=AM.AccountNo And AM.AccountTypeNo=4
									Inner Join Finance.ReferenceMain RM ON RM.VoucherDetailNo=AVD.VoucherDetailNo 
																	and RM.AccountNo=ALD.AccountNo
									Union all
									                              
									Select SB.VoucherNo -----Payment mode Cash
										  ,Abs((CM.NetAmount-SUM(CJ.Amount))) BalanceAmount
										  ,VL.VendorLocationNo,CM.DueDate
									From Finance.CashCounterMain CM 
									Inner Join Finance.PurchaseBillMain SB ON SB.PurchaseBillNo=CM.DocumentNo and SB.PaymentModeNo=1
									Inner Join Finance.CashJournalMain CJ ON CM.CashCounterNo=isnull(CJ.CashCounterNo,0)
																				   and CM.SourceDocumentFormNo=202---Purchase Bill 
									Inner Join masterdata.mVendorLocationDetail VL On VL.VendorLocationNo=CM.PayeeLocationNo                                                                  
									Group by SB.VoucherNo,VL.VendorLocationNo ,CM.DueDate,CM.NetAmount
									 )tt ON tt.VoucherNo=PBM.VoucherNo 
										 And tt.VendorLocationNo=PBM.VendorLocationNo
						                                    
					Where tt.BalanceAmount<>0   --- for pending payment
									And PBM.VendorLocationNo=@VendorLocationNo
									And PBM.DocumentDate<=@DocumentDate
									And PBM.CompanyNo=@CompanyNo
					Group by DM.DivisionName
							,DE.DeptName
						    ,PBM.Documentnoyearly 
							,PBM.DocumentDate 
							,PBM.NetAmount 
							,tt.BalanceAmount 
							,tt.DueDate 
							,L1.PrintingName
							,L2.PrintingName 
					Union All
							Select   ---- For Service Bill
											 Isnull(DM.DivisionName,'')
											,Isnull(DE.DeptName,'')
											,SBM.Documentnoyearly 
											,convert(varchar(20),SBM.DocumentDate,103)
											,Cast(SBM.NetAmount as Decimal(15,2))
											,cast(tt.BalanceAmount as Decimal(15,2))
											,convert(varchar(20),tt.DueDate,103)
											,L1.PrintingName
											,Isnull(L2.PrintingName,'')
							From Finance.ServiceBillMain as SBM
							Inner Join Finance.ServiceBillItemDetail SBI ON SBM.ServiceBillNo=SBI.ServiceBillNo-----Group by Clause For this
							Inner Join Inventory.GRNMain GRNM ON GRNM.GRNNo=SBI.GRNNo
							LEFT Join payroll.tblDivisionMaster DM ON DM.DivisionNo=GRNM.DivisionNo
							LEFT join payroll.tblDeptMaster DE ON DE.DeptNo=GRNM.DeptNo
							Inner Join Security.Login L1 On L1.LoginNo=SBM.CreatedBy
							Left Join Security.Login L2 On L2.LoginNo=SBM.AuthorizedBy
							Inner Join
									(Select AVD.VoucherNo,Abs(RM.BalanceAmount) BalanceAmount,ALD.VendorLocationNo,RM.DueDate  
									From Finance.accVoucherDetail AVD
									Inner Join Finance.ARDVendorLocationRule ALD ON ALD.AccountNo=AVD.AccountNo 
									Inner Join masterdata.mAccountMaster AM ON ALD.AccountNo=AM.AccountNo And AM.AccountTypeNo=4
									Inner Join Finance.ReferenceMain RM ON RM.VoucherDetailNo=AVD.VoucherDetailNo 
																	and RM.AccountNo=ALD.AccountNo
									Union all
									                              
									Select SB.VoucherNo -----Payment mode Cash
										  ,Abs((CM.NetAmount-SUM(CJ.Amount))) BalanceAmount
										  ,VL.VendorLocationNo,CM.DueDate
									From Finance.CashCounterMain CM 
									Inner Join Finance.ServiceBillMain SB ON SB.ServiceBillNo=CM.DocumentNo and SB.PaymentModeNo=1
									Inner Join Finance.CashJournalMain CJ ON CM.CashCounterNo=isnull(CJ.CashCounterNo,0)
																				   and CM.SourceDocumentFormNo=457---Service Bill 
									Inner Join masterdata.mVendorLocationDetail VL On VL.VendorLocationNo=CM.PayeeLocationNo 
									Group by SB.VoucherNo,VL.VendorLocationNo ,CM.DueDate,CM.NetAmount
									)tt ON tt.VoucherNo=SBM.VoucherNo 
										And tt.VendorLocationNo=SBM.VendorLocationNo
							 Where tt.BalanceAmount<>0  -----For Pending Payment 
									And SBM.VendorLocationNo=@VendorLocationNo
									And SBM.DocumentDate<=@DocumentDate
									And SBM.CompanyNo=@CompanyNo                               
							Group by  DM.DivisionName
									 ,DE.DeptName
									 ,SBM.Documentnoyearly 
									 ,SBM.DocumentDate 
									 ,SBM.NetAmount 
									 ,tt.BalanceAmount
									 ,tt.DueDate
									 ,L1.PrintingName
									 ,L2.PrintingName 
					)tt						                                    
			for xml path('tr')
			),'<td1>','<td style="text-align: center;">')
			 ,'<td2>','<td>')
			 ,'<td3>','<td>')
			 ,'<td4>','<td>')
			 ,'<td5>','<td  style="text-align: center;">')
			 ,'<td6>','<td style="text-align: Right;">')
			 ,'<td7>','<td style="text-align: Right;">')
			 ,'<td8>','<td style="text-align: center;">')
		     ,'<td9>','<td>')
			 ,'<td10>','<td>')
	)
	
								   
	set @str=@str+'</table>'
	
	
	select @str
	
END
