CREATE PROCEDURE [Finance].[VoucherSaveChildDetail]
(
	 @VoucherNo int
	,@ChildVoucherNo int
	
	,@OutputStatusNo tinyint = null output
	,@OutputStatusRemarks varchar(4000) = null output
)
AS

BEGIN
	SET NOCOUNT ON;
	
	BEGIN TRY

	INSERT INTO [Finance].[accVoucherChildDetail]
           ([ParentVoucherNo]
           ,[ChildVoucherNo])
     VALUES
           (@VoucherNo
           ,@ChildVoucherNo)

	select @OutputStatusNo=1,@OutputStatusRemarks=ident_current('[Finance].[accVoucherChildDetail]')
	
	if exists(Select 1 from Finance.accVoucherMain avm
			  where avm.VoucherNo = @VoucherNo and VoucherTypeNo = 6)
	and exists(Select 1 from Finance.accVoucherMain avm
			   inner join Finance.accVoucherDetail avd on avd.VoucherNo = avm.VoucherNo 
			   inner join Finance.accVoucherReferenceDetail avrd on avrd.VoucherDetailNo = avd.VoucherDetailNo  
			  where avm.VoucherNo = @ChildVoucherNo and avm.VoucherTypeNo = 0
			  and avrd.ReferenceTypeNo = 1
			  --and 1=2
			  )
	Begin
	 Declare @AccountNature table(AccountNo int,AccountNatureNo tinyInt)

	 Insert into @AccountNature 
	 select AM.AccountNo,Case when AG.GroupTypeNo = 26 then 1
                         when AG.GroupTypeNo = 30 then 2
						 else 4 End
	from (select distinct COANo,PostingAccountNo,PostingGroupNo 
			from masterdata.COAAccountGroupDetail) COAAGD
	inner join masterdata.mFBMain FBM on FBM.OperativeCOANo=COAAGD.COANo
									and FBM.FBTypeNo=1
	inner join masterdata.mAccountMaster AM on AM.AccountNo=COAAGD.PostingAccountNo
	inner join masterdata.mAccountGroup AG on AG.GroupNo=COAAGD.PostingGroupNo
Group by AM.AccountNo,AG.GroupTypeNo 
Order by Count(Distinct AG.GroupTypeNo) desc

		declare @Reference table(ReferenceDetailNo int,ReferenceNos varchar(max),AccountNatureNo tinyint,SlNo int identity(1,1) PRIMARY KEY)
	
		insert into @Reference
		Select CVRD.ReferenceDetailNo
		,
		(
		Select Convert(varchar,IsNULL(PRM.ReferenceNo,0) ) + '~' + Convert(varchar, ABS(PRM.SpecificCrAmount - PRM.SpecificDrAmount)) + '`'
		from  Finance.ReferenceMain  PRM
		where PRM.VoucherDetailNo = Pvd.voucherDetailNo
		For XML path('')
		) ReferenceNos
		,case when An.AccountNatureNo = 1 then case when Pvd.ByToNo = 1 then 1 else 2 End
		      when An.AccountNatureNo = 2 then Case when Pvd.BytoNo = 1 then 1 else 2 End
			  else Case when Pvd.BytoNo = 1 then 4 else 3 End End
		from Finance.accVoucherDetail CVD
		inner join Finance.accVoucherReferenceDetail CVRD on CVRD.VoucherDetailNo = CVD.VoucherDetailNo and CVRD.ReferenceTypeNo = 1
		inner join Finance.accVoucherDetail Pvd on Pvd.AccountNo = CVD.AccountNo and Pvd.VoucherNo = @VoucherNo and CVD.VoucherNo = @ChildVoucherNo
		inner join @AccountNature An on An.AccountNo = Pvd.AccountNo 

		declare @MaxCount int
		set @MaxCount = @@RowCount

		declare @Count int
		set @Count=1

		Declare @ReferenceDetailNo int,@ReferenceNos varchar(max),@AccountNatureNo tinyint
		,@RefOutputStatusNo tinyint
		,@RefOutputStatusRemarks varchar(4000)
		
		while @Count <= @MaxCount
		begin
		Select @ReferenceDetailNo = ReferenceDetailNo,
		@ReferenceNos = ReferenceNos,
		@AccountNatureNo = AccountNatureNo 
		from @Reference 
		where SlNo = @Count 

		exec [Finance].[FBRRSaveReference]
	    @ReferenceDetailNo = @ReferenceDetailNo 
		,@ReferenceNos = @ReferenceNos 
		,@AccountNatureNo = @AccountNatureNo
		,@OutputStatusNo = @RefOutputStatusNo output
		,@OutputStatusRemarks = @RefOutputStatusRemarks output

		if @RefOutputStatusNo = 0 
		   RaisError(@RefOutputStatusRemarks,16,-1)


		select @Count=@Count+1
		End

		
	End

	Select 1,@OutputStatusRemarks --@MaxCount -- @OutputStatusRemarks

	END TRY
	BEGIN CATCH
		select @OutputStatusNo=0,@OutputStatusRemarks=Error_Message()
		Select 0,@OutputStatusRemarks
	END CATCH
END
