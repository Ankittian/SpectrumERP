Stored procedure Information.ItemWiseBin: Get Stock Info. Called from Inventory.IMMIGetItemInfo.

Parameters: @ItemNo int

-- =============================================
-- Author:		Anjum
-- Create date: 2013/05/29
-- Description:	Get Stock Info
-- =============================================
CREATE PROCEDURE [Information].[ItemWiseBin] 
	-- Add the parameters for the stored procedure here
	@ItemNo int
AS
BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;

    -- Insert statements for procedure here
	declare @str varchar(max)
	declare @binHtml varchar(max)
	declare @tdCount int 


    select @str=masterdata.JSForInformation('MMPR')
    set @str=@str+masterdata.CSSForInformation('MMPR')
	set @str=@str+'<table class="altrowstable" id="alternatecolorStockInfo">'
	set @str=@str+'<Caption><Font color="Maroon"><h3 align="Left"><b>Bin Location Detail</b></h3></Font></Caption>'
    set @str=@str+'<tr><th style="width: 100px;">Sl. No.</th><th>Bin Location</th></tr>'

			set @binHtml = isnull((
			replace(
				replace((
					Select 
						ROW_NUMBER() over (order by IM.ItemNo) as td1,
						STUFF((
							SELECT ', ' + ISNULL(BC.Location, ' ') 
							From masterdata.mBinCardItemDetail BCID 
							Inner Join masterdata.mBinCard BC On BC.BinNo = BCID.BinNo
							Where BCID.ItemNo = IM.ItemNo
							FOR XML PATH(''), TYPE
						).value('.', 'NVARCHAR(MAX)'), 1, 2, '') As td2
					From masterdata.mItemMaster IM
					Where IM.ItemNo = @ItemNo
					for xml path('tr')
				), '<td1>', '<td style="text-align: Center;">'), 
				'<td2>', '<td>'
			)
		), '')

		Select @binHtml
	-- Only append row data if binHtml is not empty (meaning some Bin Locations exist)
		set @tdCount = (len(@binHtml) - len(replace(@binHtml, '<td', ''))) / len('<td')

		-- Only append rows if 2 or more <td> tags are found (i.e., data is valid)
		Select @tdCount
		if @tdCount > 1
			set @str = @str + @binHtml

		set @str = @str + '</table>'

		select @str
END
