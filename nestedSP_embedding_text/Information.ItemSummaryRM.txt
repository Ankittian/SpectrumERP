CREATE PROCEDURE [Information].[ItemSummaryRM] 
	-- Add the parameters for the stored procedure here
	@CompanyNo tinyint,
	@DocumentDate smalldatetime,
	@ItemNo int
AS
BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;

    -- Insert statements for procedure here
	

	declare @str varchar(max)
    select @str=masterdata.JSForInformation('ISRM')
    set @str=@str+masterdata.CSSForInformation('ISRM')
	set @str=@str+'<table class="altrowstable" id="alternatecolorItemSummaryRMInfo">'
	set @str=@str+'<Caption><Font color="Maroon"><h3 align="Left"><b>Item Summary</b></h3></Font></Caption>'
    set @str=@str+'<tr><th>Sl. No.</th><th>Month</th><th>Received Quantity</th><th>Received Amount</th><th>Issued Quantity</th><th>Issued Amount</th><th>Returned Quantity</th><th>Returned Amount</th></tr>'
	set @str=@str
				 +( REPLACE(
					replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					(  Select ROW_NUMBER() over (Order By Month(ld.DocumentDate)) as td1
					          ,DATENAME(MM,ld.DocumentDate)+' '+DATENAME(YY,ld.DocumentDate) as td2
							  ,SUM(isnull(ld.ReceivedQty,0)) as td3
							  ,isnull(tt2.ReceivedAmt,0) as td4
							  ,sum(isnull(tt3.[Issued Qty.],0)) as td5
							  ,sum(isnull(tt3.[Issued Amount],0)) as td6
							  ,sum(isnull(tt4.[Return Qty.],0)) as td7
							  ,sum(isnull(tt4.[Return Amt.],0)) as td8
From Inventory.InvLotDetail ld
Left Join (---Received Qty.
           Select  datename(MM,INV.DocumentDate)+' '+datename(YY,INV.DocumentDate) Date----Received Amt.
					                         ,ILS.ItemNo
					                         ,MONTH(INV.DocumentDate) M
									         ,Convert(money,sum(ilbd.LotBalanceQty * 
									              (Case when ILBD.TransactionFlag = ILS.TransactionFlag 
									                    then ILBD.VRate 
									               else ILBD.DiffRate end
									               )
									             )) ReceivedAmt
											   from Inventory.InvLotSummary ILS 
											   Inner Join Inventory.InvLotDetail INV 
											                          ON INV.InvLotSummaryNo=ILS.InvLotSummaryNo
											   Inner join Inventory.InvLotBillDetail ILBD 
											                          On ILBD.InvLotDetailNo = INV.InvLotDetailNo
											   Where ILS.TransactionFlag='DRSN' 
											         And  ILBD.BillDate <=@DocumentDate
													 And ILS.CompanyNo=@CompanyNo 
													 and ILS.ItemNo = @ItemNo---@ItemNo
													 And ILS.StockNatureNo=1 
													 And Inv.ReceivedQty>0                      
											   Group by  ILS.ItemNo
											            ,datename(MM,INV.DocumentDate)+' '
											            +datename(YY,INV.DocumentDate),MONTH(INV.DocumentDate)
												
							                )tt2 On tt2.ItemNo=ld.ItemNo
												 And tt2.Date= DATENAME(MM,ld.DocumentDate)+' '+DATENAME(YY,ld.DocumentDate)
Left Join (---Issued Qty.
			Select DATENAME(MM,ld.DocumentDate)+' '+DATENAME(YY,ld.DocumentDate) [Month]
				   ,Sum(isnull(ld.IssuedQty,0)) [Issued Qty.]
				   ,convert(money,Sum(isnull(ld.IssuedQty* ld.VRate,0))) [Issued Amount]
				   ,ld.ItemNo
			From Inventory.InvLotDetail ld
			where TransactionFlag='DRMI'
			And ItemNo=@ItemNo	
			Group by DATENAME(MM,ld.DocumentDate)+' '+DATENAME(YY,ld.DocumentDate),ld.ItemNo)tt3
			on 	tt3.ItemNo=ld.ItemNo
Left Join( ----Return Qty.
            Select DATENAME(MM,ld.DocumentDate)+' '+DATENAME(YY,ld.DocumentDate) [Month]
				  ,SUM(ld.ReceivedQty)[Return Qty.],tt2.ReceivedAmt [Return Amt.],ld.ItemNo
			From Inventory.InvLotDetail ld
			Left Join (Select  datename(MM,INV.DocumentDate)+' '+datename(YY,INV.DocumentDate) Date
					                         ,ILS.ItemNo
					                         ,MONTH(INV.DocumentDate) M
									         ,Convert(money,sum(ilbd.LotBalanceQty * 
									              (Case when ILBD.TransactionFlag = ILS.TransactionFlag 
									                    then ILBD.VRate 
									               else ILBD.DiffRate end
									               )
									             )) ReceivedAmt
											   from Inventory.InvLotSummary ILS 
											   Inner Join Inventory.InvLotDetail INV 
											                          ON INV.InvLotSummaryNo=ILS.InvLotSummaryNo
											   Inner join Inventory.InvLotBillDetail ILBD 
											                          On ILBD.InvLotDetailNo = INV.InvLotDetailNo
											   Where ILS.TransactionFlag='DRMR' 
											          And ILBD.BillDate <=@DocumentDate
													 And ILS.CompanyNo=@CompanyNo 
													 and ILS.ItemNo = @ItemNo---@ItemNo
													 And ILS.StockNatureNo=1 
													 And Inv.ReceivedQty>0                      
											   Group by  ILS.ItemNo
											            ,datename(MM,INV.DocumentDate)+' '
											            +datename(YY,INV.DocumentDate),MONTH(INV.DocumentDate)
												
							                )tt2 On tt2.ItemNo=ld.ItemNo
												 And tt2.Date= DATENAME(MM,ld.DocumentDate)+' '+DATENAME(YY,ld.DocumentDate)
where ld.ItemNo=@ItemNo and ld.TransactionFlag='DRMR'
Group By tt2.ReceivedAmt,DATENAME(MM,ld.DocumentDate)+' '+DATENAME(YY,ld.DocumentDate),ld.ItemNo
							                )tt4 On tt4.ItemNo=ld.ItemNo
where ld.ItemNo=@ItemNo and ld.TransactionFlag in ('DRSN')
and ld.CompanyNo=@CompanyNo
And ld.DocumentDate<=@DocumentDate
Group By tt2.ReceivedAmt
        ---,tt4.[Return Qty.],tt4.[Return Amt.]
        ,DATENAME(MM,ld.DocumentDate)+' '+DATENAME(YY,ld.DocumentDate),Month(ld.DocumentDate)	
Order By Month(ld.DocumentDate) 
						 for xml path('tr')
					),'<td1>','<td style="text-align: center;">'),'<td2>','<td>')
											 ,'<td3>','<td style="text-align: Right;">')
											 ,'<td4>','<td style="text-align: Right;">')
											 ,'<td5>','<td style="text-align: Right;">')
											 ,'<td6>','<td style="text-align: Right;">')
											 ,'<td7>','<td style="text-align: Right;">')
											 ,'<td8>','<td style="text-align: Right;">')
											)
	set @str=@str+'</table>'
	select @str
END
