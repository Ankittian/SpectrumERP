CREATE PROCEDURE [Information].[POPendingPurchaseRequestForDept] 
	-- Add the parameters for the stored procedure here
	@CompanyNo tinyint,
	@DocumentDate smalldatetime,
	@DeptNo int
AS
BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;

    -- Insert statements for procedure here
	

	declare @str varchar(max)
    select @str=masterdata.JSForInformation('PMPO')
    set @str=@str+masterdata.CSSForInformation('PMPO')
	set @str=@str+'<table class="altrowstable" id="alternatecolorPOPendingPurchaseRequestDeptInfo">'
	set @str=@str+'<Caption><Font color="Maroon"><h3 align="Left"><b>Pending Purchase Request</b></h3></Font></Caption>'
    set @str=@str+'<tr><th>Sl. No.</th><th>Division</th><th>Document No.</th><th>Document Date</th><th>Source Document</th><th>Source Doc. No.</th><th>Source Doc. Date</th><th>Indent Type</th><th>Requested By</th><th>Created By </th><th>Authorized By</th></tr>'
	set @str=@str
				 +( 
					REPLACE(
					REPLACE(
				    REPLACE(
				    REPLACE(
				    REPLACE(
				    REPLACE(
				    replace(
					replace(
					replace(
					replace(
					replace((    
							Select   ROW_NUMBER() over ( Order by IM.DocumentDate desc) as td1
					                ,isnull(DM.DivisionName,'') td2
					                , IM.DocumentNoYearly td3
									,Convert(Varchar(20),IM.DocumentDate,103)  td4
									,IA.IndentAgainst td5
									,Isnull(Isnull(IRD.DocumentNoYearly,SCO.DocumentNoYearly),'')  td6
									,Isnull(Convert(Varchar(30),Isnull(IRD.DocumentDate,SCO.DocumentDate),103),'')  td7
									,IT.IndentType  td8
									,Isnull(IRD.RequestedBy,'') td9
									,L.PrintingName td10
									,isnull(L1.PrintingName,'')  td11
							From Inventory.IndentMain IM
							Inner Join GlobalData.IndentAgainst IA ON IA.IndentAgainstNo=IM.IndentAgainstNo
							Inner Join Inventory.IndentItemDetail ID ON ID.IndentNo=IM.IndentNo ----Group by Clause for this
							Left Join Inventory.IndentRequisitionMain IRD On IM.RefDocumentNo=IRD.IndentRequisitionNo and IM.IndentAgainstNo=3---Material Request
							Left Join Purchase.SCOrderMain SCO ON SCO.SCOrderNo=IM.RefDocumentNo And IM.IndentAgainstNo=4----SubContract Order
							Inner Join GlobalData.IndentType IT ON IT.IndentTypeNo=IM.IndentTypeNo
							Inner Join payroll.tblDeptMaster DT ON DT.DeptNo=IM.DeptNo
							Inner Join payroll.tblDivisionMaster DM ON DM.DivisionNo=DT.DivisionNo
							Inner Join Security.Login L ON L.LoginNo=IM.CreatedBy
							Left Join Security.Login L1 On L1.LoginNo=IM.AuthorizedBy
							Where (ID.IndentQty-Isnull(ID.POQty,0)) > 0----From Item Detail 
							And IM.DocumentDate<=@DocumentDate
							And IM.DeptNo=@DeptNo
							And IM.CompanyNo=@CompanyNo	
							Group by DM.DivisionName
					                ,IM.DocumentNoYearly 
									,IM.DocumentDate  
									,IA.IndentAgainst 
									,IRD.DocumentNoYearly,SCO.DocumentNoYearly
									,IRD.DocumentDate,SCO.DocumentDate
									,IT.IndentType  
									,IRD.RequestedBy 
									,L.PrintingName 
									,L1.PrintingName  
						  for xml path('tr')),'<td1>','<td style="text-align: center;">')
						                     ,'<td2>','<td>')
											 ,'<td3>','<td>')
											 ,'<td4>','<td style="text-align: center;">')
											 ,'<td5>','<td>')
											 ,'<td6>','<td>')
											 ,'<td7>','<td style="text-align: center;">')
											 ,'<td8>','<td>')
											 ,'<td9>','<td>')
											 ,'<td10>','<td>')
											 ,'<td11>','<td>')
											)
	set @str=@str+'</table>'
	
	select @str
END
