Stored procedure Inventory.ValidateForBalQtyGEGPISRN: Validation for Gate Entry, Gate Pass In and Gate Pass Out. Called from Inventory.IGRNValidate, GateEntry.IMGEValidate.
Reads: GateEntry.InvGateEntryItemDetail, Inventory.GatePassInItemDetail, Inventory.GatePassOutItemDetail, Inventory.GatePassOutMain, Inventory.GRNItemDetail, Inventory.GRNItemIndentDetail, Inventory.SCReqItemDetail, masterdata.mItemMaster, Purchase.SCOrderAmendmentItemDetail, Purchase.SCOrderAmendmentMain, Purchase.SCOrderItemReqDetail
Parameters: @ItemDetail xml, @FormNo int, @OutputStatus bit = Null OUTPUT, @Result varchar(100) = Null OUTPUT

-- Create date: 02 April 2024
-- Description: Validation for Gate Entry, Gate Pass In and Gate Pass Out
-- ==========================================================================
CREATE PROCEDURE [Inventory].[ValidateForBalQtyGEGPISRN]
(
	@ItemDetail xml,
	@FormNo int,
	@OutputStatus bit = Null output,
	@Result varchar(100) = Null output	
)
AS

BEGIN
	SET NOCOUNT ON;	
	BEGIN TRY

Declare @InputTable Table (SCReqNo int,SCOrderNo int,GatePassInNo int,GateEntryNo int,GRNNo int,ItemNo int, MakeNo smallint,Quantity decimal(15,3))
Declare @SCRequestDetail Table (SCReqNo int,ItemNo int, MakeNo smallint,SRQty decimal(15,3))
Declare @ItemDescription varchar(max)

Insert into @InputTable
(SCReqNo ,SCOrderNo ,GatePassInNo ,GateEntryNo ,GRNNo ,ItemNo , MakeNo ,Quantity)
Select v.value('SCReqNo[1]','int') AS SCReqNo,
v.value('SCOrderNo[1]','int') AS SCOrderNo,
v.value('GatePassInNo[1]','int') AS GatePassInNo,
v.value('GateEntryNo[1]','int') AS GateEntryNo,
v.value('GRNNo[1]','int') GRNNo,
v.value('ItemNo[1]','int') AS ItemNo,
NullIF(v.value('MakeNo[1]','smallint'),0) AS MakeNo,
v.value('ChallanQty[1]','Decimal(15,3)') AS ChallanQty
FROM @ItemDetail.nodes('/DocumentElement/Table') x(v)

---Insert Service Request from Service Order
Insert into @SCRequestDetail
Select SCRD.SCReqNo 
,ID.ItemNo
,ID.MakeNo
,SCRD.Quantity
from @InputTable ID
Inner Join [Purchase].[SCOrderItemReqDetail] SCRD On SCRD.SCOrderNo = ID.SCOrderNo
												And ISNULL(SCRD.ItemNo,0) = ISNULL(ID.ItemNo,0)
												And ISNULL(SCRD.MakeNo,0) = ISNULL(ID.MakeNo,0)
Where SCRD.ItemFlowNo = 5 And ISNULL(ID.SCReqNo,0) = 0

--Insert Service request from GPI or SRN
Insert into @SCRequestDetail
Select ID.SCReqNo 
,ID.ItemNo
,ID.MakeNo
,SCRI.Quantity 
from @InputTable ID
Inner Join [Inventory].[SCReqItemDetail] SCRI On ID.SCReqNo = SCRI.SCReqNo
										And ID.ItemNo = SCRI.ItemNo
										And ISNULL(ID.MakeNo,0) = ISNULL(SCRI.MakeNo,0)
Where SCRI.ItemFlowNo = 5 And ISNULL(ID.SCReqNo,0) <> 0
Group By ID.SCReqNo ,ID.ItemNo,ID.MakeNo,SCRI.Quantity 


Select SCRI.SCReqNo,SCRI.ItemNo,SCRI.MakeNo
,SCRI.SRQty - SUM(ISNULL(GPI.PreQty,0)) - ISNULL(IT.Quantity,0) [BalanceQty]
into #OutputTable
from  @SCRequestDetail SCRI 
Left Join (
			--Gate Pass In Qty
			Select GM.SCReqNo,gid.ItemNo ,gid.MakeNo, SUM(ISNULL(GPID.ReceivedQty,0)) PreQty
			from Inventory.GatePassOutMain gm 
			Inner Join Inventory.GatePassOutItemDetail gid on gid.GatePassOutNo = gm.GatePassOutNo 
			Inner Join [Inventory].[GatePassInItemDetail] GPID On GPID.GatePassOutNo = gid.GatePassOutNo
																And GPID.ItemNo = Gid.ItemNo
																And ISNULL(GPID.MakeNo,0) = ISNULL(gid.MakeNo,0)
			Inner Join @SCRequestDetail SCRT On gm.SCReqNo = SCRT.SCReqNo
												And GPID.ItemNo = ISNULL(SCRT.ItemNo,0)
												And ISNULL(GPID.MakeNo,0) = ISNULL(SCRT.MakeNo,0)
			Left Join @InputTable IT On IT.GatePassInNo = GPID.GatePassInNo
									And IT.ItemNo = GPID.ItemNo
									And ISNULL(IT.MakeNo,0) = ISNULL(GPID.MakeNo,0)
									And IT.SCReqNo = SCRT.SCReqNo
			Where IT.GatePassInNo Is Null
			Group By GM.SCReqNo,gid.ItemNo ,gid.MakeNo

			Union All

			--SRN Qty
			Select SCRD.SCReqNo,SCRD.ItemNo,SCRD.MakeNo
			,SUM(ISNULL(GRNR.AcceptedQty,0)) PreQty
			from @SCRequestDetail SCRD 
			Inner Join Inventory.GRNItemIndentDetail GRNR On GRNR.SCReqNo = SCRD.SCReqNo
			Inner Join Inventory.GRNItemDetail GRNS On GRNS.GRNNo = GRNR.GRNNo
														And GRNS.GRNItemDetailNo = GRNR.GRNItemDetailNo
														And GRNS.ItemNo = SCRD.ItemNo
														And ISNULL(GRNS.MakeNo,0) = ISNULL(SCRD.MakeNo,0)
			Left Join @InputTable IT On IT.GRNNo = GRNS.GRNNo 
									And IT.SCReqNo = GRNR.SCReqNo
									And IT.ItemNo = GRNS.ItemNo
									And ISNULL(IT.MakeNo,0) = ISNULL(GRNS.MakeNo,0)
									And ISNULL(IT.GateEntryNo,0) = ISNULL(GRNS.GateEntryNo,0)
			Where IT.GRNNo Is Null
			Group By SCRD.SCReqNo,SCRD.ItemNo,SCRD.MakeNo

		)GPI On GPI.SCReqNo = SCRI.SCReqNo
		And GPI.ItemNo = SCRI.ItemNo
		And ISNULL(GPI.MakeNo,0) = ISNULL(SCRI.MakeNo,0)
Left Join @InputTable IT On IT.SCReqNo = SCRI.SCReqNo 
						And IT.ItemNo = SCRI.ItemNo
						And ISNULL(IT.MakeNo,0) = ISNULL(SCRI.MakeNo,0)
Group By SCRI.SCReqNo,SCRI.ItemNo,SCRI.MakeNo,SCRI.SRQty,IT.Quantity


If Exists (
				Select 1 
				from #OutputTable 
				Where BalanceQty < 0 
			)
Begin
	Select @ItemDescription = STUFF((Select ',' + IM.ItemDescription
										from #OutputTable OT
										Inner Join masterdata.mItemMaster IM On IM.ItemNo = OT.ItemNo
										Where BalanceQty < 0
										Group By Im.ItemDescription
										For xml Path ('')),1,1,'')

	If @FormNo = 62
	Begin
		select 0,'Against selected Subcontract Order, Gate Pass In exist and Received Qty can''t be greater than balance Qty for Item ' 
		+ @ItemDescription
	Return
	End

	If @FormNo = 153
	Begin
		select 0,'Against selected Gate Pass out, Gate Entry exist and Received Qty can''t be greater than balance Qty for Item ' 
		+ @ItemDescription
	End

	If @FormNo = 448
	Begin
		select 0,'Against selected Gate Pass out, Gate Pass In exist and Received Qty can''t be greater than balance Qty for Item ' 
		+ @ItemDescription
	End
End

If Exists (Select SCRI.SCOrderNo,SCRI.ItemNo,SCRI.MakeNo
			,SCRI.BalQty - ISNULL(IT.Quantity,0) - ISNULL(GE.GEQty ,0)
			from (
					Select SCRD.SCOrderNo,SCRI.ItemNo,SCRI.MakeNo,SUM(ISNULL(SCRI.BalanceQty,0)) BalQty
					from #OutputTable SCRI
					Inner Join [Purchase].[SCOrderItemReqDetail] SCRD On SCRI.SCReqNo = SCRD.SCReqNo
																And SCRI.ItemNo = SCRD.ItemNo
																And ISNULL(SCRI.MakeNo,0) = ISNULL(SCRD.MakeNo,0)
					Group By SCRD.SCOrderNo,SCRI.ItemNo,SCRI.MakeNo
					Having SUM(ISNULL(SCRI.BalanceQty,0)) >= 0
				)SCRI
			Left Join @InputTable IT On IT.SCOrderNo = SCRI.SCOrderNo
									And IT.ItemNo = SCRI.ItemNo
									And ISNULL(IT.MakeNo,0) = ISNULL(SCRI.MakeNo,0)
			Left Join (
						---GE Qty for which SRN has not been created
						Select SCAM.SCOrderNo,SCAI.ItemNo,SCAI.MakeNo
						,SUM(ISNULL(IMGE.ChallanQty,0)) [GEQty]
						from [Purchase].[SCOrderAmendmentItemDetail] SCAI 
						Inner Join Purchase.SCOrderAmendmentMain SCAM On SCAM.SCOrderAmendmentNo = SCAI.SCOrderAmendmentNo
						Inner Join GateEntry.InvGateEntryItemDetail IMGE On IMGE.SCOrderAmendmentNo = SCAM.SCOrderAmendmentNo
																		And IMGE.ItemNo = SCAI.ItemNo
																		And ISNULL(IMGE.MakeNo,0) = ISNULL(SCAI.MakeNo,0)
						Left Join Inventory.GRNItemDetail GRNS On GRNS.SCOrderAmendmentNo = IMGE.SCOrderAmendmentNo
																	And GRNS.ItemNo = IMGE.ItemNo
																	And ISNULL(GRNS.MakeNo,0) = ISNULL(IMGE.MakeNo,0)
																	And GRNS.GateEntryNo = IMGE.GateEntryNo
						Left Join (
										Select GateEntryNo,ItemNo,MakeNo 
										from @InputTable
										Where GateEntryNo Is Not Null
										Group by GateEntryNo,ItemNo,MakeNo
									)IT On IT.GateEntryNo = IMGE.GateEntryNo
									And IT.ItemNo = IMGE.ItemNo
									And ISNULL(IT.MakeNo,0) = ISNULL(IMGE.MakeNo,0)
						Where SCAI.ItemFlowNo = 5 And IT.GateEntryNo Is Null
						And GRNS.GRNItemDetailNo Is Null
						Group By SCAM.SCOrderNo,SCAI.ItemNo,SCAI.MakeNo
					)GE  On SCRI.SCOrderNo = GE.SCOrderNo
						And SCRI.ItemNo = GE.ItemNo
						And ISNULL(SCRI.MakeNo,0) = ISNULL(GE.MakeNo,0)
					Where SCRI.BalQty - ISNULL(IT.Quantity,0) - ISNULL(GE.GEQty ,0) < 0
			)
Begin 
	Select @ItemDescription =  STUFF((Select ',' + IM.ItemDescription
										from (
												Select SCRD.SCOrderNo,SCRI.ItemNo,SCRI.MakeNo,SUM(ISNULL(SCRI.BalanceQty,0)) BalQty
												from #OutputTable SCRI
												Inner Join [Purchase].[SCOrderItemReqDetail] SCRD On SCRI.SCReqNo = SCRD.SCReqNo
																							And SCRI.ItemNo = SCRD.ItemNo
																							And ISNULL(SCRI.MakeNo,0) = ISNULL(SCRD.MakeNo,0)
												Group By SCRD.SCOrderNo,SCRI.ItemNo,SCRI.MakeNo
												Having SUM(ISNULL(SCRI.BalanceQty,0)) >= 0
											)SCRI
										Left Join @InputTable IT On IT.SCOrderNo = SCRI.SCOrderNo
																And IT.ItemNo = SCRI.ItemNo
																And ISNULL(IT.MakeNo,0) = ISNULL(SCRI.MakeNo,0)
										Left Join (
													Select SCAM.SCOrderNo,SCAI.ItemNo,SCAI.MakeNo
													,SUM(ISNULL(IMGE.ChallanQty,0)) [GEQty]
													from [Purchase].[SCOrderAmendmentItemDetail] SCAI 
													Inner Join Purchase.SCOrderAmendmentMain SCAM On SCAM.SCOrderAmendmentNo = SCAI.SCOrderAmendmentNo
													Inner Join GateEntry.InvGateEntryItemDetail IMGE On IMGE.SCOrderAmendmentNo = SCAM.SCOrderAmendmentNo
																									And IMGE.ItemNo = SCAI.ItemNo
																									And ISNULL(IMGE.MakeNo,0) = ISNULL(SCAI.MakeNo,0)
													Left Join Inventory.GRNItemDetail GRNS On GRNS.SCOrderAmendmentNo = IMGE.SCOrderAmendmentNo
																								And GRNS.ItemNo = IMGE.ItemNo
																								And ISNULL(GRNS.MakeNo,0) = ISNULL(IMGE.MakeNo,0)
																								And GRNS.GateEntryNo = IMGE.GateEntryNo
													Left Join (
																	Select GateEntryNo,ItemNo,MakeNo 
																	from @InputTable
																	Where GateEntryNo Is Not Null
																	Group by GateEntryNo,ItemNo,MakeNo
																)IT On IT.GateEntryNo = IMGE.GateEntryNo
																And IT.ItemNo = IMGE.ItemNo
																And ISNULL(IT.MakeNo,0) = ISNULL(IMGE.MakeNo,0)
													Where SCAI.ItemFlowNo = 5 And IT.GateEntryNo Is Null
													And GRNS.GRNItemDetailNo Is Null
													Group By SCAM.SCOrderNo,SCAI.ItemNo,SCAI.MakeNo
												)GE  On SCRI.SCOrderNo = GE.SCOrderNo
													And SCRI.ItemNo = GE.ItemNo
													And ISNULL(SCRI.MakeNo,0) = ISNULL(GE.MakeNo,0)
												Inner Join masterdata.mItemMaster IM ON IM.ItemNo = SCRI.ItemNo
												Where SCRI.BalQty - ISNULL(IT.Quantity,0) - ISNULL(GE.GEQty ,0) < 0
												Group By IM.ItemDescription
												For xml path('')),1,1,'')
	If @FormNo = 62
	Begin
		select 0,'Against selected Subcontract Order, Gate Pass In exist and Received Qty can''t be greater than balance Qty for Item ' 
		+ @ItemDescription
	Return
	End

	If @FormNo = 153
	Begin
		select 0,'Against selected Gate Pass out, Gate Entry exist and Received Qty can''t be greater than balance Qty for Item ' 
		+ @ItemDescription
	End

	If @FormNo = 448
	Begin
		select 0,'Against selected Gate Pass out, Gate Pass In exist and Received Qty can''t be greater than balance Qty for Item ' 
		+ @ItemDescription
	End

	Return
End

	Drop Table #OutputTable
	set @OutputStatus =1
	set @Result='Success'

	END TRY
	BEGIN CATCH

	set @OutputStatus =0
	set @Result=error_message()

	Select 0,Error_Message()
	END CATCH
END
