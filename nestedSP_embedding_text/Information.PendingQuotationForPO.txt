CREATE PROCEDURE [Information].[PendingQuotationForPO] 
	-- Add the parameters for the stored procedure here
	@CompanyNo tinyint,
	@DocumentDate smalldatetime,
	@ItemNo int
	
AS
BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;

    -- Insert statements for procedure here
	

	declare @str varchar(max)
    select @str=masterdata.JSForInformation('POMQ')
    set @str=@str+masterdata.CSSForInformation('POMQ')
	set @str=@str+'<table class="altrowstable" id="alternatecolorPendingQuotationForPOInfo">'
	set @str=@str+'<Caption><Font color="Maroon"><h3 align="Left"><b>Pending Quotations For PO</b></h3></Font></Caption>'
    set @str=@str+'<tr><th>Sl. No.</th><th>Vendor Name</th><th>Quotation No.</th><th>Quotation Date</th><th>Make</th><th>Validity Date</th><th>Delivery Days</th><th>Created By</th><th>Authorized By</th></tr>'
	set @str=@str
				 +( 
					replace(
					replace(
					replace( 
					replace(
					replace(
					replace(
					replace( 
					replace(
					replace(
							(
					        Select  ROW_NUMBER() over ( Order by R.DocumentDate desc) as td1
					                ,VCD.VendorName td2
									,R.DocumentNoYearly td3
									,Convert(Varchar(20),R.[DocumentDate],103) td4
									,ISnull(MM.MakeName,'') td5
									,Convert(Varchar(20),R.[ValidityDate],103) td6
									,RQI.DeliveryDays td7
									,SL.PrintingName td8
									,Isnull(SLA.PrintingName,'') td9
							From Purchase.QuotationMain R
							Inner Join Purchase.QuotationItemDetail RQI ON RQI.QuotationNo=R.QuotationNo
							Left Join masterdata.mMakeMaster MM ON MM.MakeNo=RQI.MakeNo
							Inner join Purchase.RFQMain RFQ ON R.RFQNo =RFQ.RFQNo -- To Get [RFQ No],[RFQ Date]
							Inner JOIN masterdata.mVendorLocationDetail VCD ON R.VendorLocationNo=VCD.VendorLocationNo --[Location]
							Inner JOIN Security.Login SL on R.[CreatedBy]=SL.LoginNo
							Left JOIN Security.Login SLA on R.[AuthorizedBy]=SLA.LoginNo
							where R.RFQNo not in(
												SELECT RFQ.RFQNo	
												FROM Purchase.RevisedQuotationMain RQM
												Inner join Purchase.RFQMain RFQ ON RQM.RFQNo =RFQ.RFQNo -- To Get [RFQ No],[RFQ Date]
												Inner Join Purchase.POMain PO ON RQM.RevisedQuotationNo=PO.RefDocumentNo
																			 And  PORefDocumentTypeNo=1	 and PODocumentTypeNo=2
												)		-----Quotation					---Po typr
									And R.DocumentDate<=@DocumentDate
									And R.CompanyNo=@CompanyNo
									And RQI.ItemNo=@ItemNo
							for xml path('tr')),'<td1>','<td style="text-align: Center;">')
							                   ,'<td2>','<td>')
		   									   ,'<td3>','<td>')				
											   ,'<td4>','<td style="text-align: Center;">')
											   ,'<td5>','<td>')
		   									   ,'<td6>','<td style="text-align: Center;">')				
											   ,'<td7>','<td style="text-align: Right;">')
											   ,'<td8>','<td>')				
											   ,'<td9>','<td>')													 			
									    	)
								
				   
	set @str=@str+'</table>'
	
	select @str
END
