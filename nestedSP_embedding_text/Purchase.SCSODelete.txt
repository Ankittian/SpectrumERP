CREATE PROCEDURE [Purchase].[SCSODelete]
(

 @SCOrderNo  int,
 @SCOrderAmendmentNo  int,
 @LoginNo Smallint,
 @FormTag char(4)
 ,@OutputStatusNo tinyint = null output
 ,@OutputStatusRemarks varchar(4000)  = null output
 ,@IsCopy bit = 0
)
AS

BEGIN
	SET NOCOUNT ON;
	
	BEGIN TRY
	 BEGIN Transaction
	 
--- GET THE PREVIOUS AMENDMENT NO TO INSERT IN MAIN TABLE

if exists(select 1 from Inventory .MaterialIssueMain where 
			IssueSourceNo =3 and RefDocumentNo =@SCOrderAmendmentNo)
	raiserror('Material is issued Against this Order'	,16,-1)
else if exists(select 1 from Inventory.IndentMain where 
			IndentAgainstNo  =4 and RefDocumentNo =@SCOrderAmendmentNo)
	raiserror('Purchase requisition is prepared Against this Order'	,16,-1)

if @FormTag ='SCSO'
		begin
			if exists (select 1 from Purchase.SCOrderAmendmentMain  where SCOrderNo = @SCOrderNo 
						and AmendmentNo>0)
				RaisError('SC Order Amendment is done againt this SC Order,so you can''t delete this entry.',16,-1)		
		end	
		
		
	
	DECLARE @PrevSCOrderAmendmentNo int
	SELECT @PrevSCOrderAmendmentNo  = SCOrderAmendmentNo 
	FROM  Purchase.SCOrderAmendmentMain
	WHERE AmendmentNo =( SELECT AmendmentNo -1
	                     FROM Purchase.SCOrderAmendmentMain
	                     WHERE SCOrderAmendmentNo = @SCOrderAmendmentNo ) 
    AND SCOrderNo =( SELECT SCOrderNo 
                        FROM Purchase.SCOrderAmendmentMain
                        WHERE SCOrderAmendmentNo = @SCOrderAmendmentNo )
    
    -- DELETE MAIN TABLE IF
	
	IF @prevSCOrderAmendmentNo IS NULL -- IF ITS 0th AMENDMENT
	OR EXISTS ( SELECT 1 FROM Purchase.SCOrderAmendmentMain -- OR IF IT IS AUTHORIZED
				WHERE SCOrderAmendmentNo = @SCOrderAmendmentNo 
				AND ISNULL([AuthorizedBy],0)<>0 )
			
	BEGIN
	if @IsCopy = 1
	  set @PrevSCOrderAmendmentNo = @SCOrderAmendmentNo 
		
	-- Delete  main Details Table

Declare @SCReqDetail as table (IDNo tinyint ,SCReqNo int, ItemNo int, ActivityNo smallint,SequenceNo tinyint)
			Declare @iRow as tinyint,@iCount tinyint
			if not exists(select 1 from Purchase.SCOrderMain where SCOrderDocumentTypeNo=1 
							and SCOrderNo=@SCOrderNo)
			begin
				Insert into @SCReqDetail select row_Number() over (order by SCReqNo), SCReqNo,
					ItemNo ,ActivityNo  ,SequenceNo  from 
						Purchase.SCOrderReqDetail  where SCOrderNo= @SCOrderNo 
				set @iRow= @@rowcount		
			end		
	if (select COUNT(1) from Purchase.FreightOrderOrderDetail where SCOrderNo = @SCOrderNo 	)	=1
			delete from Purchase.FreightOrderOrderDetail where SCOrderNo = @SCOrderNo 	
			
DECLARE @SCBOQDetail as table (SCBOQNo int)
	insert into @SCBOQDetail
	select SCBOQNo from  Purchase.SCBOQMain where SCOrderAmendmentNo  =@SCOrderAmendmentNo 
	
  Delete from Purchase.SCBOQItemDetail where  SCBOQNo in(select SCBOQNo from @SCBOQDetail)
  Delete from Purchase.SCBOQItemReqDetail where   SCBOQNo in(select SCBOQNo from @SCBOQDetail)
  Delete from Purchase.SCBOQMain where   SCBOQNo in(select SCBOQNo from @SCBOQDetail)
  
  
		delete from Purchase.SCOrderAdditionItemDetail  where SCOrderNo = @SCOrderNo 
		delete from Purchase.SCOrderTransportationRoute where SCOrderNo = @SCOrderNo 
		delete from Purchase.SCOrderExpenseHead where SCOrderNo = @SCOrderNo 
					
		delete from Purchase.SCOrderBillActivityDetail where SCOrderNo = @SCOrderNo 
		delete from Purchase.SCOrderBillMarkDetail where SCOrderNo = @SCOrderNo 
		delete from Purchase.SCOrderBillDetail where SCOrderNo = @SCOrderNo 
			
		delete from Purchase.SCOrderMarkActivityDetail where SCOrderNo=@SCOrderNo
		delete from Purchase.SCOrderMarkDetail where SCOrderNo=@SCOrderNo
		Delete From Purchase.SCOrderCustomFieldDetail Where SCOrderNo=@SCOrderNo
	    Delete From Purchase.SCOrderDetail Where SCOrderNo=@SCOrderNo
	    Delete From Purchase.SCOrderItemDetail Where SCOrderNo=@SCOrderNo
	    Delete From Purchase.SCOrderItemTaxDetail Where SCOrderNo=@SCOrderNo
	    Delete From Purchase.SCOrderNotesDetail Where SCOrderNo=@SCOrderNo
	    Delete From Purchase.SCOrderPaymentTermsDetail Where SCOrderNo=@SCOrderNo
	    Delete From Purchase.SCOrderReqDetail Where SCOrderNo=@SCOrderNo
	    Delete From Purchase.SCOrderScheduleDetail Where SCOrderNo=@SCOrderNo
	    Delete From Purchase.SCOrderTaxDetail Where SCOrderNo=@SCOrderNo
	    Delete From Purchase.SCOrderTermsNConditionDetail Where SCOrderNo=@SCOrderNo
	    Delete From Purchase.SCOrderItemReqDetail  Where SCOrderNo=@SCOrderNo
	   	
	   	Declare @SCReqNo int,@ItemNo int,@ActivityNo smallint,@OutputStatus bit, @result varchar(100)
			,@SequenceNo tinyint
			
			select @iCOunt = 0
			while @iRow > @iCount
			begin
				select @SCReqNo = SCReqNo , @ItemNo = ItemNo , @ActivityNo  = ActivityNo ,
					@SequenceNo =SequenceNo 
					from @SCReqDetail  where IDNo = @iCount +1
				
				exec [Inventory].[SCReqStatus]  @SCReqNo,@ItemNo,@ActivityNo ,@SequenceNo ,	2,
					@OutputStatus output,@Result output
				
				if @OutputStatus =0
				   raiserror (@Result,16,-1)
				   
				set @iCount = @iCount + 1			
			end
			
	   	  
 END -- DELETE MAIN TABLE IF	
			 
	IF Exists(SELECT 1 FROM Purchase.SCOrderAmendmentMain  WHERE 
			  SCOrderAmendmentNo = @prevSCOrderAmendmentNo  AND ISNULL([AuthorizedBy],0)<>0) 
	AND						 
	Exists(SELECT 1 FROM Purchase.SCOrderAmendmentMain WHERE 
	       SCOrderAmendmentNo = @SCOrderAmendmentNo    AND ISNULL([AuthorizedBy],0)<>0) 
	
	BEGIN
			 	
		 INSERT INTO  [Purchase].[SCOrderAdditionItemDetail] 
				   (
				    SCOrderNo
				   ,ItemNo 
				   ,Quantity 
				   ,Rate 
				   ,BasicAmount 
				   )
		  SELECT 
					@SCOrderNo
				   ,ItemNo 
				   ,Quantity 
				   ,Rate
				   ,BasicAmount 
		  FROM Purchase.SCOrderAmendmentAdditionItemDetail 
		  WHERE SCOrderAmendmentNo = @prevSCOrderAmendmentNo
		 
		  INSERT INTO  [Purchase].[SCOrderCustomFieldDetail]
				   (
				    SCOrderNo
				   ,CustomFieldNo
				   ,FieldValue
				   )
		  SELECT 
					@SCOrderNo
				   ,CustomFieldNo
				   ,FieldValue
		  FROM Purchase.SCOrderAmendmentCustomFieldDetail
		  WHERE SCOrderAmendmentNo = @prevSCOrderAmendmentNo
		 
	  
		  INSERT INTO  [Purchase].[SCOrderNotesDetail]
				   (
				    SCOrderNo
				   ,PrintAreaNo
				   ,Notes
				   ,IsDefault
				   )

		  SELECT 
					@SCOrderNo
				   ,PrintAreaNo
				   ,Notes
				   ,IsDefault
		  FROM Purchase.SCOrderAmendmentNotesDetail
		  WHERE SCOrderAmendmentNo = @prevSCOrderAmendmentNo 
		  
		  
		  INSERT INTO  [Purchase].[SCOrderPaymentTermsDetail]
				   (
				    SCOrderNo
				   ,PaymentTypeNo
				   ,BaseDateTypeNo
				   ,PaymentPayOnNo
				   ,TermsValue
				   ,PayValue
				   ,NoOfDays
				   ,Date
				   ,AdditionalMonth
				   ,Discount
				   ,IsBGApplicable
				   ,BGAmount
				   ,IsProgressBased
				   ,Progress
				   ,Tolerance
				   ,Remark
				   ,SequenceNo
				   )
				select
				    @SCOrderNo
				   ,PaymentTypeNo
				   ,BaseDateTypeNo
				   ,PaymentPayOnNo
				   ,TermsValue
				   ,PayValue
				   ,NoOfDays
				   ,Date
				   ,AdditionalMonth
				   ,Discount
				   ,IsBGApplicable
				   ,BGAmount
				   ,IsProgressBased
				   ,Progress
				   ,Tolerance
				   ,Remark
				   ,SequenceNo
		  FROM [Purchase].[SCOrderAmendmentPaymentTermsDetail]
		  WHERE SCOrderAmendmentNo = @prevSCOrderAmendmentNo    
		  
		  INSERT INTO  [Purchase].[SCOrderItemTaxDetail]
				   (
				    SCOrderNo
				   ,Selected
				   ,ItemNo
				   ,ActivityNo
				   ,MiscChargeNo
				   ,ChargeType
				   ,Nature
				   ,ChargeOn
				   ,ChargeValue
				   ,TotalAmount
				   ,SequenceNo 
				   )
		  SELECT 
					@SCOrderNo
				   ,Selected
				   ,ItemNo
				   ,ActivityNo 
				   ,MiscChargeNo
				   ,ChargeType
				   ,Nature
				   ,ChargeOn
				   ,ChargeValue
				   ,TotalAmount
				   ,SequenceNo 
		  FROM Purchase.SCOrderAmendmentItemTaxDetail
		  WHERE SCOrderAmendmentNo = @prevSCOrderAmendmentNo 
		  
		  		  
		  INSERT INTO  [Purchase].[SCOrderTaxDetail]
				   (
				    SCOrderNo
				   ,MiscChargeNo
				   ,ChargeType
				   ,Nature
				   ,ChargeOn
				   ,ChargeValue
				   ,TotalValue
				   ,TaxCodeMiscChargesDetailNo 
				   ,IncludeInBasic
				   )
		  SELECT 
					@SCOrderNo
				   ,MiscChargeNo
				   ,ChargeType
				   ,Nature
				   ,ChargeOn
				   ,ChargeValue
				   ,TotalValue
				   ,TaxCodeMiscChargesDetailNo 
				   ,IncludeInBasic
		  FROM Purchase.SCOrderAmendmentTaxDetail
		  WHERE SCOrderAmendmentNo = @prevSCOrderAmendmentNo  
		  
		INSERT INTO  [Purchase].[SCOrderTermsNConditionDetail]
				   (
				    SCOrderNo
				   ,TermsNConditionHeadNo
				   ,TermsNCondition
				   )
		 SELECT 
					@SCOrderNo
				   ,TermsNConditionHeadNo
				   ,TermsNCondition
		  FROM Purchase.SCOrderAmendmentTermsNConditionDetail
		  WHERE SCOrderAmendmentNo = @prevSCOrderAmendmentNo  
		 
		 INSERT INTO [Purchase].[SCOrderDetail]
				   (
				    SCOrderNo
				   ,ItemNo
				   ,ActivityNo
				   ,ShortText
				   ,Quantity
				   ,Rate
				   ,NetAmount
				   ,Remarks
				   ,StatusNo
				   ,SequenceNo 
				   ,POAmendmentExpenseDetailNo
				   ,ToleranceBasisNo 
				   ,ToleranceMinus 
				   ,TolerancePlus 
				   ,ExpenseHeadNo 
				   )
				SELECT 
				    @SCOrderNo
				   ,ItemNo
				   ,ActivityNo
				   ,ShortText
				   ,Quantity
				   ,Rate
				   ,NetAmount
				   ,Remarks
				   ,StatusNo
				   ,SequenceNo
				   ,POAmendmentExpenseDetailNo
				   ,ToleranceBasisNo 
				   ,ToleranceMinus 
				   ,TolerancePlus 
				   ,ExpenseHeadNo 
		   FROM [Purchase].[SCOrderAmendmentDetail]
		   WHERE SCOrderAmendmentNo = @prevSCOrderAmendmentNo  
		   
		   INSERT INTO [Purchase].[SCOrderItemDetail]
				   (
				    SCOrderNo
				   ,ServiceItemNo
				   ,ActivityNo
				   ,ItemFlowNo
				   ,ItemNo
				   ,MakeNo
				   ,Quantity
				   ,ToleranceBasisNo 
				   ,ToleranceMinus 
				   ,TolerancePlus 
				   ,BOMRevisionSerialNo 
				   ,BOMrevisionNo 
				   ,StockQty  
				   ,SequenceNo
				   ,TechnicalGradeNo 
				   ,Rate 
				   ,ExpenseHeadNo 
				   ,BifercationPercentage 
				   ,ActualRate
				   ,HSNSACCodeNo 
				   )
			Select 
				    @SCOrderNo
				   ,ServiceItemNo
				   ,ActivityNo
				   ,ItemFlowNo
				   ,ItemNo
				   ,MakeNo
				   ,Quantity
				   ,ToleranceBasisNo 
				   ,ToleranceMinus 
				   ,TolerancePlus 
				   ,BOMRevisionSerialNo 
				   ,BOMrevisionNo 
				   ,StockQty 
				   ,SequenceNo
				   ,TechnicalGradeNo 
				   ,Rate 
				    ,ExpenseHeadNo 
				   ,BifercationPercentage 
				   ,ActualRate
				   ,HSNSACCodeNo 
		   FROM [Purchase].[SCOrderAmendmentItemDetail]
		   WHERE SCOrderAmendmentNo = @prevSCOrderAmendmentNo  
          
          INSERT INTO  [Purchase].[SCOrderReqDetail]
				   (
				    SCOrderNo
				   ,SCReqNo
				   ,ItemNo
				   ,ActivityNo
				   ,Quantity
				   ,Rate
				   ,ScheduleDate
				   ,SequenceNo
				   ,BillingItemNo
				   )
			Select
				    @SCOrderNo
				   ,SCReqNo
				   ,ItemNo
				   ,ActivityNo
				   ,Quantity
				   ,Rate
				   ,ScheduleDate
				   ,SequenceNo
				   ,BillingItemNo
			FROM [Purchase].[SCOrderAmendmentReqDetail]
		   WHERE SCOrderAmendmentNo = @prevSCOrderAmendmentNo  	   
          
          INSERT INTO [Purchase].[SCOrderItemReqDetail]
			   ([SCReqNo]
			   ,[SCOrderNo]
			   ,[ItemFlowNo]
			   ,[ItemNo]
			   ,[Quantity]
			   ,StockQty 
			   ,BOMRevisionSerialNo 
			   ,BOMrevisionNo
			   ,ActivityNo 
			   ,SequenceNo 
			   ,ServiceItemNo 
			   ,MakeNo 
			   ,TechnicalGradeNo
			   ,POAmendmentExpenseDetailNo
			   )
			 Select
			    SCReqNo 
			   ,@SCOrderNo 
			   ,ItemFlowNo 
			   ,ItemNo
			   ,Quantity
			   ,StockQty 
			   ,BOMRevisionSerialNo 
			   ,BOMrevisionNo 
			   ,ActivityNo 
			   ,SequenceNo 
			   ,ServiceItemNo 
			   ,MakeNo 
			   ,TechnicalGradeNo 
			   ,POAmendmentExpenseDetailNo
           From [Purchase].[SCOrderAmendmentItemReqDetail]
           WHERE SCOrderAmendmentNo = @prevSCOrderAmendmentNo  
          
          INSERT INTO [Purchase].[SCOrderScheduleDetail]
				   (
				    SCOrderNo
				   ,ActivityNo
				   ,ItemNo
				   ,ScheduleDate
				   ,Quantity
				   ,SequenceNo
				   )
			Select 
				    @SCOrderNo
				   ,ActivityNo
				   ,ItemNo
				   ,ScheduleDate
				   ,Quantity
				   ,SequenceNo
            FROM Purchase.[SCOrderAmendmentScheduleDetail]				   
		    WHERE SCOrderAmendmentNo = @prevSCOrderAmendmentNo  
		   
		   insert into Purchase.SCOrderMarkActivityDetail
		   (SCOrderNo , 
		   SCReqNo, 
		   WorkOrderNo, 
		   BOMRevisionSerialNo,
		    BOMRevisionNo, 
		    ActivityNo, 
		    SCQty, 
		    SCWeight)
		    SELECT     @SCOrderNo , SCReqNo, WorkOrderNo, BOMRevisionSerialNo, BOMRevisionNo, 
		    ActivityNo, SCQty, SCWeight
			FROM 
			Purchase.SCOrderAmendmentMarkActivityDetail where 
			SCOrderAmendmentNo = @prevSCOrderAmendmentNo  
		    
		    insert into Purchase.SCOrderMarkDetail(SCOrderNo , BOMRevisionSerialNo, BOMRevisionNo, SCQty)
			SELECT  @SCOrderNo , BOMRevisionSerialNo, BOMRevisionNo, SCQty
			FROM         Purchase.SCOrderAmendmentMarkDetail
			where SCOrderAmendmentNo = @prevSCOrderAmendmentNo  
		    
		    insert into Purchase.SCOrderBillDetail (SCOrderNo, BillingItemNo, SerialNo, ShortText, UnitNo,
		     Quantity, Rate, Remarks)
		     select @SCOrderNo, BillingItemNo, SerialNo, ShortText, UnitNo,
		     Quantity, Rate, Remarks
		     from Purchase.SCOrderAmendmentBillDetail 
		     where  SCOrderAmendmentNo = @prevSCOrderAmendmentNo  
			
			insert into Purchase.SCOrderBillActivityDetail(SCOrderNo, SerialNo, ActivityNo, SequenceNo)
			select @SCOrderNo, SerialNo, ActivityNo, SequenceNo
			from Purchase.SCOrderAmendmentBillActivityDetail 
			where  SCOrderAmendmentNo = @prevSCOrderAmendmentNo  
			
			insert into Purchase.SCOrderBillMarkDetail(SCOrderNo, SerialNo, BOMRevisionSerialNo, BOMRevisionNo
			, ItemNo)
			select @SCOrderNo, SerialNo, BOMRevisionSerialNo, BOMRevisionNo
			, ItemNo
			from  Purchase.SCOrderAmendmentBillMarkDetail 
			where  SCOrderAmendmentNo = @prevSCOrderAmendmentNo  
			
			insert into Purchase.SCOrderExpenseHead 
			(
			SCOrderNo, 
			POAmendmentExpenseDetailNo, 
			SCOrderItemDetNo, 
			ExpenseHeadNo, 
			BillingLocationNo, 
			BillingQtyNo, 
			FreightRateTypeNo, 
			Rate, 
			PartyNo, 
			IncludeInValuation, 
			ReferenceNo, 
			ReferenceDate, 
			RouteIdentity
			)
			select @SCOrderNo ,POAmendmentExpenseDetailNo ,SCOrderItemDetNo, 
			sed.ExpenseHeadNo, 
			sed.BillingLocationNo, 
			sed.BillingQtyNo, 
			sed.FreightRateTypeNo, 
			sed.Rate, 
			sed.PartyNo, 
			sed.IncludeInValuation, 
			sed.ReferenceNo, 
			sed.ReferenceDate, 
			sed.RouteIdentity
			from Purchase.SCOrderAmendmentExpenseHead sed
			inner join Purchase.SCOrderAmendmentItemDetail sid on sid.SCOrderAmendmentItemDetNo = sed.SCOrderAmendmentItemDetNo 
			inner join Purchase.SCOrderAmendmentMain scm on scm.SCOrderAmendmentNo = sed.SCOrderAmendmentNo 
			inner join Purchase.SCOrderItemDetail sd on sd.SCOrderNo = scm.SCOrderNo and sd.ItemNo = sid.ItemNo 
				and Coalesce(	sid.ExpenseHeadNo, sid.ActivityNo,sid.ServiceItemNo ,0) =
				 Coalesce(	sd.ExpenseHeadNo, sd.ActivityNo,sd.ServiceItemNo ,0)
				 and ISNULL(sid.makeno,0) = ISNULL(sid.makeno,0) and sid.ItemFlowNo = sd.ItemFlowNo 
			where  scm.SCOrderAmendmentNo= @prevSCOrderAmendmentNo  
			
			insert into Purchase.SCOrderTransportationRoute
			(	SCOrderNo, 
				RouteId, 
				SCOrderItemDetNo, 
				DeliveryModeNo, 
				FreightTypeNo, 
				FreightRateTypeNo, 
				FreightAmount, 
				FromLocationNo, 
				ToLocationNo, 
				WarehouseNo, 
				ToleranceBasisNo, 
				ToleranceMinus, 
				TolerancePlus, 
				Receipient, 
				IsGateEntry, 
				RouteIdentity, 
				ReceipientPartyNo, 
				SenderPartyNo, 
				NoOfTrips, 
				ParentRouteId)
				select @SCOrderNo ,
				RouteId, 
				SCOrderItemDetNo, 
				DeliveryModeNo, 
				sed.FreightTypeNo, 
				sed.FreightRateTypeNo, 
				sed.FreightAmount, 
				sed.FromLocationNo, 
				sed.ToLocationNo, 
				sed.WarehouseNo, 
				sed.ToleranceBasisNo, 
				sed.ToleranceMinus, 
				sed.TolerancePlus, 
				sed.Receipient, 
				sed.IsGateEntry, 
				sed.RouteIdentity, 
				sed.ReceipientPartyNo, 
				sed.SenderPartyNo, 
				sed.NoOfTrips, 
				sed.ParentRouteId
				from Purchase.SCOrderAmendmentTransportationRoute  sed
			inner join Purchase.SCOrderAmendmentItemDetail sid on sid.SCOrderAmendmentItemDetNo = sed.SCOrderAmendmentItemDetNo 
			inner join Purchase.SCOrderAmendmentMain scm on scm.SCOrderAmendmentNo = sed.SCOrderAmendmentNo 
			inner join Purchase.SCOrderItemDetail sd on sd.SCOrderNo = scm.SCOrderNo and sd.ItemNo = sid.ItemNo 
				and Coalesce(	sid.ExpenseHeadNo, sid.ActivityNo,sid.ServiceItemNo ,0) =
				 Coalesce(	sd.ExpenseHeadNo, sd.ActivityNo,sd.ServiceItemNo ,0)
				 and ISNULL(sid.makeno,0) = ISNULL(sid.makeno,0) and sid.ItemFlowNo = sd.ItemFlowNo 
			where  scm.SCOrderAmendmentNo= @prevSCOrderAmendmentNo  

			Delete from @SCReqDetail 
			if not exists(select 1 from Purchase.SCOrderMain where SCOrderDocumentTypeNo=1 
						and SCOrderNo =@SCOrderNo )
			begin
				Insert into @SCReqDetail select row_Number() over (order by screqno), SCReqNo ,
					ItemNo ,ActivityNo  ,SequenceNo  from 
						Purchase.SCOrderReqDetail  where SCOrderNo= @SCOrderNo 
				set @iRow= @@rowcount		
			end		
			
			 
			select @iCOunt = 0
			while @iRow > @iCount
			begin
				select @SCReqNo  = SCReqNo , @ItemNo = ItemNo , @ActivityNo = ActivityNo ,@SequenceNo=SequenceNo
				  from @SCReqDetail where  IDNo = @iCount +1
				
				exec [Inventory].[SCReqStatus] @SCReqNo,@ItemNo,@ActivityNo ,@SequenceNo ,	2,
						@OutputStatus output,@Result output
				
				if @OutputStatus =0
				   raiserror (@Result,16,-1)
				   
				set @iCOunt = @iCount + 1			
			end
						
	 -- UPDATE  main TABLE	
			 
			UPDATE  [Purchase].[SCOrderMain]
				SET 
				    AmendmentNo = SM.AmendmentNo
				   ,DivisionNo = SM.DivisionNo
				   ,DocumentNoYearly = SM.DocumentNoYearly
				   ,DocumentDate = SM.DocumentDate
				   ,DocumentStatusNo = SM.DocumentStatusNo
				   ,SCOrderDocumentTypeNo = SM.SCOrderDocumentTypeNo
				   ,SCTypeNo = SM.SCTypeNo
				   ,SCNatureNo = SM.SCNatureNo
				   ,SCSourceNo = SM.SCSourceNo
				   ,RefDocumentNo = SM.RefDocumentNo
				   ,VendorLocationNo = SM.VendorLocationNo
				   ,VendorContactNo = SM.VendorContactNo
				   ,SubcontractLocationNo = SM.SubcontractLocationNo
				   ,ValidityDate = SM.ValidityDate
				   ,CurrencyConversionNo = SM.CurrencyConversionNo
				   ,DivisibilityRate = SM.DivisibilityRate
				   ,MultiplicativeRate=SM.MultiplicativeRate
				   ,RateType = SM.RateType
				   ,ConsigneeLocationNo = SM.ConsigneeLocationNo
				   ,FreightTypeNo = SM.FreightTypeNo
				   ,FreightRateTypeNo= SM.FreightRateTypeNo
				   ,FreightAmount = SM.FreightAmount
				   ,LocationNo = SM.LocationNo 
				   ,PayToPartyLocationNo = SM.PayToPartyLocationNo
				   ,BillTopartyLocationNo = SM.BillTopartyLocationNo
				   ,PaymentModeNo = SM.PaymentModeNo
				   ,Remarks = SM.Remarks
				   ,StatusNo = SM.StatusNo
				   ,ModifiedBy = @LoginNo
				   ,ModifiedDate = Getdate()
				   ,AuthorizedBy = SM.AuthorizedBy
				   ,AuthorizedDate = SM.AuthorizedBy
				   ,ReleasedBy = SM.ReleasedBy 
				   ,ReleasedDate = SM.ReleasedBy 
				   ,BillingNatureNo = SM.BillingNatureNo
				   ,IsExpenseMapped = sm.IsExpenseMapped

				   , WarehouseNo= SM.WarehouseNo
				   , GSTDNo = SM.GSTDNo

				   ,SCValuationMethodNo = SM.SCValuationMethodNo
				   ,IsTransportationRouteApplicable = SM.IsTransportationRouteApplicable 
				   ,LevelNo = SM.LevelNo, AdditionCostTolerance = SM.AdditionCostTolerance 
		  FROM Purchase.SCOrderAmendmentMain SM
		  WHERE SCOrderAmendmentNo = @prevSCOrderAmendmentNo  
		  And [Purchase].[SCOrderMain].SCOrderNo= SM.SCOrderNo	
		   			   

		END  	
		if @IsCopy = 1
		 set @SCOrderAmendmentNo = null

-- Delete SCOrderNo Amendment Details Table
		delete from Notifications.TATAuthorizationDetail where InternalNo = @SCOrderAmendmentNo  And FormCode ='SCSA'
		delete from Purchase.SCOrderAmendmentAdditionItemDetail where SCOrderAmendmentNo=@SCOrderAmendmentNo
		delete from Purchase.SCOrderAmendmentTransportationRoute where SCOrderAmendmentNo=@SCOrderAmendmentNo
		delete from Purchase.SCOrderAmendmentExpenseHead where SCOrderAmendmentNo=@SCOrderAmendmentNo

		delete from Purchase.SCOrderAmendmentBillActivityDetail where SCOrderAmendmentNo=@SCOrderAmendmentNo
		delete from Purchase.SCOrderAmendmentBillMarkDetail  where SCOrderAmendmentNo=@SCOrderAmendmentNo
		delete from Purchase.SCOrderAmendmentBillDetail  where SCOrderAmendmentNo=@SCOrderAmendmentNo
		
		delete from Purchase.SCOrderAmendmentMarkActivityDetail where SCOrderAmendmentNo=@SCOrderAmendmentNo
		delete from Purchase.SCOrderAmendmentMarkDetail where SCOrderAmendmentNo=@SCOrderAmendmentNo
        Delete From Purchase.SCOrderAmendmentCustomFieldDetail Where SCOrderAmendmentNo=@SCOrderAmendmentNo
