CREATE PROCEDURE [Finance].[FIFIAuthorize]  
	-- Add the parameters for the stored procedure here
	@FreightInvoiceNo int,
	@Isprocessing bit =0 
	, @Result bit = 0 output
    ,@ResultMsg varchar(1000) ='' output
	 
AS
BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;
begin try

	Declare @PartyBillNo varchar(30), @PartyBillDate smalldatetime,	 @DocumentNoYearly varchar(30), @DocumentDate smalldatetime ,@VoucherNo Int
	Declare @SetRateZero Decimal(15,2)=1
	
	SELECT   @PartyBillNo = InvoiceNo   , @DocumentNoYearly = DocumentNoYearly, 
		@PartyBillDate = InvoiceDate   , @DocumentDate = DocumentDate ,@VoucherNo =VoucherNo 
		from Finance.FreightInvoiceMain where FreightInvoiceNo  = @FreightInvoiceNo


	If Exists (Select 1
			   From (
							Select GD.POAmendmentNo
							From Finance.FreightInvoiceDocumentDetails DD
							Inner Join GateEntry.InvGateEntryItemDetail GD on GD.GateEntryNo=DD.GateEntryNo
							Where DD.FreightInvoiceNo=@FreightInvoiceNo And DD.GateEntryNo is Not Null
							Group By GD.POAmendmentNo								
							Union All
							Select GD.POAmendmentNo
							From Finance.FreightInvoiceDocumentDetails DD
							Inner Join Inventory.GRNItemDetail GD on GD.GRNNo=DD.GRNNo
							Where DD.FreightInvoiceNo=@FreightInvoiceNo And DD.GRNNo is Not Null
							Group By GD.POAmendmentNo
					)DD 
			   Inner Join Purchase.POAmendmentMain PA on PA.POAmendmentNo=DD.POAmendmentNo 
			   Where PA.PORefDocumentTypeNo=9 -- Against DO
				)
	BEGIN
		set @SetRateZero=0
	END
			
if exists(select 1 from Finance.FreightInvoiceDocumentDetails a, GateEntry.InvGateEntryMain b where a.GateEntryNo = b.GateEntryNo 
				and b.ModuleNo = 19  and FreightInvoiceNo = @FreightInvoiceNo )
and exists(select 1 from Finance.FreightInvoiceDocumentDetails a, GateEntry.InvGateEntryMain b where a.GateEntryNo = b.GateEntryNo 
				and b.ModuleNo = 2 and FreightInvoiceNo = @FreightInvoiceNo  )				
begin
	raiserror ('Discrete And Inventory Gateentry can''t be pass at same time',16,-1)
end

-- for direct
 if exists(select 1 from Finance.FreightInvoiceMain fm Where FreightInvoiceNo =@FreightInvoiceNo 
			and AuthorizedBy is not null and fm.AdviceTypeNo <>2 ) --exclude On behalf of party )
			 
begin
 
if  exists (select 1 from masterdata.mItemOtherInformation ioi,Finance.FreightInvoiceItemDetail gd where gd.ItemNo = ioi.ItemNo
					and gd.FreightInvoiceNo = @FreightInvoiceNo and StockValueMethodNo =4)
begin
	if exists (select 1 from Finance.StockMonthlyLockingMain slm 
				inner join  Finance.ARDItemRule ard on ard.AccountNo = slm.StockAccountNo 
				where ard.ItemNo in(select ItemNo from Finance.FreightInvoiceItemDetail where FreightInvoiceNo = @FreightInvoiceNo ) 
					and @DocumentDate <= slm.DocumentDate )
		raiserror('Month is locked via Stock Monthly Closing(ISML)',16,-1)		

end	

 
	
	if @VoucherNo is null	
		select @VoucherNo= cjm.VoucherNo from Finance.FreightInvoiceMain gm 
			inner join Finance.CashCounterMain ccm on ccm.DocumentNo = gm.FreightInvoiceNo and
					ccm.SourceDocumentFormNo =450 and gm.PaymentModeNo =1
		inner join Finance.CashJournalMain cjm on cjm.CashCounterNo =ccm.CashCounterNo
		where FreightInvoiceNo= @FreightInvoiceNo 
		
			 
			insert into Inventory.InvLotBillDetail (InvLotSummaryNo, InvLotDetailNo, ItemNo, MakeNo, BillNo,
			 BillNoYearly, BillDate, PartyBillNo, PartyBillDate, TransactionFlag, BasicValue, BillQty
				,ItemRate ,DiffRate,CreatedDate,VoucherNo ,LotBalanceQty  )
			select ls.InvLotSummaryNo, bd.InvLotDetailNo ,bd.ItemNo , bd.MakeNo ,@FreightInvoiceNo ,
					fm.DocumentNoYearly , fm.DocumentDate , fm.InvoiceNo,fm.InvoiceDate ,'FIFI'
					,(ls.ReceivedQty *(((( FreightAmount - isnull(rej.amount,0)  - ISNULL(DeductionAmount,0)- ISNULL(OtherDeduction,0)) * isnull(nullif((ls.ReceivedQty* convert(decimal(18,10),isnull(RatePerUnit,ls.VRate)))/isnull(Total,1),0),1) )  /ls.ReceivedQty )- isnull(RatePerUnit,0) ))*@SetRateZero
					 ,ls.ReceivedQty 
					 ,( ((( FreightAmount   - isnull(rej.amount,0) - ISNULL(DeductionAmount,0)- ISNULL(OtherDeduction,0)) * isnull(nullif((ls.ReceivedQty*  convert(decimal(18,10),isnull(RatePerUnit,ls.VRate)))/isnull(Total,1),0),1) )  /ls.ReceivedQty )- isnull(RatePerUnit,0))*@SetRateZero  Rate
					 ,0 ,GETDATE () + 0.001 ,@VoucherNo ,ls.BalanceQty 
				from 
				Finance.FreightInvoiceMain fm  
				Inner join Finance.FreightInvoiceDocumentDetail fd on fd.FreightInvoiceNo = fm.FreightInvoiceNo   
				inner join Finance.FreightInvoiceDocumentDetails fds on fds.FreightInvoiceNo = fd.FreightInvoiceNo 
					and (fds.GateEntryNo = fd.RefDocNo or fds.GRNNo = fd.RefDocNo or fds.SalesReturnNo = fd.RefDocNo  )
				outer apply ( select sum(PA.PayableAmt) [Amount]
									from Inventory.GRNItemDetail ID
										inner join Inventory.GRNMain M on M.GRNNo=ID.GRNNo
										cross apply (select ID.ChallanQty*isnull(nullif(convert(decimal(18,6),ID.Rate),0),0.0001)*fd.FreightAmount /
															(select SUM(case when IDIn.AcceptedQty+ISNULL(IDIn.FOCQty,0)=0 then IDIn.ChallanQty 
																		else IDIn.AcceptedQty+ISNULL(IDIn.FOCQty,0) end*isnull(nullif(convert(decimal(18,6),IDIn.Rate),0),0.0001)) 
																from Inventory.GRNItemDetail IDIn
																where IDIn.GateEntryNo=ID.GateEntryNo) FreightAmount) RDNI
										cross apply (select (RDNI.FreightAmount - isnull(ID.FreightAmount,0)) PayableAmt)PA
									where M.AuthorizedBy is not null and (ID.GRNNo = fds.GRNNo or ID.GateEntryNo = fds.GateEntryNo)
										and ID.AcceptedQty+ISNULL(ID.FOCQty,0) = 0
										and M.AuthorizedDate <= fm.AuthorizedDate -- 15 01 2020
									having sum(PA.PayableAmt)<>0) rej
				inner join Inventory.InvLotSummary  ls on ((ls.GateEntryNo = fds.GateEntryNo
						or ls.DocumentNo = fds.GRNNo) and ls.TransactionFlag in('IGRN','DRSN','RSLE')  or (ls.DocumentNo = fds.SalesReturnNo and ls.transactionflag= 'SDSR'))
				 
				Inner join Inventory.InvLotDetail ld on ld.InvLotSummaryNo = ls.InvLotSummaryNo and ld.TransactionFlag = ls.TransactionFlag 
				Inner join Inventory.InvLotBillDetail bd on ld.InvLotSummaryNo = bd.InvLotSummaryNo and bd.InvLotDetailNo = ld.InvLotDetailNo 
											and bd.TransactionFlag = ld.TransactionFlag 
				Left join Inventory.InvLotTaxDetail td on td.InvLotSummaryNo = bd.InvLotSummaryNo and td.InvLotDetailNo = bd.InvLotDetailNo 
							 and td.InvLotBillDetailNo = bd.InvLotBillDetailNo and td.MiscChargeNo is null and td.ExpenseHeadNo is null and td.RatePerUnit <>0
				Inner join masterdata.mItemMaster im on im.ItemNo = ls.ItemNo 
				outer apply (select  SUM(pls.ReceivedQty * convert(decimal(18,10),isnull(pitd.RatePerUnit,pls.VRate))) Total from 
									Inventory.InvLotSummary pls
									inner join Inventory.InvLotDetail pld on pld.InvLotSummaryNo = pls.InvLotSummaryNo and pld.TransactionFlag = pls.TransactionFlag 
									Inner join Inventory.InvLotBillDetail pbd on pbd.InvLotSummaryNo = pld.InvLotSummaryNo 
											AND pbd.InvLotDetailNo = pld.InvLotDetailNo and pbd.TransactionFlag = pld.TransactionFlag 
											and pls.TransactionFlag = pbd.TransactionFlag  and pbd.IsIncluded   = 1
									left join Inventory.InvLotTaxDetail pitd on pitd.invlotbilldetailno = pbd.invlotbilldetailno 
												and pitd.miscchargeno is null and pitd.expenseheadno is null and pitd.RatePerUnit<>0
									  where  ( ((pls.GateEntryNo = ls.GateEntryNo Or pls.DocumentNo = fds.GRNNo )  
									  and pls.TransactionFlag in('IGRN','DRSN','RSLE')) or (pls.DocumentNo = fds.SalesReturnNo and pls.TransactionFlag ='SDSR')) -- and pbd.IsIncluded =1 commeneted on  20 mar 2014 by akhil
									  and PLS.IsRejectedLot=0--- Added by Alok on 26 oct 2023 (191023002)
										having SUM(pls.ReceivedQty * pls.VRate)>0) Per
				
				where
				ld.ReceivedQty >0 and ld.StockTypeNo =1 and 	ls.StockNatureNo =1 and bd.IsIncluded =1 and
				fd.FreightInvoiceNo =@FreightInvoiceNo
				and not exists(select 1 from inventory.InvLotBillDetail where TransactionFlag ='FIFI' and BillNo = @FreightInvoiceNo and
								InvLotSummaryNo =ls.InvLotSummaryNo  )
				and Not exists(select 1 from Inventory.GRNItemDetail gd where gd.GRNNo = ls.DocumentNo and gd.ItemNo = ls.ItemNo 
						and isnull(gd.MakeNo,0) = isnull(ls.MakeNo,0) and isnull(gd.TechnicalGradeNo,0) = isnull(ls.TechnicalGradeNo,0) 
						and isnull(gd.POAmendmentNo,0) = isnull(ls.POAmendmentNo ,0) and isnull(gd.GateEntryNo,0) = isnull(ls.GateEntryNo,0) 
						and isnull(gd.DispatchChallanNo,0) = isnull(ls.DispatchChallanNo,0) and isnull(gd.SalesOrderNo,0) = isnull(ls.SalesOrderNo ,0)
						and isnull(gd.InvStockTransferNo,0) = isnull( ls.invStockTransferNo ,0)
						and gd.ReceivedQty - isnull(gd.RejectedQty,0) = 0 )-- exclude full rejectd entry
			
			insert into Inventory.InvLotTaxDetail(InvLotSummaryNo,InvLotDetailNo,InvLotBillDetailNo, 
			ItemNo, MakeNo, Qty, RatePerUnit, Amount, InOutType, ChargeTypeNo)
			select InvLotSummaryNo,InvLotDetailNo,InvLotBillDetailNo, ItemNo, MakeNo, BillQty, ItemRate, isnull(ItemRate* BillQty,0) , 1, 2 
			from 	 
			Inventory.InvLotBillDetail bd
			where BillNo = @FreightInvoiceNo and bd.TransactionFlag ='FIFI'
			
				
			insert into Inventory.InvLotTaxDetail(MiscChargeNo,InvLotSummaryNo,InvLotDetailNo,InvLotBillDetailNo, 
			ItemNo, MakeNo, Qty, RatePerUnit, Amount, InOutType, ChargeTypeNo,IncludeInBasic)
			select ftd.MiscChargeNo, bd.InvLotSummaryNo, bd.InvLotDetailNo, bd.InvLotBillDetailNo,bd.ItemNo,bd.MakeNo,
			ls.receivedQty,
				 ((( ftd.TotalValue *  ((ls.ReceivedQty* convert(decimal(18,10),ls.VRate))/isnull(Total,1)) * (FreightAmount - ISNULL(DeductionAmount,0)- ISNULL(OtherDeduction,0)))/TotAmount) /  (ls.ReceivedQty) )*@SetRateZero
				 ,((( ( ftd.TotalValue *  ((ls.ReceivedQty* convert(decimal(18,10),ls.VRate))/isnull(Total,1)) * (FreightAmount - ISNULL(DeductionAmount,0)- ISNULL(OtherDeduction,0)) /TotAmount )) / (ls.ReceivedQty))* ls.receivedQty  )*@SetRateZero
				 , 1, ChargeType ,ftd.IncludeInBasic 
			from 
				Finance.FreightInvoiceTaxDetail ftd
				inner join Finance.FreightInvoiceDocumentDetail fd on fd.FreightInvoiceNo= ftd.FreightInvoiceNo 
				inner join Finance.FreightInvoiceDocumentDetails fds on fds.FreightInvoiceNo = fd.FreightInvoiceNo 
					and (fds.GateEntryNo = fd.RefDocNo or fds.GRNNo = fd.RefDocNo )
				inner join (select sum(FreightAmount - ISNULL(DeductionAmount,0)- ISNULL(OtherDeduction,0)) TotAmount,FreightInvoiceNo from Finance.FreightInvoiceDocumentDetail
							where FreightInvoiceNo=@FreightInvoiceNo group by FreightInvoiceNo) tot
						on tot.FreightInvoiceNo= fd.FreightInvoiceNo 
				inner join Inventory.InvLotSummary  ls on (ls.GateEntryNo = fds.GateEntryNo
						Or ls.DocumentNo = fds.GRNNo) and ls.TransactionFlag in('IGRN','DRSN','RSLE')
				Inner join inventory.invlotbilldetail bd on bd.InvLotSummaryNo = ls.InvLotSummaryNo and 
							bd.BillNo =@FreightInvoiceNo and bd.transactionFlag='FIFI'
				inner join masterdata.mMiscCharges mc on mc.MiscChargeNo = ftd.MiscChargeNo 
				outer apply (select  SUM(pls.ReceivedQty * convert(decimal(18,10),pls.VRate)) Total from 
									Inventory.InvLotSummary pls
									Inner join Inventory.InvLotBillDetail pbd on pls.InvLotSummaryNo = pbd.InvLotSummaryNo 
											and pls.TransactionFlag = pbd.TransactionFlag 
									  where   (GateEntryNo = ls.GateEntryNo Or DocumentNo = fds.GRNNo)  
									  and pls.TransactionFlag in('IGRN','DRSN','RSLE') --and pbd.IsIncluded =1 commentedd on 20 mar 2014
										having SUM(pls.ReceivedQty * pls.VRate)>0) Per
				where
				fd.FreightInvoiceNo =@FreightInvoiceNo and ls.ReceivedQty>0 
				 and not exists(select 1 from inventory.InvLotBillDetail a,Inventory.InvLotTaxDetail b where
							a.InvLotBillDetailNo = b.InvLotBillDetailNo and TransactionFlag ='FIFI' and BillNo = @FreightInvoiceNo and 
								a.InvLotSummaryNo  =bd.InvLotSummaryNo and b.MiscChargeNo = ftd.MiscChargeNo  )

		
			Update bd 
			 set DiffRate = bd.ItemRate + isnull(DiffVrate,0) *@SetRateZero
			, VRate = (ls.Vrate + ISNULL(bd.ItemRate,0) +  ISNULL(DiffVrate,0))  * @SetRateZero
			, DiffInLRate = td.DiffinLRate *@SetRateZero
	 		from Inventory.InvLotBillDetail bd
	 		inner join inventory.invlotSummary ls on ls.invlotsummaryno = bd.invlotSummaryNo
	 		outer apply (select  isnull(sum(case when mc.CalculationNature is null then td.RatePerUnit 
	 							when td.IncludeInBasic =1 and isnull(mc.CalculationNature,1) =1  then td.RatePerUnit 
								when td.IncludeInBasic =1 and mc.CalculationNature =2 then 0 -td.RatePerUnit
								else 0 end),0) DiffVrate,
								 isnull(sum(case when isnull(mc.CalculationNature,1) =1  then td.RatePerUnit 
								when mc.CalculationNature =2 then 0 -(td.RatePerUnit  )
								else 0 end),0) DiffinLRate
					from  Inventory.InvLotTaxDetail td
						Inner join masterdata.mMiscCharges mc on mc.MiscChargeNo = td.MiscChargeNo 
					 where td.InvLotSummaryNo =bd.InvLotSummaryNo and td.InvLotDetailNo = bd.InvLotDetailNo 
								and td.InvLotBillDetailNo = bd.InvLotBillDetailNo  And ChargeTypeNo = 2 ) td
	 		where bd.BillNo = @FreightInvoiceNo and bd.TransactionFlag ='FIFI' 
	 		
	 		update ls set VRate = (ls.VRate + ISNULL(DiffRate,0))*@SetRateZero 
	 		, LRate = (ls.LRate + ISNULL(DiffInLRate ,0))*@SetRateZero
	 		from
	 		Inventory.InvLotSummary ls
	 		inner join Inventory.InvLotBillDetail bd on bd.InvLotSummaryNo = ls.InvLotSummaryNo 
	 		where bd.BillNo = @FreightInvoiceNo and bd.TransactionFlag ='FIFI' 
			
		  	update td set RatePerUnit = (RatePerUnit + isnull(DiffRate,0))*@SetRateZero,
			Amount = (Amount + (isnull(DiffRate,0) * BillQty ))*@SetRateZero
			from
			Inventory.InvLotSummaryTaxDetail td
			inner join Inventory.InvLotBillDetail bd on bd.InvLotSummaryNo = td .InvLotSummaryNo 
			where  bd.BillNo = @FreightInvoiceNo and bd.TransactionFlag ='FIFI'  and td.MiscChargeNo is null and td.ExpenseHeadNo is null
			
			update td set RatePerUnit = (td.RatePerUnit + isnull(DiffRate,0))*@SetRateZero,
			Amount = td.Amount + (isnull(DiffRate,0) * BillQty )
			from
			Inventory.InvLotSummaryTaxDetail td
			inner join Inventory.InvLotBillDetail bd on bd.InvLotSummaryNo = td .InvLotSummaryNo 
			INNER JOIN Inventory.InvLotTaxDetail itd on itd .InvLotBillDetailNo = bd.InvLotBillDetailNo
			where  bd.BillNo = @FreightInvoiceNo and bd.TransactionFlag ='FIFI' and td.MiscChargeNo = itd.MiscChargeNo 
			
			insert into Inventory.InvLotSummaryTaxDetail(InvLotSummaryNo, MiscChargeNo, Qty, RatePerUnit,Amount, InOutType, ChargeTypeNo)
			select bd.InvLotSummaryNo, MiscChargeNo, Qty, RatePerUnit, Amount, InOutType, ChargeTypeNo
			from
			Inventory.InvLotBillDetail bd
			inner join Inventory.InvLotTaxDetail td on td.InvLotBillDetailNo = bd.InvLotBillDetailNo 
			where BillNo = @FreightInvoiceNo and bd.TransactionFlag ='FIFI' and MiscChargeNo is Not null
			and not exists(select 1 from Inventory.InvLotSummaryTaxDetail where InvLotSummaryNo = td.InvLotSummaryNo and MiscChargeNo = td.MiscChargeNo)
			
			insert into Inventory.InvLotSummaryTaxDetail(InvLotSummaryNo, MiscChargeNo, Qty, RatePerUnit,Amount, InOutType, ChargeTypeNo)
			select bd.InvLotSummaryNo, null, BillQty , DiffRate, BillQty* DiffRate, 1, 2
			from
			Inventory.InvLotBillDetail bd
			where BillNo = @FreightInvoiceNo and bd.TransactionFlag ='FIFI' 
			and not exists(select 1 from Inventory.InvLotSummaryTaxDetail where InvLotSummaryNo = bd.InvLotSummaryNo 
			and MiscChargeNo is null and ExpenseHeadNo is null)
	 
 

      
begin


	 
		IF OBJECT_ID('tempdb..#IntransitDetail') IS NOT NULL
			DROP TABLE #IntransitDetail

		CREATE TABLE #IntransitDetail (
			invlotsummaryNo    int,
			IntransitQty       decimal(12,3),
			TempDiff           decimal(18,6),
			TempLandedRate     decimal(18,6),
			IsInclude          bit
		)
		Declare @PORoute as table (PONo int,MaxRouteIdentity decimal(10,8)) 
		
		Insert into @PORoute 
		select ptr.PONo , MAX(ptr.RouteIdentity ) from 
		   Purchase.POTransportationRoute ptr
		    inner join Purchase.POAmendmentMain pam on pam.PONo = ptr.PONo 
			inner join Inventory.InvLotSummary ls on ls.POAmendmentNo = pam.POAmendmentNo 
			inner join Inventory.InvLotBillDetail bd on bd.InvLotSummaryNo = ls.InvLotSummaryNo 
			where bd.BillNo = @FreightInvoiceNo and bd.TransactionFlag ='FIFI'
			and convert(tinyint,ptr.RouteIdentity )= convert(tinyint,ls.RouteIdentity)
			and ptr.RouteIdentity - ls.RouteIdentity between 
	CONVERT(decimal(18,8),'0.'+REPLICATE ('0', LEN(CONVERT(int,CONVERT(float,REVERSE (ls.RouteIdentity))))) +'1')
	 and CONVERT(decimal(18,8),'0.'+REPLICATE ('0', LEN(CONVERT(int,CONVERT(float,REVERSE (ls.RouteIdentity))))) +'9')
			group by  ptr.PONo
								
 
	
	if exists(select 1 from @PORoute)
	begin		
		if not exists (select 1 from Purchase.POExpenseHead peh , Purchase.POExpenseHeadOutputDetail poed ,@PORoute pr
					where peh.POExpenseDetailNo = poed.POExpenseDetailNo and peh.PONo = pr.PONo  )
		begin	
				If object_Id('tempdb..#TempCTE4') is not NULL
				   Drop Table #TempCTE4

				;with CTE4
					as
					(select ld.InvLotSummaryNo ParentInvLotSummaryNo  ,ld.InvLotSummaryNo GrandParentInvLotSummaryNo , Cast(1 as int) Level 
					from Inventory.InvLotDetail ld
					inner join Inventory.InvLotBillDetail bd on bd.InvLotSummaryNo = ld.InvLotSummaryNo 
					where bd.BillNo = @FreightInvoiceNo and bd.TransactionFlag ='FIFI'
					and ld.TransactionFlag ='IMST' group by ld.InvLotSummaryNo
					union all
					select ls.InvLotSummaryNo  ,GrandParentInvLotSummaryNo, Level + 1
					from CTE4 , Inventory.InvlotSummaryRefDetail ls
					where ls.ParentInvLotSummaryNo = CTE4.ParentInvLotSummaryNo  and ls.InvLotSummaryNo <>ls.ParentInvLotSummaryNo 
					)

				Select * into #TempCTE4 from CTE4 

				CREATE INDEX IX_TempCTE4_Parent 
				ON #TempCTE4 (ParentInvLotSummaryNo) 
				INCLUDE (GrandParentInvLotSummaryNo, Level);

				If object_Id('tempdb..#TempCTEFinal') is not NULL
				   Drop Table #TempCTEFinal

		
				;with CTE5
				as 
				(
				Select ILS.InvLotSummaryNo ParentInvLotSummaryNo,ILS.InvLotSummaryNo GrandParentInvLotSummaryNo 
				from #TempCTE4 CTE4
				inner join Inventory.InvLotSummary ILS on ILS.InvLotSummaryNo = CTE4.ParentInvLotSummaryNo and CTE4.ParentInvLotSummaryNo <> CTE4.GrandParentInvLotSummaryNo 
				where CTE4.ParentInvLotSummaryNo = (Select Min(ParentInvLotSummaryNo) from #TempCTE4 CTE42 where CTE42.ParentInvLotSummaryNo <> CTE4.GrandParentInvLotSummaryNo ) 
				--Select 'CTE4',* from CTE4
				Union all
				select ls.InvLotSummaryNo  ,GrandParentInvLotSummaryNo
				from CTE5 , Inventory.InvlotSummaryRefDetail ls
				where ls.ParentInvLotSummaryNo = CTE5.ParentInvLotSummaryNo  and ls.InvLotSummaryNo <>ls.ParentInvLotSummaryNo 
				)

				Select * into #TempCTEFinal from CTE5 
				CREATE INDEX IX_TempCTEFinal_Parent 
				ON #TempCTEFinal (ParentInvLotSummaryNo)
				INCLUDE (GrandParentInvLotSummaryNo);


				;with CTE
					as
					(select ld.InvLotSummaryNo ParentInvLotSummaryNo  ,ld.InvLotSummaryNo GrandParentInvLotSummaryNo  
					from Inventory.InvLotDetail ld
					inner join Inventory.InvLotBillDetail bd on bd.InvLotSummaryNo = ld.InvLotSummaryNo 
					where bd.BillNo = @FreightInvoiceNo and bd.TransactionFlag ='FIFI'
					and ld.TransactionFlag ='IMST' group by ld.InvLotSummaryNo
					union all
					select ls.InvLotSummaryNo  ,GrandParentInvLotSummaryNo 
					from CTE , Inventory.InvlotSummaryRefDetail ls
					where ls.ParentInvLotSummaryNo = CTE.ParentInvLotSummaryNo  and ls.InvLotSummaryNo <>ls.ParentInvLotSummaryNo 
					)

				Select * into #TempCTECurr from CTE 
				
				CREATE INDEX IX_TempCTECurr_Parent 
				ON #TempCTECurr (ParentInvLotSummaryNo)
				INCLUDE (GrandParentInvLotSummaryNo);
					
		insert into #IntransitDetail
		select ls.InvLotSummaryNo ,case when IMSTtt.Intranit >IMSTtt.IMSTQty then null else IMSTtt.IMSTQty-isnull(IMSTtt.Intranit,0) end
			,case when tt.Intranit >tt.IMSTQty then (bd.DiffRate * tt.IMSTQty)/tt.Intranit else  bd.DiffRate end
			,case when tt.Intranit >tt.IMSTQty then (bd.DiffRate* tt.IMSTQty)/tt.Intranit else   bd.DiffRate end
			--, case when LS.InvLotSummaryNo = lls.InvLotSummaryNo  then 0 else 1 end
			,case when LS.RouteIdentity = isnull(maxROuteID,ls.RouteIdentity)  then 0 else 1 end
			 from 
			inventory.InvLotSummary LS
			inner Join #TempCTECurr CTE on CTE.ParentInvLotSummaryNo=LS.InvLotSummaryNo
			Inner join Purchase.POAmendmentMain pm on pm.POAmendmentNo = LS.POAmendmentNo 
			Inner join Purchase.POAmendmentMain llspm on pm.PONo= llspm.PONo 
			inner join Inventory.InvLotSummary lls on lls.POAmendmentNo = llspm.POAmendmentNo 
			Inner join inventory.InvlotSummaryRefDetail LSRD on LSRD.InvLotSummaryNo = LS.InvLotSummaryNo

Create table #temp (InvLotSummaryNo int,SNo smallint, DiffRate decimal(18,6),DiffInLRate decimal(18,6)	, Lrate  decimal(18,6) ,VRate  decimal(18,6) , 
			InvLotBillDetailNo int  , ItemNo int	, transactionflag char(4), ItemRate decimal(18,6) primary key (InvLotSummaryNo,SNo) ) 
		Insert into #temp Select InvLotSummaryNo, DENSE_RANK () over (Partition By InvLotSummaryNo order by CreatedDate,invLotBillDetailNo) as SNo,
			DiffRate,DiffInLRate,ItemRate + ISNULL(DiffInLRate,0) Lrate, VRate,InvLotBillDetailNo,ItemNo,transactionflag,ItemRate 
			From Inventory.InvLotBillDetail 
            where  TransactionFlag not in ('SDDC','SDEI','SDCI') 
            and invlotsummaryNo in(select invlotsummaryNo from Inventory.InvLotBillDetail where BillNo =@FreightInvoiceNo and TransactionFlag ='FIFI' )
         
         declare @Vrate decimal(18,6), @Lrate decimal(18,6),@InvlotSummaryNo int,@DiffRate  decimal(18,6)    
         Update #temp Set 
          @Lrate= case when Sno=1 then Itemrate + DiffInLRate else @Lrate + DiffInLRate end,
       	  Lrate=case when Sno=1 then Itemrate + DiffInLRate else  @Lrate end ,
       	  @Vrate= case when Sno=1 then Itemrate + DiffRate else @Vrate + DiffRate end,
       	  Vrate=case when Sno=1 then Itemrate + DiffRate else  @Vrate end,
       	 @InvlotSummaryNo=InvlotSummaryNo
		From #temp
        update ls set LRate =t.Lrate,VRate =t.Vrate
        from #temp t,Inventory.InvLotBillDetail   bd,Inventory.InvLotSummary ls where(T.InvLotBillDetailNo = bd.InvLotBillDetailNo) 
       And (t.Lrate <> ls.LRate or t.Vrate <> ls.Vrate ) And ls.InvLotSummaryNo = bd.InvLotSummaryNo and sNo = (select MAX(SNo) from #temp where InvLotSummaryNo= ls.InvLotSummaryNo )
       update ls set VRate =t.Vrate  from #temp t,Inventory.InvLotBillDetail   bd,Inventory.InvLotSummary ls where(T.InvLotBillDetailNo = bd.InvLotBillDetailNo)
       and ls.VRate <>t.Vrate and ls.InvLotSummaryNo = bd.InvLotSummaryNo and sNo = (select MAX(SNo) from #temp where InvLotSummaryNo= ls.InvLotSummaryNo )
      update bd set VRate =t.Vrate  from #temp t,Inventory.InvLotBillDetail   bd where(T.InvLotBillDetailNo = bd.InvLotBillDetailNo) and t.Vrate <>bd.VRate
      update ld set VRate =ls.Vrate,LRate = ls.LRate from #temp t,Inventory.InvLotDetail ld , Inventory.InvLotSummary ls
