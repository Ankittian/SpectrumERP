						and rm.ReturnNo = rid.ReturnNo 
				where mid.itemno = @ItemNo and isnull(mid.MakeNo,0) = isnull(@MakeNo,0) and 
					--isnull(mid.TechnicalGradeNo,0) = isnull(@TechnicalGradeNo,0) and comment on 23 feb 2013 as return doesnt have TechincalGrade Column (have to add it later)
					(mid.CostCenterNo is null or mid.CostCenterNo = @CostCenterNo) and rm.returnNo = @DocumentNo
				SET @iRowCount = @@ROWCOUNT
				delete from @invTempDetail where BalanceQty<=0
					   
			
		    set @iCount = 0
			WHILE @ReturnQty > 0
					BEGIN
						SELECT  @TempinvLotSummaryNo = invlotsummaryNo,@IssuedQty = balanceQty,
						@TempinvLotDetailNo =invlotDetailNo from
						@invTempDetail where  IDNo = @iCount + 1

							IF isnull(@IssuedQty,0) > isnull(@ReturnQty,0)
								BEGIN
									SET @TempReturnQty = isnull(@ReturnQty,0)	
									SET @ReturnQty = 0
								END
							ELSE
								BEGIN
									SET @TempReturnQty = isnull(@IssuedQty,0)
									SET @ReturnQty = isnull(@ReturnQty,0) - isnull(@IssuedQty,0)
								END	
					INSERT INTO Inventory.InvLotDetail (CompanyNo,YearNo,InvLotSummaryNo,ParentInvLotSummaryNo,
					[ItemNo],[MakeNo],[LotNo],[DocumentNo],[DocumentNoYearly],[DocumentDate]
					,[TransactionFlag],[AccountingReceivedQty],[StockTypeNo],[ReceivedQty],[IssuedQty],[ItemRate]
					,[LRate],[VRate] ,[IsOwnership]
					,[VendorNo],[IsReserved],[IsCapital],[ProjectNo],[DivisionNo],[DeptNo],[CostCenterNo],[BinNo]
					,[WarehouseNo], [IndentTypeNo],[ReturnQty],TechnicalGradeNo ,CreatedDate,VoucherNo)
					
					select @CompanyNo,@YearNo ,ls.InvLotSummaryNo,ls.ParentInvLotSummaryNo,
					 ls.ItemNo, ls.MakeNo, ls.LotNo, @DocumentNo, @DocumentNoYearly, @DocumentDate
					,@TransactionFlag, null ,@StockTypeNo,null ,null,ls.[ItemRate],ls.[LRate],ls.[VRate] ,
					ls.IsOwnership,null,
					ls.IsReserved, ls.IsCapital, ls.ProjectNo, @DivisionNo, @DeptNo, @CostCenterNo , @BinNo
					,@WarehouseNo, null,@TempReturnQty,@TechnicalGradeNo, GETDATE (),@VoucherNo
					from 
					inventory.InvLotDetail ls where invlotDetailNo=@TempinvLotDetailNo

				UPDATE Inventory.InvLOTSummary SET ReturnQty =  ISNULL(ReturnQty,0) +  isnull(@TempReturnQty,0)
							,BalanceQty = isnull(BalanceQty,0) +  isnull(@TempReturnQty,0)
							WHERE InvLotSummaryNo = @TempinvlotSummaryNo
					
						SET @iCount = @iCount  + 1
			  END		-- WHILE @ReturnQty > 0
			end --- 'IMMR'
			
			
			else
			begin
				Declare @ErrorMsg Varchar(1000),@balQty decimal(15,3) 
				
				Select @ErrorMsg='insufficient Stock ' + convert(varchar,@InvLotSummaryNo) + ' for selected lot ' + IsNull(' of "' + IM.FullCode  + ' - ' + ls.LotNo  +
										'"','') 
				From masterdata.mItemMaster IM,Inventory.InvLotSummary ls
				Where ls.ItemNo = IM.ItemNo and ls.InvLotSummaryNo = @InvLotSummaryNo and  im.ItemNo=@ItemNo
			  declare @tempWarehouseno smallint
			  select @balQty= balanceQty  ,@tempWarehouseno = WarehouseNo 
			    from Inventory.invlotSummary where invlotsummaryNo= @invlotsummaryNo
			   if @WarehouseNo<>@tempWarehouseno 
			   begin
					set @ErrorMsg= @ErrorMsg + ' warehouse house mismatch '
					raiserror(@ErrorMsg,16,-1)
				end
				 
				if @IssuedQty > @balQty
				begin
					set @ErrorMsg= @ErrorMsg + ' Balance Qty ' + CONVERT(varchar(10),@balQty)
					raiserror(@ErrorMsg,16,-1)
				end
				else if exists(select 1 from Inventory.InvLotDetail  where 
						invlotsummaryNo= @invlotsummaryNo 
					having isnull(sum(ReceivedQty),0) + isnull(SUM(ReturnQty),0) - isnull(SUM(IssuedQty),0) - @IssuedQty <0)
				begin
					set @ErrorMsg= @ErrorMsg + ' Lot Balance Qty ' + CONVERT(varchar(10),@balQty)
					raiserror(@ErrorMsg,16,-1)
				end
				else if exists(
								select 1 
								from Inventory.InvLotSummary ls 
								Where ls.InvLotSummaryNo =@InvLotSummaryNo And ls.IsReserved = 1 
							)And @InvLotReserveNo Is Null
				begin
					Declare @ErrMsg Varchar(1000)

					Select 	@ErrMsg ='Lot is reserved for item ' + im.ItemDescription 
					From Inventory.InvLotSummary LS
					Inner Join masterdata.mItemMaster im on im.ItemNo=LS.ItemNo
					Where LS.InvLotSummaryNo=@InvLotSummaryNo

					raiserror(@ErrMsg,16,-1)
				end 
				else if exists(select 1 from Inventory.InvLotSummaryReserveDetail  where 
						InvLotReserveNo = @InvLotReserveNo 
						and BalanceQty - @IssuedQty <0)
				begin
					raiserror(@ErrorMsg,16,-1)
				end
				else if exists(select 1 from Inventory.InvLotSummary ls
								cross apply (select sum(balanceqty) BalQty , sum(BalancePcs) BalPcs 
									from Inventory.InvLotsummaryreserveDetail  where InvLotSummaryNo =ls.InvlotsummaryNo) tt
								where ls.Isreserved=1 and (abs(ls.BalanceQty - tt.BalQty)>.1 or abs(ls.Balancepcs - tt.BalPcs)>0 ) and IsReserved =1 
								and ls.InvLotSummaryNo =@InvLotSummaryNo )
							begin
								Declare @IName Varchar(1000)
								Select 	@IName ='Lot is reserved by some other user for item ' + im.ItemDescription 
								From Inventory.InvLotSummary LS
								Inner Join masterdata.mItemMaster im on im.ItemNo=LS.ItemNo
								Where LS.InvLotSummaryNo=@InvLotSummaryNo
							  raiserror(@IName,16,-1)
							end 

				 
				if @StockValueMethodNo=1 -- standard rate
					select @VRate =Rate  from masterdata.ItemRateValuationDetail 
					where PriceControlNo =1 and 
					@DocumentDate between FromDate and ISNULL(ToDate,@DocumentDate )
					and ItemNo =@ItemNo 
					and isnull(MakeNo,0) =isnull(@MakeNo,0) 
					and isnull(TechnicalGradeNo,0) =ISNULL( @TechnicalGradeNo ,0)
					and MAPValuationNo in(select 
					MAS.MAPValuationNo from masterdata.MAPValuationSetting MAS
					inner join masterdata.MAPValuationSettingDetail MASD on MASD.MAPValuationNo = MAS.MAPValuationNo 
					where MAS.CompanyNo = @CompanyNo and
					(MASD.DivisionNo is null or MASD.DivisionNo =@DivisionNo ) and
					(MASD.WarehouseNo = @WareHouseNo ))
				else if @StockValueMethodNo = 4 -- MAP
					select @VRate =Rate  from masterdata.ItemRateValuationDetail 
					where PriceControlNo =2 and 
					@DocumentDate between FromDate and ISNULL(ToDate,@DocumentDate )
					and ItemNo =@ItemNo 
					and isnull(MakeNo,0) =isnull(@MakeNo,0) 
					and isnull(TechnicalGradeNo,0) =ISNULL( @TechnicalGradeNo ,0)
					and MAPValuationNo in(select 
					MAS.MAPValuationNo from masterdata.MAPValuationSetting MAS
					inner join masterdata.MAPValuationSettingDetail MASD on MASD.MAPValuationNo = MAS.MAPValuationNo 
					where MAS.CompanyNo = @CompanyNo and
					(MASD.DivisionNo is null or MASD.DivisionNo =@DivisionNo ) and
					(MASD.WarehouseNo = @WareHouseNo ))
				
				INSERT INTO Inventory.InvLotDetail (CompanyNo, YearNo, InvLotSummaryNo, ItemNo, MakeNo, 
				LotNo, DocumentNo, DocumentNoYearly, DocumentDate, TransactionFlag, AccountingReceivedQty, 
				StockTypeNo, ReceivedQty, IssuedQty, IsOwnership, VendorNo, IsReserved, IsCapital, 
				ProjectNo, DivisionNo, DeptNo,CostCenterNo, BinNo, WarehouseNo, ParentinvLotSummaryNo,
				TechnicalGradeNo,CreatedDate ,ItemRate ,VRate,VoucherNo,SalesOrderAmendmentNo,IssuedPcs,MAPRate,LRate)
				
				select ls.CompanyNo, @YearNo, @InvLotSummaryNo, ls.ItemNo, ls.MakeNo, ls.LotNo, @DocumentNo, 
				@DocumentNoYearly, @DocumentDate, @TransactionFlag, null, @StockTypeNo, null, @IssuedQty, 
				ls.IsOwnership, ls.VendorNo, ls.IsReserved, @IsCapital, @ProjectNo, 
				ls.DivisionNo, @DeptNo, @CostCenterNo, @BinNo, ls.WarehouseNo,lsrd.ParentInvLotSummaryNo ,
				@TechnicalGradeNo,GETDATE (), ls.ItemRate , isnull(@VRate,ls.VRate),@VoucherNo
				,Case When @TransactionFlag  ='SDDC' 
						  Or @TransactionFlag  ='SDCI' 
						  Or @TransactionFlag  ='SDEI' 
					  Then @SalesOrderAmendmentNo  
					  Else Null 
				  END -- Dispatch Hold SO in Item Detail thats why
				  ,@IssuedPcs , ls.MAPRate ,ls.LRate 
				from inventory.InvLotSummary ls
				inner join Inventory.InvLotDetail ld on ld.InvLotSummaryNo = ls.InvLotSummaryNo 
						and ld.StockTypeNo = @StockTypeNo and ld.ReceivedQty >0
				inner join Inventory.InvlotSummaryRefDetail lsrd on lsrd.InvLotSummaryNo = ls.InvLotSummaryNo		
				where ls.invlotsummaryno = @invlotsummaryNo
				
				set @InvLotDetailNo = ident_current('Inventory.InvLotDetail')
			if @TransactionFlag = 'SDDC' or  @TransactionFlag = 'SDEI' or  @TransactionFlag = 'SDCI'
			or  @TransactionFlag = 'SDCS' or  @TransactionFlag = 'SDPS' or  @TransactionFlag = 'SDPK' 
			begin
			
			exec [Inventory].[InvLotBillTaxTransaction]
			@ItemNo =@ItemNo  , @MakeNo = @MakeNo ,@TechnicalGradeNo = @TechnicalGradeNo ,@CompanyNo=@CompanyNo ,
			@DivisionNo =@DivisionNo ,@TransactionFlag =@TransactionFlag ,@DocumentNo = @DocumentNo ,
			@WareHouseNo =@WareHouseNo ,@POAmendmentNo = @POAmendmentNo ,@SCOrderAmendmentNo=@SCOrderAmendmentNo ,
			@Status = @Status  output,@Result = @Result  output, @InvLotSummaryNo = @InvLotSummaryNo ,
			@RouteIdentity =@RouteIdentity,
			@InvLotDetailNo = @InvLotDetailNo ,	@InvLotBillDetailNo = @InvLotBillDetailNo output
			
		if @Status <> 1
				raiserror (@Result,16,-1)	
			end	
			
				update Inventory.invlotsummary set BalanceQty = BalanceQty - @IssuedQty 
										,BalancePcs = BalancePcs - @IssuedPcs 
										,IssuedQty = isnull(IssuedQty,0) + @IssuedQty
										,IssuedPcs = isnull(IssuedPcs,0) + @IssuedPcs 
				where invLotsummaryNo = @invLotSummaryNo 
				--and StockNatureNO = @StockNatureNo -- comment on 12 sep 2013
				
				
				update Inventory.InvLotSummaryReserveDetail  set BalanceQty = BalanceQty - @IssuedQty 
										,BalancePcs = BalancePcs - @IssuedPcs 
										,IssuedQty = isnull(IssuedQty,0) + @IssuedQty
										,IssuedPcs = isnull(IssuedPcs,0) + @IssuedPcs
										
				where InvLotReserveNo = @InvLotReserveNo
				
				
			if @TransactionFlag ='IMST'	and @RouteIdentity is not null
			begin
				update Inventory.InvLotDetail set InTransitBalanceQty = IssuedQty where 
						InvLotDetailNo =@InvLotDetailNo 
			end
				
				
			end
				
	  END --end of	IF @TransacionType = 'U'	 

 set @Status =1
 set @result ='Success'

END TRY
    BEGIN CATCH
		 set @Status = 0
		 set @result =ERROR_MESSAGE()
		
	END CATCH
