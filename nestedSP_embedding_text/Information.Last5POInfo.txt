CREATE PROCEDURE [Information].[Last5POInfo]  
	-- Add the parameters for the stored procedure here
	@CompanyNo tinyint,
	@DocumentDate smalldatetime,
	@ItemNo int
AS
BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;
	

    -- Insert statements for procedure here
	declare @str varchar(max)
    select @str=masterdata.JSForInformation('PRFQ')
    set @str=@str+masterdata.CSSForInformation('PRFQ')
	set @str=@str+'<table class="altrowstable" id="alternatecolorLastPOInfo">'
	set @str=@str+'<Caption><Font color="Maroon"><h3 align="Left"><b>Last 5 Purchase Order</b></h3></Font></Caption>'
    set @str=@str+'<tr><th>Sl. No.</th><th>Vendor Name</th><th>Document No.</th><th>Document Date</th><th>MaKe</th><th>Purchase Rate</th><th>Landed Rate</th><th>Discount</th></tr>'
	set @str=@str
				 +(
					replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					replace((   
					         Select Top 5 ROW_NUMBER() over ( Order by PM.DocumentDate desc) as td1
                               ,MVC.VendorName td2
					           ,PM.DocumentNoYearly As [td3]
					           ,convert(varchar,PM.DocumentDate,103) As [td4]
					           ,isnull(MM.MakeName,'') As [td5]
							   ,Cast(PO.Rate as Decimal(15,2)) [td6]
							   ,Cast((SuM(LR.LRate-Po.Rate))+PO.Rate as decimal(15,2))  As [td7]---Agreegate Function Group by Clause For it
																								---Landed rate include Po Rate also For each route So First Eliminate the Po rate From it 
																								---and the perform sum on it and then add Porate in it after 
							   ,CONVERT(decimal(12,2),ISNULL(ttPOITD.Discount,0)) as [td8]
                  			   From Purchase.POMain PM 
                  			   Inner Join Purchase.POItemDetail PO on PO.PONo=PM.PONo
							   Inner Join Purchase.POItemLRate LR on LR.PONo=PO.PONo			---In Case Of Route Identity Multiple landed Rate For Same item,Make and technical grade
																And LR.ItemNo=PO.ItemNo
																And ISNULL(LR.MakeNo,0)=isnull(PO.MakeNo,0)
																And isnull(LR.TechnicalGradeNo,0)=isnull(PO.TechnicalGradeNo,0)                
							   Left Join masterdata.mMakeMaster MM on MM.MakeNo=PO.MakeNo
							   Inner Join masterdata.mVendorLocationDetail MVC on MVC.VendorLocationNo = PM.VendorLocationNo			                               
							   Left Join (
										  SELECT PONo,ItemNo,MakeNo,Sum(TotalAmount) as Discount
										  from Purchase.POItemTaxDetail POITD
										  INNER JOIN  masterdata.mMiscCharges MC ON MC.MiscChargeNo=POITD.MiscChargeNo 
										  Inner Join masterdata.mTaxGroup TXG On TXG.TaxGroupNo=MC.TaxGroupNo
										  AND TXG.TaxCategoryNo = 11 ---for discount
										  Group By PONo,ItemNo,MakeNo --can be repeated for diff technical grade
										 )ttPOITD ON isnull(ttPOITD.PONo,0)=isnull(PO.PONo,0)  
	                                                AND ttPOITD.ItemNo= PO.ItemNo
                                                    AND ISNULL(ttPOITD.MakeNo,0)= ISNULL(PO.MakeNo,0)                                                    
							   Where PM.DocumentDate<=@DocumentDate
							   And PO.ItemNo=@ItemNo
							   And PM.AuthorizedBy is not null
							   Group By MVC.VendorName,PM.DocumentNoYearly,PM.DocumentDate
							            ,MM.MakeName,PO.Rate,ttPOITD.Discount
							   Order by PM.DocumentDate desc
							   For xml path('tr')),'<td1>','<td style="text-align: center;">'),'<td2>','<td>')
												   ,'<td3>','<td>')
												   ,'<td4>','<td style="text-align: center;">')
										           ,'<td5>','<td>')
										 		   ,'<td6>','<td style="text-align: right;">')
										 		   ,'<td7>','<td style="text-align: right;">')
										 		   ,'<td8>','<td style="text-align: right;">'))
				   
	set @str=@str+'</table>'
	
	select @str

END
