CREATE PROCEDURE [Information].[RejectionRatioForVendor] 
	-- Add the parameters for the stored procedure here
	@CompanyNo tinyint,
	@DocumentDate smalldatetime,
	@VendorLocationNo int,
	@YearNo tinyint
AS
BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;

    -- Insert statements for procedure here

	declare @str varchar(max)
    select @str=masterdata.JSForInformation('PMPO')
    set @str=@str+masterdata.CSSForInformation('PMPO')
	set @str=@str+'<table class="altrowstable" id="alternatecolorPORejectionRatioInfo">'
	set @str=@str+'<Caption><Font color="Maroon"><h3 align="Left"><b>Rejection Ratio</b></h3></Font></Caption>'
    set @str=@str+'<tr><th>Sl. No.</th><th>Month</th><th>Total Rejection %</th></tr>'
	set @str=@str
				 +( replace(
				    replace(
					replace(( 
							
						   Select  ROW_NUMBER() over (Order by ML.SortingMonth) as td1
						          ,ML.MonthName td2
								  ,isnull(IM.td2,0) td3
						   From masterdata.mFYear YR
						   Inner Join masterdata.MonthList ML ON YR.YearNo=ML.YearNo
						   LEFT Join
						   ( 
							
							Select MONTH(GRN.DocumentDate) [Month]
								  ,GRN.YearNo
								  ,cast(((SUM(Isnull(GRI.RejectedQty,0))/SUM(GRI.ReceivedQty))*100) as decimal(5,2)) td2 
							From Inventory.GRNMain GRN
							Inner Join Inventory.GRNItemDetail GRI ON GRN.GRNNo=GRI.GRNNo
							Where GRN.YearNo=@YearNo
							And GRN.CompanyNo=@CompanyNo
							And GRN.DocumentDate<=@DocumentDate
							And GRN.VendorLocationNo=@VendorLocationNo and GRN.FormCode='IGRN'
							Group by MONTH(GRN.DocumentDate),GRN.YearNo
							)IM ON IM.Month=ML.MonthNo
								And YR.YearNo=IM.YearNo
							Where YR.YearNo=@YearNo
							Order by ML.SortingMonth	
							for xml path('tr'))
							,'<td1>','<td style="text-align: Center;">')
							,'<td2>','<td>')
						    ,'<td3>','<td style="text-align: right;">'))
								
				   
	set @str=@str+'</table>'
	
	select @str
END
