-- =============================================
-- Author:		Alok
-- Create date: 15 Dec 2011
-- Description:	Save QC Parameters
-- =============================================
CREATE PROCEDURE [masterdata].[MDQCSave]
		@QCParameterNo smallint
       ,@QCParamCode Char(4)
       ,@QCParamName varchar(30)
       ,@QCUnit smallint
       ,@StandardValue decimal(9,3)
       ,@StandardValue2 decimal(9,3)
       ,@ToleranceValue decimal(9,3)
       ,@ToleranceType tinyint
       ,@ToleranceUnit smallint
       ,@DeductionOn tinyint
       ,@FormulaType tinyint
       ,@DeductionRate tinyint
       ,@DeductionType tinyint
       ,@DoubleDeductionFrom decimal(9,3)
	   ,@QualityParameterNo int = null
       ,@LoginNo SmallInt
       ,@DocumentStatusNo tinyint = 30
	   ,@QCSNo tinyint=Null
	   ,@OutputStatus bit = Null output
	   ,@OutputRemarks varchar(100) = Null output
	   ,@DeductionCalculationOn tinyint = NULL
AS
BEGIN
	
	SET NOCOUNT ON;
	BEGIN TRY

	Declare @ColName varchar(30),
	@Query varchar(Max),@PrevQCSNo tinyint

	Select @PrevQCSNo = QCSNo
	from masterdata.mQCParameters
	Where QCParameterNo = @QCParameterNo

		If Exists(
						Select 1
						from masterdata.mQCParameters
						Where QCParameterNo <> ISNULL(@QCParameterNo,0)
						And QCSNo = ISNULL(@QCSNo,0)
					)
		Begin
			RaisError('Two QC can not have same QC Para. Sequence.',16,-1)
		End

		Select @ColName = Case When LEN(@PrevQCSNo) >= 2 Then 'QC' + Convert(varchar(10),@PrevQCSNo) + 'StandardValue' 
		Else 'QC0' + Convert(varchar(10),@PrevQCSNo) + 'StandardValue' End 

		If @QCSNo <> @PrevQCSNo
		Begin
			Set @Query =' 
			If Exists (
							Select 1
							from Inventory.InvLotSummaryQCDetail
							Where ' + @ColName + ' Is Not Null
						)
			Begin
				RaisError(''Lot exists against this QC Para. Sequence No.'',16,-1)
			End'


			set @Query = REPLACE(@Query,'@ColName',@ColName)
			exec (@Query)
		End

		If Isnull(@QCParameterNo,0)=0 
			BEGIN
				Declare @Code varchar(6)
				
				SELECT @Code='QP'+ RIGHT(REPLICATE('0', 4) +cast(isnull(max(Convert(int,substring(Code,3,4))),0) + 1 as varchar(4)),4)  
			FROM [masterdata].[mQCParameters]
				
				INSERT INTO [masterdata].[mQCParameters]
					   ([Code]
					   ,[QCParameterName]
					   ,[UnitNo]
					   ,[StandardValue]
					   ,[StandardValue2]
					   ,[ToleranceValue]
					   ,[ToleranceTypeNo]
					   ,[ToleranceUnit]
					   ,[DeductionOn]
					   ,[FormulaType]
					   ,[DeductionRate]
					   ,[DeductionType]
					   ,[DoubleDeductionFrom]
					   ,[CreatedBy]
					   ,[CreatedDate]
					   ,AuthorizedBy 
					   ,AuthorizedDate 
					   ,DocumentStatusNo
					   ,QualityParameterNo
					   ,QCSNo
					   ,DeductionCalculationOn)
				values (@Code	
					   ,@QCParamName
					   ,@QCUnit 
					   ,@StandardValue
					   ,@StandardValue2
					   ,@ToleranceValue
					   ,@ToleranceType
					   ,@ToleranceUnit
					   ,@DeductionOn
					   ,@FormulaType
					   ,@DeductionRate
					   ,@DeductionType
					   ,@DoubleDeductionFrom
					   ,@LoginNo
					   ,Getdate()
					   ,case when @DocumentStatusNo=30 then @LoginNo else null end
					   ,case when @DocumentStatusNo=30 then GETDATE ()else null end
					   ,@DocumentStatusNo
					   ,@QualityParameterNo
					   ,@QCSNo
					   ,@DeductionCalculationOn)
			
			Select @QCParameterNo= ident_current('[masterdata].[mQCParameters]')

			insert into [masterdata].mQCParametersCompanyDetail (QCParameterNo ,CompanyNo , Inactive )
				select @QCParameterNo,companyno ,0
					from masterdata.mCompanyMaster 
					
			END
		ELSE
			BEGIN
				
				--New Code
				 
				UPDATE [masterdata].[mQCParameters]
				SET    [QCParameterName] = @QCParamName
					  ,[UnitNo] = @QCUnit
					  ,[StandardValue] = @StandardValue
					  ,[StandardValue2]=@StandardValue2
					  ,[ToleranceValue] = @ToleranceValue
					  ,[ToleranceTypeNo]=@ToleranceType
					  ,[ToleranceUnit] = @ToleranceUnit
					  ,[DeductionOn] = @DeductionOn
					  ,[FormulaType] = @FormulaType
					  ,[DeductionRate] = @DeductionRate
					  ,[DeductionType] = @DeductionType
					  ,[DoubleDeductionFrom] = @DoubleDeductionFrom
					  ,[ModifiedBy] = @LoginNo
					  ,[ModifiedDate] = GetDate()
					  ,AuthorizedBy = case when @DocumentStatusNo=30 then @LoginNo else null end
					  ,AuthorizedDate = case when @DocumentStatusNo=30 then GETDATE ()else null end
					  ,DocumentStatusNo= @DocumentStatusNo
					  ,QualityParameterNo = @QualityParameterNo
					  ,QCSNo = @QCSNo
					  ,DeductionCalculationOn = @DeductionCalculationOn 
				where QCParameterNo = @QCParameterNo 	
				  
			END
	
			Select 1,@QCParameterNo
			Select @OutputStatus = 1,@OutputRemarks = 'Success'
		
	END TRY
	BEGIN CATCH
			Select 0,Error_message()
			Select @OutputStatus = 0,@OutputRemarks = Error_message()
	END CATCH

    -- Insert statements for procedure here
	SELECT 1
END
