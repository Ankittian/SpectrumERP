CREATE PROCEDURE [Production].[PMLEFinancePosting]
	@OtherDataInXml xml
	,@VendorLocationNo int
	,@BasicAmount Decimal(18,6)
	,@VoucherDate smalldatetime
	,@MiscChargeNos xml
	,@NetAmount Decimal(18,6)
	,@CompanyNo tinyint = null
	,@ItemNos varchar(max)
	,@ServiceTaxAbatementRuleNo int = null
	,@FBNo smallint = null
AS

BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;
begin try	
		declare @ItemTable table (ItemNo int,BasicAmount money,Id int,Qty decimal(12,3),Rate money)
		insert into @ItemTable
		select Col1, sum(convert(money,Col2)), Col3, sum(CONVERT(decimal(12,3),Col4)), sum(CONVERT(money,Col5))
		from masterdata.Split_RowColumn(@ItemNos,'`','~')
		group by Col1, Col3	

		create table #Posting ([Add to Basic] bit,Particular varchar(500),[Debit Amount] decimal(15,2) , 
		[Credit Amount] decimal(15,2),[Account Name] int, [Ref. Detail] varchar(50),[Cost Center] varchar(50),
		[Amount] decimal(15,2), [Type] tinyint,SlNo tinyint ,MiscChargeNo int , TaxCodeMiscChargesDetailNo int,
			ItemNos varchar(5000))


		 Insert into #Posting 	
			select cast(0 as bit) [Add to Basic] ,IM.ItemDescription+' Stock A/C' Particular 
			,CONVERT(Decimal(15,2),sum(I.BasicAmount)) [Debit Amount] ,CAST(null as Decimal(15,2)) [Credit Amount] 
				,IR.AccountNo [Account Name] 
				,cast(null as varchar) [Ref. Detail] ,cast(null as varchar) [Cost Center] 
				,sum(I.BasicAmount) Amount ,1 [Type] 
				, row_number() over (Order by IR.accountno)  + (select isnull(max(SLNo),0) from #Posting) SlNo 
				,CAST(null as smallint) MiscChargeNo,null TaxCodeMiscChargesDetailNo,CAST(null as varchar(5000))ItemNos
			from @ItemTable I
				inner join masterdata.mItemMaster IM on IM.ItemNo=I.ItemNo
				left join (select ItemNo,IR.AccountNo--,AM.AccountName
							from Finance.ARDItemRule IR
							--inner join masterdata.mAccountMaster AM on AM.AccountNo=IR.AccountNo
							where IR.AccountTypeNo=14) IR on IR.ItemNo=I.ItemNo
			group by IM.ItemDescription+' Stock A/C',IR.AccountNo,isnull(cast(IR.AccountNo as int),I.ItemNo)
			union all
			select cast(0 as bit) [Add to Basic] ,IM.ItemDescription+' Consumption A/C' Particular 
				,CAST(null as Decimal(15,2)) [Debit Amount] ,CONVERT(Decimal(15,2),sum(v.value('Amount[1]','money'))) [Credit Amount] 
				,IR.AccountNo [Account Name] 
				,cast(null as varchar) [Ref. Detail] ,cast(null as varchar) [Cost Center] 
				,sum(v.value('Amount[1]','money')) Amount ,2 [Type] 
				, row_number() over (Order by IR.accountno)  + (select isnull(max(SLNo),0) from #Posting) SlNo 
				,CAST(null as smallint) MiscChargeNo,null TaxCodeMiscChargesDetailNo,CAST(null as varchar(5000))ItemNos
			FROM @OtherDataInXml.nodes('/DocumentElement/Table') x(v)
				inner join masterdata.mItemMaster IM on IM.ItemNo=v.value('Particular[1]','int')
				left join (select ItemNo,IR.AccountNo--,AM.AccountName
							from Finance.ARDItemRule IR
							--inner join masterdata.mAccountMaster AM on AM.AccountNo=IR.AccountNo
							where IR.AccountTypeNo=15) IR on IR.ItemNo=IM.ItemNo
			where v.value('Flag[1]','tinyint')=1																
			group by IM.ItemDescription+' Consumption A/C',IR.AccountNo,isnull(cast(IR.AccountNo as int),IM.ItemNo)


  if exists(select 1 from #Posting having abs(SUM([Credit Amount])- SUM([Debit Amount]))<= .99)
 begin
		update PD set [Debit Amount] = [Debit Amount] + ISNULL(diff,0)
		from
		#Posting PD
		cross join (select SUM([Credit Amount])- SUM([Debit Amount]) Diff from #Posting) tt
		where PD.SlNo  = 1
	
	update PD set [Credit Amount] = [Credit Amount] + ISNULL(diff,0)
		from
		#Posting PD
		cross join (select SUM([Debit Amount])- SUM([Credit Amount]) Diff from #Posting) tt
		where PD.SlNo  = 1
	
	update #Posting set [Debit Amount]=abs([Credit Amount]) ,[Credit Amount]=null,Type =1 ,Amount=abs([Credit Amount])
			where SlNo = 1 and [Credit Amount]<0
	
	update #Posting set [Credit Amount]=abs([Debit Amount]) ,Type =2 ,Amount=abs([Debit Amount]),[Debit Amount]=null
			where SlNo = 1 and [Debit Amount]<0
						
 end
 
 
select * from #Posting  where [Debit Amount]>0 or [Credit Amount]>0



END TRY
	
	BEGIN CATCH
		select ERROR_MESSAGE()
		INSERT INTO [masterdata].[mErrorLog]
			   ([ErrorDate]
			   ,[ErrorNo]
			   ,[ErrorSeverity]
			   ,[ErrorState]
			   ,[ErrorProcedure]
			   ,[ErrorLine]
			   ,[ErrorDescription]
			   ,[UserName]
			   ,[FormCode])
		VALUES
			   (getdate()			   
			   ,ERROR_NUMBER()
			   ,ERROR_SEVERITY()
			   ,ERROR_STATE()
			   ,ERROR_PROCEDURE()
			   ,ERROR_LINE()
			   ,ERROR_MESSAGE()
			   ,SUSER_SNAME()
			   ,'PBPB')
	END CATCH
	
END
