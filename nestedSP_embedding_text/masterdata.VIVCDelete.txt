CREATE PROCEDURE [masterdata].[VIVCDelete]
	@VendorNo int
	,@ReturnStatus tinyint = null output
	,@ReturnRemarks varchar(max) = null output
as
BEGIN

Begin Try

	SET NOCOUNT ON;
	-- 16.5.19 case of rrierp as they delete vendorlocation which are linked in ARD
		if (Select COUNT(1) from Finance.ARDVendorLocationRule 
			Where ARDNo in(
							Select ARDNo  
							From Finance.ARDVendorLocationRule  
							Where VendorLocationNo in( Select VendorLocationNo
													   From Masterdata.mVendorLocationDetail 
													   Where VendorNo =@VendorNo
													   ))
							) >1
		BEGIN
			Select 0,'Unable to delete this entry,used in ARD'
			Return
		END	
			
	Begin Transaction

		-- Delete ProcureNetMapping
		Delete From ProcureNet.MapSupplierVendorLocation 
		Where VendorLocationNo in (select VendorLocationNo From Masterdata.mVendorLocationDetail Where VendorNo =@VendorNo)
		
		--- Delete Vendor Bank Detail
		Delete From Masterdata.mVendorLocationBankDetailCompanyDetail 
		where VendorBankNo in(select VendorBankNo from Masterdata.mVendorLocationBankDetail Where VendorNo =@VendorNo)
	
		Delete From Masterdata.mVendorLocationBankDetail Where VendorNo =@VendorNo

		--- Delete Vendor Contact Person
		Delete From Masterdata.mVendorLocationContactDetailCompanyDetail 
		where VendorContactNo in(select VendorContactNo from Masterdata.mVendorLocationContactDetail Where VendorNo =@VendorNo)
	
		Delete From Masterdata.mVendorLocationContactDetail Where VendorNo =@VendorNo

		-- Delete Vendor Location Information
		Delete From Masterdata.mVendorLocationInformation Where VendorNo =@VendorNo
			
		--- Delete Vendor Location	
		Delete From Masterdata.mVendorLocationCustomFieldDetail 
		Where VendorLocationNo in (select VendorLocationNo From Masterdata.mVendorLocationDetail Where VendorNo =@VendorNo)

        --- Delete Vendor Location from Vendor Group
        Delete from masterdata.mVendorGroupDetail 
        where VendorlocationNo in (Select VendorLocationNo from masterdata.mVendorLocationDetail where VendorNo = @VendorNo)  
		
		Delete From Masterdata.mVendorLocationDetailCompanyDetail  
		Where VendorLocationNo in (select VendorLocationNo From Masterdata.mVendorLocationDetail Where VendorNo =@VendorNo)
		
		delete from masterdata.mItemVendorPartDetail where VendorNo =@VendorNo
		
		Delete from masterdata.mVendorLocationDivisionDetail where VendorNo = @VendorNo 

		Delete From Masterdata.mVendorLocationDetail Where VendorNo =@VendorNo
		
		Delete from Masterdata.mVendorCardCompanyDetail Where VendorNo =@VendorNo		  
		
		--- Delete Vendor Code
		Delete from Masterdata.mVendorCard Where VendorNo =@VendorNo
		 
	Commit Transaction

select @ReturnStatus=1
		,@ReturnRemarks='Success' 

select @ReturnStatus,@ReturnRemarks

End Try
Begin Catch
	 Rollback Transaction	
	 select @ReturnStatus=0
		,@ReturnRemarks=Error_message()

	 select @ReturnStatus,@ReturnRemarks
End Catch
END
