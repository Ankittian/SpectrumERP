Stored procedure Purchase.SCSOAuthorize: To calculate Primary Output Item rate @time of Order. Called from Inventory.IMMIDelete, Notifications.TANRSaveUserApproval.
Writes: Notifications.TATAuthorizationDetail, Purchase.SCOrderAmendmentItemDetail, Purchase.SCOrderAmendmentMain, Purchase.SCOrderItemDetail, Purchase.SCOrderMain
Reads: DiscreteRM.MaterialIssueJobCardDetail, DiscreteRM.MaterialIssueLotDetail, DiscreteRM.MaterialIssueLotJobCardDetail, DiscreteRM.MaterialIssueMain, Inventory.InvLotSummary, Inventory.ItemStandardRateDetail, Inventory.MaterialIssueItemDetail, Inventory.MaterialIssueLotDetail, Inventory.MaterialIssueMain, Inventory.SCReqWorkOrderDetail, masterdata.ItemRateValuationDetail, masterdata.MAPValuationSetting, masterdata.MAPValuationSettingDetail, Notifications.MultiAuthorizationSetting, Notifications.TATAuthorizationDetail, Purchase.SCOrderAmendmentItemDetail, Purchase.SCOrderAmendmentMain, Purchase.SCOrderAmendmentReqDetail, Purchase.SCOrderDetail, Purchase.SCOrderItemDetail, Purchase.SCOrderReqDetail, Security.LoginRoleDetail, Security.LoginTempRoleDetail, Security.mRoleMasterFormDetail
Parameters: @SCOrderAmendmentNo int, @DocumentDate smalldatetime, @OutputStatusNo bit = null OUTPUT, @OutputStatusRemarks varchar(2000) = null OUTPUT
Calls procedures: Project.ProjectBudgetValidation, Purchase.SCSOSendAuthorizationMail

-- Description: To calculate Primary Output Item rate @time of Order

CREATE PROCEDURE [Purchase].[SCSOAuthorize]
(
	@SCOrderAmendmentNo  int,
	@DocumentDate smalldatetime,
	@OutputStatusNo bit = null output 
	,@OutputStatusRemarks varchar(2000) = null  output
)
AS

BEGIN
	SET NOCOUNT ON;
	
	BEGIN TRY
	 declare @MAPValuationNo smallint,@SCOrderNo int,@AuthorizedBy int,@CompanyNo tinyint ,@DivisionNo int ,@DocumentStatusNo tinyint
	 ,@AuthDetailNo int ,@CreatedBy smallint
	
		
	 select @SCOrderNo  = SCOrderNo  ,@AuthorizedBy = AuthorizedBy,@CompanyNo =CompanyNo  ,@DivisionNo = DivisionNo  ,@DocumentStatusNo = DocumentStatusNo
		,@CreatedBy = sam.CreatedBy  
		from Purchase.SCOrderAmendmentMain sam where SCOrderAmendmentNo = @SCOrderAmendmentNo 
	Declare  @UserCount int 
	Declare @NextUsers table(SNo smallint,AuthDetailNo int,LevelNo int,LoginNo int) 
	Declare @UserNO int,@LevelNo smallint,@UserCounter int =1
	 ---------- Multi Level Authorization Query
 	   If Exists (Select 1  From Notifications.MultiAuthorizationSetting where CompanyNo =@CompanyNo and ISNULL(DivisionNo,0) = ISNULL(@DivisionNo,0)
 					and FormCode ='SCSA') and @DocumentStatusNo = 20
 		BEGIN
 		
 			 if Exists (Select 1 From Notifications.TATAuthorizationDetail where FormCode ='SCSA' and InternalNo = @SCOrderAmendmentNo 
 							and IsReady = 1 and IsEmailSent=0)
				BEGIN
						-- Send Email
					
					Insert into @NextUsers
					select ROW_NUMBER () over (order by LoginNo) SlNo,AuthDetailNo, LevelNo, LoginNo  from (
					Select LevelNo,LoginNo,AuthDetailNo
					 From Notifications.TATAuthorizationDetail where FormCode ='SCSA' and InternalNo = @SCOrderAmendmentNo 
						and IsReady = 1 and IsEmailSent =0	and LoginNo is not null
					union 
					Select LevelNo,lrd.LoginNo,AuthDetailNo
					 From Notifications.TATAuthorizationDetail tad
					 inner join Security.mRoleMasterFormDetail rfd on rfd.RoleMasterNo = tad.RoleNo and rfd.FormNo in( 188,199)
					 inner join Security.LoginRoleDetail lrd on rfd.RoleMasterNo = lrd.RoleMasterNo 
					 where FormCode ='SCSA' and InternalNo = @SCOrderAmendmentNo 
						and IsReady = 1 and IsEmailSent =0	and tad.RoleNo  is not null	
					union 
					Select LevelNo,lrd.LoginNo,AuthDetailNo
					 From Notifications.TATAuthorizationDetail tad
					 inner join Security.mRoleMasterFormDetail rfd on rfd.RoleMasterNo = tad.RoleNo and rfd.FormNo in( 188,199)
					 inner join Security.LoginTempRoleDetail lrd on rfd.RoleMasterNo = lrd.RoleMasterNo 
					 where FormCode ='SCSA' and InternalNo = @SCOrderAmendmentNo and CONVERT(smalldatetime,convert(varchar(15), GETDATE(),106))
						 between lrd.FromDate and ISNULL( lrd.ToDate,CONVERT(smalldatetime,convert(varchar(15), GETDATE(),106)))
						and IsReady = 1 and IsEmailSent =0	and tad.RoleNo  is not null	) tt
						group by LevelNo, LoginNo 	,AuthDetailNo							
					
					
					Set @UserCount = @@ROWCOUNT	
					
					While @UserCounter <= @UserCount
					 Begin
					
						 Select @UserNO=LoginNo , @LevelNo = LevelNo,@AuthDetailNo=AuthDetailNo From @NextUsers Where SNo = @UserCounter
								
						 Exec [Purchase].[SCSOSendAuthorizationMail] @SCOrderAmendmentNo,@UserNo,@LevelNo ,@AuthDetailNo
						 
						Update Notifications.TATAuthorizationDetail  set IsEmailSent =1 where FormCode = 'SCSA' and InternalNo = @SCOrderAmendmentNo 
								and LevelNo = @LevelNo and IsReady =1
						 Set @UserCounter=@UserCounter+1 
					 End
		 
				END							   
				
				 
			 					  	
 		 END
 	  else If Exists (Select 1  From Notifications.MultiAuthorizationSetting where CompanyNo =@CompanyNo and ISNULL(DivisionNo,0) = ISNULL(@DivisionNo,0)
 					and FormCode ='SCSA') and Exists (Select 1 From Notifications.TATAuthorizationDetail where FormCode ='SCSA' 
 						and InternalNo = @SCOrderAmendmentNo and IsRejectedMailSend =1) -- rejection mail
 		BEGIN
 			 
 			 if Exists (Select 1 From Notifications.TATAuthorizationDetail where FormCode ='SCSA' and InternalNo = @SCOrderAmendmentNo 
 							and IsRejectedMailSend =1)
				BEGIN
						-- Send Email
					 
					Insert into @NextUsers
					select ROW_NUMBER () over (order by LoginNo) SlNo,AuthDetailNo, LevelNo, LoginNo  from (
					Select LevelNo,LoginNo,AuthDetailNo
					 From Notifications.TATAuthorizationDetail where FormCode ='SCSA' and InternalNo = @SCOrderAmendmentNo 
						and IsRejectedMailSend =1 and LoginNo is not null
					union 
					Select LevelNo,lrd.LoginNo,AuthDetailNo
					 From Notifications.TATAuthorizationDetail tad
					 inner join Security.mRoleMasterFormDetail rfd on rfd.RoleMasterNo = tad.RoleNo and rfd.FormNo in( 188,199)
					 inner join Security.LoginRoleDetail lrd on rfd.RoleMasterNo = lrd.RoleMasterNo 
					 where FormCode ='SCSA' and InternalNo = @SCOrderAmendmentNo 
						and IsRejectedMailSend =1 and tad.RoleNo  is not null	
					union 
					Select LevelNo,lrd.LoginNo,AuthDetailNo
					 From Notifications.TATAuthorizationDetail tad
					 inner join Security.mRoleMasterFormDetail rfd on rfd.RoleMasterNo = tad.RoleNo and rfd.FormNo in( 188,199)
					 inner join Security.LoginTempRoleDetail lrd on rfd.RoleMasterNo = lrd.RoleMasterNo 
					 where FormCode ='SCSA' and InternalNo = @SCOrderAmendmentNo and CONVERT(smalldatetime,convert(varchar(15), GETDATE(),106))
						 between lrd.FromDate and ISNULL( lrd.ToDate,CONVERT(smalldatetime,convert(varchar(15), GETDATE(),106)))
						and IsRejectedMailSend =1 and tad.RoleNo  is not null	) tt
						group by LevelNo, LoginNo 	,AuthDetailNo							
					
					
					Set @UserCount = @@ROWCOUNT	
					
					
					While @UserCounter <= @UserCount
					 Begin
					
						 Select @UserNO=LoginNo , @LevelNo = LevelNo,@AuthDetailNo=AuthDetailNo From @NextUsers Where SNo = @UserCounter
								
						 Exec [Purchase].[SCSOSendAuthorizationMail] @SCOrderAmendmentNo,@UserNo,@LevelNo ,@AuthDetailNo ,1
						 
						Update Notifications.TATAuthorizationDetail  set IsRejectedMailSend =0 where FormCode = 'SCSA' and InternalNo = @SCOrderAmendmentNo 
								and LevelNo = @LevelNo and IsRejectedMailSend =1
						 Set @UserCounter=@UserCounter+1 
					 End
				
				
				
				END							   
				
				 
			 					  	
 		 END 	 		
 		
 	 -- Sending mail if last level got approved
 	  If exists (	 Select 1 
 					 From Notifications.TATAuthorizationDetail TAT
 					 Inner join Purchase.SCOrderAmendmentMain SOAM on SOAM.SCOrderAmendmentNo = TAT.InternalNo
 					 where InternalNo = @SCOrderAmendmentNo and FormCode = 'SCSA' and SOAM.DocumentStatusNo = 30 -- Authorized
						   And 
						   TAT.LevelNo = (
							Select MAX(LevelNo)
							From Notifications.TATAuthorizationDetail
							Where InternalNo=@SCOrderAmendmentNo	and FormCode= 'SCSA'
						)  and TAT.StatusNo = 13 -- Approve
				 )
	  BEGIN 
		Exec [Purchase].[SCSOSendAuthorizationMail]  @SCOrderAmendmentNo = @SCOrderAmendmentNo,@LoginNo = 0 ,@LevelNo = 0 ,@AuthDetailNo =0 ,@IsRejected=0
	  END			
 		
	 update scm set IsProductionOrder=1  from  
		Purchase.SCOrderAmendmentMain scm  
		inner join Purchase.SCOrderAmendmentReqDetail sd on sd.SCOrderAmendmentNo = scm.SCOrderAmendmentNo   
		inner join Inventory.SCReqWorkOrderDetail wd on wd.SCReqNo = sd.SCReqNo and wd.WorkOrderNo  is not null  
		where   scm.SCOrderAmendmentNo = @SCOrderAmendmentNo 
	 
	  update scm set IsProductionOrder=1  from  
		Purchase.SCOrderMain scm  
		inner join Purchase.SCOrderReqDetail sd on sd.SCOrderNo = scm.SCOrderNo   
		inner join Inventory.SCReqWorkOrderDetail wd on wd.SCReqNo = sd.SCReqNo and wd.WorkOrderNo  is not null  
		where  scm.SCOrderNo = @SCOrderNo 
		
	---- Project Budget Validation
	BEGIN	
		Declare 	@OutputStatusNo1 tinyint = null ,@OutputStatusRemarks1 varchar(4000) = null 	
		exec [Project].[ProjectBudgetValidation] Null,@SCOrderAmendmentNo,null,@OutputStatusNo1 output,@OutputStatusRemarks1 output

		If @OutputStatusNo1=0
		BEGIN
			Raiserror(@OutputStatusRemarks1,16,-1) 	
		END		
	END
					
		
	if @AuthorizedBy is null
	begin
		select 1,'Authorized'
		return
	end
		
	select @MAPValuationNo= MAS.MAPValuationNo  from Purchase.SCOrderAmendmentMain sam 
	inner join  masterdata.MAPValuationSetting MAS on MAS.CompanyNo = sam.CompanyNo 
		inner join masterdata.MAPValuationSettingDetail MASD on MASD.MAPValuationNo = MAS.MAPValuationNo 
		where 
		MAS.CompanyNo = sam.CompanyNo 
		and (MASD.DivisionNo is null or MASD.DivisionNo =sam.DivisionNo )
		and MASD.WarehouseNo = sam.WarehouseNo
		and sam.SCOrderAmendmentNo = @SCOrderAmendmentNo 
	
	Declare @iRow as int,@iCount int,@BetSeqNo tinyint, @ServiceItemNo int, @ActivityNo smallint, 
			@SequenceNo tinyint,@itemFlowNo tinyint,@ItemNo int,@MakeNo smallint,@TechnicalGradeNo tinyint,
			@ActualRate decimal(18,6) ,@ExpenseHeadNo smallint ,@SCValuationMethodNo tinyint,
			@Rate decimal(18,6),@Amount money 
	Declare @Activity as table (IDNo smallint,ServiceItemNo int,ActivityNo smallint,SequenceNo tinyint
					,Qty decimal(12,3),ItemFlowNo tinyint,ItemNo int,MakeNo smallint,TechnicalGradeNo tinyint
					,ExpenseHeadNo smallint , SCValuationMethodNo tinyint)
	
	
	insert into @Activity 
	select ROW_NUMBER () over(order by ItemFlowNo asc,SequenceNo,ServiceItemNo), sid.ServiceItemNo ,ActivityNo 
		,SequenceNo ,Quantity  ,sid.ItemFlowNo ,sid.ItemNo ,sid.MakeNo ,sid.TechnicalGradeNo,sid.ExpenseHeadNo 
			,sam.SCValuationMethodNo 
	from 
	Purchase.SCOrderAmendmentMain sam
	inner join Purchase.SCOrderAmendmentItemDetail sid on sid.SCOrderAmendmentNo = sam.SCOrderAmendmentNo 
	
	where 
	--sam.SCTypeNo <>1 -- exclude service  commented on 13 05 2013
	sam.SCOrderAmendmentNo = @SCOrderAmendmentNo 
	--and ItemFlowNo <>1
	and ItemFlowNo <> 5
	order by ItemFlowNo 
	 
	set @iRow = @@ROWCOUNT 
	set @iCount =0
	 
	while @iRow >@iCount 
	begin
		select @ServiceItemNo= ServiceItemNo ,@ActivityNo = ActivityNo ,@SequenceNo = isnull(SequenceNo,0) ,
				@itemFlowNo =ItemFlowNo ,@ItemNo = ItemNo ,@MakeNo = MakeNo ,@TechnicalGradeNo = TechnicalGradeNo 
				,@ExpenseHeadNo = ExpenseHeadNo ,@SCValuationMethodNo= SCValuationMethodNo
		  from @Activity where IDNo = @iCount + 1
	
	if @BetSeqNo is null
	 set @BetSeqNo=0
	
	set @ActualRate=0		
	if @SCValuationMethodNo =2 and  @itemFlowNo in(3,4) -- standard Rate
	begin
		select @Rate =Rate  from Inventory.ItemStandardRateDetail where StardardRateNo =
			(select max(StardardRateNo) from
		Inventory.ItemStandardRateDetail where CompanyNo = @CompanyNo 
				and ItemNo = @ItemNo and ISNULL(MakeNo ,0) = ISNULL(@MakeNo,0) 
				and ISNULL(TechnicalGradeNo ,0) = ISNULL(@TechnicalGradeNo,0)
				and EffectiveFrom = (select max(EffectiveFrom) from Inventory.ItemStandardRateDetail 
						where CompanyNo =@CompanyNo and EffectiveFrom <= @DocumentDate  
				and ItemNo = @ItemNo and ISNULL(MakeNo ,0) = ISNULL(@MakeNo,0) 
				and ISNULL(TechnicalGradeNo ,0) = ISNULL(@TechnicalGradeNo,0)
				))
		if 	@Rate is null
			raiserror('No standard Rate is Define for selected item.',16,-1)	
	end	
	
		 
		 if @itemFlowNo not in(3,4)
		 begin
			 if exists(select 1 from Inventory.MaterialIssueMain IM
				inner join Purchase.SCOrderAmendmentMain sam on sam.SCOrderAmendmentNo = IM.RefDocumentNo
				INNER join Inventory.MaterialIssueItemDetail mid on mid.IssueNo = IM.IssueNo 
				where IssueSourceNo =3 and sam.SCOrderNo = @SCOrderNo 
				and mid.ItemNo = @ItemNo and ISNULL(mid.MakeNo ,0) = ISNULL(@MakeNo,0)
				 and isnull(mid.TechnicalGradeNo,0) = isnull(@TechnicalGradeNo,0)
				 
					)
			begin
				 
				select @ActualRate = round(sum(mild.IssuedQty * ls.VRate)  /SUM(mild.IssuedQty) ,2)
				from Inventory.MaterialIssueMain IM
				inner join Purchase.SCOrderAmendmentMain sam on sam.SCOrderAmendmentNo = IM.RefDocumentNo 
				inner join Purchase.SCOrderItemDetail soid on soid.SCOrderNo = sam.SCOrderNo 
				inner join Inventory.MaterialIssueItemDetail misd on misd.IssueNo = IM.IssueNo 
				inner join Inventory.MaterialIssueLotDetail  mild on mild.IssueItemDetailNo = misd.IssueItemDetailNo 
				inner join Inventory.InvLotSummary ls on ls.InvLotSummaryNo = mild.InvLotSummaryNo 

				where IssueSourceNo =3 and sam.SCOrderNo = @SCOrderNo 
					and misd.ItemNo = soid.ItemNo and isnull(misd.MakeNo,0) = isnull(soid.MakeNo,0) 
						and isnull(misd.TechnicalGradeNo,0) = isnull(soid.TechnicalGradeNo ,0)
					 and Coalesce(	soid.ExpenseHeadNo, soid.ActivityNo,soid.ServiceItemNo ,0) =
						coalesce(@ExpenseHeadNo, @ActivityNo,@ServiceItemNo ,0)
						and ISNULL(soid.SequenceNo,0) between @BetSeqNo and @SequenceNo
						and misd.ItemNo = @ItemNo and isnull(misd.MakeNo,0) = ISNULL(@MakeNo,0) 
						and isnull(misd.TechnicalGradeNo,0) = isnull(@TechnicalGradeNo,0)
						and ItemFlowNo <>2
				 
			end	
			else if exists(select 1 from DiscreteRM.MaterialIssueMain IM
				inner join DiscreteRM.MaterialIssueJobCardDetail mijd on mijd.IssueNo = IM.IssueNo 
				inner join Purchase.SCOrderAmendmentMain sam on sam.SCOrderAmendmentNo = mijd.SCOrderAmendmentNo 
				INNER join DiscreteRM.MaterialIssueLotDetail mid on mid.IssueNo = IM.IssueNo 
				where sam.SCOrderNo = @SCOrderNo 
				and im.ItemNo = @ItemNo and (im.MakeNo = @MakeNo or im.makeno is null)
					)
			begin
				 
				select @ActualRate = round(sum(misd.IssuedWt * ls.VRate)  /SUM(misd.IssuedWt) ,2)
				from DiscreteRM.MaterialIssueMain IM
				inner join DiscreteRM.MaterialIssueLotJobCardDetail misd on misd.IssueNo = IM.IssueNo 
				inner join Purchase.SCOrderAmendmentMain sam on sam.SCOrderAmendmentNo = misd.SCOrderAmendmentNo 
				inner join Purchase.SCOrderItemDetail soid on soid.SCOrderNo = sam.SCOrderNo 
				inner join Inventory.InvLotSummary ls on ls.InvLotSummaryNo= misd.InvLotSummaryNo

				where sam.SCOrderNo = @SCOrderNo 
					and IM.ItemNo = soid.ItemNo and (IM.MakeNo = soid.MakeNo or im.makeno is null)
					 and Coalesce(	soid.ExpenseHeadNo, soid.ActivityNo,soid.ServiceItemNo ,0) =
						coalesce(@ExpenseHeadNo, @ActivityNo,@ServiceItemNo ,0)
						and ISNULL(soid.SequenceNo,0) between @BetSeqNo and @SequenceNo
						and IM.ItemNo = @ItemNo and isnull(IM.MakeNo,0) = ISNULL(@MakeNo,0) 
						and ItemFlowNo <>2
				 
			end		
			 else	
			 begin	

				 select @ActualRate  = sum(sid.Quantity * ird.Rate )/SUM(sid.Quantity)  from 
				 Purchase.SCOrderItemDetail sid
				 inner join Purchase.SCOrderDetail sd on sd.SCOrderNo = sid.SCOrderNo 
					and coalesce(sd.ExpenseHeadNo, sd.ActivityNo,sd.ItemNo ,0) =
						coalesce(sid.ExpenseHeadNo, sid.ActivityNo,sid.ServiceItemNo ,0)
							and isnull(sid.SequenceNo,0) = isnull(sd.SequenceNo,0)
				 inner join masterdata.ItemRateValuationDetail ird on ird.ItemNo = sid.ItemNo 
						and ISNULL(sid.MakeNo,0)= ISNULL(ird.MakeNo,0) 
						and ISNULL(sid.TechnicalGradeNo ,0)= ISNULL(ird.TechnicalGradeNo ,0) 
				 where sid.SCOrderNo = @SCOrderNo  
				 and ird.MAPValuationNo = @MAPValuationNo and PriceControlNo = 2 -- MAP
				  and Coalesce(	sd.ExpenseHeadNo, sd.ActivityNo,sd.ItemNo ,0) =
						coalesce(@ExpenseHeadNo, @ActivityNo,@ServiceItemNo ,0)
						and ISNULL(sd.SequenceNo,0) between @BetSeqNo and @SequenceNo
				  and sid.ItemNo = @ItemNo 
				  and isnull(sid.MakeNo,0) = ISNULL(@MakeNo,0) 
				  and isnull(sid.TechnicalGradeNo,0) = isnull(@TechnicalGradeNo,0)
				  and sid.ItemFlowNo <>2
			 
			end
			 
			 update Purchase.SCOrderAmendmentItemDetail set 
					ActualRate = @ActualRate
					,Rate = @ActualRate
			  where SCOrderAmendmentNo = @SCOrderAmendmentNo
			   and Coalesce(ExpenseHeadNo, ActivityNo,ServiceItemNo ,0) =
						coalesce(@ExpenseHeadNo, @ActivityNo,@ServiceItemNo ,0)
					and isnull(SequenceNo,0) = isnull(@SequenceNo,0)
			  and ItemFlowNo =@ItemFlowNo and ItemFlowNo <>5
			  and ItemNo = @ItemNo 
			  and isnull(MakeNo,0) = ISNULL(@MakeNo,0) 
			  and isnull(TechnicalGradeNo,0) = isnull(@TechnicalGradeNo,0)
			  
			 update Purchase.SCOrderItemDetail set 
				 ActualRate =  @ActualRate 
				 ,Rate = @ActualRate
			  where SCOrderNo = @SCOrderNo  and ItemFlowNo <>5
			   and Coalesce(ExpenseHeadNo, ActivityNo,ServiceItemNo ,0) =
						coalesce(@ExpenseHeadNo, @ActivityNo,@ServiceItemNo ,0)
					and isnull(SequenceNo,0) = isnull(@SequenceNo,0)
			  and ItemFlowNo =@ItemFlowNo
			  and ItemNo = @ItemNo 
			  and isnull(MakeNo,0) = ISNULL(@MakeNo,0) 
			  and isnull(TechnicalGradeNo,0) = isnull(@TechnicalGradeNo,0)
				  
			 end
		 
		 else
		 begin
			
			
			
			
			select @Amount= sum(Quantity * ActualRate) from Purchase.SCOrderAmendmentItemDetail
			where SCOrderAmendmentNo = @SCOrderAmendmentNo
			  and coalesce (ActivityNo,ExpenseHeadNo,ServiceItemNo,0)=
				coalesce (@ActivityNo,@ExpenseHeadNo,@ServiceItemNo,0) 
				and ISNULL(SequenceNo,0) = ISNULL(@SequenceNo,0)
			     and ItemFlowNo = 1
			
			 
			select @Amount=@Amount -ISNULL(sum(Quantity * ActualRate),0) 
			from Purchase.SCOrderAmendmentItemDetail
			where SCOrderAmendmentNo = @SCOrderAmendmentNo
			 and Coalesce(ExpenseHeadNo, ActivityNo,ServiceItemNo ,0) =
						coalesce(@ExpenseHeadNo, @ActivityNo,@ServiceItemNo ,0)
					and isnull(SequenceNo,0) = isnull(@SequenceNo,0)
			 and ItemFlowNo = 2  -- returnable ISsue
			  
			 
			  update Purchase.SCOrderAmendmentItemDetail set 
			  Rate =round(case @SCValuationMethodNo when  1 then Rate when 2 then @Rate else  
					(@Amount * isnull(BifercationPercentage*0.01,1))/ Quantity end,2) ,
			  ActualRate = round((@Amount * isnull(BifercationPercentage*0.01,1))/ Quantity ,2)
			  where SCOrderAmendmentNo = @SCOrderAmendmentNo
			  and coalesce(ActivityNo,ExpenseHeadNo,ServiceItemNo,0)=
						coalesce(@ActivityNo,@ExpenseHeadNo,@ServiceItemNo,0)
				and isnull(	SequenceNo,0) = isnull(	@SequenceNo,0)
			  and ItemNo = @ItemNo 
			  and isnull(MakeNo,0) = isnull(@MakeNo ,0)
			  and isnull(TechnicalGradeNo,0) = isnull(@TechnicalGradeNo ,0)
			  and ItemFlowNo =@itemFlowNo   and ItemFlowNo <>5
			
			 update Purchase.SCOrderItemDetail set
			  Rate = round(case @SCValuationMethodNo when  1 then Rate when 2 then @Rate else  
					(@Amount * isnull(BifercationPercentage*0.01,1))/ Quantity end,2) ,
			  ActualRate = round((@Amount * isnull(BifercationPercentage*0.01,1))/ Quantity ,2)
			  where SCOrderNo = @SCOrderNo
			  and coalesce(ActivityNo,ExpenseHeadNo,ServiceItemNo,0)=
						coalesce(@ActivityNo,@ExpenseHeadNo,@ServiceItemNo,0)
				and isnull(	SequenceNo,0) = isnull(	@SequenceNo,0)
			  and ItemNo = @ItemNo 
			  and isnull(MakeNo,0) = isnull(@MakeNo ,0)
			  and isnull(TechnicalGradeNo,0) = isnull(@TechnicalGradeNo ,0)
			  and ItemFlowNo =@itemFlowNo  and ItemFlowNo <>5
		  
		 end
		  
	 
	 			
		  select @BetSeqNo = SequenceNo + 1
		  from @Activity where IDNo = @iCount + 1
		  
		set @iCount = @iCount + 1
	end ---while @iRow >@iCount 
		
		

		select @OutputStatusNo =1
 select 1,'Sucess'
	
	END TRY
	BEGIN CATCH
		Select 0,Error_Message()
		select @OutputStatusNo = 0 , @OutputStatusRemarks = error_message()
	END CATCH
END
