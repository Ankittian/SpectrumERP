Stored procedure Project.ProjectBudgetValidation: Validate for budget amount of Project. Called from Purchase.SCSOAuthorize, Purchase.PMPOAuthorize.
Reads: Inventory.GRNItemDetail, Inventory.GRNItemIndentDetail, Inventory.IndentMain, Inventory.SCReqMain, Project.BudgetRevisionMain, Project.BudgetRevisionWBSDetail, Project.MaterialPlanningWBSDetail, Project.PrjWBSDetail, Project.PrjWBSMain, Project.PrjWBSReportingResourceDetail, Project.PrjWBSRevisionDetail, Project.PrjWBSRevisionMain, Project.ProjectMain, Project.WBSMain, Purchase.ChangePOStatusItemDetails, Purchase.ChangePOStatusMain, Purchase.POAmendmentIndentDetail, Purchase.POAmendmentItemDetail, Purchase.POAmendmentMain, Purchase.SCOrderAmendmentMain, Purchase.SCOrderAmendmentReqDetail
Parameters: @POAmendmentNo int, @SCOrderAmendmentNo int, @WBSReportingNo Int, @OutputStatusNo tinyint = null OUTPUT, @OutputStatusRemarks varchar(4000) = null OUTPUT

-- =============================================

-- Create date: 05 July 2022
-- Description:	Validate for budget amount of Project
-- =============================================
CREATE PROCEDURE [Project].[ProjectBudgetValidation]
	@POAmendmentNo int,
	@SCOrderAmendmentNo int,
	@WBSReportingNo Int,
	@OutputStatusNo tinyint = null output,
	@OutputStatusRemarks varchar(4000) = null output
AS
BEGIN 		
SET NOCOUNT ON;
Begin Try

Declare @WBSInputData as Table(WBSRevisionSerialNo Int,WBSNo int)
Declare @WBSList As Table (WBSSerialNo Int,WBSNo int,ProjectNo Int,BudgetAmount Money,ConsumedBudget Money,ChildConsumedBudget Money,LevelNo TinyInt)
Declare @BudgetUtilized As Table (WBSSerialNo Int,WBSNo int,Form Varchar(10),Amount Money,Id Int)

BEGIN
-- Input (PO)
	Insert into @WBSInputData
	Select	MPWD.WBSRevisionSerialNo,RM.WBSNo	
	From [Purchase].[POAmendmentMain] POAM
	Inner Join (    
				Select PM.PONo,MAX(PM.AmendmentNo) [AmendmentNo]
				From [Purchase].[POAmendmentMain] PM
				Group by PM.PONo 	
				) LSTP on LSTP.PONo = POAM.PONo	and LSTP.AmendmentNo = POAM.AmendmentNo
	Inner Join Purchase.POAmendmentIndentDetail  PIND on PIND.POAmendmentNo = POAM.POAmendmentNo
	Inner Join Purchase.POAmendmentItemDetail PAID on PAID.POAmendmentNo=PIND.POAmendmentNo and PAID.ItemNo=PIND.ItemNo
												  And IsNull(PAID.MakeNo,0)=IsNull(PIND.MakeNo,0)
												  And IsNull(PAID.TechnicalGradeNo,0)=IsNull(PIND.TechnicalGradeNo,0)
												  And PAID.StatusNo<>4 -- Don't load Cancel Line item, we will take it from GRN												  
	Inner Join Inventory.IndentMain INDM on INDM.IndentNo = PIND.IndentNo
	Inner Join Project.MaterialPlanningWBSDetail MPWD on MPWD.MPWBSDetailNo = INDM.RefDocumentNo 
														 and INDM.IndentAgainstNo = 5	 -- against material planning
	Inner Join Project.PrjWBSRevisionMain RM on RM.WBSRevisionNo=MPWD.WBSRevisionNo														 
	Where POAm.POAmendmentNo=@POAmendmentNo
	Group By MPWD.WBSRevisionSerialNo,RM.WBSNo
	
-- Input (SCSO)
	Insert into @WBSInputData
	Select	MPWD.WBSRevisionSerialNo,RM.WBSNo		
	From [Purchase].[SCOrderAmendmentMain] SOAM
	Inner Join (    Select PM.SCOrderNo,MAX(PM.AmendmentNo) [AmendmentNo]
					From [Purchase].[SCOrderAmendmentMain] PM
					Group by PM.SCOrderNo 	
				) LSTP on LSTP.SCOrderNo = SOAM.SCOrderNo	and LSTP.AmendmentNo = SOAM.AmendmentNo	
	Inner Join Purchase.SCOrderAmendmentReqDetail  SARD on SARD.SCOrderAmendmentNo = SOAM.SCOrderAmendmentNo														 
	Inner Join Inventory.SCReqMain SCRM on SCRM.SCReqNo = SARD.SCReqNo												 
	Inner Join Project.MaterialPlanningWBSDetail MPWD on MPWD.MPWBSDetailNo = SCRM.MPWBSDetailNo 
														  and SCRM.SourceDocumentNo = 3	---- against material planning
	Inner Join Project.PrjWBSRevisionDetail WRD on MPWD.WBSRevisionSerialNo = WRD.WBSRevisionSerialNo  
														and MPWD.WBSRevisionNo  = WRD.WBSRevisionNo 
														and WRD.WBSTypeNo = 4 --- Procurement
	Inner Join Project.PrjWBSRevisionMain RM on RM.WBSRevisionNo=MPWD.WBSRevisionNo		 								 
	Where SOAM.SCOrderAmendmentNo=@SCOrderAmendmentNo
	Group by MPWD.WBSRevisionSerialNo,RM.WBSNo		
	
-- Input (Reporting)	
	Insert into @WBSInputData	
	Select RD.WBSRevisionSerialNo,RM.WBSNo	
	From Project.PrjWBSReportingResourceDetail RD	
	Inner Join Project.PrjWBSRevisionMain RM on RD.WBSRevisionNo=RM.WBSRevisionNo
	Where RD.WBSReportingNo=@WBSReportingNo
	Group By RD.WBSRevisionSerialNo,RM.WBSNo
END


-- Get All current and parent WBS Id
;With CTE as
(
	Select WB.WBSId,WB.WBSSerialNo,WB.wbsNo,CONVERT(Bit,1) IsCurrent
	From Project.PrjWBSDetail WB
	Inner Join @WBSInputData WD on WD.WBSRevisionSerialNo=WB.WBSSerialNo and WD.WBSNo=WB.WBSNo
	Union All
	Select WB1.WBSId,WB1.WBSSerialNo,WB1.wbsNo,CONVERT(Bit,0) IsCurrent
	From Project.PrjWBSDetail WB	
	Inner Join CTE C on C.WBSSerialNo=WB.WBSSerialNo and C.WBSNo=WB.WBSNo
	Inner Join Project.PrjWBSDetail WB1 on WB.ParentWBSSerialNo=WB1.WBSSerialNo And WB.ParentWBSNo=WB1.WBSNo	
),
CTE1 as
(
	Select PWB.WBSId,PWB.WBSSerialNo,PWB.wbsNo,CONVERT(Bit,1) IsCurrent
	From Project.PrjWBSDetail WB
	Inner Join Project.PrjWBSDetail PWB on PWB.WBSNo=WB.WBSNo and PWB.WBSSerialNo=WB.ParentWBSSerialNo
	Inner Join @WBSInputData WD on WD.WBSRevisionSerialNo=WB.WBSSerialNo and WD.WBSNo=WB.WBSNo
	Union All
	Select WB1.WBSId,WB1.WBSSerialNo,WB1.wbsNo,CONVERT(Bit,1) IsCurrent
	From Project.PrjWBSDetail WB	
	Inner Join CTE1 C on C.WBSSerialNo=WB.WBSSerialNo and C.WBSNo=WB.WBSNo
	Inner Join Project.PrjWBSDetail WB1 on WB1.ParentWBSSerialNo=WB.WBSSerialNo And WB1.ParentWBSNo=WB.WBSNo	
)

-- Get Target Budget Amount
Insert into @WBSList(WBSSerialNo ,WBSNo ,ProjectNo ,BudgetAmount,LevelNo)
Select C.WBSSerialNo,C.WBSNo,WB.ProjectNo,BGT.TargetBudget
,(LEN(WD.WBSId) - Len(Replace(WD.WBSId,'.',''))+1)
From(

		Select C.WBSSerialNo,C.WBSNo
		From CTE C
		Group By C.WBSSerialNo,C.WBSNo
		Union
		Select C1.WBSSerialNo,C1.WBSNo
		From CTE1 C1
		Group By C1.WBSSerialNo,C1.WBSNo		
) C
Inner Join Project.PrjWBSDetail WD on WD.WBSSerialNo=C.WBSSerialNo And WD.WBSNo=C.WBSNo
Inner Join Project.WBSMain WB on WB.WBSNo=C.WBSNo
Left Join (
			Select RWD.WBSRevisionSerialNo,WRM.WBSNo,Max(RWD.TargetBudget) TargetBudget
			From Project.BudgetRevisionMain RM
			Inner Join Project.BudgetRevisionWBSDetail RWD on RWD.BudgetRevisionNo=RM.BudgetRevisionNo
			Inner Join Project.PrjWBSRevisionMain WRM on WRM.WBSRevisionNo=RWD.WBSRevisionNo
			Inner Join (
							Select RM.ProjectNo
							From @WBSInputData WID
							Inner Join Project.PrjWBSMain RM on RM.WBSNo=WID.WBSNo
							Group by RM.ProjectNo
						)W on W.ProjectNo=RM.ProjectNo
			Where RM.RevisionNo =(
									Select MAX(RM1.RevisionNo)
									From Project.BudgetRevisionMain RM1
									Where RM1.BudgetNo=RM.BudgetNo
									And RM1.AuthorizedBy is Not Null
								 )
			Group by RWD.WBSRevisionSerialNo,WRM.WBSNo	
			)BGT on BGT.WBSRevisionSerialNo=C.WBSSerialNo And BGT.WBSNo=C.WBSNo
Group by C.WBSSerialNo,C.WBSNo,WB.ProjectNo,BGT.TargetBudget,WD.WBSId


					
-- Get PO Details
-- Get SCO Details
-- Get Resource Detail


-- PO Cost

	Insert into @BudgetUtilized 
	Select	
	WL.WBSSerialNo,WL.WBSNo,'PMPO' Form	 
	,Sum(PIND.POQty*PIND.PORate) Amount	
	,POAM.PONo
	From [Purchase].[POAmendmentMain] POAM
	Inner Join (    
				Select PM.PONo,MAX(PM.AmendmentNo) [AmendmentNo]
				From [Purchase].[POAmendmentMain] PM
				Group by PM.PONo 	
				) LSTP on LSTP.PONo = POAM.PONo	and LSTP.AmendmentNo = POAM.AmendmentNo
	Inner Join Purchase.POAmendmentIndentDetail  PIND on PIND.POAmendmentNo = POAM.POAmendmentNo
	Inner Join Purchase.POAmendmentItemDetail PAID on PAID.POAmendmentNo=PIND.POAmendmentNo and PAID.ItemNo=PIND.ItemNo
												  And IsNull(PAID.MakeNo,0)=IsNull(PIND.MakeNo,0)
												  And IsNull(PAID.TechnicalGradeNo,0)=IsNull(PIND.TechnicalGradeNo,0)
												  And PAID.StatusNo<>4 -- Don't load Cancel Line item, we will take it from GRN												  
	Inner Join Inventory.IndentMain INDM on INDM.IndentNo = PIND.IndentNo
	Inner Join Project.MaterialPlanningWBSDetail MPWD on MPWD.MPWBSDetailNo = INDM.RefDocumentNo 
														 and INDM.IndentAgainstNo = 5	 -- against material planning
	Inner Join Project.PrjWBSRevisionMain RM on RM.WBSRevisionNo=MPWD.WBSRevisionNo														 
	Inner Join @WBSList WL on WL.WBSSerialNo=MPWD.WBSRevisionSerialNo And WL.WBSNo=RM.WBSNo	
	Group By POAM.PONo,WL.WBSSerialNo,WL.WBSNo	

-- PO Amount From GRN (For Cancel PO)
	Insert into @BudgetUtilized
	Select 	MPWD.WBSRevisionSerialNo ,RM.WBSNo,'IGRN' Form,Sum(GIID.AcceptedQty*PID.Rate) Amount,GID.GRNNo
	From Inventory.GRNItemDetail GID
	inner join Inventory.GRNItemIndentDetail GIID on GIID.GRNItemDetailNo = GID.GRNItemDetailNo 
	inner join Purchase.POAmendmentItemDetail PID on PID.POAmendmentNo = GID.POAmendmentNo 
												  And PID.ItemNo = GID.ItemNo 
												  And ISNULL(PID.MakeNo,0) = ISNULL(GID.MakeNo,0)
												  And ISNULL(GID.TechnicalGradeNo,0) = ISNULL(PID.TechnicalGradeNo ,0) 
												  And PID.StatusNo =4
	inner join Purchase.POAmendmentMain POM on POM.POAmendmentNo = PID.POAmendmentNo
	Inner Join Inventory.IndentMain INDM on INDM.IndentNo = GIID.IndentNo
	Inner Join Project.MaterialPlanningWBSDetail MPWD on MPWD.MPWBSDetailNo = INDM.RefDocumentNo
														 and INDM.IndentAgainstNo = 5	 -- against material planning
	Inner Join Project.PrjWBSRevisionMain RM on RM.WBSRevisionNo=MPWD.WBSRevisionNo		 								 
	Inner Join @WBSList WL on WL.WBSSerialNo=MPWD.WBSRevisionSerialNo And WL.WBSNo=RM.WBSNo
	Where not exists (
					Select 1 
					From Purchase.ChangePOStatusMain  a
					Inner join  Purchase.POAmendmentMain b on b.POAmendmentNo  = a.POAmendmentNo 
					Inner join Purchase.ChangePOStatusItemDetails  c on c.ChangePOStatusNo = a.ChangePOStatusNo
					Where b.PONo = pom.PONo and c.ItemNo = pid.ItemNo and c.IsReleaseIndent = 0
					)
	Group By MPWD.WBSRevisionSerialNo ,RM.WBSNo,GID.GRNNo
	
-- Service Order
	Insert into @BudgetUtilized
	Select	WL.WBSSerialNo,WL.WBSNo	,'SCSO' Form,Sum(SARD.Quantity * SARD.Rate) Amount,SOAM.SCOrderNo			
	From [Purchase].[SCOrderAmendmentMain] SOAM
	Inner Join (    Select PM.SCOrderNo,MAX(PM.AmendmentNo) [AmendmentNo]
					From [Purchase].[SCOrderAmendmentMain] PM
					Group by PM.SCOrderNo 	
				) LSTP on LSTP.SCOrderNo = SOAM.SCOrderNo	and LSTP.AmendmentNo = SOAM.AmendmentNo	
	Inner Join Purchase.SCOrderAmendmentReqDetail  SARD on SARD.SCOrderAmendmentNo = SOAM.SCOrderAmendmentNo														 
	Inner Join Inventory.SCReqMain SCRM on SCRM.SCReqNo = SARD.SCReqNo												 
	Inner Join Project.MaterialPlanningWBSDetail MPWD on MPWD.MPWBSDetailNo = SCRM.MPWBSDetailNo 
														  and SCRM.SourceDocumentNo = 3	---- against material planning
	Inner Join Project.PrjWBSRevisionDetail WRD on MPWD.WBSRevisionSerialNo = WRD.WBSRevisionSerialNo  
														and MPWD.WBSRevisionNo  = WRD.WBSRevisionNo 
														and WRD.WBSTypeNo = 4 --- Procurement
	Inner Join Project.PrjWBSRevisionMain RM on RM.WBSRevisionNo=MPWD.WBSRevisionNo		 								 
	Inner Join @WBSList WL on WL.WBSSerialNo=MPWD.WBSRevisionSerialNo And WL.WBSNo=RM.WBSNo
	Group by SOAM.SCOrderNo,WL.WBSSerialNo,WL.WBSNo	

-- Resource 
	Insert Into @BudgetUtilized
	Select WL.WBSSerialNo,WL.WBSNo,'RSRR',RD.Amount,RD.WBSReportingNo
	From Project.PrjWBSReportingResourceDetail RD	
	Inner Join Project.PrjWBSRevisionMain RM on RD.WBSRevisionNo=RM.WBSRevisionNo
	Inner Join @WBSList WL on WL.WBSSerialNo=RD.WBSRevisionSerialNo and WL.WBSNo=RM.WBSNo		
	
	Update WL
	Set WL.ConsumedBudget=WB.Amount
	From @WBSList WL
	Inner Join Project.PrjWBSDetail WD on WD.WBSSerialNo=WL.WBSSerialNo and WD.WBSNo=WL.WBSNo
	Left join (	
				Select  WBSSerialNo,WBSNo,Sum(Amount) Amount
				From @BudgetUtilized
				Group by WBSSerialNo,WBSNo
			  )	WB on WB.WBSSerialNo=WL.WBSSerialNo And WB.WBSNo=WL.WBSNo
	
	--Select WL.WBSNo,WL.WBSSerialNo,WD.WBSId,WL.BudgetAmount [Budget],WB.Amount [Utilized Budget]
	--From @WBSList WL
	--Inner Join Project.PrjWBSDetail WD on WD.WBSSerialNo=WL.WBSSerialNo and WD.WBSNo=WL.WBSNo
	--Left join (	
	--			Select  WBSSerialNo,WBSNo,Sum(Amount) Amount
	--			From @BudgetUtilized
	--			Group by WBSSerialNo,WBSNo
	--		  )	WB on WB.WBSSerialNo=WL.WBSSerialNo And WB.WBSNo=WL.WBSNo
	--Order by WBSNo,WD.SequenceNo		  
	
	
	Declare @MaxLevelNo TinyInt
	Select 	@MaxLevelNo=MAX(LevelNo) 
	From @WBSList
	
	While @MaxLevelNo>0
	BEGIN	
		Update WL
		set WL.ChildConsumedBudget=CHLD.ChildConsumedBudget
		From @WBSList WL
		Inner Join Project.PrjWBSDetail PWD on PWD.WBSSerialNo=WL.WBSSerialNo And PWD.WBSNo	=wl.WBSNo
		Outer Apply (
					 Select Sum(IsNull(WL1.ConsumedBudget,0)+ IsNull(WL1.ChildConsumedBudget,0)) ChildConsumedBudget
					 From @WBSList WL1 
					 Inner Join Project.PrjWBSDetail PWD1 on PWD1.WBSSerialNo=WL1.WBSSerialNo And PWD1.WBSNo =WL1.WBSNo
					 Where WL.ProjectNo=WL1.ProjectNo					 
					 And SUBSTRING(PWD1.WBSId,0,Len(PWD1.WBSId)- CHARINDEX('.',Reverse(PWD1.WBSId),0)+1)=PWD.WBSId
					 ) CHLD
		Where WL.LevelNo=@MaxLevelNo
		
		Select @MaxLevelNo=@MaxLevelNo-1
	END

	Declare @Message Varchar(1000)=''
	
	If Exists (
				Select 1
				From @WBSList WL
				Inner Join Project.PrjWBSDetail WD on WD.WBSSerialNo=WL.WBSSerialNo and WD.WBSNo=WL.WBSNo	
				Where WL.BudgetAmount< ISNULL(WL.ConsumedBudget,0)+ISNULL(WL.ChildConsumedBudget,0)			
			 )
	BEGIN
	--
		Select @Message = @Message  + ' ' + PM.DocumentNoYearly  + ' (' + Prj.WBSIds +  ')'		
		From @WBSInputData WID
		Inner Join Project.PrjWBSMain RM on RM.WBSNo=WID.WBSNo
		Inner Join Project.ProjectMain PM on PM.ProjectNo=RM.ProjectNo
		Cross Apply (
						Select STUFF((
									Select ',' +  WD.WBSId
									From @WBSList WL
									Inner Join Project.PrjWBSDetail WD on WD.WBSSerialNo=WL.WBSSerialNo and WD.WBSNo=WL.WBSNo	
									Inner Join Project.ProjectMain PM on PM.ProjectNo=WL.ProjectNo
									Where WL.BudgetAmount< ISNULL(WL.ConsumedBudget,0)+ISNULL(WL.ChildConsumedBudget,0)
									And WL.ProjectNo=PM.ProjectNo
									For Xml Path('')),1,1,'') WBSIds
					)Prj
		Group by PM.DocumentNoYearly,Prj.WBSIds
		
		Select @Message='Budget Amount should not exceeded for following project and WBS ID' +@Message
	END
	
	
	If Len(@Message)>0
	BEGIN
		Select @OutputStatusNo = 0, @OutputStatusRemarks = @Message
	END
	ELSe
	BEGIN
		Select @OutputStatusNo = 1, @OutputStatusRemarks = 'Success'
	END
	
	--Select WL.WBSNo,WL.WBSSerialNo,WD.WBSId
	--,WL.BudgetAmount [Budget]
	--,WL.ConsumedBudget
	--,WL.ChildConsumedBudget
	--,WL.LevelNo
	--From @WBSList WL
	--Inner Join Project.PrjWBSDetail WD on WD.WBSSerialNo=WL.WBSSerialNo and WD.WBSNo=WL.WBSNo	
	--Where WL.BudgetAmount< ISNULL(WL.ConsumedBudget,0)+ISNULL(WL.ChildConsumedBudget,0)
	--Order by WBSNo,WD.SequenceNo
	
	

End Try
Begin Catch
		select 0,ERROR_MESSAGE()
		INSERT INTO [masterdata].[mErrorLog]
				   ([ErrorDate]
				   ,[ErrorNo]
				   ,[ErrorSeverity]
				   ,[ErrorState]
				   ,[ErrorProcedure]
				   ,[ErrorLine]
				   ,[ErrorDescription]
				   ,[UserName]
				   ,[FormCode])
			 VALUES
				   (getdate()
				   ,ERROR_NUMBER()
				   ,ERROR_SEVERITY()
				   ,ERROR_STATE()
				   ,ERROR_PROCEDURE()
				   ,ERROR_LINE()
				   ,ERROR_MESSAGE()
				   ,SUSER_SNAME()
				   ,'BudgetValidation')
End Catch
    
END
