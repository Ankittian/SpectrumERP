Stored procedure Purchase.SCOStatus: To Change Status OF Sc Order Status. Called from Inventory.IGRNDelete.
Writes: Purchase.SCOrderAmendmentDetail, Purchase.SCOrderAmendmentMain, Purchase.SCOrderDetail, Purchase.SCOrderMain
Reads: Inventory.GRNSCOrderDetail, Purchase.SCOrderAmendmentMain, Purchase.SCOrderDetail, Purchase.SCOrderReqDetail
Parameters: @SCOrderNo int, @ItemNo int, @ActivityNo smallint, @SequenceNo tinyint, @StatusNo tinyint, @OutputStatus bit OUTPUT, @Result varchar(100) OUTPUT, @ExpenseHeadNo smallint = null
Calls procedures: Inventory.SCReqStatus
Author: Akhil, created 22 oct 2012
Code:
-- =============================================
-- Author:		Akhil
-- Create date: 22 oct 2012
-- Description:	To Change Status OF Sc Order Status
-- =============================================
-- Purchase.SCOStatus 107 ,null, null,null ,2,null,null,7 
CREATE PROCEDURE [Purchase].[SCOStatus] 
	-- Add the parameters for the stored procedure here
	@SCOrderNo int , 
	@ItemNo int ,
	@ActivityNo smallint,
	@SequenceNo tinyint,
	@StatusNo tinyint,
	@OutputStatus bit output,
	@Result varchar(100) output,
	@ExpenseHeadNo smallint = null
AS
BEGIN
begin try

Declare @GRNQty decimal(12,3)
Select @GRNQty = SUM(Quantity)
				From Inventory.GRNSCOrderDetail
				where SCOrderAmendmentNo in(Select SCOrderAmendmentNo From Purchase.SCOrderAmendmentMain SAM
										where SAM.SCOrderNo = @SCOrderNo)
				AND ( ExpenseheadNo = @ExpenseHeadNo or ServiceItemNo = @ItemNo or (ISNULL(SequenceNo,0) = ISNULL(@SequenceNo,0)
											and ActivityNo= @ActivityNo ))

set @GRNQty= isnull( @GRNQty,0)


if (@StatusNo = 2 or @StatusNo =4)
BEGIN

	UPDATE Purchase.SCOrderDetail SET StatusNo =2 
	Where SCOrderNo =@SCOrderNo AND (ExpenseheadNo = @ExpenseHeadNo or ItemNo = @ItemNo or(ISNULL(SequenceNo,0) =ISNULL(@SequenceNo,0) 
			and  ActivityNo = @ActivityNo)) and Quantity < = isnull(@GRNQty,0) 

	UPDATE Purchase.SCOrderAmendmentDetail SET StatusNo =2 
	Where SCOrderAmendmentNo in(Select SCOrderAmendmentNo From Purchase.SCOrderAmendmentMain SAM
										where SAM.SCOrderNo = @SCOrderNo)
	AND (ExpenseheadNo = @ExpenseHeadNo or ItemNo = @ItemNo or(ISNULL(SequenceNo,0) =ISNULL(@SequenceNo,0) 
	and  ActivityNo = @ActivityNo)) and Quantity < = isnull(@GRNQty,0)
			
	Update Purchase.SCOrderDetail set StatusNo = 1
					where SCOrderNo= @SCOrderNo and 
					(ExpenseheadNo = @ExpenseHeadNo or ItemNo = @ItemNo or (isnull(SequenceNo,0) = isnull(@SequenceNo,0)
					 and ActivityNo = @ActivityNo)) and Quantity > isnull(@GRNQty,0)  and StatusNo =2


	Update Purchase.SCOrderAmendmentDetail set StatusNo = 1
					where SCOrderAmendmentNo in(Select SCOrderAmendmentNo From Purchase.SCOrderAmendmentMain SAM
										where SAM.SCOrderNo = @SCOrderNo)
					and (ExpenseheadNo = @ExpenseHeadNo or ItemNo = @ItemNo or (isnull(SequenceNo,0) = isnull(@SequenceNo,0)
					 and ActivityNo = @ActivityNo)) and Quantity > isnull(@GRNQty,0)  and StatusNo =2
					 					
	Update Purchase.SCOrderDetail set StatusNo = 8
					where SCOrderNo = @SCOrderNo 
					and (ExpenseheadNo = @ExpenseHeadNo or ItemNo = @ItemNo or (isnull(SequenceNo,0) = isnull(@SequenceNo,0)
					and ActivityNo = @ActivityNo))
					and isnull(@GRNQty,0)>0   and StatusNo =1 
					
	Update Purchase.SCOrderAmendmentDetail set StatusNo = 8
					where SCOrderAmendmentNo in(Select SCOrderAmendmentNo From Purchase.SCOrderAmendmentMain SAM
										where SAM.SCOrderNo = @SCOrderNo)
					and (ExpenseheadNo = @ExpenseHeadNo or ItemNo = @ItemNo or (isnull(SequenceNo,0) = isnull(@SequenceNo,0)
					and ActivityNo = @ActivityNo))
					and isnull(@GRNQty,0)>0   and StatusNo =1 
					
end -- if @StatusNo = 2 or @StatusNo =4

if @StatusNo <> 2
	 begin
		Update Purchase.SCOrderDetail set  StatusNo = @StatusNo
					where SCOrderNo  = @SCOrderNo 
					and (ExpenseheadNo = @ExpenseHeadNo or ItemNo = @ItemNo or (isnull(SequenceNo,0) = isnull(@SequenceNo,0)
					and ActivityNo = @ActivityNo))

		Update Purchase.SCOrderAmendmentDetail set  StatusNo = @StatusNo
					where SCOrderAmendmentNo in(Select SCOrderAmendmentNo From Purchase.SCOrderAmendmentMain SAM
										where SAM.SCOrderNo = @SCOrderNo)
					and (ExpenseheadNo = @ExpenseHeadNo or ItemNo = @ItemNo or (isnull(SequenceNo,0) = isnull(@SequenceNo,0)
					and ActivityNo = @ActivityNo))
					
	 end --if @StatusNo <> 2

if @StatusNo= 4  or @StatusNo =7 -- short Close
begin
	
	
	Declare @SCReqNo int, @iRow tinyint ,@iCount tinyint
	Declare @SCReqDetail as table (IDNo tinyint,SCReqNo int)
	insert into @SCReqDetail 
	 select row_number() over (Order By SCOrderReqDetNo), SCReqNo
		from Purchase.SCOrderReqDetail where 
		SCOrderNo =@SCOrderNo  and ItemNo = @Itemno and 
		(ItemNo = @ItemNo or (isnull(SequenceNo,0) = isnull(@SequenceNo,0)
					and ActivityNo = @ActivityNo))
	
	set @iRow = @@rowcount
	set @iCount = 0
	while @iRow >@iCount
	begin
	 select @SCReqNo = SCReqNo from @SCReqDetail where IdNo = @iCOunt + 1
			
	exec [Inventory].[SCReqStatus] @SCReqNo ,@ItemNo ,@ActivityNo,@SequenceNo , @StatusNo,
			@OutputStatus output, @Result output
	if @OutputStatus <> 1
		raiserror(@Result,16,-1)
		
		set @iCOunt = @iCOunt + 1
	end	-- while @iRow >@iCount
End -- if @StatusNo= 4



	if exists(select 1 from Purchase.SCOrderDetail where SCOrderNo = @SCOrderNo and statusNo =1)
	 begin
	   if exists (select 1 from Inventory.GRNSCOrderDetail a,Purchase.SCOrderAmendmentMain b where
					a.SCOrderAmendmentNo = b.SCOrderAmendmentNo and SCOrderNo = @SCOrderNo)
				
		 begin
		  update Purchase.SCOrderMain set StatusNo = 8 where  SCOrderNo  = @SCOrderNo
		  update Purchase.SCOrderAmendmentMain set StatusNo = 8 where  SCOrderNo = @SCOrderNo
		 end
	   else
	   begin
			update Purchase.SCOrderMain set StatusNo = 1 where  SCOrderNo  = @SCOrderNo
		  update Purchase.SCOrderAmendmentMain set StatusNo = 1 where  SCOrderNo = @SCOrderNo
		end
	 end	
	 else if exists(select 1 from Purchase.SCOrderDetail where SCOrderNo = @SCOrderNo and statusNo =8)
	 begin
	   if exists (select 1 from Inventory.GRNSCOrderDetail a,Purchase.SCOrderAmendmentMain b where
					a.SCOrderAmendmentNo = b.SCOrderAmendmentNo and SCOrderNo  = @SCOrderNo )
		 begin
		   update Purchase.SCOrderMain set StatusNo = 8 where  SCOrderNo  = @SCOrderNo
		  update Purchase.SCOrderAmendmentMain set StatusNo = 8 where  SCOrderNo = @SCOrderNo
		 end
	   else
	   begin
		  update Purchase.SCOrderMain set StatusNo = 1 where  SCOrderNo  = @SCOrderNo
		  update Purchase.SCOrderAmendmentMain set StatusNo = 1 where  SCOrderNo = @SCOrderNo
		end
	 end	
	 else if exists(select 1 from Purchase.SCOrderDetail where SCOrderNo = @SCOrderNo and Statusno = 3 )-- Hold 
		begin
		  update Purchase.SCOrderMain set StatusNo = 3 where  SCOrderNo  = @SCOrderNo
		  update Purchase.SCOrderAmendmentMain set StatusNo = 3 where  SCOrderNo = @SCOrderNo
		end
		
	 else if exists(select 1 from Purchase.SCOrderDetail where SCOrderNo = @SCOrderNo and Statusno = 4 )--cancel
	 	begin
		  update Purchase.SCOrderMain set StatusNo = 4 where  SCOrderNo  = @SCOrderNo
		  update Purchase.SCOrderAmendmentMain set StatusNo = 4 where  SCOrderNo = @SCOrderNo
		end
	 	
	else if exists(select 1 from Purchase.SCOrderDetail where SCOrderNo = @SCOrderNo and Statusno=7)--shortclose
	 	begin
		  update Purchase.SCOrderMain set StatusNo = 7 where  SCOrderNo  = @SCOrderNo
		  update Purchase.SCOrderAmendmentMain set StatusNo = 7 where  SCOrderNo = @SCOrderNo
		end				
	
	else if exists(select 1 from Purchase.SCOrderDetail where SCOrderNo = @SCOrderNo  and Statusno=2)--completed
		begin
		  update Purchase.SCOrderMain set StatusNo = 2 where  SCOrderNo  = @SCOrderNo
		  update Purchase.SCOrderAmendmentMain set StatusNo = 2 where  SCOrderNo = @SCOrderNo
		end
		
set @OutputStatus =1
set @Result='Success'
end try

begin catch
set @OutputStatus =0
set @Result=error_message()

		select 0, error_message()
end Catch


END
