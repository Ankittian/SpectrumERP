CREATE PROCEDURE [Finance].[AutoAdjustVoucher] 
	@VoucherNo int
	,@OutputStatusNo tinyint = null output
   ,@OutputStatusRemarks varchar(4000) = null output
AS
BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;
begin try


declare @CompanyNo  tinyint,@NewVoucherNo  int,@DocumentDate  smalldatetime,@DocumentNoYearly  varchar(30),@DocumentStatusNo  tinyint,@FBNo  smallint
,@VoucherTypeNo  tinyint,@TransactionFormCode  varchar(10),@CurrencyConversionNo  smallint,@DivisibilityRate decimal(9,6),@MultiplicativeRate decimal(9,6)
,@SpecificAmount  money,@BaseAmount  decimal(25,6),@Narration  varchar(2000),@ReversalDate  smalldatetime,@ReleasedBy  smallint,@AuthorizedBy  smallint
,@DocumentSettingNo  smallint,@LoginNo   SmallInt,@SourceDocumentNo int,@SourceDocumentDate smalldatetime,@SourceDocumentNoYearly varchar(30)
,@SourceDocumentFormCode varchar(10), @RegularInterFBNo tinyint = 1
,@VendorDocumentNo varchar(50),@VendorDocumentDate smalldatetime,@ServiceTaxAbatementRuleNo smallint,@VendorLocationNo int ,@BillAmount money 

if not exists(select 1
 from Finance.accVoucherReferenceDetail rd 
inner join Finance.accVoucherDetail vd on vd.VoucherDetailNo = rd.VoucherDetailNo  
inner join Finance.accVoucherMain vm on vm.VoucherNo = vd.VoucherNo 
inner join Finance.AutoAdjustReferenceVoucher aa on aa.CompanyNo = vm.CompanyNo 
inner join Finance.ReferenceMain rm on rm.ReferenceNo = rd.ReferenceNo 
where rd.ReferenceTypeNo= 2 and aa.VoucherTypeNo = vm.VoucherTypeNo
 and vd.VoucherNo = @VoucherNo and vm.AuthorizedBy is not null
and abs(rm.BalanceAmount )<= aa.Amount and abs(rm.BalanceAmount )>0 )
begin
	select 1,'Success'
	return
end


select @CompanyNo = CompanyNo 
,@DocumentDate = FormattedDocumentDate 
,@DocumentNoYearly = DocumentNoYearly 
,@DocumentStatusNo = '30'
,@FBNo = FBNo 
,@VoucherTypeNo = 0
,@TransactionFormCode = 'ATRV'
,@CurrencyConversionNo = CurrencyConversionNo 
,@DivisibilityRate = DivisibilityRate 
,@MultiplicativeRate = MultiplicativeRate 
,@SpecificAmount = 0
,@BaseAmount = 0
,@Narration = 'Being amount Auto adjusted'
,@ReversalDate = NULL
,@ReleasedBy = NULL
,@AuthorizedBy = AuthorizedBy 
,@LoginNo = AuthorizedBy 
,@SourceDocumentNo = VoucherNo 
,@SourceDocumentDate = FormattedDocumentDate 
,@SourceDocumentNoYearly = DocumentNoYearly 
,@SourceDocumentFormCode = 'ATRV'
,@RegularInterFBNo = RegularInterFBNo 
,@VendorDocumentNo = NULL
,@VendorDocumentDate = NULL
,@ServiceTaxAbatementRuleNo = NULL
,@VendorLocationNo = NULL
,@BillAmount = 0
from Finance.accVoucherMain where VoucherNo = @VoucherNo 

select @DocumentSettingNo =ds.DocumentSettingNo
			from masterdata.DocumentSetting ds
				inner join masterdata.DocumentFormDetail dfd on dfd.DocumentSettingNo=ds.DocumentSettingNo
															and dfd.FormNo=126
				inner join masterdata.DocumentFinanceBookDetail fbd on fbd.DocumentSettingNo=ds.DocumentSettingNo
															and fbd.FBNo=@FBNo 
			where ds.CompanyNo=@CompanyNo and ds.Inactive=0 and dfd.IsDefault =1 



exec Finance.VoucherSaveMain
@CompanyNo = @CompanyNo 
,@VoucherNo = null 
,@DocumentDate = @DocumentDate 
,@DocumentNoYearly = @DocumentNoYearly 
,@DocumentStatusNo = '30'
,@FBNo = @FBNo 
,@ProjectAmendmentNo = NULL
,@VoucherTypeNo = @VoucherTypeNo 
,@TransactionFormCode = 'ATRV'
,@CurrencyConversionNo = @CurrencyConversionNo 
,@DivisibilityRate = @DivisibilityRate 
,@MultiplicativeRate = @MultiplicativeRate 
,@SpecificAmount = @SpecificAmount
,@BaseAmount = @BaseAmount
,@Narration = @Narration
,@ReversalDate = NULL
,@ReleasedBy = NULL
,@AuthorizedBy = @AuthorizedBy
,@DocumentSettingNo = @DocumentSettingNo 
,@LoginNo = @LoginNo 
,@SourceDocumentNo = @SourceDocumentNo 
,@SourceDocumentDate = @SourceDocumentDate 
,@SourceDocumentNoYearly = @SourceDocumentNoYearly 
,@SourceDocumentFormCode = 'ATRV'
,@RegularInterFBNo =@RegularInterFBNo 
,@VendorDocumentNo = NULL
,@VendorDocumentDate = NULL
,@ServiceTaxAbatementRuleNo = NULL
,@VendorLocationNo = NULL
,@BillAmount = 0
,@OutputStatusNo = @OutputStatusNo output
,@OutputStatusRemarks = @OutputStatusRemarks  output

if @OutputStatusNo= 0 
 raiserror(@OutputStatusRemarks,16,-1)
 
set @NewVoucherNo = @OutputStatusRemarks  
 
 declare @ReferenceNo int, @BalanceAmount money, @AccountNo int , @SerialNo smallint,@DrAmount money,@CrAmount money, @ByToNo tinyint
	,@VoucherDetailNo int , @ReferenceNoYearly varchar(30), @ReferenceDate smalldatetime,@CreditDays  tinyint,@DueDate  smalldatetime
	, @SpecificReferenceAmount money, @AdjusmentAccountNo int
	
 set @SerialNo= 1
declare cur cursor static for
select rm.ReferenceNo ,rm.BalanceAmount ,vd.AccountNo , SerialNo , rd.ReferenceNoYearly, rd.ReferenceDate, rd.CreditDays,rd.DueDate
		,aa.AdjusmentAccountNo
 from Finance.accVoucherReferenceDetail rd 
inner join Finance.accVoucherDetail vd on vd.VoucherDetailNo = rd.VoucherDetailNo  
inner join Finance.accVoucherMain vm on vm.VoucherNo = vd.VoucherNo 
inner join Finance.AutoAdjustReferenceVoucher aa on aa.CompanyNo = vm.CompanyNo 
inner join Finance.ReferenceMain rm on rm.ReferenceNo = rd.ReferenceNo 
where rd.ReferenceTypeNo= 2 and aa.VoucherTypeNo = vm.VoucherTypeNo
 and vd.VoucherNo = @VoucherNo
and abs(rm.BalanceAmount )<= aa.Amount 



open cur
fetch next from cur into @ReferenceNo, @BalanceAmount, @AccountNo ,@SerialNo , @ReferenceNoYearly, @ReferenceDate, @CreditDays,@DueDate
		,@AdjusmentAccountNo
while @@FETCH_STATUS =0 
begin
 set @VoucherDetailNo =0 
	if @BalanceAmount>0
	begin
	 set @CrAmount = @BalanceAmount 
	 set @DrAmount =0 
	 set @ByToNo = 2
	end 
	else
	begin
	  set @DrAmount  = abs(@BalanceAmount )
	  set @CrAmount =0
	  set @ByToNo = 1
	end  
if not exists(select 1 from Finance.accVoucherDetail where VoucherNo = @NewVoucherNo and AccountNo = @AccountNo )
begin	  
	exec Finance.VoucherSaveDetail
		@VoucherNo = @NewVoucherNo 
		,@SerialNo = @SerialNo 
		,@AccountNo = @AccountNo
		,@BaseDrAmount = @DrAmount
		,@BaseCrAmount = @CrAmount
		,@SpecificDrAmount =@DrAmount
		,@SpecificCrAmount = @CrAmount
		,@ByToNo = @ByToNo
		,@TaxCodeMiscChargesDetailNo = NULL
		,@OutputStatusNo = @OutputStatusNo  output
		,@OutputStatusRemarks = @OutputStatusRemarks output
		
	If @OutputStatusNo= 0 
		raiserror(@OutputStatusRemarks,16,-1)
 
	set @VoucherDetailNo = @OutputStatusRemarks 
end
else
begin
	select @VoucherDetailNo = vd.VoucherDetailNo from Finance.accVoucherDetail vd where AccountNo = @AccountNo and VoucherNo = @NewVoucherNo 
	update Finance.accVoucherDetail set BaseCrAmount = BaseCrAmount + @CrAmount , SpecificCrAmount = SpecificCrAmount + @CrAmount
			, BaseDrAmount = BaseDrAmount + @DrAmount , SpecificDrAmount = SpecificDrAmount + @DrAmount 
			where VoucherDetailNo=@VoucherDetailNo
			
end	
 
exec Finance.VoucherSaveReferenceDetail
@VoucherDetailNo = @VoucherDetailNo
,@ReferenceNo = @ReferenceNo
,@ReferenceNoYearly = @ReferenceNoYearly 
,@ReferenceDate = @ReferenceDate 
,@CreditDays = @CreditDays 
,@DueDate = @DueDate 
,@ReferenceTypeNo = '2'
,@SpecificDrAmount = @DrAmount 
,@SpecificCrAmount = @CrAmount 
,@BaseDrAmount = @DrAmount 
,@BaseCrAmount = @CrAmount 
,@TransactionFormCode = 'ATRV'
,@SourceDocumentNo = @NewVoucherNo 
,@SourceDocumentDate = @DocumentDate 
,@SourceDocumentNoYearly = @DocumentNoYearly 
,@Remarks = 'Being amount Adjusted'
,@SpecificReferenceAmount = @DrAmount 
,@BaseReferenceAmount = @DrAmount 
,@SourceDocumentFormCode = 'ATJV'
,@AccountNo = @AccountNo 
,@CompanyNo = @CompanyNo 
,@FBNo = @FBNo 
,@ByToNo = @ByToNo 
,@CurrencyConversionNo = @CurrencyConversionNo 
,@DivisibilityRate = @DivisibilityRate 
,@MultiplicativeRate = @MultiplicativeRate 
,@OutputStatusNo =@OutputStatusNo  output
,@OutputStatusRemarks =@OutputStatusRemarks  output
	If @OutputStatusNo= 0 
		raiserror(@OutputStatusRemarks,16,-1)
		
	fetch next from cur into @ReferenceNo, @BalanceAmount, @AccountNo ,@SerialNo , @ReferenceNoYearly, @ReferenceDate, @CreditDays,@DueDate
		,@AdjusmentAccountNo

end
close cur
deallocate cur

if @AdjusmentAccountNo<>0
begin
	select @CrAmount=case when SUM(basedramount-BaseCrAmount )>0 then SUM(basedramount-BaseCrAmount ) else 0 end
		,@DrAmount =  case when SUM(BaseCrAmount- basedramount)>0 then SUM(BaseCrAmount - basedramount ) else 0 end
		from Finance.accVoucherDetail where VoucherNo = @NewVoucherNo
	if @DrAmount>0 
	 set @ByToNo= 1
	else 
	 set @ByToNo =2
	
	set @SerialNo = @SerialNo + 1
	exec Finance.VoucherSaveDetail
		@VoucherNo = @NewVoucherNo 
		,@SerialNo = @SerialNo
		,@AccountNo = @AdjusmentAccountNo
		,@BaseDrAmount = @DrAmount
		,@BaseCrAmount = @CrAmount
		,@SpecificDrAmount =@DrAmount
		,@SpecificCrAmount = @CrAmount
		,@ByToNo = @ByToNo
		,@TaxCodeMiscChargesDetailNo = NULL
		,@OutputStatusNo = @OutputStatusNo  output
		,@OutputStatusRemarks = @OutputStatusRemarks output
		
	If @OutputStatusNo= 0 
		raiserror(@OutputStatusRemarks,16,-1)
 
 
	update Finance.accVoucherMain set SpecificAmount = @DrAmount + @CrAmount , BaseAmount  = @DrAmount + @CrAmount 
		,BillAmount = @DrAmount + @CrAmount  
	where VoucherNo = @NewVoucherNo 
	
	
end

exec Finance.VoucherSaveStatuory @VoucherNo = @NewVOucherno
exec Finance.VoucherSaveAfterInsert @VoucherNo =@NewVoucherNo

exec [Finance].[VoucherSaveChildDetail]@VoucherNo = @VoucherNo,@ChildVoucherNo = @NewVoucherNo 
	,@OutputStatusNo =@OutputStatusNo  output
	,@OutputStatusRemarks =@OutputStatusRemarks  output
	
	If @OutputStatusNo= 0 
		raiserror(@OutputStatusRemarks,16,-1)
		
 
select @OutputStatusNo=1 ,@OutputStatusRemarks= 'Sucess' 

end try

begin catch
 select @OutputStatusRemarks
		select @OutputStatusNo=1 ,@OutputStatusRemarks= error_message()
end catch

end
