CREATE PROCEDURE [Finance].[FOAMFillValuesOrderNo]
(
	@OrderTypeNo tinyint
	,@OrderNo int
)
AS

BEGIN
	SET NOCOUNT ON;
	
	BEGIN TRY
			
		--declare @Return table (AccountType varchar(50),Particular varchar(500),AccountNo smallint,AccountTypeNo tinyint
		--						,Id int,Flag varchar(10))

		DECLARE @ItemNos xml
		DECLARE @MiscChargeNos xml
		DECLARE @ExpenseHeadNos xml
		DECLARE @FreightTypeNos xml
		DECLARE @PaymentTermNos xml
		DECLARE @VendorLocationNo int
		DECLARE @DocumentDate smalldatetime
		declare @ServiceItemNos xml
		DECLARE @CostHeadNos xml
		
		declare @DivisionNo int
				,@CompanyNo tinyint
				
		if @OrderTypeNo=1
		begin
			declare @IndentTypeNo tinyint
					,@OutputItemDetail xml
					
			select @VendorLocationNo=VendorLocationNo
					,@DocumentDate=DocumentDate 
					,@FreightTypeNos='<DocumentElement><Table><Freight_x0020_Type>'+CONVERT(varchar(5),FreightTypeNo)+'</Freight_x0020_Type><Route_x0020_Id>1.1</Route_x0020_Id></Table></DocumentElement>'
					,@IndentTypeNo = IndentTypeNo
					,@DivisionNo = isnull(DivisionNo,0)
					,@CompanyNo = CompanyNo
			from Purchase.POMain where PONo=@OrderNo
			
			select @ItemNos=(select [Item_x0020_Code] from (select ItemNo [Item_x0020_Code] 
							from Purchase.POItemDetail 
							where PONo=@OrderNo
							union
							select bd.itemNo  from 
							 Purchase.SCBOQMain bm
							 inner join Purchase.SCBOQitemdetail bd on bd.SCBOQNo= bm.SCBOQNo
							 where bm.pono = @OrderNo) tt group  by [Item_x0020_Code]
							for xml path('Table'),Root ('DocumentElement'))
			 
			select @MiscChargeNos=(select MiscChargeNo [Charge_x0020_Name] 
											,TaxCodeMiscChargesDetailNo [TDC_x0020_Account_x0020_Rule]
									from Purchase.POTaxDetail 
									where PONo=@OrderNo
									for xml path('Table'),Root ('DocumentElement'))
							
			select @ExpenseHeadNos=(select ExpenseheadNo [Expense_x0020_Head] 
									from Purchase.POExpenseHead
									where PONo=@OrderNo
									for xml path('Table'),Root ('DocumentElement'))
							
			select @FreightTypeNos=(select FreightTypeNo [Freight_x0020_Type],RouteId [Route_x0020_Id]
									from Purchase.POTransportationRoute
									where PONo=@OrderNo
									for xml path('Table'),Root ('DocumentElement'))
			where @FreightTypeNos is null
										
			select @PaymentTermNos=(select PaymentTypeNo [Payment_x0020_Type] 
									from Purchase.POPaymentTermsDetail
									where PONo=@OrderNo
									for xml path('Table'),Root ('DocumentElement'))
							 
			select @OutputItemDetail=(select HOD.ItemNo [Item_x0020_Description] 
									from Purchase.POExpenseHeadOutputDetail HOD
										inner join Purchase.POExpenseHead EH on EH.POExpenseDetailNo=HOD.POExpenseDetailNo
									where PONo=@OrderNo
									for xml path('Table'),Root ('DocumentElement'))
													 
			--insert into @Return
			EXECUTE [Purchase].[PMPOFillValuesAccountAssignment] 
			   @ItemNos = @ItemNos
			  ,@MiscChargeNos = @MiscChargeNos
			  ,@ExpenseHeadNos = @ExpenseHeadNos
			  ,@FreightTypeNos = @FreightTypeNos
			  ,@PaymentTermNos = @PaymentTermNos
			  ,@VendorLocationNo = @VendorLocationNo
			  ,@DocumentDate = @DocumentDate
			  ,@PONo = @OrderNo
				,@IndentTypeNo = @IndentTypeNo
				,@DivisionNo = @DivisionNo 
				,@CompanyNo = @CompanyNo 
				,@OutputItemDetail = @OutputItemDetail
		end
		else if @OrderTypeNo=2
		begin
			declare	@SalesCategoryNo tinyint
					,@SalesOrderTypeNo tinyint
	
			select @VendorLocationNo=CustomerLocationNo
					,@DocumentDate=DocumentDate 
					,@FreightTypeNos='<DocumentElement><Table><Freight_x0020_Type>'+CONVERT(varchar(5),FreightTypeNo)+'</Freight_x0020_Type><Route_x0020_Id>1.1</Route_x0020_Id></Table></DocumentElement>'
					,@SalesCategoryNo = SalesCategoryNo
					,@SalesOrderTypeNo = SalesOrderTypeNo 
			from Sales.SalesOrderMain where SalesOrderNo=@OrderNo
			
			select @ItemNos=(select ItemNo [Item_x0020_Code]
									,ProcessingTypeNo [Processing_x0020_Type]
							from Sales.SalesOrderItemDetail 
							where SalesOrderNo=@OrderNo
							group by ItemNo ,ProcessingTypeNo
							for xml path('Table'),Root ('DocumentElement'))
			 
			select @MiscChargeNos=(select MiscChargeNo [Charge_x0020_Name] 
											,TaxCodeMiscChargesDetailNo [TDC_x0020_Account_x0020_Rule]
									from Sales.SalesOrderTaxDetail 
									where SalesOrderNo=@OrderNo
									for xml path('Table'),Root ('DocumentElement'))
																								
			select @PaymentTermNos=(select SalesPaymentTypeNo [Payment_x0020_Type] 
									from Sales.SalesOrderPaymentTermDetail
									where SalesOrderNo=@OrderNo
									for xml path('Table'),Root ('DocumentElement'))

			select @CostHeadNos=(select CostHeadNo [Cost_x0020_Head] 
									from Sales.SalesOrderExportCostDetail
									where SalesOrderNo=@OrderNo
									for xml path('Table'),Root ('DocumentElement'))

			
							 
			--insert into @Return
			EXECUTE [sales].[SDSOFillValuesAccountAssignment] 
			   @ItemNos = @ItemNos
			  ,@MiscChargeNos = @MiscChargeNos
			  ,@FreightTypeNos = @FreightTypeNos
			  ,@PaymentTermNos = @PaymentTermNos
			  ,@CustomerLocationNo = @VendorLocationNo
			  ,@DocumentDate = @DocumentDate
			  ,@SalesOrderNo = @OrderNo
			  ,@SalesCategoryNo = @SalesCategoryNo
			  ,@SalesOrderTypeNo = @SalesOrderTypeNo 
			  ,@CostHeadNos = @CostHeadNos
		end
		else if @OrderTypeNo=3
		begin
			declare @IsExpenseMapped bit
					,@SCNatureNo tinyint
					,@SCTypeNo tinyint
					
			select @VendorLocationNo=VendorLocationNo
					,@DocumentDate=DocumentDate 
					,@FreightTypeNos='<DocumentElement><Table><Freight_x0020_Type>'+CONVERT(varchar(5),FreightTypeNo)+'</Freight_x0020_Type><Route_x0020_Id>1.1</Route_x0020_Id></Table></DocumentElement>'
					,@IsExpenseMapped = IsExpenseMapped
					,@SCNatureNo = SCNatureNo
					,@SCTypeNo = SCTypeNo
			from Purchase.SCOrderMain where SCOrderNo=@OrderNo
			
			select @ItemNos=(select ItemNo [Item_x0020_Code] 
									,ItemFlowNo [Item_x0020_Flow]
							from Purchase.SCOrderItemDetail 
							where SCOrderNo=@OrderNo
							for xml path('Table'),Root ('DocumentElement'))
			 
			select @MiscChargeNos=(select MiscChargeNo [Charge_x0020_Name] 
											,TaxCodeMiscChargesDetailNo [TDC_x0020_Account_x0020_Rule]
									from Purchase.SCOrderTaxDetail 
									where SCOrderNo=@OrderNo
									for xml path('Table'),Root ('DocumentElement'))
							
			select @ExpenseHeadNos=(select POEH.ExpenseheadNo [ExpenseHeadNo] 
									from Purchase.SCOrderExpenseHead SCOEH
										inner join Purchase.POAmendmentExpenseHead POEH on POEH.POAmendmentExpenseDetailNo=SCOEH.POAmendmentExpenseDetailNo
									where SCOrderNo=@OrderNo
									for xml path('Table'),Root ('DocumentElement'))
							
			select @PaymentTermNos=(select PaymentTypeNo [Payment_x0020_Type] 
									from Purchase.SCOrderPaymentTermsDetail
									where SCOrderNo=@OrderNo
									for xml path('Table'),Root ('DocumentElement'))						 
			
			if @SCTypeNo=3
			begin 
				select @ServiceItemNos=(select ActivityNo [Code] 
									from Purchase.SCOrderDetail
									where SCOrderNo=@OrderNo
									for xml path('Table'),Root ('DocumentElement'))
			end
			else
			begin		 
				select @ServiceItemNos=(select ItemNo [Code] 
									from Purchase.SCOrderDetail
									where SCOrderNo=@OrderNo
									for xml path('Table'),Root ('DocumentElement'))
			end
													 
			--insert into @Return
			EXECUTE [Purchase].[SCSOFillValuesAccountAssignment] 
			   @ItemNos = @ItemNos
			  ,@ServiceItemNos = @ServiceItemNos
			  ,@MiscChargeNos = @MiscChargeNos
			  ,@ExpenseHeadNos = @ExpenseHeadNos
			  ,@FreightTypeNos = @FreightTypeNos
			  ,@PaymentTermNos = @PaymentTermNos
			  ,@VendorLocationNo = @VendorLocationNo
			  ,@DocumentDate = @DocumentDate
			  ,@SCOrderNo = @OrderNo
			  ,@IsExpenseMapped = @IsExpenseMapped
			  ,@SCNatureNo = @SCNatureNo
			  ,@SCTypeNo = @SCTypeNo
		end
		else if @OrderTypeNo=5
		begin
			--insert into @Return
			Select
			AT.AccountType [Account Type],
			AD.Particular [Particular],
			AD.AccountNo [Account Name],
			AD.AccountTypeNo [AccountTypeNo],
			AD.Id [Id],
			AD.Flag [Flag]
			from Purchase.FreightOrderAccountAssignmentDetail AD
				inner join GlobalData.AccountType AT on AT.AccountTypeNo=AD.AccountTypeNo
			where AD.FreightOrderNo=@OrderNo
				and @OrderTypeNo=4
		end
		else if @OrderTypeNo=5
		begin
			declare @ResourceNos xml
			
			select @VendorLocationNo=CustomerLocationNo
					,@DocumentDate=DocumentDate 
			from [CustomerService].[ServiceOrderMain] where ServiceOrderNo=@OrderNo
			
			select @ItemNos=(select ItemNo [Item_x0020_Code] 
							from [CustomerService].[ServiceOrderPartsList] 
							where ServiceOrderNo=@OrderNo
							for xml path('Table'),Root ('DocumentElement'))
			 							
			select @ResourceNos=(select EAMResourceNo [Resource_x0020_Name]
									from [CustomerService].ServiceOrderResourceDetail
									where ServiceOrderNo=@OrderNo
									for xml path('Table'),Root ('DocumentElement'))
							
			select @PaymentTermNos=(select SalesPaymentTypeNo [Payment_x0020_Type] 
									from [CustomerService].ServiceOrderPaymentTermDetail
									where ServiceOrderNo=@OrderNo
									for xml path('Table'),Root ('DocumentElement'))
							 
			select @ServiceItemNos=(select ItemNo [Service_x0020_Description]
									from [CustomerService].ServiceOrderServiceDetail
									where ServiceOrderNo=@OrderNo
									for xml path('Table'),Root ('DocumentElement'))
													 
			--insert into @Return
			EXECUTE [CustomerService].[CMSOFillValuesAccountAssignment] 
			   @ItemNos = @ItemNos
			  ,@ServiceItemNos = @ServiceItemNos
			  ,@PaymentTermNos = @PaymentTermNos
			  ,@CustomerLocationNo = @VendorLocationNo
			  ,@DocumentDate = @DocumentDate
			  ,@ServiceOrderNo = @OrderNo
			  ,@ResourceNos = @ResourceNos
		
		end
		else if @OrderTypeNo=6
		begin
			select @VendorLocationNo=CustomerLocationNo
					,@DocumentDate=DocumentDate 
			from Sales.SalesServiceOrderMain where SalesServiceOrderNo=@OrderNo
			
			select @ItemNos=(select ItemNo [Item_x0020_Code]
							from Sales.SalesServiceOrderDetail 
							where SalesServiceOrderNo=@OrderNo
							group by ItemNo
							for xml path('Table'),Root ('DocumentElement'))
			 																								
			select @PaymentTermNos=(select SalesPaymentTypeNo [Payment_x0020_Type] 
									from Sales.SalesServiceOrderPaymentTermDetail
									where SalesServiceOrderNo=@OrderNo
									for xml path('Table'),Root ('DocumentElement'))
							 
			--insert into @Return
			EXECUTE [sales].[SDSEFillValuesAccountAssignment] 
			   @ItemNos = @ItemNos
			  ,@PaymentTermNos = @PaymentTermNos
			  ,@CustomerLocationNo = @VendorLocationNo
			  ,@DocumentDate = @DocumentDate
			  ,@SalesServiceOrderNo = @OrderNo
		end
		else
		Select
			cast(null as varchar(50)) [Account Type],
			cast(null as varchar(100)) [Particular],
			cast(null as int) [Account Name],
			cast(null as tinyint) [AccountTypeNo],
			cast(null as int) [Id],
			cast(null as varchar(10)) [Flag]
		where 1=2			
		--0
		/*
		select AccountType [Account Type],
			Particular [Particular],
			AccountNo [Account Name],
			AccountTypeNo [AccountTypeNo],
			Id [Id],
			Flag [Flag] 
		from @Return
		*/
		
	END TRY
	BEGIN CATCH
		INSERT INTO [masterdata].[mErrorLog]
			   ([ErrorDate]
			   ,[ErrorNo]
			   ,[ErrorSeverity]
			   ,[ErrorState]
			   ,[ErrorProcedure]
			   ,[ErrorLine]
			   ,[ErrorDescription]
			   ,[UserName]
			   ,[FormCode])
		VALUES
			   (getdate()
			   ,ERROR_NUMBER()
			   ,ERROR_SEVERITY()
			   ,ERROR_STATE()
			   ,ERROR_PROCEDURE()
			   ,ERROR_LINE()
			   ,ERROR_MESSAGE()
			   ,SUSER_SNAME()
			   ,'FOAM')
	END CATCH
END
