CREATE PROCEDURE [Information].[ServiceOrderPendingMIForDept]  
	-- Add the parameters for the stored procedure here
	@CompanyNo tinyint,
	@DocumentDate smalldatetime,
	@DeptNo int
AS
BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;

    -- Insert statements for procedure here
	declare @str varchar(max)
    select @str=masterdata.JSForInformation('IMMI')
    set @str=@str+masterdata.CSSForInformation('IMMI')
	set @str=@str+'<table class="altrowstable" id="alternatecolorServiceOrderPendingMIInfo">'
	set @str=@str+'<Caption><Font color="Maroon"><h3 align="Left"><b>Pending Issue  (Service Order)</b></h3></Font></Caption>'
    set @str=@str+'<tr><th>Sl. No.</th><th>Document No.</th><th>Document Date</th><th>Source  Document</th><th>Ref. Doc. No.</th><th>Ref. Doc. Date</th><th>SubContract Type</th><th>SubContract Nature</th><th>Vendor/Contractor</th><th>SubContract Location</th><th>Created By</th><th>Authorized By</th></tr>'
	set @str=@str
				 +( replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					replace((Select ROW_NUMBER() over ( Order by SCOM.DocumentDate desc) as td1
					               ,SCOM.[DocumentNoYearly] td2
								   ,Convert(Varchar(20),SCOM.[DocumentDate],103) td3
								   ,sd.SourceDocument td4
								   ,Isnull(Isnull(Isnull(qm.DocumentNoYearly
								           ,SR.DocumentNoYearly),SA.DocumentNoYearly),'')td5
								   ,Isnull(Convert(Varchar(20)
								   ,Isnull(Isnull(qm.DocumentDate,SR.DocumentDate),SA.DocumentDate),103),'') td6  
								   ,sct.SCTypeName td7
								   ,SCN.SCNature td8
								   ,VC.Name td9  
								   ,sl.SubcontractLocation td10                                  
								   ,L1.PrintingName td11
								   ,Isnull(L2.PrintingName,'') td12
							From Purchase.SCOrderMain as SCOM
							Inner Join masterdata.mVendorLocationDetail as VLD On SCOM.VendorLocationNo=VLD.VendorLocationNo
							Inner Join masterdata.mVendorCard as VC On VC.VendorNo = VLD.VendorNo 
							Inner Join GlobalData.SubcontractLocation sl on sl.SubcontractLocationNo = SCOM.SubcontractLocationNo
							Inner Join GlobalData.SCSourceDocument  sd ON sd.SCSourceNo = SCOM.SCSourceNo 
							Inner join GlobalData.SCType sct on sct.SCTypeNo = SCOM.SCTypeNo
							Inner join GlobalData.SubcontractNature SCN on SCN.SCNatureNo=SCOM.SCNatureNo  
							Left Join purchase.RevisedSCQuotationMain  qm on SCOM.RefDocumentNo  = qm.RevisedSCQuotationNo
																		  And SCOM.SCSourceNo=1 ----Quotation
							Left Join Purchase.SCOrderReqDetail SCR ON SCR.SCOrderNo=SCOM.SCOrderNo----Contain Item Wise Detail Group by clause for this
							Left Join inventory.SCReqMain SR on SCR.SCReqNo = SR.SCReqNo 
															And SCOM.SCSourceNo=2-------Service Request
							Left Join Purchase.SCAgreementMain  SA on SA.SCAgreementNo=SCOM.RefDocumentNo
																And SCOM.SCSourceNo= 3 -----Agreement
					
							Left Join Inventory.MaterialIssueMain MI ON MI.RefDocumentNo=SCOM.SCOrderNo
																     And MI.IssueSourceNo=3 ---SubContract Order
							Inner Join Security.Login L1 On L1.LoginNo=SCOM.CreatedBy
							Left Join Security.Login L2 On L2.LoginNo=SCOM.AuthorizedBy
							Where SCOM.SCOrderDocumentTypeNo=2---SubContract Order
							And SCOM.CompanyNo=@CompanyNo 
							And SCOM.DocumentDate<=@DocumentDate
							And SR.DeptNo=@DeptNo
							And MI.IssueNo Is Null
							Group By SCOM.[DocumentNoYearly] 
								   ,SCOM.[DocumentDate] 
								   ,sd.SourceDocument 
								   ,qm.DocumentNoYearly,SR.DocumentNoYearly,SA.DocumentNoYearly
								   ,SCOM.SCSourceNo,qm.DocumentDate,SR.DocumentDate,SA.DocumentDate  
								   ,sct.SCTypeName 
								   ,SCN.SCNature
								   ,VC.Name 
								   ,sl.SubcontractLocation                            
								   ,L1.PrintingName 
								   ,L2.PrintingName 
							for xml path('tr')),'<td1>','<td style="text-align: center;">')
							               ,'<td2>','<td>')
										   ,'<td3>','<td style="text-align: center;">')
										   ,'<td4>','<td>')
										   ,'<td5>','<td>')
										   ,'<td6>','<td style="text-align: center;">')
										   ,'<td7>','<td>')
										   ,'<td8>','<td>')
										   ,'<td9>','<td>')
										   ,'<td10>','<td>')
										   ,'<td11>','<td>')
										   ,'<td12>','<td>')
										   )
				   
	set @str=@str+'</table>'
	
	select @str
	
END
