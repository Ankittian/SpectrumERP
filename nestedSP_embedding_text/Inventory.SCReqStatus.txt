Stored procedure Inventory.SCReqStatus: to Chage Status OF Indent. Called from Purchase.SCOStatus, Purchase.SCSODelete.
Writes: Inventory.SCReqActivityDetail, Inventory.SCReqMain, Inventory.SCReqServiceDetail
Reads: Inventory.GRNSCOrderDetail, Inventory.GRNSCOrderRequisitionDetail, Inventory.SCReqActivityDetail, Inventory.SCReqServiceDetail, Purchase.SCOrderAmendmentMain, Purchase.SCOrderAmendmentReqDetail, Purchase.SCOrderDetail, Purchase.SCOrderMain, Purchase.SCOrderReqDetail, Purchase.SCRFQReqDetail
Parameters: @SCReqNo int, @ItemNo int, @ActivityNo smallint, @SequenceNo tinyint, @StatusNo tinyint, @OutputStatus bit OUTPUT, @Result varchar(100) OUTPUT
Author: akhil, created 22 oct 2012
Code:
-- =============================================
-- Author:		akhil
-- Create date: 22 oct 2012
-- Description:	to Chage Status OF Indent
-- =============================================
CREATE PROCEDURE [Inventory].[SCReqStatus] 
	-- Add the parameters for the stored procedure here
	@SCReqNo int,
	@ItemNo int,
	@ActivityNo smallint,
	@SequenceNo tinyint,
	@StatusNo tinyint,
	@OutputStatus bit output,
	@Result varchar(100) output
AS
BEGIN
begin try
		Declare @SCQty decimal(10,3), @GRNQty decimal(10,3)
	
	if (@StatusNo = 2 or @StatusNo = 4) -- Cancel or complete status
	begin
		
		select @SCQty = sum(srd.Quantity)
			from 
			Purchase.SCOrderMain sam inner join Purchase.SCOrderReqDetail srd on srd.SCOrderNo= sam.SCOrderNo
			inner join Purchase.SCOrderDetail sad on sad.SCOrderNo = sam.SCOrderNo and 
			(srd.ItemNo = sad.ItemNo or (isnull(srd.SequenceNo,0) = isnull(sad.SequenceNo,0)
					 and srd.ActivityNo = srd.ActivityNo))
			where srd.SCReqNo = @SCReqNo  and sad.StatusNo <>4 and
			(sad.ItemNo = @ItemNo or (isnull(sad.SequenceNo,0) = isnull(@SequenceNo,0)
					and sad.ActivityNo = @ActivityNo))
			and SCOrderDocumentTypeNo <>1 -- not for LOI

		select @GRNQty = sum(gd.Quantity)
				from 
				Purchase.SCOrderAmendmentMain sam 
				inner join Purchase.SCOrderReqDetail srd on srd.SCOrderNo= sam.SCOrderNo
				inner join Purchase.SCOrderDetail sad on sad.SCOrderNo = sam.SCOrderNo and 
			(srd.ItemNo = sad.ItemNo or (isnull(srd.SequenceNo,0) = isnull(sad.SequenceNo,0)
					 and srd.ActivityNo = srd.ActivityNo))
			inner join Inventory.GRNSCOrderDetail gid on 
						gid.SCOrderAmendmentNo = sam.SCOrderAmendmentNo and 
						(gid.ServiceItemNo = sad.ItemNo or (isnull(gid.SequenceNo,0) = isnull(sad.SequenceNo,0)
					 and gid.ActivityNo = srd.ActivityNo))
			inner join Inventory.GRNSCOrderRequisitionDetail gd on gid.GRNSCOrderDetailNo = gd.GRNSCOrderDetailNo 
							and gd.SCReqNo = srd.SCReqNo 
			
			where srd.SCReqNo =@SCReqNo  and sad.StatusNo = 4 and
					(sad.ItemNo = @ItemNo or (isnull(sad.SequenceNo,0) = isnull(@SequenceNo,0)
					and sad.ActivityNo = @ActivityNo))
				and SCOrderDocumentTypeNo<>1 -- not for LOI

		if @ActivityNo is not null
		begin
								
			Update Inventory.SCReqActivityDetail set StatusNo = 2
					where SCReqNo  = @SCReqNo and 
					isnull(SequenceNo,0) = isnull(@SequenceNo,0)
					and ActivityNo = @ActivityNo and Quantity <= ISNULL(@SCQty,0) + ISNULL(@GRNQty,0)
					
			Update Inventory.SCReqActivityDetail set StatusNo = 8
					where SCReqNo  = @SCReqNo and 
					isnull(SequenceNo,0) = isnull(@SequenceNo,0)
					and ActivityNo = @ActivityNo
					and Quantity > ISNULL(@SCQty,0) + ISNULL(@GRNQty,0) and ISNULL(@SCQty,0) >0 and StatusNo =1
					
					
			Update Inventory.SCReqActivityDetail set StatusNo = 1
					where SCReqNo  = @SCReqNo and 
					isnull(SequenceNo,0) = isnull(@SequenceNo,0)
					and ActivityNo = @ActivityNo
					and Quantity > ISNULL(@SCQty,0) + ISNULL(@GRNQty,0) and StatusNo =2
								
		 end
		 else
		 begin
			 Update Inventory.SCReqServiceDetail set StatusNo = 2
					where SCReqNo  = @SCReqNo and 
					itemno = @ItemNo and Quantity <= ISNULL(@SCQty,0) + ISNULL(@GRNQty,0)
					
			Update Inventory.SCReqServiceDetail set StatusNo = 8
					where SCReqNo  = @SCReqNo and 
					itemno = @ItemNo and Quantity > ISNULL(@SCQty,0) + ISNULL(@GRNQty,0) and 
						ISNULL(@SCQty,0) >0 and StatusNo =1
					
					
			Update Inventory.SCReqServiceDetail set StatusNo = 1
					where SCReqNo  = @SCReqNo and 
					itemno = @ItemNo 
					and Quantity > ISNULL(@SCQty,0) + ISNULL(@GRNQty,0) and StatusNo =2
		 end
	 
	 end -- if (@StatusNo = 2 or @StatusNo = 4)
	if @StatusNo <> 2
	begin
		if @ActivityNo is not null
			Update Inventory.SCReqActivityDetail set  StatusNo = @StatusNo
					where SCReqNo  = @SCReqNo and 
					isnull(SequenceNo,0) = isnull(@SequenceNo,0)
					and ActivityNo = @ActivityNo
		 else
		   	Update Inventory.SCReqServiceDetail  set  StatusNo = @StatusNo
					where SCReqNo  = @SCReqNo and itemno = @ItemNo 
	 end --  if @StatusNo <> 2
	 
	 -- Preferences list Authpoized/Hold/Cancel/Short Close/Completed
if @ActivityNo is not null
begin
	 
	if exists(select 1 from Inventory.SCReqActivityDetail where SCReqNo = @SCReqNo and statusNo =1)
	 begin
	   if exists (select 1 from Purchase.SCRFQReqDetail where SCReqNo = @SCReqNo)
		update Inventory.SCReqMain set StatusNo = 8 where SCReqNo = @SCReqNo
		
	   else if exists (select 1 from Purchase.SCOrderAmendmentReqDetail where SCReqNo =@SCReqNo)
		 update Inventory.SCReqMain set StatusNo = 8 where SCReqNo = @SCReqNo
	   else
		 update Inventory.SCReqMain set StatusNo = 1 where SCReqNo = @SCReqNo
		
	 end	
	 else if exists(select 1 from Inventory.SCReqActivityDetail where SCReqNo = @SCReqNo and statusNo =8)
	 begin
	   if exists (select 1 from Purchase.SCRFQReqDetail where SCReqNo = @SCReqNo)
		update Inventory.SCReqMain set StatusNo = 8 where SCReqNo = @SCReqNo
	   else if exists (select 1 from Purchase.SCOrderAmendmentReqDetail where SCReqNo =@SCReqNo)
		 update Inventory.SCReqMain set StatusNo = 8 where SCReqNo = @SCReqNo
	   else
	     update Inventory.SCReqMain set StatusNo = 1 where SCReqNo = @SCReqNo
		
	 end	
	 else if exists(select 1 from Inventory.SCReqActivityDetail where SCReqNo = @SCReqNo and Statusno = 3 )-- Hold 
		update Inventory.SCReqMain set StatusNo = 3 where SCReqNo = @SCReqNo
		
	 else if exists(select 1 from Inventory.SCReqActivityDetail where SCReqNo = @SCReqNo and Statusno = 4 )--cancel
	 	update Inventory.SCReqMain set StatusNo =4 where SCReqNo = @SCReqNo
	 	
	else if exists(select 1 from Inventory.SCReqActivityDetail where SCReqNo = @SCReqNo and Statusno=7)--shortclose
	 	update Inventory.SCReqMain set StatusNo =7 where SCReqNo = @SCReqNo					
	 	
	else if exists(select 1 from Inventory.SCReqActivityDetail where SCReqNo = @SCReqNo and Statusno=2)--completed
		update Inventory.SCReqMain set StatusNo =2 where SCReqNo = @SCReqNo	

end
else
begin


if exists(select 1 from Inventory.SCReqServiceDetail where SCReqNo = @SCReqNo and statusNo =1)
	 begin
	   if exists (select 1 from Purchase.SCRFQReqDetail where SCReqNo = @SCReqNo)
		update Inventory.SCReqMain set StatusNo = 8 where SCReqNo = @SCReqNo
		
	   else if exists (select 1 from Purchase.SCOrderAmendmentReqDetail where SCReqNo =@SCReqNo)
		 update Inventory.SCReqMain set StatusNo = 8 where SCReqNo = @SCReqNo
	   else
		 update Inventory.SCReqMain set StatusNo = 1 where SCReqNo = @SCReqNo
		
	 end	
	 else if exists(select 1 from Inventory.SCReqServiceDetail where SCReqNo = @SCReqNo and statusNo =8)
	 begin
	   if exists (select 1 from Purchase.SCRFQReqDetail where SCReqNo = @SCReqNo)
		update Inventory.SCReqMain set StatusNo = 8 where SCReqNo = @SCReqNo
		
	   else if exists (select 1 from Purchase.SCOrderAmendmentReqDetail where SCReqNo =@SCReqNo)
		 update Inventory.SCReqMain set StatusNo = 8 where SCReqNo = @SCReqNo
	   else
	     update Inventory.SCReqMain set StatusNo = 1 where SCReqNo = @SCReqNo
		
	 end	
	 else if exists(select 1 from Inventory.SCReqServiceDetail where SCReqNo = @SCReqNo and Statusno = 3 )-- Hold 
		update Inventory.SCReqMain set StatusNo = 3 where SCReqNo = @SCReqNo
		
	 else if exists(select 1 from Inventory.SCReqServiceDetail where SCReqNo = @SCReqNo and Statusno = 4 )--cancel
	 	update Inventory.SCReqMain set StatusNo =4 where SCReqNo = @SCReqNo
	 	
	else if exists(select 1 from Inventory.SCReqServiceDetail where SCReqNo = @SCReqNo and Statusno=7)--shortclose
	 	update Inventory.SCReqMain set StatusNo =7 where SCReqNo = @SCReqNo					
	 	
	else if exists(select 1 from Inventory.SCReqServiceDetail where SCReqNo = @SCReqNo and Statusno=2)--completed
		update Inventory.SCReqMain set StatusNo =2 where SCReqNo = @SCReqNo	
		
end


set @OutputStatus =1
set @Result='Success'
end try
begin catch
set @OutputStatus =0
set @Result=error_message()

		select 0, error_message()
end Catch


END
