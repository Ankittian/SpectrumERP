CREATE PROCEDURE [Utility].[CreateMissingVoucherRef]
@AccountNo int,
@FBNo smallint,
@ReferenceTypeNo tinyint
AS
BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;
begin try
 declare @ReferenceDate smalldatetime ,@SpecificDrAmount  money , @SpecificCrAmount money ,@Amount money , @companyNo Tinyint
	, @CurrencyConversionNo smallint , @DivisibilityRate decimal(9,6) , @MultiplicativeRate decimal(9,6) , @OutputStatusNo tinyint 
	,@OutputStatusRemarks varchar(4000)
 
					
if @ReferenceTypeNo =1
begin
	
	Insert into Finance.accVoucherReferenceDetail ( VoucherDetailNo, ReferenceNo, ReferenceNoYearly, ReferenceDate, CreditDays, DueDate, ReferenceTypeNo, SpecificDrAmount, SpecificCrAmount, BaseDrAmount, BaseCrAmount)
				select  vd.VoucherDetailNo,null ReferenceNo,null ReferenceNoYearly,null ReferenceDate,null CreditDays,null DueDate,1 ReferenceTypeNo, 
				vd.SpecificDrAmount, vd.SpecificCrAmount, vd.BaseDrAmount, vd.BaseCrAmount 
				from Finance.accVoucherMain vm
				inner join  finance.accvoucherdetail vd on vm.VoucherNo = vd.VoucherNo 
				inner join masterdata.maccountmaster am on am.AccountNo = vd.AccountNo 
				where am.IsReferenceApp = 1 and am.accountno = @AccountNo and vm.FBNo = isnull(@FBNo ,vm.FBNo)
				and not exists(select 1 from Finance.accVoucherReferenceDetail where VoucherDetailNo = vd.VoucherDetailNo )


declare cur cursor static for 
	select ab.OpeningDate  ,OpeningDrAmount , OpeningCrAmount ,OpeningAmount , ab.FBNo  , fb.CompanyNo  , CurrencyConversionNo ,DivisibilityRate , MultiplicativeRate 
	from [Finance].[AccountOpeningBalance] ab
	 inner join masterdata.mFBMain fb on fb.FBNo = ab.FBNo 
	  where AccountNo = @AccountNo and ab.FBNo = isnull(@FBNo ,ab.FBNo)
	  and not exists(select 1 from finance.referencemain  where  SourceDocumentFormNo= 133 and accountno = @accountno and 
					fbno = ab.fbno )
open cur
fetch next from cur into @ReferenceDate ,@SpecificDrAmount  , @SpecificCrAmount ,@Amount , @FBNo ,@companyNo , @CurrencyConversionNo ,@DivisibilityRate , @MultiplicativeRate 

while @@fetch_status=0
begin

exec [Finance].[VoucherSaveReferenceDetail] @VoucherDetailNo = null ,@ReferenceNo  =null ,@ReferenceNoYearly  ='OPBL' ,@ReferenceDate  = @ReferenceDate
	,@CreditDays  = null ,@DueDate  = @ReferenceDate ,@ReferenceTypeNo  =3 ,@SpecificDrAmount = @SpecificDrAmount ,@SpecificCrAmount  = @SpecificCrAmount
	,@BaseDrAmount  = @SpecificDrAmount,@BaseCrAmount = @SpecificCrAmount ,@TransactionFormCode ='FMOB', @SourceDocumentNo = @AccountNo 
	,@SourceDocumentDate = @ReferenceDate ,@SourceDocumentNoYearly ='OPBL' ,@Remarks =null , @SpecificReferenceAmount = @Amount
	,@BaseReferenceAmount = @Amount , @SourceDocumentFormCode ='FMOB',@AccountNo = @AccountNo ,@CompanyNo = @companyNo 
	,@ByToNo =1 , @CurrencyConversionNo = @CurrencyConversionNo, @DivisibilityRate  = @DivisibilityRate ,@MultiplicativeRate = @MultiplicativeRate
	,@FBNo = @FBNo ,@OutputStatusNo = @OutputStatusNo output
	,@OutputStatusRemarks = @OutputStatusRemarks output
	if @OutputStatusNo= 0
		raiserror(@OutputStatusRemarks,16,-1)

fetch next from cur into @ReferenceDate ,@SpecificDrAmount  , @SpecificCrAmount ,@Amount , @FBNo ,@companyNo , @CurrencyConversionNo ,@DivisibilityRate , @MultiplicativeRate 

end
close cur 
deallocate cur

end
else if @ReferenceTypeNo = 2
begin
	
	
 	DECLARE @iRow int,@iCount int =1 , @VoucherDetailNo int ,@ReferenceDetailNo int
	select ROW_NUMBER () over (order by vd.VoucherDetailNo) SNNo ,  vd.VoucherDetailNo into #VOucher
	from Finance.accVoucherMain vm
	inner join  finance.accvoucherdetail vd on vm.VoucherNo = vd.VoucherNo 
	inner join masterdata.maccountmaster am on am.AccountNo = vd.AccountNo 
	where am.IsReferenceApp = 1 and am.accountno = @AccountNo and vm.FBNo =  isnull(@FBNo ,vm.FBNo) 
	and not exists(select 1 from Finance.accVoucherReferenceDetail where VoucherDetailNo = vd.VoucherDetailNo )
	set @iRow = @@ROWCOUNT 
	while @iRow >= @iCount
	begin
		select @VoucherDetailNo = VoucherDetailNo from #Voucher where SNNo = @iCount 
		
		Insert into Finance.accVoucherReferenceDetail ( VoucherDetailNo, ReferenceNo, ReferenceNoYearly, ReferenceDate, CreditDays, DueDate, ReferenceTypeNo, SpecificDrAmount, SpecificCrAmount, BaseDrAmount, BaseCrAmount)
				select  vd.VoucherDetailNo,null ReferenceNo,null ReferenceNoYearly,null ReferenceDate,null CreditDays,null DueDate,1 ReferenceTypeNo, 
				vd.SpecificDrAmount, vd.SpecificCrAmount, vd.BaseDrAmount, vd.BaseCrAmount 
				from Finance.accVoucherMain vm
				inner join  finance.accvoucherdetail vd on vm.VoucherNo = vd.VoucherNo 
				inner join masterdata.maccountmaster am on am.AccountNo = vd.AccountNo 
				where am.IsReferenceApp = 1 and am.accountno = @AccountNo and vm.FBNo =  isnull(@FBNo ,vm.FBNo) 
				and not exists(select 1 from Finance.accVoucherReferenceDetail where VoucherDetailNo = vd.VoucherDetailNo )
				and vd.VoucherDetailNo = @VoucherDetailNo 
		set @ReferenceDetailNo = IDENT_CURRENT ('Finance.accVoucherReferenceDetail')
			
			
			Exec utility.ConvertOnAccountToNewReference @ReferenceDetailNo	
	
		set @iCount = @iCount + 1
	end
	 
 
declare curRef cursor static for 
	select ab.OpeningDate  ,OpeningDrAmount , OpeningCrAmount ,OpeningAmount , ab.FBNo  , fb.CompanyNo  , CurrencyConversionNo ,DivisibilityRate , MultiplicativeRate 
	from [Finance].[AccountOpeningBalance] ab
	 inner join masterdata.mFBMain fb on fb.FBNo = ab.FBNo 
	  where AccountNo = @AccountNo and ab.FBNo = isnull(@FBNo ,ab.FBNo)
	  and not exists(select 1 from finance.referencemain  where  SourceDocumentFormNo= 133 and accountno = @accountno and 
					fbno = ab.fbno )
open curRef
fetch next from curRef into @ReferenceDate ,@SpecificDrAmount  , @SpecificCrAmount ,@Amount , @FBNo ,@companyNo , @CurrencyConversionNo ,@DivisibilityRate , @MultiplicativeRate 

while @@fetch_status=0
begin

exec [Finance].[VoucherSaveReferenceDetail] @VoucherDetailNo = null ,@ReferenceNo  =null ,@ReferenceNoYearly  ='OPBL' ,@ReferenceDate  = @ReferenceDate
	,@CreditDays  = null ,@DueDate  = @ReferenceDate ,@ReferenceTypeNo  =3 ,@SpecificDrAmount = @SpecificDrAmount ,@SpecificCrAmount  = @SpecificCrAmount
	,@BaseDrAmount  = @SpecificDrAmount,@BaseCrAmount = @SpecificCrAmount ,@TransactionFormCode ='FMOB', @SourceDocumentNo = @AccountNo 
	,@SourceDocumentDate = @ReferenceDate ,@SourceDocumentNoYearly ='OPBL' ,@Remarks =null , @SpecificReferenceAmount = @Amount
	,@BaseReferenceAmount = @Amount , @SourceDocumentFormCode ='FMOB',@AccountNo = @AccountNo ,@CompanyNo = @companyNo 
	,@ByToNo =1 , @CurrencyConversionNo = @CurrencyConversionNo, @DivisibilityRate  = @DivisibilityRate ,@MultiplicativeRate = @MultiplicativeRate
	,@FBNo = @FBNo ,@OutputStatusNo = @OutputStatusNo output
	,@OutputStatusRemarks = @OutputStatusRemarks output
	if @OutputStatusNo= 0
		raiserror(@OutputStatusRemarks,16,-1)

fetch next from curRef into @ReferenceDate ,@SpecificDrAmount  , @SpecificCrAmount ,@Amount , @FBNo ,@companyNo , @CurrencyConversionNo ,@DivisibilityRate , @MultiplicativeRate 

end
close curRef
deallocate curRef


 end

end try

begin catch
	select 0,ERROR_MESSAGE ()
end catch
 
END
