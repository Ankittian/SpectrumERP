CREATE PROCEDURE [Information].[PendingPurchaseRequestForDept] 
	-- Add the parameters for the stored procedure here
	@CompanyNo tinyint,
	@DocumentDate smalldatetime,
	@DeptNo int
AS
BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;

    -- Insert statements for procedure here
	
    
	declare @str varchar(max)
    select @str=masterdata.JSForInformation('MMPR')
    set @str=@str+masterdata.CSSForInformation('MMPR')
	set @str=@str+'<table class="altrowstable" id="alternatecolorPUPRInfo">'
	set @str=@str+'<Caption><Font color="Maroon"><h3 align="Left"><b>Pending Purchase Request</b></h3></Font></Caption>'
    set @str=@str+'<tr><th>Sl. No.</th><th>Division</th><th>Document No.</th><th>Document Date</th><th>Material Request No.</th><th>Material Request Date</th><th>Request Type</th><th>Requested By</th><th>Created By</th><th>Authorized By</th></tr>'
	set @str=@str
				 +(REPLACE( 
				    REPLACE(
				    REPLACE(
				    REPLACE(
				    REPLACE(
				    replace(
					replace(
					replace(
					replace(
					replace((    
								 Select  ROW_NUMBER() over ( Order by IM.DocumentDate desc) as td1
										,Isnull(DM.DivisionName,'') td2
										,IM.DocumentNoYearly td3
										,convert(varchar(20),IM.DocumentDate,103) td4
										,IRD.DocumentNoYearly  td5
										,convert(varchar(20),IRD.DocumentDate,103)  td6
										,IT.IndentType  td7
										,IRD.RequestedBy td8
										,L.PrintingName td9
										,isnull(L1.PrintingName,'')  td10		    
								From Inventory.IndentMain IM
										Inner Join Inventory.IndentItemDetail ID ON ID.IndentNo=IM.IndentNo
										Inner Join Inventory.IndentRequisitionMain IRD On IM.RefDocumentNo=IRD.IndentRequisitionNo 
										                                               and IM.IndentAgainstNo=3
										Inner Join GlobalData.IndentType IT ON IT.IndentTypeNo=IM.IndentTypeNo
										Inner Join payroll.tblDeptMaster DT ON DT.DeptNo=IM.DeptNo
										Left Join payroll.tblDivisionMaster DM ON DM.DivisionNo=DT.DivisionNo
										Inner Join Security.Login L ON L.LoginNo=IM.CreatedBy
										Left Join Security.Login L1 On L1.LoginNo=IM.AuthorizedBy
								Where (ID.IndentQty-isnull(ID.POQty,0)) > 0 -----Pending Purchase Request
										  And IM.DeptNo=@DeptNo 
										  And IM.CompanyNo=@CompanyNo 
										  And IM.DocumentDate<=@DocumentDate
								Group by IM.DocumentNoYearly ,IM.DocumentDate ,IRD.DocumentNoYearly ,IRD.DocumentDate
										,IT.IndentType ,L.PrintingName ,L1.PrintingName ,DM.DivisionName ,IRD.RequestedBy
								for xml path('tr')
			              ),'<td1>','<td style="text-align: center;">'),'<td2>','<td>')
								 ,'<td3>','<td>')
								,'<td4>','<td style="text-align: center;">')
								 ,'<td5>','<td>')
								,'<td6>','<td  style="text-align: center;">')
								,'<td7>','<td>')
								,'<td8>','<td>')
								,'<td9>','<td>')
								,'<td10>','<td>'))
								
				   
	set @str=@str+'</table>'
	
	select @str
END
