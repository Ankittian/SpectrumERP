CREATE PROCEDURE [Customize].[MDNEFinancePosting] 
	@OtherDataInXml xml ,
	@CompanyNo tinyint  
AS
BEGIN

	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;

begin try 

create table #Posting ([Add to Basic] bit,Particular varchar(500),[Debit Amount] decimal(15,2) , 
		[Credit Amount] decimal(15,2),[Account Name] int, [Ref. Detail] varchar(50),[Cost Center] varchar(50),
		[Amount] decimal(15,2), [Type] tinyint,SlNo tinyint ,MiscChargeNo int , TaxCodeMiscChargesDetailNo int,
			ItemNos varchar(200))




 declare @idoc int
EXEC sp_xml_preparedocument @idoc OUTPUT, @OtherDataInXml 

	insert into #Posting ([Add to Basic] , Particular,[Debit Amount],[Credit Amount],[Account Name],SlNo , [Amount],[Type])
	
	select 0 , case when [Debit Amount] >0 then 'Debited Account' else 'Creditted Account' end
		, [Debit Amount],[Credit Amount],[Account Name], ROW_NUMBER () over( order by [Account Name])
			, ISNULL([Debit Amount],0) + ISNULL([Credit Amount],0) 
			,case when [Debit Amount] >0 then 1 else 2 end 
 FROM OPENXML (@idoc, '/DocumentElement/Table',2)  
	with ([Type] tinyint,[Account Name] Int,[Debit Amount] decimal(12,2),[Credit Amount] decimal(12,2)) cte
		where  [Account Name] is not null

EXEC sp_xml_removedocument @idoc
 
  
 if exists(select 1 from #Posting having abs(SUM([Credit Amount])- SUM([Debit Amount]))<=.5)
 begin
	update PD set [Debit Amount] = [Debit Amount] + ISNULL(diff,0)
		from
		#Posting PD
		cross join (select SUM([Credit Amount])- SUM([Debit Amount]) Diff from #Posting) tt
		where PD.SlNo  = 1
	
	update PD set [Credit Amount] = [Credit Amount] + ISNULL(diff,0)
		from
		#Posting PD
		cross join (select SUM([Debit Amount])- SUM([Credit Amount]) Diff from #Posting) tt
		where PD.SlNo  = 1
			
 end
 
 
select * from #Posting
 where isnull([Debit Amount],0) >0 or isnull([Credit Amount],0) >0
   
   
   
END TRY
	
	BEGIN CATCH
		select ERROR_MESSAGE()
		INSERT INTO [masterdata].[mErrorLog]
			   ([ErrorDate]
			   ,[ErrorNo]
			   ,[ErrorSeverity]
			   ,[ErrorState]
			   ,[ErrorProcedure]
			   ,[ErrorLine]
			   ,[ErrorDescription]
			   ,[UserName]
			   ,[FormCode])
		VALUES
			   (getdate()
			   ,ERROR_NUMBER()
			   ,ERROR_SEVERITY()
			   ,ERROR_STATE()
			   ,ERROR_PROCEDURE()
			   ,ERROR_LINE()
			   ,ERROR_MESSAGE()
			   ,SUSER_SNAME()
			   ,'MDNE')
	END CATCH
END
