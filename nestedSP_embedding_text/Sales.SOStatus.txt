Stored procedure Sales.SOStatus: To Change Status OF Sales Order. Called from Inventory.IGRNDelete, Sales.SDSODelete.
Writes: masterdata.mCostCenterMasterCompanyDetail, Sales.SalesOrderAmendmentItemDetail, Sales.SalesOrderAmendmentMain, Sales.SalesOrderItemDetail, Sales.SalesOrderMain
Reads: Inventory.GRNItemIndentDetail, Inventory.GRNMain, masterdata.mCostCenterMaster, Sales.CommercialInvoiceItemDetails, Sales.CommercialInvoiceMain, Sales.DispatchChallanItemDetail, Sales.InvoiceItemDetails, Sales.InvoiceMain, Sales.SalesOrderAmendmentItemDetail, Sales.SalesOrderAmendmentMain, Sales.SalesOrderItemDetail, Sales.SalesOrderMain, Sales.Setting
Parameters: @SalesOrderNo int, @ItemNo int, @MakeNo smallint, @CombinationNo Int, @StatusNo tinyint, @OutputStatus bit OUTPUT, @Result varchar(100) OUTPUT

-- Description:	To Change Status OF Sales Order
-- =============================================

CREATE PROCEDURE [Sales].[SOStatus] 	
	@SalesOrderNo int , 
	@ItemNo int ,
	@MakeNo smallint,	
	@CombinationNo Int,
	@StatusNo tinyint,
	@OutputStatus bit output,
	@Result varchar(100) output	
AS
BEGIN
begin try

Declare @IsSOCloseAfterTolerance Bit

Select @IsSOCloseAfterTolerance=IsNull(ST.IsSOCloseAfterTolerance,0)
From Sales.Setting ST
Inner Join Sales.SalesOrderMain SOM on SOM.CompanyNo=ST.CompanyNo
Where SOM.SalesOrderNo=@SalesOrderNo

SEt @IsSOCloseAfterTolerance = NullIf(@IsSOCloseAfterTolerance,0)

If Exists (Select 1 From Sales.SalesOrderMain Where SalesOrderNo=@SalesOrderNo
			And FormTag='SDSO')
BEGIN			

	Declare @DispatchQty Decimal(12,3),@SOCombinationNo Int 
	
	If Exists(
				Select 1
				From Sales.SalesOrderItemDetail SD
				Where SD.SalesOrderNo=@SalesOrderNo
				And SD.ItemNo=@ItemNo
				And ISNULL(SD.MakeNo,0)=ISNULL(@MakeNo,0)
				And SD.ItemAttributeBookingNatureNo=1
			 ) -- Fixed
	BEGIN
			Set @SOCombinationNo=@CombinationNo
	END
	ELSE
	BEGIN	-- Variant
			Select @SOCombinationNo=SD.CombinationNo
			From Sales.SalesOrderItemDetail SD
			Where SD.SalesOrderNo=@SalesOrderNo
			And SD.ItemNo=@ItemNo
			And ISNULL(SD.MakeNo,0)=ISNULL(@MakeNo,0)
	END

	
		Select @DispatchQty=Qty-isnull(ReturnQty,0)-isnull(CancelQty,0) --(Dispatch - Return - Cancel)
		From [Sales].[FuncSOPreDispatchQty](@SalesOrderNo,@ItemNo,@MakeNo,@SOCombinationNo)

	
	if @StatusNo = 2 
	BEGIN								
			-- Formula
			-- If (SOQty - Tolerance Minus) <= Dispatch Qty (DC,EI,CI) Then Status will be closed
			-- New Requirement from ACPL
			-- Don't Close SO untill we deliver upto tolerance qty
			-- We are taking one setting for it.
			-- New Formula 
			-- If New Setting is enabled then we can dispatch upto  tolerance (+) qty
			
			UPDATE Sales.SalesOrderItemDetail
			SET StatusNo =2 
			Where SalesOrderNo=@SalesOrderNo AND ItemNo = @ItemNo And ISNULL(MakeNo,0) =ISNULL(@MakeNo,0) 
			And ISNULL(CombinationNo,0)=ISNULL(@SOCombinationNo,0)			
			And Quantity -
			Isnull(Case When ToleranceBasisNo = 1 Then ToleranceMinus Else ToleranceMinus * Quantity * 0.01 end,0) 
			+Isnull(Case When ToleranceMinus Is Null And @IsSOCloseAfterTolerance =1 Then Case When ToleranceBasisNo = 1 Then TolerancePlus Else TolerancePlus * Quantity * 0.01 end Else 0 End,0) 
			< = isnull(@DispatchQty,0)
						
			UPDATE SD
			SET StatusNo =2 
			From Sales.SalesOrderAmendmentItemDetail SD
			Inner Join Sales.SalesOrderAmendmentMain SM on SM.SalesOrderAmendmentNo=SD.SalesOrderAmendmentNo
														And SM.SalesOrderNo=@SalesOrderNo
			Where  SD.ItemNo = @ItemNo And ISNULL(SD.MakeNo,0) =ISNULL(@MakeNo,0) 
			And ISNULL(SD.CombinationNo,0)=ISNULL(@SOCombinationNo,0)			
			And SD.Quantity 
			-isnull(case when SD.ToleranceBasisNo = 1 then SD.ToleranceMinus else SD.ToleranceMinus * Quantity * 0.01 end,0) 
			+Isnull(Case When SD.ToleranceMinus Is Null And @IsSOCloseAfterTolerance =1 Then Case When SD.ToleranceBasisNo = 1 Then SD.TolerancePlus Else SD.TolerancePlus * Quantity * 0.01 end Else 0 End,0)
			< = isnull(@DispatchQty,0)
			
			UPDATE Sales.SalesOrderItemDetail
			SET StatusNo =1
			Where SalesOrderNo=@SalesOrderNo AND ItemNo = @ItemNo And ISNULL(MakeNo,0) =ISNULL(@MakeNo,0) 
			And ISNULL(CombinationNo,0)=ISNULL(@SOCombinationNo,0)			
			And Quantity -isnull(case when ToleranceBasisNo = 1 then ToleranceMinus else ToleranceMinus * Quantity * 0.01 end,0) 
			+Isnull(Case When ToleranceMinus Is Null And @IsSOCloseAfterTolerance =1 Then Case When ToleranceBasisNo = 1 Then TolerancePlus Else TolerancePlus * Quantity * 0.01 end Else 0 End,0)
			> isnull(@DispatchQty,0)
			--And isnull(@DispatchQty,0)=0
			And StatusNo In (2,8)

			UPDATE SD
			SET StatusNo =1 
			From Sales.SalesOrderAmendmentItemDetail SD
			Inner Join Sales.SalesOrderAmendmentMain SM on SM.SalesOrderAmendmentNo=SD.SalesOrderAmendmentNo
														And SM.SalesOrderNo=@SalesOrderNo
			Where  SD.ItemNo = @ItemNo And ISNULL(SD.MakeNo,0) =ISNULL(@MakeNo,0) 
			And ISNULL(SD.CombinationNo,0)=ISNULL(@SOCombinationNo,0)			
			And SD.Quantity -isnull(case when SD.ToleranceBasisNo = 1 then SD.ToleranceMinus else SD.ToleranceMinus * Quantity * 0.01 end,0) 
			+Isnull(Case When SD.ToleranceMinus Is Null And @IsSOCloseAfterTolerance =1 Then Case When SD.ToleranceBasisNo = 1 Then SD.TolerancePlus Else SD.TolerancePlus * Quantity * 0.01 end Else 0 End,0)
			> isnull(@DispatchQty,0)			
			--And isnull(@DispatchQty,0)=0
			And SD.StatusNo In (2,8)
			
			UPDATE Sales.SalesOrderItemDetail
			SET StatusNo =8
			Where SalesOrderNo=@SalesOrderNo AND ItemNo = @ItemNo And ISNULL(MakeNo,0) =ISNULL(@MakeNo,0) 
			And ISNULL(CombinationNo,0)=ISNULL(@SOCombinationNo,0)			
			And ISNULL(@DispatchQty,0)>0
			And StatusNo=1

			UPDATE SD
			SET StatusNo =8 
			From Sales.SalesOrderAmendmentItemDetail SD
			Inner Join Sales.SalesOrderAmendmentMain SM on SM.SalesOrderAmendmentNo=SD.SalesOrderAmendmentNo
														And SM.SalesOrderNo=@SalesOrderNo
			Where  SD.ItemNo = @ItemNo And ISNULL(SD.MakeNo,0) =ISNULL(@MakeNo,0) 
			And ISNULL(SD.CombinationNo,0)=ISNULL(@SOCombinationNo,0)			
			And ISNULL(@DispatchQty,0)>0
			And SD.StatusNo=1		

			 
	END -- if @StatusNo = 2 

	 if @StatusNo <> 2
	 begin
	 
		Update ID		
		Set StatusNo=@StatusNo
		From Sales.SalesOrderAmendmentItemDetail ID
		Inner Join Sales.SalesOrderAmendmentMain SAM on SAM.SalesOrderAmendmentNo=ID.SalesOrderAmendmentNo
		Where SAM.SalesOrderNo=@SalesOrderNo AND ItemNo = @ItemNo And ISNULL(MakeNo,0) =ISNULL(@MakeNo,0) 
		And ISNULL(CombinationNo,0)=ISNULL(@SOCombinationNo,0)					
		
		Update Sales.SalesOrderItemDetail
		Set StatusNo=@StatusNo
		Where SalesOrderNo=@SalesOrderNo AND ItemNo = @ItemNo And ISNULL(MakeNo,0) =ISNULL(@MakeNo,0) 
		And ISNULL(CombinationNo,0)=ISNULL(@SOCombinationNo,0)					
	 end

	 If exists(select 1 from Sales.SalesOrderItemDetail where SalesOrderNo = @SalesOrderNo And statusNo In (1,8))
		-- If Any item is in initial or authorized mode then 1st if
	 begin
	   if Exists ( 
				   Select 1
				   From Sales.DispatchChallanItemDetail DID
				   Inner Join Sales.SalesOrderAmendmentMain SAM on SAM.SalesOrderAmendmentNo=DID.SalesOrderAmendmentNo
				   Where SAM.SalesOrderNo=@SalesOrderNo
				 ) Or
		   Exists(
		   		   Select 1
				   From Sales.CommercialInvoiceItemDetails ID
				   Inner join Sales.CommercialInvoiceMain CM on CM.CommercialInvoiceNo=ID.CommercialInvoiceNo
				   Inner Join Sales.SalesOrderAmendmentMain SAM on SAM.SalesOrderAmendmentNo=ID.SalesOrderAmendmentNo
				   Where SAM.SalesOrderNo=@SalesOrderNo
				   And CM.SourceDocumentNo=2				   
				  ) Or
		   Exists(
		   		   Select 1
				   From Sales.InvoiceItemDetails ID
				   Inner join Sales.InvoiceMain CM on CM.InvoiceNo=ID.InvoiceNo
				   Inner Join Sales.SalesOrderAmendmentMain SAM on SAM.SalesOrderAmendmentNo=ID.SalesOrderAmendmentNo
				   Where SAM.SalesOrderNo=@SalesOrderNo
				   And CM.SourceDocumentNo in (2,13,23)
				  ) Or
			Exists(
		   		   Select 1
				   From Inventory.GRNItemIndentDetail GIID
				   Inner join Inventory.GRNMain GRNM on GRNM.GRNNo =GIID.GRNNo
				   Inner Join Sales.SalesOrderAmendmentMain SAM on SAM.SalesOrderAmendmentNo=GIID.SalesOrderAmendmentNo
				   Where SAM.SalesOrderNo=@SalesOrderNo
						And GRNM.GRNTypeNo= 2				   
				  )
		BEGIN
			  update Sales.SalesOrderMain set StatusNo = 8 where  SalesOrderNo = @SalesOrderNo
			  update Sales.SalesOrderAmendmentMain set StatusNo = 8 where  SalesOrderNo = @SalesOrderNo		
		END
		ELSE
		BEGIN
			   update Sales.SalesOrderMain set StatusNo = 1 where  SalesOrderNo = @SalesOrderNo
			   update Sales.SalesOrderAmendmentMain set StatusNo = 1 where  SalesOrderNo = @SalesOrderNo
		END		
	END
	 Else if exists(select 1 from Sales.SalesOrderItemDetail where  SalesOrderNo = @SalesOrderNo and Statusno = 3 )-- Hold 
		begin
			update Sales.SalesOrderMain set StatusNo = 3 where  SalesOrderNo = @SalesOrderNo
			update Sales.SalesOrderAmendmentMain set StatusNo = 3 where  SalesOrderNo = @SalesOrderNo
		end
		
	 else if exists(select 1 from Sales.SalesOrderItemDetail where  SalesOrderNo = @SalesOrderNo and Statusno = 4 )--cancel
	 	begin
			update Sales.SalesOrderMain set StatusNo =4 where  SalesOrderNo = @SalesOrderNo
			update Sales.SalesOrderAmendmentMain set StatusNo = 4 where  SalesOrderNo = @SalesOrderNo
		end
	 	
	else if exists(select 1 from Sales.SalesOrderItemDetail where  SalesOrderNo = @SalesOrderNo and Statusno=7)--shortclose
	 	begin
			update Sales.SalesOrderMain set StatusNo = 7 where  SalesOrderNo = @SalesOrderNo
			update Sales.SalesOrderAmendmentMain set StatusNo = 7 where  SalesOrderNo = @SalesOrderNo
		end				
	
	else if exists(select 1 from Sales.SalesOrderItemDetail where  SalesOrderNo = @SalesOrderNo and Statusno=2)--completed
		begin
			update Sales.SalesOrderMain set StatusNo = 2 where  SalesOrderNo = @SalesOrderNo
			update Sales.SalesOrderAmendmentMain set StatusNo =  2 where  SalesOrderNo = @SalesOrderNo
		end
End
	
	Declare @CompanyNo TinyInt
	Select @CompanyNo=companyNo From Sales.SalesOrderMain Where SalesOrderNo=@SalesOrderNo
		
	-- for RCF	
	If exists(select 1 from Sales.Setting where CompanyNo =@CompanyNo and IsCreateCostCenterINSO =1)		
	And Exists (Select 1 From Sales.SalesOrderMain where SalesOrderNo=@SalesOrderNo and StatusNo in (2,7))
	BEGIN
			-- for RCF
			 Declare @CostCenterNo int
			 Select @CostCenterNo = CostCenterNo 
			 From masterdata.mCostCenterMaster cm 
			 Inner Join  Sales.SalesOrderMain sm on cm.CostCenterName = sm.CustomerOrderNo And sm.SalesOrderNo = @SalesOrderNo
			 
			 Update masterdata.mCostCenterMasterCompanyDetail 
			 Set Inactive =1 
			 Where CostCenterNo = @CostCenterNo --and CompanyNo = @CompanyNo 
	END			 
			 		
set @OutputStatus =1
set @Result='Success'


end try

begin catch
set @OutputStatus =0
set @Result=error_message()

		select 0, error_message()
end Catch
END
