Stored procedure Purchase.POStatus: To Change Status OF PO. Called from Inventory.IGRNDelete.
Writes: Purchase.POAmendmentItemDetail, Purchase.POAmendmentMain, Purchase.POItemDetail, Purchase.POMain
Reads: DiscreteRM.SINLotOrderDetail, DiscreteRM.SINMain, Inventory.GRNItemDetail, Inventory.GRNMain, masterdata.mItemMaster, Purchase.POAmendmentMain, Purchase.POIndentDetail, Purchase.POItemDetail, Purchase.POMain, Purchase.POQtySetting, Purchase.SCBOQItemDetail, Purchase.SCBOQMain
Parameters: @PONo int, @ItemNo int, @MakeNo smallint, @TechnicalGradeNo tinyint, @StatusNo tinyint, @OutputStatus bit OUTPUT, @Result varchar(100) OUTPUT
Calls procedures: Inventory.IndentStatus

Code:
-- =============================================
-- Author:		Akhil
-- Create date: 20 feb 2012
-- Description:	To Change Status OF PO
-- =============================================
-- [Purchase].[POStatus]  415, 22,null,null,2,null,null
CREATE PROCEDURE [Purchase].[POStatus]
	-- Add the parameters for the stored procedure here
	@PONo int , 
	@ItemNo int ,
	@MakeNo smallint,
	@TechnicalGradeNo tinyint,
	@StatusNo tinyint,
	@OutputStatus bit output,
	@Result varchar(100) output 
AS
BEGIN
begin try


-- 	select * from GlobalData.POBillingQty pq
--created by akhil
--- case pqs.POBalancedQty when 2 then ChallanQty when 4 then ReceivedQty else AcceptedQty end
			
Declare @GRnQty decimal(12,3), @SINQty decimal(12,3)
--Select @GRNQty = SUM((FirstCF*AcceptedQty)/SecondCF)  -- commented on 13 feb 2013
Select @GRNQty = SUM(case pqs.POClosedOnQty when  1 then ChallanQty -ISNULL(RejectedQty,0) 
					when  2 then ChallanQty -ISNULL(RejectedQty,0) 
					when 4 then ReceivedQty -ISNULL(RejectedQty,0) else AcceptedQty end) 
				From [Inventory].[GRNItemDetail] gid
				inner join masterdata.mItemMaster im on  im.ItemNo= gid.ItemNo 
				inner join Inventory.GRNMain gm on gm.GRNNo = gid.GRNNo 
				inner join Purchase.POQtySetting pqs on pqs.CompanyNo = gm.CompanyNo and im.ItemTypeNo = pqs.ItemTypeNo 
				where (RouteIdentity is null or RouteIdentity % .1=0) 
					and POAmendmentNo in(Select POAmendmentNo From Purchase.POAmendmentMain PM 
										where PONo = @PONo and isnull(PM.IsManuallyClosing,0)=0)
				AND gid.ItemNo = @ItemNo And ISNULL(MakeNo,0) = ISNULL(@MakeNo,0)
				and  ISNULL(TechnicalGradeNo,0) = ISNULL(@TechnicalGradeNo ,0)

				
Select @SINQty =  SUM(case pqs.POClosedOnQty  when 1 then ChallanQty when 2 then ChallanQty else Qty end) 
				From DiscreteRM.SINLotOrderDetail gid
				inner join masterdata.mItemMaster im on  im.ItemNo= gid.ItemNo
				inner join DiscreteRM.SINMain si on si.SINNo = gid.SINNo 
				inner join Purchase.POQtySetting pqs on pqs.CompanyNo = si.CompanyNo and im.ItemTypeNo = pqs.ItemTypeNo 
				where POAmendmentNo in(Select POAmendmentNo From Purchase.POAmendmentMain PM 
										where PONo = @PONo and isnull(PM.IsManuallyClosing,0)=0)
				AND gid.ItemNo = @ItemNo And ISNULL(MakeNo,0) = ISNULL(@MakeNo,0)

set @GRNQty= isnull( @GRNQty,0) + isnull(@SINQty,0)


if @StatusNo = 2 or @StatusNo =4
BEGIN

	UPDATE Purchase.POItemDetail SET StatusNo =2 
	Where PONo=@PONo AND ItemNo = @ItemNo And (ISNULL(MakeNo,0) =ISNULL(@MakeNo,0)  or makeno is null)
			and  ISNULL(TechnicalGradeNo,0) = ISNULL(@TechnicalGradeNo ,0)
			and Quantity -isnull(case when ToleranceBasisNo = 1 then IsNull(ToleranceMinus,0) else IsNull(ToleranceMinus,0) * Quantity * 0.01 end,0) < = isnull(@GRNQty,0) 

	UPDATE Purchase.POAmendmentItemDetail SET StatusNo =2 
	Where POAmendmentNo In (SELECT POAmendmentNo FROM Purchase.POAmendmentMain WHERE PONo=@PONo )
	AND ItemNo = @ItemNo And (ISNULL(MakeNo,0) =ISNULL(@MakeNo,0)  or makeno is null)
	and  ISNULL(TechnicalGradeNo,0) = ISNULL(@TechnicalGradeNo ,0)
		and Quantity - isnull(case when ToleranceBasisNo = 1 then IsNull(ToleranceMinus,0) else IsNull(ToleranceMinus,0) * Quantity * 0.01 end,0) < = isnull(@GRNQty,0) 
					
	Update Purchase.POItemDetail set StatusNo = 1
					where PONo = @PONo and ItemNo = @ItemNo and (isnull(MakeNo,0) = isnull(@MakeNo,0) or makeno is null)
					and  ISNULL(TechnicalGradeNo,0) = ISNULL(@TechnicalGradeNo ,0)
					and Quantity -isnull(case when ToleranceBasisNo = 1 then IsNull(ToleranceMinus,0) else IsNull(ToleranceMinus,0) * Quantity * 0.01 end,0) > isnull(@GRNQty,0)  and StatusNo =2
					
	Update Purchase.POAmendmentItemDetail set StatusNo = 1
					where POAmendmentNo In (SELECT POAmendmentNo FROM Purchase.POAmendmentMain WHERE PONo=@PONo )
					and ItemNo = @ItemNo and (isnull(MakeNo,0) = isnull(@MakeNo,0) or makeno is null)
					and  ISNULL(TechnicalGradeNo,0) = ISNULL(@TechnicalGradeNo ,0)
					and Quantity -isnull(case when ToleranceBasisNo = 1 then IsNull(ToleranceMinus,0) else IsNull(ToleranceMinus,0) * Quantity * 0.01 end,0) > isnull(@GRNQty,0)  and StatusNo =2

	Update Purchase.POItemDetail set StatusNo = 8
					where PONo = @PONo and ItemNo = @ItemNo and (isnull(MakeNo,0) = isnull(@MakeNo,0) or makeno is null)
					and  ISNULL(TechnicalGradeNo,0) = ISNULL(@TechnicalGradeNo ,0)
					and isnull(@GRNQty,0)>0   and StatusNo =1 
					
	Update Purchase.POAmendmentItemDetail set StatusNo = 8
					where POAmendmentNo In (SELECT POAmendmentNo FROM Purchase.POAmendmentMain WHERE PONo=@PONo )
					and ItemNo = @ItemNo and (isnull(MakeNo,0) = isnull(@MakeNo,0) or makeno is null)
					and  ISNULL(TechnicalGradeNo,0) = ISNULL(@TechnicalGradeNo ,0)
					and  isnull(@GRNQty,0) >0 and StatusNo =1 
										
   --- Updating Item Detail Status to In Progress, if GRN has already created (Only for case of PO Manully closing) 
   	if exists(select 1 from Purchase.POItemDetail PID
			  inner join Purchase.POMain PM on PM.PONo = PID.PONo 
			  Where PID.PONo = @PONo and PID.ItemNo = @ItemNo and (isnull(PID.MakeNo,0) = isnull(@MakeNo,0) )
			  and  ISNULL(PID.TechnicalGradeNo,0) = ISNULL(@TechnicalGradeNo ,0)	
			  and PID.statusNo =1
			  and IsNULL(PM.IsManuallyClosing,0) = 1 )
	 begin
	   if exists (
				  Select 1 
				  From Inventory.GRNItemDetail GID
				  Inner Join Purchase.POAmendmentMain PA on PA.POAmendmentNo=GID.POAmendmentNo and IsNULL(PA.IsManuallyClosing,0) = 1 
				  inner join Purchase.POItemDetail PID on PA.PONo = PID.PONo
												   and PID.ItemNo = GID.ItemNo and (isnull(PID.MakeNo,0) = isnull(GID.MakeNo,0))
												   and  ISNULL(PID.TechnicalGradeNo,0) = ISNULL(GID.TechnicalGradeNo ,0)
				  Where PA.PONo = @PoNo
				  And PID.ItemNo = @ItemNo and (isnull(PID.MakeNo,0) = isnull(@MakeNo,0) or PID.makeno is null)
												   and  ISNULL(PID.TechnicalGradeNo,0) = ISNULL(@TechnicalGradeNo ,0)	
												   and PID.statusNo =1	
				  )
		  or exists (
					 Select 1 
					 From DiscreteRM.SINLotOrderDetail SLOD
					 inner join Purchase.POAmendmentMain POAM on POAM.POAmendmentNo = SLOD.POAmendmentNo and IsNULL(POAM.IsManuallyClosing,0) = 1 
					 inner join Purchase.POItemDetail PAID on PAID.PONo = POAM.PONo
												   And PAID.ItemNo = SLOD.ItemNo and (isnull(PAID.MakeNo,0) = isnull(SLOD.MakeNo,0))					 
					 Where POAM.PONo = @PoNo
					 And PAID.ItemNo = @ItemNo and (isnull(PAID.MakeNo,0) = isnull(@MakeNo,0) or PAID.makeno is null)	
												   and PAID.statusNo =1	
					 )				
		 begin
				update PID 
				set PID.statusNo = 8 
				from Purchase.POItemDetail PID 
				inner join Purchase.POMain PM on PM.PONo = PID.PONo 
				where 	PID.PONo = @PoNo
					and PID.ItemNo = @ItemNo and (isnull(PID.MakeNo,0) = isnull(@MakeNo,0))
												   and  ISNULL(PID.TechnicalGradeNo,0) = ISNULL(@TechnicalGradeNo ,0)	
												   and PID.statusNo =1
				and IsNULL(PM.IsManuallyClosing,0) = 1									 

				update PID 
				set PID.statusNo = 8 
				from Purchase.POAmendmentItemDetail PID 
				inner join Purchase.POAmendmentMain PM on PM.POAmendmentNo = PID.POAmendmentNo 
				where 	PM.PONo = @PoNo
					and PID.ItemNo = @ItemNo and (isnull(PID.MakeNo,0) = isnull(@MakeNo,0))
					and  ISNULL(PID.TechnicalGradeNo,0) = ISNULL(@TechnicalGradeNo ,0)	
					and PID.statusNo =1
					and IsNULL(PM.IsManuallyClosing,0) = 1									 

		 end
	 End
   ----

      --- Updating Item Detail Status to In Progress, if GRN has already created (Only for case of PO Manully closing) 
   	if exists(select 1 from Purchase.POItemDetail PID
			  inner join Purchase.POMain PM on PM.PONo = PID.PONo 
			  where PID.PONo = @PONo 
			  and PID.ItemNo = @ItemNo and (isnull(PID.MakeNo,0) = isnull(@MakeNo,0) )
			  and  ISNULL(PID.TechnicalGradeNo,0) = ISNULL(@TechnicalGradeNo ,0)	
			  and PID.statusNo in (2,8)
			  )
	 begin
	   if Not exists (
					  Select 1 
					  From Inventory.GRNItemDetail GID
					  inner join Purchase.POAmendmentMain POAM on POAM.POAmendmentNo = GID.POAmendmentNo 
					  inner join Purchase.POItemDetail PID on PID.PONo=POAM.PONo
												   and PID.ItemNo = GID.ItemNo and (isnull(PID.MakeNo,0) = isnull(GID.MakeNo,0))
												   and  ISNULL(PID.TechnicalGradeNo,0) = ISNULL(GID.TechnicalGradeNo ,0)	
					  Where POAM.PONO = @PoNo
					  And PID.ItemNo = @ItemNo and (isnull(PID.MakeNo,0) = isnull(@MakeNo,0) or PID.makeno is null)
					  And  ISNULL(PID.TechnicalGradeNo,0) = ISNULL(@TechnicalGradeNo ,0)	
					  And PID.statusNo in (2,8)
						)
		  and Not exists (
							Select 1 
							From DiscreteRM.SINLotOrderDetail SLOD
							inner join Purchase.POAmendmentMain POAM on POAM.POAmendmentNo = SLOD.POAmendmentNo 
							inner join Purchase.POItemDetail PAID on PAID.PONo = POAM.PONo
												   and PAID.ItemNo = SLOD.ItemNo and (isnull(PAID.MakeNo,0) = isnull(SLOD.MakeNo,0))							
							where POAM.PONo = @PoNo
							and PAID.ItemNo = @ItemNo and (isnull(PAID.MakeNo,0) = isnull(@MakeNo,0) or PAID.makeno is null)	
							and PAID.statusNo in (2,8)	
							)				
		 begin

				update PID 
				set PID.statusNo = 1 
				from Purchase.POItemDetail PID 
				inner join Purchase.POMain PM on PM.PONo = PID.PONo 
				where 	PID.PONo = @PoNo
					and PID.ItemNo = @ItemNo and (isnull(PID.MakeNo,0) = isnull(@MakeNo,0))
												   and  ISNULL(PID.TechnicalGradeNo,0) = ISNULL(@TechnicalGradeNo ,0)	
												   and PID.statusNo in (2,8)									 

				update PID 
				set PID.statusNo = 1 
				from Purchase.POAmendmentItemDetail PID 
				inner join Purchase.POAmendmentMain PM on PM.POAmendmentNo = PID.POAmendmentNo 
				where 	PM.PONo = @PoNo
					and PID.ItemNo = @ItemNo and (isnull(PID.MakeNo,0) = isnull(@MakeNo,0))
												   and  ISNULL(PID.TechnicalGradeNo,0) = ISNULL(@TechnicalGradeNo ,0)	
												   and PID.statusNo in (2,8)									 

		 end
	 End
   ----

end -- if @StatusNo = 2 or @StatusNo =4

if @StatusNo <> 2
	 begin
		Update Purchase.POItemDetail set  StatusNo = @StatusNo
					where PONo = @PONo and ItemNo = @ItemNo and (isnull(MakeNo,0) = isnull(@MakeNo,0) or makeno is null)
					and  ISNULL(TechnicalGradeNo,0) = ISNULL(@TechnicalGradeNo ,0)

		Update Purchase.POAmendmentItemDetail set StatusNo = @StatusNo
					where POAmendmentNo In (SELECT POAmendmentNo FROM Purchase.POAmendmentMain WHERE PONo=@PONo )
					and ItemNo = @ItemNo and (isnull(MakeNo,0) = isnull(@MakeNo,0) or makeno is null)
					and  ISNULL(TechnicalGradeNo,0) = ISNULL(@TechnicalGradeNo ,0)
	 end --if @StatusNo <> 2

if @StatusNo= 4 or @StatusNo =7
begin
	
	
	Declare @IndentNo int,@IndentMakeNo smallint, @iRow tinyint ,@iCount tinyint,@IndentTechnicalGradeNo tinyint
	Declare @IndentDetail as table (IdNo tinyint,IndentNo int, IndentMakeNo smallint,IndentTechnicalGradeNo tinyint)
	insert into @IndentDetail 
	 select row_number() over (Order By POIndentDetailNo), IndentNo , IndentMakeNo ,IndentTechnicalGradeNo
		from Purchase.POIndentDetail where 
		PoNo = @PONo and ItemNo = @Itemno and (isnull(MakeNo,0) = isnull(@MakeNo,0) or makeno is null)
	
	set @iRow = @@rowcount
	set @iCount = 0
	while @iRow >@iCount
	begin
	 select @IndentNo = IndentNo , @IndentMakeNo = IndentMakeNo,@IndentTechnicalGradeNo =IndentTechnicalGradeNo 
	 from @IndentDetail where IdNo = @iCOunt + 1
			
	exec [Inventory].[IndentStatus] @IndentNo ,@ItemNo ,@IndentMakeNo , 2,
			@OutputStatus output, @Result output,@IndentTechnicalGradeNo 
	if @OutputStatus <> 1
		raiserror(@Result,16,-1)
		
		set @iCOunt = @iCOunt + 1
	end	-- while @iRow >@iCount
End -- if @StatusNo= 4

--- BOQ Case
If Exists (Select 1 from Purchase.SCBOQMain BQ where PONo=@PONo)
BEGIN
		--As per discussion with Team 23 Mar 2024, we will not complete the po item detail and po main in case of BOQ.
		--User need to short-close the PO through "PCPS" form.Because we didn't know how many times user wants to create the BOQ.
		Declare @BOQPOSummary As Table(PONo Int,ItemNo Int,MakeNo SmallInt,TechnicalGradeNo SmallInt,GRNQty Decimal(15,3))
			
		Insert Into @BOQPOSummary
		Select BID.PONo,BID.POItemNo,BID.POMakeNo,BID.POTechGradeNo
		,SUM(gid.GRNQty) [GRNQty]
		From (
					Select BQ.ItemNo [POItemNo],BQ.MakeNo [POMakeNo],BQ.TechnicalGradeNo [POTechGradeNo],BQ.PONo,ID.ItemNo,ID.MakeNo,ID.TechnicalGradeNo
					From Purchase.SCBOQItemDetail ID
					Inner Join Purchase.SCBOQMain BQ on BQ.SCBOQNo=ID.SCBOQNo
					Where BQ.PONo  = @PONo
						And BQ.ItemNo = @ItemNo
						And ISNULL(BQ.MakeNo,0) = ISNULL(@MakeNo,0)
						And ISNULL(BQ.TechnicalGradeNo,0) = ISNULL(@TechnicalGradeNo,0)
					Group by BQ.ItemNo,BQ.MakeNo,BQ.TechnicalGradeNo,BQ.PONo,ID.ItemNo,ID.MakeNo,ID.TechnicalGradeNo
				) BID 
		LEFT Join (
					Select PONo,gid.ItemNo,gid.MakeNo,gid.TechnicalGradeNo,
							 SUM(case pqs.POClosedOnQty when  1 then ChallanQty -ISNULL(RejectedQty,0) 
								when  2 then ChallanQty -ISNULL(RejectedQty,0) 
								when 4 then ReceivedQty -ISNULL(RejectedQty,0) else AcceptedQty end) GRNQty
					From [Inventory].[GRNItemDetail] gid
					inner join masterdata.mItemMaster im on  im.ItemNo= gid.ItemNo 
					inner join Inventory.GRNMain gm on gm.GRNNo = gid.GRNNo 
					inner join Purchase.POQtySetting pqs on pqs.CompanyNo = gm.CompanyNo and im.ItemTypeNo = pqs.ItemTypeNo 
					Inner join Purchase.POAmendmentMain Pm on Pm.POAmendmentNo = gid.POAmendmentNo
					Where (RouteIdentity is null ) 
					and pm.PONo = @PONo and isnull(PM.IsManuallyClosing,0)=0				
					Group By PONo,gid.ItemNo,gid.MakeNo,gid.TechnicalGradeNo
				)gid on gid.PONo = BID.PONo
				And gid.ItemNo = BID.ItemNo
				And ISNULL(gid.MakeNo,0)= ISNULL(BID.MakeNo,0)
				And ISNULL(gid.TechnicalGradeNo,0) = ISNULL(BID.TechnicalGradeNo,0)
		Group by BID.PONo,BID.POItemNo,BID.POMakeNo,BID.POTechGradeNo
		
		IF Exists (Select 1 From @BOQPOSummary Where GRNQty>0)
		BEGIN --- Inprogress 
			Update PID
			set PID.StatusNo=8
			From Purchase.POItemDetail PID
			Inner Join @BOQPOSummary BQ on BQ.PONo=PID.PONo And BQ.ItemNo =PID.ItemNo
										And IsNull(BQ.MakeNo,0) =IsNull(PID.MakeNo,0)
										And IsNull(BQ.TechnicalGradeNo,0) =IsNull(PID.TechnicalGradeNo,0)
			Where  pid.StatusNo =1 							
										
			Update PID
			set PID.StatusNo=8
			From Purchase.POAmendmentItemDetail PID
			Inner Join @BOQPOSummary BQ on PID.POAmendmentNo in(Select PA.POAmendmentNo From Purchase.POAmendmentMain PA where PONo=@PONo)
										And BQ.ItemNo =PID.ItemNo
										And IsNull(BQ.MakeNo,0) =IsNull(PID.MakeNo,0)
										And IsNull(BQ.TechnicalGradeNo,0) =IsNull(PID.TechnicalGradeNo,0)		
			Where  pid.StatusNo =1 							
										
		END		
		Else -- Authorized if no item found
		BEGIN		
			Update PID
			set PID.StatusNo=1
			From Purchase.POItemDetail PID
			Inner Join @BOQPOSummary BQ on BQ.PONo=PID.PONo And BQ.ItemNo =PID.ItemNo
										And IsNull(BQ.MakeNo,0) =IsNull(PID.MakeNo,0)
										And IsNull(BQ.TechnicalGradeNo,0) =IsNull(PID.TechnicalGradeNo,0)
			Where PID.StatusNo in (8,2)
										
			Update PID
			set PID.StatusNo=1
			From Purchase.POAmendmentItemDetail PID
			Inner Join @BOQPOSummary BQ on PID.POAmendmentNo in(Select PA.POAmendmentNo From Purchase.POAmendmentMain PA where PONo=@PONo)
										And BQ.ItemNo =PID.ItemNo
										And IsNull(BQ.MakeNo,0) =IsNull(PID.MakeNo,0)
										And IsNull(BQ.TechnicalGradeNo,0) =IsNull(PID.TechnicalGradeNo,0)	
		    Where PID.StatusNo in (8,2)
		END				
END



	if exists(select 1 from Purchase.POItemDetail where PONo = @PONo and statusNo =1)
	 begin
	   if exists (select 1 from Inventory.GRNItemDetail a,Purchase.POAmendmentMain b where
					a.POAmendmentNo = b.POAmendmentNo and PONo = @PoNo)
		  or exists (select 1 from DiscreteRM.SINLotOrderDetail a,Purchase.POAmendmentMain b where
					a.POAmendmentNo = b.POAmendmentNo and PONo = @PoNo)				
		 begin
		  update Purchase.POMain set StatusNo = 8 where  PONo = @PONo
		  update Purchase.POAmendmentMain set StatusNo = 8 where  PONo = @PONo
		 end
	   else
	   begin
			update Purchase.POMain set StatusNo = 1 where  PONo = @PONo
			update Purchase.POAmendmentMain set StatusNo = 1 where  PONo = @PONo
		end
	 end	
	 else if exists(select 1 from Purchase.POItemDetail where PONo = @PONo and statusNo =8)
	 begin
	   if exists (select 1 from Inventory.GRNItemDetail a,Purchase.POAmendmentMain b where
					a.POAmendmentNo = b.POAmendmentNo and PONo = @PoNo)
		   or exists (select 1 from DiscreteRM.SINLotOrderDetail a,Purchase.POAmendmentMain b where
					a.POAmendmentNo = b.POAmendmentNo and PONo = @PoNo)			
		 begin
		  update Purchase.POMain set StatusNo = 8 where  PONo = @PONo
		  update Purchase.POAmendmentMain set StatusNo = 8 where  PONo = @PONo
		 end
	   else
	   begin
			update Purchase.POMain set StatusNo = 1 where  PONo = @PONo
			update Purchase.POAmendmentMain set StatusNo = 1 where  PONo = @PONo
		end
	 end	
	 else if exists(select 1 from Purchase.POItemDetail where  PONo = @PONo and Statusno = 3 )-- Hold 
		begin
			update Purchase.POMain set StatusNo = 3 where  PONo = @PONo
			update Purchase.POAmendmentMain set StatusNo = 3 where  PONo = @PONo
		end
		
	 else if exists(select 1 from Purchase.POItemDetail where  PONo = @PONo and Statusno = 4 )--cancel
	 	begin
			update Purchase.POMain set StatusNo =4 where  PONo = @PONo
			update Purchase.POAmendmentMain set StatusNo = 4 where  PONo = @PONo
		end
	 	
	else if exists(select 1 from Purchase.POItemDetail where  PONo = @PONo and Statusno=7)--shortclose
	 	begin
			update Purchase.POMain set StatusNo = 7 where  PONo = @PONo
			update Purchase.POAmendmentMain set StatusNo = 7 where  PONo = @PONo
		end				
	
	else if exists(select 1 from Purchase.POItemDetail where  PONo = @PONo and Statusno=2)--completed
		begin
			update Purchase.POMain set StatusNo = 2 where  PONo = @PONo
			update Purchase.POAmendmentMain set StatusNo =  2 where  PONo = @PONo
		end
		
set @OutputStatus =1
set @Result='Success'
end try

begin catch
set @OutputStatus =0
set @Result=error_message()

		select 0, error_message()
end Catch


END
