CREATE PROCEDURE [Inventory].[IGRNDropShipmentFinancePosting] 
	@ItemDetail xml ,
	@CompanyNo tinyint
	
AS
BEGIN

	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;

begin try

Declare @ItemNos as varchar(max)

Create table #ItemDetailDrop (ReferenceNo int,ReferenceType varchar(4),ItemNo int,MakeNo smallint,
		TechnicalGradeNo tinyint,AcceptedQty decimal(15,3), Rate decimal(15,2),GateEntryNo int 
		,RouteIdentity Decimal(10,8) , IncludedTax decimal(18,6) , ChallanQty decimal(15,3),ReceivedQty decimal(15,3)
		)
Create table #IncludedTaxDrop(IncludedAmount decimal(18,6),ItemNo int,MakeNo smallint,TechnicalGradeNo tinyint)
Create table #FreightDrop(FreightAmount decimal(18,6),ItemNo int,MakeNo smallint,TechnicalGradeNo tinyint)
Create table #ExpensesDrop(ExpenseAmount decimal(18,6),AccountNo int,ItemNo int,
					MakeNo smallint,TechnicalGradeNo tinyint, ExpenseAccountNo int)
create table #PostingDrop ([Add to Basic] bit,Particular varchar(500),[Debit Amount] decimal(15,2) , 
		[Credit Amount] decimal(15,2),[Account Name] int, [Ref. Detail] varchar(50),[Cost Center] varchar(50),
		[Amount] decimal(15,2), [Type] tinyint,SlNo tinyint ,MiscChargeNo int , TaxCodeMiscChargesDetailNo int,
			ItemNos varchar(200))
					
	 ;with cte as 
	(
		SELECT v.value('ReferenceNo[1]','int') AS ReferenceNo,
		v.value('ReferenceType[1]','varchar(4)') ReferenceType,
		v.value('ItemNo[1]','int') ItemNo,
		v.value('MakeNo[1]','smallint') MakeNo,
		v.value('TechnicalGradeNo[1]','tinyint') TechnicalGradeNo,
		v.value('AcceptedQty[1]','decimal(15,3)') AcceptedQty,
		v.value('Rate[1]','decimal(18,6)') Rate,
		v.value('GateEntryNo[1]','int') GateEntryNo,
		v.value('RouteIdentity[1]','Decimal(10,8)') RouteIdentity,
		v.value('IncludedTax[1]','decimal(18,6)') IncludedTax,
		v.value('ChallanQty[1]','decimal(15,3)') ChallanQty,
		v.value('ReceivedQty[1]','decimal(15,3)') ReceivedQty
		FROM @ItemDetail.nodes('/DocumentElement/Table') x(v)
	)
	Insert into #ItemDetailDrop (ReferenceNo ,ReferenceType ,ItemNo ,MakeNo ,TechnicalGradeNo ,AcceptedQty , 
	Rate ,GateEntryNo ,RouteIdentity , IncludedTax , ChallanQty,ReceivedQty)
	select ReferenceNo ,ReferenceType ,ItemNo ,MakeNo ,TechnicalGradeNo ,AcceptedQty , 
	Rate ,GateEntryNo ,RouteIdentity , IncludedTax , ChallanQty ,ReceivedQty 
	from cte where ItemNo is not null
	
  
 set @ItemNos = STUFF((select '`' + convert(varchar(7),ItemNo) +'~' + convert(varchar(25),SUM(AcceptedQty * Rate))		  from #ItemDetailDrop group by ItemNo for xml path ('')),1,1,'')

 ---- expense Heads
 
begin
insert into #ExpensesDrop 
	select sum(PEH.Rate * case PEH.BillingQtyNo when 1 then Id.ChallanQty 
						  when 2 then Id.ChallanQty 
						  when 4 then ID.ReceivedQty else   
						  id.AcceptedQty end),
			PAAD.AccountNo ,Id.ItemNo ,Id.MakeNo,Id.TechnicalGradeNo ,PAEAD.AccountNo 
		from 
		Purchase.POAmendmentMain PAM
		inner join Purchase.POAmendmentExpenseHead PEH on PAM.POAmendmentNo= PEH.POAmendmentNo
		inner join Purchase.POAmendmentItemDetail PID on PID.POAmendmentNo  = PEH.POAmendmentNo
		inner join #ItemDetailDrop ID on ID.ReferenceType ='PMPO' and ID.ItemNo= PID.ItemNo and 
						isnull(ID.MakeNo,0) = isnull(PID.MakeNo,0) and
						isnull(ID.TechnicalGradeNo,0) = isnull(PID.TechnicalGradeNo,0) and
						ID.ReferenceNo = PAM.POAmendmentNo 
		left join Purchase.POAccountAssignmentDetail PAAD on PAAD.PONo = PAM.PONo and PAAD.Flag='EH'
					and PAAD.Id = PEH.ExpenseheadNo and PAAD.AccountTypeNo =54
		left join Purchase.POAccountAssignmentDetail PAEAD on PAEAD.PONo = PAM.PONo and PAEAD.Flag='EH'
					and PAEAD.Id = PEH.ExpenseheadNo and PAEAD.AccountTypeNo = 39
		where
		PEH.IncludeInValuation =1 and
		PEH.FreightRateTypeNo  =1 and -- per unit
		peh.BillingLocationNo = pam.ToLocationNo and
		(PEH.itemno is null or(PEH.ItemNo = PID.ItemNo and 
						isnull(PEH.MakeNo,0) = isnull(PID.MakeNo,0) and
						isnull(PEH.TechnicalGradeNo,0) = isnull(PID.TechnicalGradeNo,0) ))
		group by  PAAD.AccountNo, Id.ItemNo ,Id.MakeNo,Id.TechnicalGradeNo  
		,PAEAD.AccountNo

insert into #ExpensesDrop 
		select sum(ExpenseAmount) ,AccountNo ,ItemNo,MakeNo ,TechnicalGradeNo  ,expenseAccountNo 
		from (
		select 		 
		case PEH.BillingQtyNo when 1 then Id.ChallanQty 
						  when 2 then Id.ChallanQty 
						  when 4 then ID.ReceivedQty else   
						  id.AcceptedQty end * ((PEH.Rate * pid.Quantity * pid.Rate)/
				(select  sum(pd.Quantity * pd.Rate) 
						from Purchase.POAmendmentItemDetail pd where pd.POAmendmentNo = PAM.POAmendmentNo )
				  )/ pid.Quantity ExpenseAmount	
			, PAAD.AccountNo AccountNo ,Id.ItemNo ,Id.MakeNo,Id.TechnicalGradeNo
			, PAEAD.AccountNo  expenseAccountNo
		from 
		Purchase.POAmendmentMain PAM
		inner join Purchase.POAmendmentExpenseHead PEH on PAM.POAmendmentNo= PEH.POAmendmentNo 
		inner join Purchase.POAmendmentItemDetail PID on PID.POAmendmentNo= PEH.POAmendmentNo 
		inner join #ItemDetailDrop ID on ID.ReferenceType ='PMPO' and ID.ItemNo= PID.ItemNo and 
						isnull(ID.MakeNo,0) = isnull(PID.MakeNo,0) and
						isnull(ID.TechnicalGradeNo,0) = isnull(PID.TechnicalGradeNo,0) and
						ID.ReferenceNo = PAM.POAmendmentNo 
		left join Purchase.POAccountAssignmentDetail PAAD on PAAD.PONo = PAM.PONo and PAAD.Flag='EH'
					and PAAD.Id = PEH.ExpenseheadNo and PAAD.AccountTypeNo =54
		left join Purchase.POAccountAssignmentDetail PAEAD on PAEAD.PONo = PAM.PONo and PAEAD.Flag='EH'
					and PAEAD.Id = PEH.ExpenseheadNo and PAEAD.AccountTypeNo = 39
		where
		PEH.IncludeInValuation =1 and
		PEH.FreightRateTypeNo  =2 and -- Fixed Amount
		peh.BillingLocationNo = pam.ToLocationNo and
		(PEH.itemno is null or(PEH.ItemNo = PID.ItemNo and 
						isnull(PEH.MakeNo,0) = isnull(PID.MakeNo,0) and
						isnull(PEH.TechnicalGradeNo,0) = isnull(PID.TechnicalGradeNo,0) ))
		) tt
		group by
		AccountNo,ItemNo ,MakeNo,TechnicalGradeNo ,expenseAccountNo 

insert into #ExpensesDrop 
	select sum(ExpenseAmount) ,AccountNo,ItemNo ,MakeNo,TechnicalGradeNo ,ExpenseAccountNo 
	from(
	select 
	sum( PEH.Rate * cnt * AcceptedQty * id.Rate )/
		(select sum(AcceptedQty * Rate) from #ItemDetailDrop where GateEntryNo is Null 
						or GateEntryNo = Cid.GateEntryNo) ExpenseAmount
	,PAAD.AccountNo AccountNo ,Id.ItemNo ,Id.MakeNo,Id.TechnicalGradeNo ,PAEAD.AccountNo ExpenseAccountNo 
		from 
		Purchase.POAmendmentMain PAM
		inner join (select count(distinct isnull(GateEntryNo,1)) cnt ,ReferenceNo,GateEntryNo from #ItemDetailDrop id 
	group by ReferenceNo ,GateEntryNo )Cid
				  on Cid.ReferenceNo = PAM.POAmendmentNo 
		inner join Purchase.POAmendmentExpenseHead PEH on PAM.POAmendmentNo= PEH.POAmendmentNo 
		inner join Purchase.POAmendmentItemDetail PID on PID.POAmendmentNo  = PEH.POAmendmentNo
		inner join #ItemDetailDrop ID on ID.ReferenceType ='PMPO' and ID.ItemNo= PID.ItemNo and 
						isnull(ID.MakeNo,0) = isnull(PID.MakeNo,0) and
						isnull(ID.TechnicalGradeNo,0) = isnull(PID.TechnicalGradeNo,0) and
						ID.ReferenceNo = PAM.POAmendmentNo 
						and Cid.ReferenceNo = ID.ReferenceNo and ID.AcceptedQty >0
						and isnull(ID.GateEntryNo,0) = isnull(Cid.GateEntryNo,0) 
		left join Purchase.POAccountAssignmentDetail PAAD on PAAD.PONo = PAM.PONo and PAAD.Flag='EH'
					and PAAD.Id = PEH.ExpenseheadNo and PAAD.AccountTypeNo =54
		left join Purchase.POAccountAssignmentDetail PAEAD on PAEAD.PONo = PAM.PONo and PAEAD.Flag='EH'
					and PAEAD.Id = PEH.ExpenseheadNo and PAAD.AccountTypeNo =39
		where
		PEH.IncludeInValuation =1 and
		PEH.FreightRateTypeNo  =3 and -- per trip
		peh.BillingLocationNo = pam.ToLocationNo and
		(PEH.itemno is null or(PEH.ItemNo = PID.ItemNo and 
						isnull(PEH.MakeNo,0) = isnull(PID.MakeNo,0) and
						isnull(PEH.TechnicalGradeNo,0) = isnull(PID.TechnicalGradeNo,0) ))
		group by  PAAD.AccountNo, Id.ItemNo ,Id.MakeNo,Id.TechnicalGradeNo 	,Cid.GateEntryNo,PAEAD.AccountNo 
		)tt group by AccountNo ,ItemNo,MakeNo ,TechnicalGradeNo , ExpenseAccountNo 

end		

-- Included Taxes
begin
insert into #IncludedTaxDrop 
select sum(IncludedAmount),ItemNo ,MakeNo ,TechnicalGradeNo   from (
select sum(case when MC.calculationnature=1 then   (TotalAmount  /pid.Quantity ) * ID.AcceptedQty
				when calculationnature =2 then 0-(TotalAmount  /pid.Quantity ) * ID.AcceptedQty else 0 end)
				IncludedAmount, ID.ItemNo ,ID.MakeNo ,ID.TechnicalGradeNo  
				
		from #ItemDetailDrop ID
		inner join Purchase.POAmendmentTaxDetail pd on pd.POAmendmentNo = ID.ReferenceNo 
		inner join Purchase.POAmendmentItemTaxDetail ptd on ptd.POAmendmentNo = pd.POAmendmentNo 
					and pd.MiscChargeNo = ptd.MiscChargeNo 
		inner join Purchase.POAmendmentItemDetail pid on pid.POAmendmentNo = pd.POAmendmentNo 		
		inner join purchase.POAmendmentMain pam on pam.POAmendmentno = pid.POAmendmentno
		inner join masterdata.mMiscCharges MC on MC.MiscChargeNo = pd.MiscChargeNo 
		 			and (pd.IncludeInBasic  =1 and pd.ChargeType = 2)  
		where ID.ItemNo = ptd.ItemNo and ISNULL(id.MakeNo,0) = ISNULL(ptd.MakeNo,0)	
			and ISNULL(id.TechnicalGradeNo,0) = isnull(ptd.TechnicalGradeNo ,0)
			and pid.ItemNo = ptd.ItemNo and ISNULL(pid.MakeNo,0) = ISNULL(ptd.MakeNo,0)	
			and ISNULL(pid.TechnicalGradeNo,0) = isnull(ptd.TechnicalGradeNo ,0)
			and ID.ReferenceType  ='PMPO' and isnull(ID.IncludedTax ,0)=0 
			group by ID.ItemNo ,ID.MakeNo ,ID.TechnicalGradeNo 
Union all
select sum(case when MC.calculationnature=1 then  0- (TotalAmount  /pid.Quantity ) * ID.AcceptedQty
				when calculationnature =2 then  (TotalAmount  /pid.Quantity ) * ID.AcceptedQty else 0 end)
				IncludedAmount, ID.ItemNo ,ID.MakeNo ,ID.TechnicalGradeNo  
				
		from #ItemDetailDrop ID
		inner join Purchase.POAmendmentTaxDetail pd on pd.POAmendmentNo = ID.ReferenceNo 
		inner join Purchase.POAmendmentItemTaxDetail ptd on ptd.POAmendmentNo = pd.POAmendmentNo 
					and pd.MiscChargeNo = ptd.MiscChargeNo 
		inner join Purchase.POAmendmentItemDetail pid on pid.POAmendmentNo = pd.POAmendmentNo 		
		inner join purchase.POAmendmentMain pam on pam.POAmendmentno = pid.POAmendmentno
		inner join masterdata.mMiscCharges MC on MC.MiscChargeNo = pd.MiscChargeNo 
				and (pd.IncludeInBasic  =0 and ptd.ChargeType =1)
		where ID.ItemNo = ptd.ItemNo and ISNULL(id.MakeNo,0) = ISNULL(ptd.MakeNo,0)	
			and ISNULL(id.TechnicalGradeNo,0) = isnull(ptd.TechnicalGradeNo ,0)
			and pid.ItemNo = ptd.ItemNo and ISNULL(pid.MakeNo,0) = ISNULL(ptd.MakeNo,0)	
			and ISNULL(pid.TechnicalGradeNo,0) = isnull(ptd.TechnicalGradeNo ,0)
			and ID.ReferenceType  ='PMPO' and isnull(ID.IncludedTax ,0)=0 
			group by ID.ItemNo ,ID.MakeNo ,ID.TechnicalGradeNo ) tt
	group by ItemNo ,MakeNo ,TechnicalGradeNo		
			
insert into #IncludedTaxDrop 
	select sum(IncludedTax) ,ItemNo ,MakeNo,TechnicalGradeNo from #ItemDetailDrop 
			where IncludedTax<>0
		group by ItemNo ,MakeNo,TechnicalGradeNo 
end 			
 
insert into #FreightDrop 
	select sum(FreightAmount),  ID.ItemNo ,ID.MakeNo , ID.TechnicalGradeNo 	 from (
    select  sum(fd.PayableAmount + ISNULL(fd.AdvanceAmount,0)) * (ID.AcceptedQty * ID.Rate)/
				(select SUM(AcceptedQty* rate) from #ItemDetailDrop where GateEntryNo = id.GateEntryNo) FreightAmount
		 , ID.ItemNo ,ID.MakeNo , ID.TechnicalGradeNo 		
		 from
	Finance.FreightInvoiceMain fm 
	inner join Finance.FreightInvoiceDocumentDetail fd on fd.FreightInvoiceNo = fm.FreightInvoiceNo 
	inner join Finance.FreightInvoiceDocumentDetails fds on fds.FreightInvoiceNo = fd.FreightInvoiceNo 
					and (fds.GateEntryNo = fd.RefDocNo or fds.GRNNo = fd.RefDocNo )
	inner join #ItemDetailDrop ID on ID.GateEntryNo = fds.GateEntryNo 	and ID.AcceptedQty >0			
	where fm.AuthorizedBy is not null
	group by ID.ItemNo ,ID.MakeNo , ID.TechnicalGradeNo ,id.GateEntryNo,ID.AcceptedQty , ID.Rate) ID
	group by   ID.ItemNo ,ID.MakeNo , ID.TechnicalGradeNo 	

-- FREIGHT ORDER
if not exists(select 1 from #FreightDrop) 
	and exists(select 1 from #ItemDetailDrop id, Purchase.POAmendmentMain pam ,Purchase.FreightOrderOrderDetail fd 
				where pam.POAmendmentNo = ID.ReferenceNo and pam.PONo =fd.PONo)
begin
insert into #FreightDrop 
select SUM(FreightAmt), ItemNo , MakeNo , TechnicalGradeNo  from (
select (foa.FreightAmount- isnull(DeductionAmount ,0)) * ID.AcceptedQty * ID.Rate 
			/(select SUM(AcceptedQty*rate) from #ItemDetailDrop where ReferenceNo = ID.ReferenceNo ) FreightAmt
			, ID.ItemNo ,ID.MakeNo ,ID.TechnicalGradeNo 
		from [Purchase].[funcFreightOrderAmount] (@ItemDetail) foa
		inner join #ItemDetailDrop ID on ID.ReferenceNo = foa.POAmendmentNo and ID.ReferenceType = 'PMPO'
		 where ID.AcceptedQty >0  
		 ) tt group by  ItemNo , MakeNo , TechnicalGradeNo 
end


---- No Freight Route
if exists(select 1 from #ItemDetailDrop where RouteIdentity is null)
begin
insert into #FreightDrop 
select SUM(FreightAmt), ItemNo , MakeNo , TechnicalGradeNo  from (
select  (foa.FreightAmount- isnull(DeductionAmount ,0)) * ID.AcceptedQty * ID.Rate 
			/(select SUM(AcceptedQty*rate) from #ItemDetailDrop where ReferenceNo = ID.ReferenceNo ) FreightAmt
			, ID.ItemNo ,ID.MakeNo ,ID.TechnicalGradeNo 
		from [Purchase].[funcFreightOrderAmount] (@ItemDetail) foa
		inner join #ItemDetailDrop ID on ID.ReferenceNo = foa.SCOrderAmendmentNo  and ID.ReferenceType = 'SCSO'
		inner join Purchase.SCOrderAmendmentMain pam on pam.SCOrderAmendmentNo = ID.ReferenceNo 
			where ID.AcceptedQty >0
			and exists(select 1 from Purchase.FreightOrderOrderDetail fd where fd.SCOrderNo = pam.SCOrderNo)
		 ) tt group by  ItemNo , MakeNo , TechnicalGradeNo 
		
  if exists(select 1 from #ItemDetailDrop where ReferenceType ='PMPO')
  begin
	insert into #FreightDrop 
	select sum(FreightAmount * id.AcceptedQty) ,ItemNo,MakeNo ,TechnicalGradeNo  from 
	Purchase.POAmendmentMain  pm
	inner join #ItemDetailDrop id on id.ReferenceNo = pm.POAmendmentNo 
	where pm.FreightTypeNo =1 -- To Pay only
	and FreightRateTypeNo  =1 -- per unit
	and ReferenceType ='PMPO'
	and RouteIdentity is null
	and not exists(select 1 from Finance.FreightInvoiceDocumentDetails fd where fd.GateEntryNo = id.GateEntryNo )
	and not exists(select 1 from Purchase.FreightOrderOrderDetail fd where fd.PONo = PM.PONo)
	group by ItemNo,MakeNo ,TechnicalGradeNo
	
	insert into #FreightDrop 
	select sum(FreightAmount),ItemNo, MakeNo , TechnicalGradeNo
	from(
	select  id.AcceptedQty * ((FreightAmount * pid.Quantity * pid.Rate)/
				(select SUM(pd.Quantity * pd.Rate) 
						from Purchase.POAmendmentItemDetail pd where pd.POAmendmentNo = pm.POAmendmentNo )
				  )/ pid.Quantity FreightAmount
	
	   ,id.ItemNo,id.MakeNo ,id.TechnicalGradeNo  from 
	Purchase.POAmendmentMain  pm
	inner join #ItemDetailDrop id on id.ReferenceNo = pm.POAmendmentNo and id.AcceptedQty >0
	inner join Purchase.POAmendmentItemDetail pid on pid.POAmendmentNo = pm.POAmendmentNo 
			and pid.ItemNo =id.ItemNo and isnull(pid.MakeNo,0) = isnull(id.MakeNo,0)
			and isnull(pid.TechnicalGradeNo,0) = isnull(id.TechnicalGradeNo ,0)
	where pm.FreightTypeNo =1 -- To Pay only
	and FreightRateTypeNo  =2 -- Fixed
	and ReferenceType ='PMPO'
	and RouteIdentity is null
	and not exists(select 1 from Finance.FreightInvoiceDocumentDetails fd where fd.GateEntryNo = id.GateEntryNo )
	and not exists(select 1 from Purchase.FreightOrderOrderDetail fd where fd.PONo = PM.PONo)
	) tt
	group by   ItemNo, MakeNo , TechnicalGradeNo
	
	insert into #FreightDrop 
	Select sum(FreightAmount),ItemNo,MakeNo ,TechnicalGradeNo 
	from (
	select sum(FreightAmount * cnt * i.AcceptedQty * i.Rate )/
		(select sum(AcceptedQty * Rate) from #ItemDetailDrop where GateEntryNo is null or GateEntryNo = id.GateEntryNo)
		 FreightAmount
	,ItemNo,MakeNo ,TechnicalGradeNo  from 
	Purchase.POAmendmentMain  pm
	inner join (select count(distinct isnull(GateEntryNo,1)) cnt ,ReferenceNo,GateEntryNo from #ItemDetailDrop id 
	where not exists(select 1 from Finance.FreightInvoiceDocumentDetails fd where fd.GateEntryNo = id.GateEntryNo)
	group by ReferenceNo ,GateEntryNo )
				id on id.ReferenceNo = pm.POAmendmentNo 
	inner join #ItemDetailDrop I on i.ReferenceNo = id.ReferenceNo and isnull(i.GateEntryNo,0)= isnull(id.GateEntryNo,0) and i.AcceptedQty >0
	where pm.FreightTypeNo =1 -- To Pay only
	and FreightRateTypeNo  =3 -- per trip
	and RouteIdentity is null
	and ReferenceType ='PMPO'
	and not exists(select 1 from Purchase.FreightOrderOrderDetail fd where fd.PONo = PM.PONo)
	group by ItemNo,MakeNo ,TechnicalGradeNo,id.GateEntryNo) tt
	group by ItemNo,MakeNo ,TechnicalGradeNo
  end
end

Insert into #PostingDrop 
		select cast(0 as bit) [Add to Basic]
				,'Purchase Account' as Particular
				,convert(decimal(15,2),sum(Amount + ISNULL(IncludedAmount,0))) [Debit Amount]
				,convert(decimal(15,2),null) [Credit Amount]
				, ItemAC.AccountNo [Account Name]
				,cast(null as varchar) [Ref. Detail]
				,cast(null as varchar) [Cost Center]
				,SUM(Amount + ISNULL(IncludedAmount,0)) [Amount]
				,1 [Type]
				,ROW_NUMBER()over(order by ItemAC .AccountNo ) SlNo
				,null MiscChargeNo
				,null TaxCodeMiscChargesDetailNo
				,'' ItemNos
			from (select  SUM(AcceptedQty * Rate) Amount ,ItemNo ,MakeNo,TechnicalGradeNo  
							from #ItemDetailDrop 
							group by ItemNo,MakeNo,TechnicalGradeNo)I
				left join (SELECT pad.AccountNo ,PID.ItemNo FROM Purchase.POAccountAssignmentDetail pad
						inner join Purchase.POAmendmentMain pam on pam.PONo = pad.PONo
						inner join #ItemDetailDrop PID on PID.ReferenceNo = pam.POAmendmentNo 
						where PID.ReferenceType ='PMPO' and pad.Flag ='IM' and pad.Id = PID.ItemNo
						and pad.AccountTypeNo =20
						group by pad.AccountNo ,PID.ItemNo
						) ItemAC on ItemAC.ItemNo= I.ItemNo 
			left join #IncludedTaxDrop tt on I.ItemNo = tt.ItemNo 
							and ISNULL(I.MakeNo,0) = ISNULL(tt.MakeNo,0)	
							and ISNULL(I.TechnicalGradeNo,0) = isnull(tt.TechnicalGradeNo ,0)
			group by  ItemAC.AccountNo 

insert into #PostingDrop  
	 select cast(0 as bit) [Add to Basic]
				,'GR Suspense A/c' as Particular
				,convert(decimal(15,2), null) [Debit Amount]
				,convert(decimal(15,2),SUM(Amount + ISNULL(tt.IncludedAmount,0) )) [Credit Amount]
				, ItemAC.AccountNo [Account Name]
				,cast(null as varchar) [Ref. Detail]
				,cast(null as varchar) [Cost Center]
				,SUM(Amount+ ISNULL(tt.IncludedAmount,0))  [Amount]
				,2 [Type]
				,ROW_NUMBER()over(order by  ItemAC.AccountNo ) 
						+ (select isnull(max(SlNo),0) from #PostingDrop) SlNo
				,null MiscChargeNo
				,null TaxCodeMiscChargesDetailNo
				,'' ItemNos
			from (select sum(AcceptedQty * Rate) Amount ,ItemNo ,MakeNo,TechnicalGradeNo  
					from #ItemDetailDrop
					 group by ItemNo,MakeNo,TechnicalGradeNo) I
			left join #IncludedTaxDrop tt on I.ItemNo = tt.ItemNo 
							and ISNULL(I.MakeNo,0) = ISNULL(tt.MakeNo,0)	
							and ISNULL(I.TechnicalGradeNo,0) = isnull(tt.TechnicalGradeNo ,0)
			left join (SELECT pad.AccountNo ,PID.ItemNo FROM Purchase.POAccountAssignmentDetail pad
						inner join Purchase.POAmendmentMain pam on pam.PONo = pad.PONo
						inner join #ItemDetailDrop PID on PID.ReferenceNo = pam.POAmendmentNo 
						where PID.ReferenceType ='PMPO' and pad.Flag ='IM11' and pad.Id = PID.ItemNo
						and pad.AccountTypeNo =54
						group by pad.AccountNo ,PID.ItemNo
						) ItemAC on ItemAC.ItemNo= I.ItemNo 
			group by  ItemAC.AccountNo 
			
insert into #PostingDrop 
	  select cast(0 as bit) [Add to Basic]
				,'Expense Account A/c' as Particular
				,convert(decimal(15,2), sum(E.ExpenseAmount)) [Debit Amount]
				,convert(decimal(15,2),NUll) [Credit Amount]
				, e.ExpenseAccountNo [Account Name]
				,cast(null as varchar) [Ref. Detail]
				,cast(null as varchar) [Cost Center]
				,sum(E.ExpenseAmount) [Amount]
				,1 [Type]
				,ROW_NUMBER()over(order by E.ExpenseAccountNo) + (select isnull(max(SlNo),0) from #PostingDrop) SlNo
				,null MiscChargeNo
				,null TaxCodeMiscChargesDetailNo
				,'' ItemNos
			from #ExpensesDrop E
			group by  e.ExpenseAccountNo
			having
			sum(E.ExpenseAmount)<>0
			  	
insert into #PostingDrop 
	  select cast(0 as bit) [Add to Basic]
				,'Expense Suspense Account' as Particular
				,convert(decimal(15,2), null) [Debit Amount]
				,convert(decimal(15,2),sum(E.ExpenseAmount)) [Credit Amount]
				, e.AccountNo [Account Name]
				,cast(null as varchar) [Ref. Detail]
				,cast(null as varchar) [Cost Center]
				,sum(E.ExpenseAmount) [Amount]
				,2 [Type]
				,ROW_NUMBER()over(order by E.AccountNo) + (select isnull(max(SlNo),0) from #PostingDrop) SlNo
				,null MiscChargeNo
				,null TaxCodeMiscChargesDetailNo
				,'' ItemNos
			from #ExpensesDrop E
			group by  e.AccountNo
			having
			sum(E.ExpenseAmount)<>0

insert into #PostingDrop 
	 select cast(0 as bit) [Add to Basic]
				,'Freight Expense Account' as Particular
				,convert(decimal(15,2),case when sum(I.FreightAmount) >0 then sum(I.FreightAmount) else null end) [Debit Amount]
				,convert(decimal(15,2),case when sum(I.FreightAmount)>0 then null else ABS(sum(I.FreightAmount)) end) [Credit Amount]
				, IR.AccountNo [Account Name]
				,cast(null as varchar) [Ref. Detail]
				,cast(null as varchar) [Cost Center]
				,abs(sum(I.FreightAmount)) [Amount]
				,case when sum(I.FreightAmount)>0 then 1 else 2 end [Type]
				,ROW_NUMBER()over(order by IR.AccountNo) + (select isnull(max(SlNo),0) from #PostingDrop) SlNo
				,null MiscChargeNo
				,IR.TaxCodeMiscChargesDetailNo
				,'' ItemNos
			from #FreightDrop I
			outer apply (select top 1 AccountNo ,TaxCodeMiscChargesDetailNo
						from
						(select isnull(IR.AccountNo ,ARAR.AccountNo) AccountNo , 
						TCD.TaxCodeMiscChargesDetailNo from 
					masterdata.mTaxCode tc
					inner join masterdata.mTaxCodeMiscChargesDetail TCD on TCD.TaxCodeNo = tc.TaxCodeNo 
	inner join Finance.ARDTDCAccountRule ARAR on ARAR.TaxCodeMiscChargesDetailNo = TCD.TaxCodeMiscChargesDetailNo
		inner join Finance.ARDItemRule IR on IR.ARDNo = ARAR.ARDNo and IR.ItemNo = I.ItemNo 
					where tc.TaxCategoryNo =10 and TaxAccountTypeNo =27
					and tc.TaxAccountBasisNo=1
					union all
					select AccountNo, mcd.TaxCodeMiscChargesDetailNo
					from
					masterdata.mTaxCode tc 
					inner join masterdata.mTaxCodeMiscChargesDetail mcd on tc.taxcodeno=mcd.taxcodeno
					inner join Finance.ARDTDCAccountRule ar on 
						mcd.TaxCodeMiscChargesDetailNo=ar.TaxCodeMiscChargesDetailNo
					where tc.TaxAccountBasisNo=2
					 and tc.TaxCategoryNo =10 and TaxAccountTypeNo =27
					) tt) IR 
			 
				group by IR.AccountNo,IR.TaxCodeMiscChargesDetailNo
				having
				sum(I.FreightAmount)<>0
					
insert into #PostingDrop 
	 select cast(0 as bit) [Add to Basic]
				,'Freight Account' as Particular

				,null MiscChargeNo
				,null TaxCodeMiscChargesDetailNo
				,'' ItemNos

						inner join #ItemDetailDrop id on id.ReferenceNo= pam.POAmendmentNo

						group by ad.AccountNo,id.ItemNo
						) IR on IR.ItemNo=I.ItemNo
			group by IR.AccountNo
			having 
			sum(I.FreightAmount)<>0
 		
 if exists(select 1 from #PostingDrop having abs(SUM([Credit Amount])- SUM([Debit Amount]))<=.5)
