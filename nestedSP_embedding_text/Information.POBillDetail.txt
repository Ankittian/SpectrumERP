CREATE PROCEDURE [Information].[POBillDetail] 
	-- Add the parameters for the stored procedure here
	@CompanyNo tinyint,
	@DocumentDate smalldatetime,
	@PONo int
	
AS
BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;

    -- Insert statements for procedure here
	

	declare @str varchar(max)
    select @str=masterdata.JSForInformation('PMPO')
    set @str=@str+masterdata.CSSForInformation('PMPO')
	set @str=@str+'<table class="altrowstable" id="alternatecolorPOBillDetail">'
	set @str=@str+'<Caption><Font color="Maroon"><h3 align="Left"><b>Bill Detail</b></h3></Font></Caption>'
    set @str=@str+'<tr><th>Sl. No.</th><th>Bill Type</th><th>Particular</th><th>Doc. No.</th><th>Doc. Date</th><th>Party Name</th><th>Party Bill No.</th><th>Party Bill Date</th><th>Party Bill Amount</th><th>Ref. Amount</th><th>Paid Amount</th><th>Balance Amount</th><th>Due Date</th><th>Route Id</th></tr>'
	set @str=@str
				 +( 
					
					replace(
					replace( 
					replace(
					
					replace(
					replace( 
					replace(
					
					replace(
					replace(
					replace( 
					
					replace( 
					replace(
					replace(
					
					replace(
					replace(
							(
								Select  ROW_NUMBER() over ( Order by BD.PartyBillDate desc) as td1
										,ISNULL( BL.[Bill Type],'' ) td2
										,ISNULL( Case When BD.TransactionFlag= 'FASB'
													then EXHM.ExpenseHeadName
													Else BL.Particular
													End 
												,'') As td3
										 ,ISNULL( BD.BillNoYearly,'') td4
										 ,Isnull( Convert(varchar(20),BD.BillDate ,103),'') as td5
										 ,Isnull( VLD.Name,'') td6
										 ,Isnull( BD.PartyBillNo,'') td7
										 ,ISNULL( Convert(varchar(20), BD.PartyBillDate ,103),'') td8
										 ,Isnull( CONVERT(Decimal(15,2),ISNULL( ISNULL(PB.NetAmount,SBM.NetAmount ) ,FIM.PayableAmount) ),0) td9
										 ,Isnull( CONVERT(Decimal(15,2),POM.NetAmount),0) td10
										 ,ISNULL( CONVERT(Decimal(15,2), Isnull(PM1.PaidAmt,PM2.PaidAmt)),0.00) td11
										,ISNULL( ISNULL(PB.NetAmount,SBM.NetAmount ) ,FIM.PayableAmount) - ISNULL( ISNULL(PM1.PaidAmt,PM2.PaidAmt),0.00) td12
										,ISNULL( Convert(varchar(20), PS.DueDate ,103),'') td13
										, Isnull( CONVERT( varchar(500), LS.RouteIdentity ),'')td14
										
								From Inventory.InvLotSummary LS 
								Inner Join Inventory.InvLotBillDetail BD on BD.InvLotSummaryNo=LS.InvLotSummaryNo
																		   And LS.TransactionFlag='IGRN' 
																		   And BD.TransactionFlag<>'IGRN'
								Inner Join (	Select 'PBPB' As [TransactionFlag], 'Purchase Bill' [Bill Type],'Purchase' as Particular 
												Union 
												Select 'FASB' As [TransactionFlag], 'Service Bill' [Bill Type],'' as Particular 
												Union 
												Select 'FIFI' As [TransactionFlag], 'Freight Bill' [Bill Type],'Freight Charges' as Particular 
											)BL On BD.TransactionFlag = BL.TransactionFlag 
								Left Join Finance.PurchaseBillMain PB On BD.BillNo = PB.PurchaseBillNo
																		And BD.TransactionFlag = 'PBPB'
								Left Join Finance.ServiceBillMain SBM On BD.BillNo = SBM.ServiceBillNo
																		And BD.TransactionFlag = 'FASB'										
								Left Join Finance.FreightInvoiceMain FIM On BD.BillNo = FIM.FreightInvoiceNo
																			And BD.TransactionFlag = 'FIFI'																		
								Left Join Finance.PaymentSchedule PS on (PB.PurchaseBillNo = PS.DocumentNo And PS.DocumentType = 'PBPB')	
																		Or (SBM.ServiceBillNo = PS.DocumentNo And PS.DocumentType = 'FASB')										
																		Or (FIM.FreightInvoiceNo = PS.DocumentNo And PS.DocumentType = 'FIFI')										
								Left Join (	Select CC.DocumentNo,SUM(JM.Amount) PaidAmt,CC.SourceDocumentFormNo
											from Finance.CashCounterMain CC
											Left Join Finance.CashJournalMain JM on JM.CashCounterNo=CC.CashCounterNo
											Group By CC.DocumentNo,CC.NetAmount,CC.SourceDocumentFormNo
											) PM1 On (PB.PurchaseBillNo = PM1.DocumentNo And PM1.SourceDocumentFormNo = 202)
													 Or (SBM.ServiceBillNo = PM1.DocumentNo And PM1.SourceDocumentFormNo = 457)
													 Or (FIM.FreightInvoiceNo = PM1.DocumentNo And PM1.SourceDocumentFormNo = 450)

								Outer Apply (	Select VD.VoucherNo
													,RM.SpecificCrAmount+RM.SpecificDrAmount-abs(RM.BalanceAmount)  PaidAmt
												From Inventory.InvLotBillDetail BD1
												Cross Apply( Select MAX(VoucherDetailNo ) VoucherDetailNo
															 From Finance.accVoucherDetail VD
															 Inner Join masterdata.mAccountMaster AM on VD.AccountNo = AM.AccountNo
																								And AccountTypeNo = 4
															 Where VD.VoucherNo = BD1.VoucherNo
															) tt1												    
												Inner Join  Finance.accVoucherDetail VD  On VD.VoucherNo = BD1.VoucherNo
																						And VD.VoucherDetailNo = tt1.VoucherDetailNo        			
												Left Join Finance.ReferenceMain RM On RM.VoucherDetailNo = tt1.VoucherDetailNo
																				And RM.AccountNo = vd.AccountNo
											
												Where BD1.VoucherNo = FIM.VoucherNo
												And BD.BillNo =  BD1.BillNo        
												   ) PM2  
								Inner Join Purchase.POAmendmentMain PM on PM.POAmendmentNo=LS.POAmendmentNo	
								Inner Join Purchase.POMain POM	On PM.PONo = POM.PONo
								Left Join masterdata.mVendorCard VLD On LS.VendorNo= VLD.VendorNo
								Left Join  masterdata.mExpenseHead EXHM On BD.ExpenseheadNo = EXHM.ExpenseHeadNo
								Where POM.PONo = @PONo
										And POM.DocumentDate = @DocumentDate
										And POM.CompanyNo = @CompanyNo
								Group By BL.[Bill Type]
										 ,Case When BD.TransactionFlag= 'FASB'
												then EXHM.ExpenseHeadName
												Else BL.Particular
										  End 
										 ,BD.BillNoYearly 
										 ,BD.BillDate 
										 ,VLD.Name   
										 ,BD.PartyBillNo 
										 ,BD.PartyBillDate 
										,PB.NetAmount
										,SBM.NetAmount  
										,FIM.PayableAmount
										 ,POM.NetAmount 
										 ,PM1.PaidAmt,PM2.PaidAmt
										,PS.DueDate 
										,LS.RouteIdentity
								
										
							for xml path('tr')),'<td1>','<td style="text-align: Center;">')
							                   ,'<td2>','<td>')
		   									   ,'<td3>','<td>')				
											   ,'<td4>','<td>')
											   ,'<td5>','<td>')
		   									   ,'<td6>','<td>')			
											   ,'<td7>','<td>')
											    ,'<td8>','<td style="text-align: Right;">')
											   ,'<td9>','<td style="text-align: Right;">')
											   ,'<td10>','<td style="text-align: Right;">')
											    ,'<td11>','<td>')
											    ,'<td12>','<td style="text-align: Right;">')
											    ,'<td13>','<td>')
											    ,'<td14>','<td>')
									    	)
								
				   
	set @str=@str+'</table>'
	
	select @str
END
