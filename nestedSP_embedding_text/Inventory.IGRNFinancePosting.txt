CREATE PROCEDURE [Inventory].[IGRNFinancePosting] 
	@ItemDetail xml ,
	@CompanyNo tinyint ,
	@VendorLocationNo int  
	
AS

BEGIN

	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;

begin try

Declare @IsInterUnit bit , @IsManagingExpenseAccount bit  ,@ItemNos as varchar(max),@IncludedAmount money

Create table #ItemDetail (TableID int not null Identity(1,1) primary key, ReferenceNo int,ReferenceType varchar(4),ItemNo int,MakeNo smallint,
		TechnicalGradeNo tinyint,AcceptedQty decimal(15,3), Rate decimal(18,6),GateEntryNo int,DispatchChallanNo int 
		,RouteIdentity decimal(10,8) , IncludedTax decimal(18,6) , ChallanQty decimal(15,3),ReceivedQty decimal(15,3)
		,TransporterNo int,MAXRouteIdentity decimal(10,8), MINRouteIdentity decimal(10,8)
		,FromLocationNo smallint,ToLocationNo smallint,IsReturnThroughDC bit,SourceDocumentNo tinyint,ExchangeRate decimal(18,6) 
		)
Create table #IncludedTax(IncludedAmount decimal(18,6),ItemNo int,MakeNo smallint,TechnicalGradeNo tinyint
							,RefNo int)
Create table #Freight(FreightAmount decimal(18,6),ItemNo int,MakeNo smallint,TechnicalGradeNo tinyint
					,IsPaid bit,RefNo int)
Create table #Expenses(ExpenseAmount decimal(18,6),AccountNo int,ItemNo int,
					MakeNo smallint,TechnicalGradeNo tinyint, ExpenseAccountNo int ,RefNo int)
create table #Posting ([Add to Basic] bit,Particular varchar(500),[Debit Amount] decimal(15,2) , 
		[Credit Amount] decimal(15,2),[Account Name] int, [Ref. Detail] varchar(50),[Cost Center] varchar(50),
		[Amount] decimal(15,2), [Type] tinyint,SlNo tinyint ,MiscChargeNo int , TaxCodeMiscChargesDetailNo int,
			ItemNos varchar(200))
					
if exists (select 1 from masterdata.mVendorCard vc
			inner join masterdata.mVendorLocationDetail vld on vld.VendorNo = vc.VendorNo 
			where vld.VendorLocationNo =@VendorLocationNo and vc.VendorOfCompanyNo is not null
				and VendorOfCompanyNo<>@CompanyNo )
begin
	select @IsInterUnit=1
end
 
 	
 	
 declare @idoc int
EXEC sp_xml_preparedocument @idoc OUTPUT, @ItemDetail 
-- SELECT statement that uses the OPENXML rowset provider. 
Insert into #ItemDetail (ReferenceNo ,ReferenceType ,ItemNo ,MakeNo ,TechnicalGradeNo ,AcceptedQty , 
	Rate ,GateEntryNo ,DispatchChallanNo,RouteIdentity , IncludedTax , ChallanQty,ReceivedQty	,TransporterNo ,
	FromLocationNo ,ToLocationNo ,MAXRouteIdentity , MINRouteIdentity,IsReturnThroughDC,SourceDocumentNo,ExchangeRate)
	select ReferenceNo ,ReferenceType ,cte.ItemNo ,MakeNo ,TechnicalGradeNo ,
		case when s.GRNPassOnQty = 2 and im.ItemTypeNo = 2 then ChallanQty else AcceptedQty end , 
	Rate ,GateEntryNo ,DispatchChallanNo,RouteIdentity , IncludedTax , ChallanQty ,ReceivedQty ,TransporterNo ,FromLocationNo
	,ToLocationNo
	--,(select MAX(ptd.RouteIdentity) from Purchase.POAmendmentTransportationRoute ptd
	--	where ptd.POAmendmentNo =cte.ReferenceNo and ReferenceNo is not null and cte.ReferenceType ='PMPO'
	--								and ptd.RouteIdentity /10 = cte.RouteIdentity /10)
	,(select isnull(max(ptd.RouteIdentity ),cte.RouteIdentity) from Purchase.POAmendmentTransportationRoute ptd
				where ptd.POAmendmentNo =cte.ReferenceNo and ReferenceNo is not null and cte.ReferenceType ='PMPO'
				and ptd.RouteIdentity - cte.RouteIdentity between 
				CONVERT(decimal(18,8),'0.'+REPLICATE ('0', LEN(CONVERT(int,CONVERT(float,REVERSE (cte.RouteIdentity))))) +'1')
					and CONVERT(decimal(18,8),'0.'+REPLICATE ('0', LEN(CONVERT(int,CONVERT(float,REVERSE (cte.RouteIdentity))))) +'9'))
	,null
	 ,IsReturnThroughDC,cte.SourceDocumentNo,cte.ExchangeRate
	 
FROM OPENXML (@idoc, '/DocumentElement/Table',2)  
	with (ReferenceNo int,ReferenceType varchar(4),ItemNo int,MakeNo smallint,
		TechnicalGradeNo tinyint,AcceptedQty decimal(15,3), Rate decimal(18,6),GateEntryNo int,DispatchChallanNo int 
		,RouteIdentity decimal(10,8), IncludedTax decimal(18,6) , ChallanQty decimal(15,3),ReceivedQty decimal(15,3)
		,TransporterNo int,MAXRouteIdentity Decimal(10,8), MINRouteIdentity decimal(10,8)
		,FromLocationNo smallint,ToLocationNo smallint,IsReturnThroughDC bit,SourceDocumentNo tinyint,ExchangeRate decimal(18,6)) cte
	inner join Inventory.Setting s on s.CompanyNo = @CompanyNo	
	inner join masterdata.mItemMaster im on im.ItemNo = cte.ItemNo
		where  cte.ItemNo is not null and AcceptedQty >0 

EXEC sp_xml_removedocument @idoc
 
 
 if exists(select 1 from Purchase.POAmendmentMain pam
			inner join #ItemDetail id on id.ReferenceNo = pam.POAmendmentNo and ReferenceType ='PMPO'
			and pam.IndentTypeNo =1)-- capital
	select @IsManagingExpenseAccount=0
else
	select @IsManagingExpenseAccount = IsManagingExpenseAccount  from Finance.Setting where CompanyNo = @CompanyNo  


 if exists (select 1 from #ItemDetail id inner join Purchase.POAmendmentMain pam on 
		pam.POAmendmentNo = id.ReferenceNo and id.ReferenceType ='PMPO' and pam.IndentTypeNo =5) -- drop shipment
 begin
	exec [Inventory].[IGRNDropShipmentFinancePosting] @ItemDetail=@ItemDetail ,@CompanyNo = @CompanyNo 
	return
 end	 
  
 set @ItemNos = STUFF((select '`' + convert(varchar(7),ItemNo) +'~' + convert(varchar(25),SUM(AcceptedQty * Rate))
			from #ItemDetail group by ItemNo for xml path ('')),1,1,'')

 ---- expense Heads
begin
insert into #Expenses 
	select sum(PEH.Rate * case PEH.BillingQtyNo when 1 then Id.ChallanQty when 2 then Id.ChallanQty 
						  when 4 then ID.ReceivedQty else   
						  id.AcceptedQty end),
						  PAAD.AccountNo ,Id.ItemNo ,Id.MakeNo,Id.TechnicalGradeNo ,PAEAD.AccountNo
						  ,ID.ReferenceNo
		from 
		Purchase.POAmendmentMain PAM
		inner join Purchase.POAmendmentExpenseHead PEH on PAM.POAmendmentNo= PEH.POAmendmentNo
		outer apply (select  APEOD.ItemNo from  Purchase.POAmendmentExpenseHeadOutputDetail APEOD
						inner join Purchase.POAmendmentExpenseHead APEH on APEH.POAmendmentExpenseDetailNo = APEOD.POAmendmentExpenseDetailNo 
						where  APEH.POAmendmentNo = PEH.POAmendmentNo  
							And APEOD.ItemNo in (Select i2.Itemno from #ItemDetail i2)
						group by  APEOD.ItemNo 						
						) PEOD
		inner join masterdata.mExpenseHead eh on eh.ExpenseHeadNo = PEH.ExpenseheadNo and eh.NatureNo <>2
		inner join Purchase.POAmendmentItemDetail PID on PID.POAmendmentNo  = PEH.POAmendmentNo
		inner join #ItemDetail ID on ID.ReferenceType ='PMPO' and (ID.ItemNo= PEOD.ItemNo or ID.ItemNo = PID.ItemNo) and 
						isnull(ID.MakeNo,0) = isnull(PID.MakeNo,0) and
						isnull(ID.TechnicalGradeNo,0) = isnull(PID.TechnicalGradeNo,0) and
						ID.ReferenceNo = PAM.POAmendmentNo 
		left join Purchase.POAccountAssignmentDetail PAAD on PAAD.PONo = PAM.PONo and PAAD.Flag='EH'
					and PAAD.Id = PEH.ExpenseheadNo and PAAD.AccountTypeNo =54
		left join Purchase.POAccountAssignmentDetail PAEAD on PAEAD.PONo = PAM.PONo and PAEAD.Flag='EH'
					and PAEAD.Id = PEH.ExpenseheadNo and PAEAD.AccountTypeNo = 39
		where
		PEH.IncludeInValuation =1 and
		PEH.FreightRateTypeNo  =1 and -- per unit
		isnull(PEH.RouteIdentity ,0)= isnull(id.RouteIdentity,0) and
		(PEH.itemno is null or(PEH.ItemNo = PID.ItemNo and 
						isnull(PEH.MakeNo,0) = isnull(PID.MakeNo,0) and
						isnull(PEH.TechnicalGradeNo,0) = isnull(PID.TechnicalGradeNo,0) ))
		group by  PAAD.AccountNo,Id.ItemNo ,Id.MakeNo,Id.TechnicalGradeNo,ID.ReferenceNo
		,PAEAD.AccountNo option (keep Plan)

insert into #Expenses 
		select sum(ExpenseAmount) ,AccountNo ,ItemNo,MakeNo ,TechnicalGradeNo  ,expenseAccountNo ,ReferenceNo
		from (
		select 		 
		case PEH.BillingQtyNo when 1 then Id.ChallanQty when 2 then Id.ChallanQty 
						  when 4 then ID.ReceivedQty else   
						  id.AcceptedQty end * ((PEH.Rate * pid.Quantity * pid.Rate)/
				(select  sum(pd.Quantity * pd.Rate) 
						from Purchase.POAmendmentItemDetail pd 
						inner join (Select PAEH.POAmendmentNo,PAEH.ItemNo,PAEH.MakeNo,PAEH.TechnicalGradeNo from Purchase.POAmendmentExpenseHead PAEH 
						            Group by PAEH.POAmendmentNo,PAEH.ItemNo,PAEH.MakeNo,PAEH.TechnicalGradeNo   
									  ) FT on FT.POAmendmentNo = pd.POAmendmentNo 
									  and ((FT.ItemNo is not NULL
										  and PEH.ItemNo is not NULL	
										  and pd.ItemNo = ft.ItemNo 
										  and IsNULL(pd.MakeNo,0) = IsNULL(ft.MakeNo,0)
										  and IsNULL(pd.TechnicalGradeNo,0) = IsNULL(ft.TechnicalGradeNo,0)
										  and pd.ItemNo = pid.ItemNo 
										  and IsNULL(pd.MakeNo,0) = IsNULL(pid.MakeNo,0)
										  and IsNULL(pd.TechnicalGradeNo,0) = IsNULL(pid.TechnicalGradeNo,0))									      
										  or (FT.ItemNo is NULL and PEH.ItemNo is NULL))
						where pd.POAmendmentNo = PAM.POAmendmentNo )
				  )/ pid.Quantity ExpenseAmount	
			, PAAD.AccountNo  ,Id.ItemNo ,Id.MakeNo,Id.TechnicalGradeNo
			, PAEAD.AccountNo ExpenseAccountNo,ID.ReferenceNo
		from 
		Purchase.POAmendmentMain PAM
		inner join Purchase.POAmendmentExpenseHead PEH on PAM.POAmendmentNo= PEH.POAmendmentNo 
		outer apply (select APEOD.ItemNo from  Purchase.POAmendmentExpenseHeadOutputDetail APEOD
						inner join Purchase.POAmendmentExpenseHead APEH on APEH.POAmendmentExpenseDetailNo = APEOD.POAmendmentExpenseDetailNo 
						where  APEH.POAmendmentNo = PEH.POAmendmentNo group by  APEOD.ItemNo ) PEOD
		inner join masterdata.mExpenseHead eh on eh.ExpenseHeadNo = PEH.ExpenseheadNo and eh.NatureNo <>2
		inner join Purchase.POAmendmentItemDetail PID on PID.POAmendmentNo= PEH.POAmendmentNo 
		inner join #ItemDetail ID on ID.ReferenceType ='PMPO' and (ID.ItemNo= PEOD.ItemNo or ID.ItemNo = PID.ItemNo) and 
						isnull(ID.MakeNo,0) = isnull(PID.MakeNo,0) and
						isnull(ID.RouteIdentity,0) = isnull(PEH.RouteIdentity,0) and
						isnull(ID.TechnicalGradeNo,0) = isnull(PID.TechnicalGradeNo,0) and
						ID.ReferenceNo = PAM.POAmendmentNo 
		left join Purchase.POAccountAssignmentDetail PAAD on PAAD.PONo = PAM.PONo and PAAD.Flag='EH'
					and PAAD.Id = PEH.ExpenseheadNo and PAAD.AccountTypeNo =54
		left join Purchase.POAccountAssignmentDetail PAEAD on PAEAD.PONo = PAM.PONo and PAEAD.Flag='EH'
					and PAEAD.Id = PEH.ExpenseheadNo and PAEAD.AccountTypeNo = 39
		where
		PEH.IncludeInValuation =1 and
		PEH.FreightRateTypeNo  =2 and -- Fixed Amount
		isnull(PEH.RouteIdentity,0)=ISNULL(id.RouteIdentity,0) and
		(PEH.itemno is null or(PEH.ItemNo = PID.ItemNo and 
						isnull(PEH.MakeNo,0) = isnull(PID.MakeNo,0) and
						isnull(PEH.TechnicalGradeNo,0) = isnull(PID.TechnicalGradeNo,0) ))
		) tt
		group by
		AccountNo,ItemNo ,MakeNo,TechnicalGradeNo ,expenseAccountNo ,ReferenceNo option(keep plan)

insert into #Expenses 
	select sum(ExpenseAmount) ,AccountNo,ItemNo ,MakeNo,TechnicalGradeNo ,ExpenseAccountNo ,ReferenceNo
	from(
	select 
	sum( PEH.Rate * cnt * AcceptedQty * id.Rate )/
		(select sum(AcceptedQty * Rate) from #ItemDetail where GateEntryNo is Null 
						or GateEntryNo = Cid.GateEntryNo) ExpenseAmount
			, PAAD.AccountNo AccountNo ,Id.ItemNo ,Id.MakeNo,Id.TechnicalGradeNo 
			,PAEAD.AccountNo ExpenseAccountNo ,id.ReferenceNo
		from 
		Purchase.POAmendmentMain PAM
		inner join (select count(distinct isnull(GateEntryNo,1)) cnt ,ReferenceNo,GateEntryNo from #ItemDetail id 
	group by ReferenceNo ,GateEntryNo )Cid
				  on Cid.ReferenceNo = PAM.POAmendmentNo 
		inner join Purchase.POAmendmentExpenseHead PEH on PAM.POAmendmentNo= PEH.POAmendmentNo 
		outer apply (select APEOD.ItemNo from  Purchase.POAmendmentExpenseHeadOutputDetail APEOD
						inner join Purchase.POAmendmentExpenseHead APEH on APEH.POAmendmentExpenseDetailNo = APEOD.POAmendmentExpenseDetailNo 
						where  APEH.POAmendmentNo = PEH.POAmendmentNo group by  APEOD.ItemNo ) PEOD
		inner join masterdata.mExpenseHead eh on eh.ExpenseHeadNo = PEH.ExpenseheadNo and eh.NatureNo <>2
		inner join Purchase.POAmendmentItemDetail PID on PID.POAmendmentNo  = PEH.POAmendmentNo
		inner join #ItemDetail ID on ID.ReferenceType ='PMPO' and (ID.ItemNo= PEOD.ItemNo or ID.ItemNo = PID.ItemNo) and 
						isnull(ID.MakeNo,0) = isnull(PID.MakeNo,0) and
						isnull(ID.TechnicalGradeNo,0) = isnull(PID.TechnicalGradeNo,0) and
						ID.ReferenceNo = PAM.POAmendmentNo 
						and Cid.ReferenceNo = ID.ReferenceNo and ID.AcceptedQty >0
						and isnull(ID.GateEntryNo,0) = isnull(Cid.GateEntryNo,0) 
		left join Purchase.POAccountAssignmentDetail PAAD on PAAD.PONo = PAM.PONo and PAAD.Flag='EH'
					and PAAD.Id = PEH.ExpenseheadNo and PAAD.AccountTypeNo =54
		left join Purchase.POAccountAssignmentDetail PAEAD on PAEAD.PONo = PAM.PONo and PAEAD.Flag='EH'
					and PAEAD.Id = PEH.ExpenseheadNo and PAEAD.AccountTypeNo =39
		where
		PEH.IncludeInValuation =1 and
		PEH.FreightRateTypeNo  =3 and -- per trip
		isnull(PEH.RouteIdentity,0) = isnull(id.RouteIdentity,0) and
		(PEH.itemno is null or(PEH.ItemNo = PID.ItemNo and 
						isnull(PEH.MakeNo,0) = isnull(PID.MakeNo,0) and
						isnull(PEH.TechnicalGradeNo,0) = isnull(PID.TechnicalGradeNo,0) ))
		group by PAAD.AccountNo,Id.ItemNo ,Id.MakeNo,Id.TechnicalGradeNo,Cid.GateEntryNo,PAEAD.AccountNo,
				ID.ReferenceNo
		)tt group by AccountNo ,ItemNo,MakeNo ,TechnicalGradeNo , ExpenseAccountNo,ReferenceNo option(keep Plan)

end		

-- Included Taxes
begin
insert into #IncludedTax 
select sum(IncludedAmount),ItemNo ,MakeNo ,TechnicalGradeNo ,RefNo  from (
select sum(case when MC.calculationnature=1 then   (TotalAmount  /pid.Quantity ) * ID.AcceptedQty
				when calculationnature =2 then 0-(TotalAmount  /pid.Quantity ) * ID.AcceptedQty else 0 end)
				IncludedAmount, ID.ItemNo ,ID.MakeNo ,ID.TechnicalGradeNo  ,ID.ReferenceNo RefNo
				
		from #ItemDetail ID
		inner join Purchase.POAmendmentTaxDetail pd on pd.POAmendmentNo = ID.ReferenceNo 
		inner join Purchase.POAmendmentItemTaxDetail ptd on ptd.POAmendmentNo = pd.POAmendmentNo 
					and pd.MiscChargeNo = ptd.MiscChargeNo 
		inner join Purchase.POAmendmentItemDetail pid on pid.POAmendmentNo = pd.POAmendmentNo 		
		inner join purchase.POAmendmentMain pam on pam.POAmendmentno = pid.POAmendmentno
		inner join masterdata.mMiscCharges MC on MC.MiscChargeNo = pd.MiscChargeNo 
		 			and (pd.IncludeInBasic  =1 and pd.ChargeType = 2)  
		where ID.ItemNo = ptd.ItemNo and ISNULL(id.MakeNo,0) = ISNULL(ptd.MakeNo,0)	
			and ISNULL(id.TechnicalGradeNo,0) = isnull(ptd.TechnicalGradeNo ,0)
			and pid.ItemNo = ptd.ItemNo and ISNULL(pid.MakeNo,0) = ISNULL(ptd.MakeNo,0)	
			and ISNULL(pid.TechnicalGradeNo,0) = isnull(ptd.TechnicalGradeNo ,0)
			and ID.ReferenceType  ='PMPO' 
			--and ISNULL(ID.RouteIdentity,0) =isnull(MINRouteIdentity ,0)
			and ISNULL(ID.RouteIdentity,1.1) % .1 = 0
			and isnull(ID.IncludedTax ,0)=0 
			group by ID.ItemNo ,ID.MakeNo ,ID.TechnicalGradeNo ,ID.ReferenceNo
Union all
select sum(case when MC.calculationnature=1 then  0- (TotalAmount  /pid.Quantity ) * ID.AcceptedQty
				when calculationnature =2 then  (TotalAmount  /pid.Quantity ) * ID.AcceptedQty else 0 end)
				IncludedAmount, ID.ItemNo ,ID.MakeNo ,ID.TechnicalGradeNo  ,ID.ReferenceNo
				
		from #ItemDetail ID
		inner join Purchase.POAmendmentTaxDetail pd on pd.POAmendmentNo = ID.ReferenceNo 
		inner join Purchase.POAmendmentItemTaxDetail ptd on ptd.POAmendmentNo = pd.POAmendmentNo 
					and pd.MiscChargeNo = ptd.MiscChargeNo 
		inner join Purchase.POAmendmentItemDetail pid on pid.POAmendmentNo = pd.POAmendmentNo 		
		inner join purchase.POAmendmentMain pam on pam.POAmendmentno = pid.POAmendmentno
		inner join masterdata.mMiscCharges MC on MC.MiscChargeNo = pd.MiscChargeNo 
				and (pd.IncludeInBasic  =0 and ptd.ChargeType =1)
		where ID.ItemNo = ptd.ItemNo and ISNULL(id.MakeNo,0) = ISNULL(ptd.MakeNo,0)	
			and ISNULL(id.TechnicalGradeNo,0) = isnull(ptd.TechnicalGradeNo ,0)
			and pid.ItemNo = ptd.ItemNo and ISNULL(pid.MakeNo,0) = ISNULL(ptd.MakeNo,0)	
			and ISNULL(pid.TechnicalGradeNo,0) = isnull(ptd.TechnicalGradeNo ,0)
			and ID.ReferenceType  ='PMPO' 
			--and ISNULL(ID.RouteIdentity,0) =isnull(MINRouteIdentity ,0)
			and ISNULL(ID.RouteIdentity,1.1) % .1 = 0
			and isnull(ID.IncludedTax ,0)=0 
			group by ID.ItemNo ,ID.MakeNo ,ID.TechnicalGradeNo,ID.ReferenceNo ) tt
	group by ItemNo ,MakeNo ,TechnicalGradeNo,RefNo  option (keep plan)
			
insert into #IncludedTax 
	select sum(IncludedTax) ,ItemNo ,MakeNo,TechnicalGradeNo,0 from #ItemDetail 
			where IncludedTax<>0
		group by ItemNo ,MakeNo,TechnicalGradeNo   option (keep plan)
end 			
 
insert into #Freight 
	select sum(FreightAmount),  ID.ItemNo ,ID.MakeNo , ID.TechnicalGradeNo ,1,ReferenceNo from (
    select  sum(fd.PayableAmount + ISNULL(fd.AdvanceAmount,0)) * (ID.AcceptedQty * ID.Rate)/ TotalAmt
		 + IsNULL((Select  ((SUM(IsNULL(Case when MC.CalculationNature = 2 then -1 else 1 End * FITD.TotalValue,0))* (ID.AcceptedQty * ID.Rate))/ TotalAmt) 
		                    * 
							(sum(fd.PayableAmount + ISNULL(fd.AdvanceAmount,0))/ TotalFreightAmt)  
		       from Finance.FreightInvoiceTaxDetail FITD
			  inner join masterdata.mMiscCharges MC on MC.MiscChargeNo = FITD.MiscChargeNo 
			  inner join Finance.FreightInvoiceDocumentDetail fidd on fidd.FreightInvoiceNo = fITD.FreightInvoiceNo 
	          inner join Finance.FreightInvoiceDocumentDetails fidds on fidds.FreightInvoiceNo = fidd.FreightInvoiceNo 
					and (fidds.GateEntryNo = fidd.RefDocNo or fidds.GRNNo = fidd.RefDocNo )
	             where ID.GateEntryNo = fidds.GateEntryNo and FITD.IncludeInBasic = 1),0) 	

	     FreightAmount
		 , ID.ItemNo ,ID.MakeNo , ID.TechnicalGradeNo ,ID.ReferenceNo 
		 from
	Finance.FreightInvoiceMain fm 
	inner join Finance.FreightInvoiceDocumentDetail fd on fd.FreightInvoiceNo = fm.FreightInvoiceNo 
	inner join Finance.FreightInvoiceDocumentDetails fds on fds.FreightInvoiceNo = fd.FreightInvoiceNo 
					and (fds.GateEntryNo = fd.RefDocNo or fds.GRNNo = fd.RefDocNo )
	inner join #ItemDetail ID on ID.GateEntryNo = fds.GateEntryNo 	and ID.AcceptedQty >0			
	inner join (select SUM(AcceptedQty* rate) TotalAmt,GateEntryNo 
					from #ItemDetail where rate<>0 group by GateEntryNo) GEtt on GEtt.GateEntryNo = ID.GateEntryNo 
	Outer Apply ( Select sum(frdd.PayableAmount + ISNULL(frdd.AdvanceAmount,0)) TotalFreightAmt from Finance.FreightInvoiceDocumentDetails frdds
				  inner join Finance.FreightInvoiceDocumentDetail frdd on frdd.FreightInvoiceNo = frdds.FreightInvoiceNo 	
	              where frdds.GateEntryNo = ID.GateEntryNo) FT 
	where fm.AuthorizedBy is not null
	group by ID.ItemNo ,ID.MakeNo , ID.TechnicalGradeNo ,id.GateEntryNo,ID.AcceptedQty , ID.Rate  ,TotalAmt,TotalFreightAmt 
	,ID.ReferenceNo 
	) ID
	group by   ID.ItemNo ,ID.MakeNo , ID.TechnicalGradeNo,ReferenceNo 	option (keep Plan)

-- FREIGHT ORDER
if not exists(select 1 from #Freight) 
	and exists(select 1 from #ItemDetail id, Purchase.POAmendmentMain pam ,Purchase.FreightOrderOrderDetail fd 
				where pam.POAmendmentNo = ID.ReferenceNo and ISNULL(fd.RouteIdentity,0) = isnull(id.RouteIdentity ,0) 
					and id.ReferenceType ='PMPO' and pam.PONo =fd.PONo)
begin
insert into #Freight 
select SUM(FreightAmt), ItemNo , MakeNo , TechnicalGradeNo ,0,ReferenceNo from (
select (foa.FreightAmount- isnull(DeductionAmount ,0)) * ID.AcceptedQty * ID.Rate /TotalAmount  FreightAmt
			, ID.ItemNo ,ID.MakeNo ,ID.TechnicalGradeNo,ID.ReferenceNo
		from [Purchase].[funcFreightOrderAmount] (@ItemDetail) foa
		inner join #ItemDetail ID on ID.ReferenceNo = foa.POAmendmentNo and ID.ReferenceType = 'PMPO'
		inner join (select SUM(AcceptedQty*rate) TotalAmount ,ReferenceNo from #ItemDetail group by ReferenceNo)REtt
				on REtt.ReferenceNo = ID.ReferenceNo 
		 where ID.AcceptedQty >0  
		 ) tt group by  ItemNo , MakeNo , TechnicalGradeNo,ReferenceNo option (keep plan)
end


---- No Freight Route
if exists(select 1 from #ItemDetail where RouteIdentity is null)
begin
insert into #Freight 
select SUM(FreightAmt), ItemNo , MakeNo , TechnicalGradeNo ,0 ,ReferenceNo from (
select  (foa.FreightAmount- isnull(DeductionAmount ,0)) * ID.AcceptedQty * ID.Rate 
			/(select SUM(AcceptedQty*rate) from #ItemDetail where ReferenceNo = ID.ReferenceNo ) FreightAmt
			, ID.ItemNo ,ID.MakeNo ,ID.TechnicalGradeNo ,id.ReferenceNo
		from [Purchase].[funcFreightOrderAmount] (@ItemDetail) foa
		inner join #ItemDetail ID on ID.ReferenceNo = foa.SCOrderAmendmentNo  and ID.ReferenceType = 'SCSO'
		inner join Purchase.SCOrderAmendmentMain pam on pam.SCOrderAmendmentNo = ID.ReferenceNo 
			where ID.AcceptedQty >0
			and exists(select 1 from Purchase.FreightOrderOrderDetail fd where fd.SCOrderNo = pam.SCOrderNo)
		 ) tt group by  ItemNo , MakeNo , TechnicalGradeNo ,ReferenceNo
		
  if exists(select 1 from #ItemDetail where ReferenceType ='PMPO')
  begin
	insert into #Freight 
	select sum(FreightAmount * id.AcceptedQty) ,ItemNo,MakeNo ,TechnicalGradeNo ,0,ID.ReferenceNo from 
	Purchase.POAmendmentMain  pm
	inner join #ItemDetail id on id.ReferenceNo = pm.POAmendmentNo 
	where pm.FreightTypeNo =1 -- To Pay only
	and FreightRateTypeNo  =1 -- per unit
