CREATE PROCEDURE [Project].[WBSPSendAuthorizationMail] --206,1,1,96
	@WBSRevisionNo int,
	@LoginNo int,
	@LevelNo smallint,
	@AuthDetailNo int ,
	@IsRejected bit = 0 
AS
BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;

	  		
	Declare @Email Varchar(100),			
			@Body Varchar(Max),
			@Subject VArchar(Max),
			@CompanyName varchar(100),
			@PortalURL varchar(500),
			@WBSNo int ,
			@LocalURL varchar(500),
			@StaticURL varchar(500),
			@ProjectDocumentDate smalldatetime,
			@ProjectDocumentNoYearly varchar(30),
			@ProjectDescription varchar(300),
			@MailProfile Varchar(50),	
			@LoginName Varchar(100)=''
			,@PhoneNo varchar(50)
			,@sResponse varchar(800)
			,@SmsSubject varchar(Max) = ''   -- Subject For Sms 
			,@FormName varchar(100) = '' --Form Name for Subject and Message of Notification 
			,@FormTag varchar(4) = '' --FormTag for creating JSON
	Declare @SMSTable as table(col varchar(100))
			
	Select @CompanyName = CM.PrintingName		   
	From Project.PrjWBSRevisionMain  SA
	Inner Join masterdata.mCompanyMaster CM on CM.CompanyNo=SA.CompanyNo
	Where WBSRevisionNo = @WBSRevisionNo 
	
	Select @LocalURL = LocalHost,
		   @MailProfile=MailProfile,
		   @StaticURL =StaticHost
	From Notifications.AuthorizationPageSetting
			
	Select @Email = Email,@LoginName = PrintingName , @PhoneNo = Mobile
	From Security.Login 
	Where LoginNo = @LoginNo

	Select @FormName = FM.FormCaption,@FormTag = FM.FormCode  
	from masterdata.mFormMaster FM
	inner join Project.PrjWBSRevisionMain SOAM on SOAM.WBSRevisionNo = @WBSRevisionNo and FM.FormCode = 'WBSR'



Declare @URLPara Varchar(Max)='',@ForwardURL varchar(max) ='url:../MultiAuthDetail?Param=#SDSOJSON', @LoginDetail varchar(200)

Select @AuthDetailNo = AuthDetailNo
from Notifications.TATAuthorizationDetail
Where LevelNo = @LevelNo and 
		LoginNo = @LoginNo and
		InternalNo = @WBSRevisionNo and
		FormCode = 'WBSR'
					
	If @Email is not null and @IsRejected = 0
	 Begin
			Select @WBSNo  = WBSRevisionNo ,			   
			@ProjectDocumentDate = PM.DocumentDate	,
			@ProjectDocumentNoYearly  = PM.DocumentNoYearly,
			@ProjectDescription = PM.ProjectDescription 
			From Project.PrjWBSRevisionMain  POM 
			inner join Project.ProjectMain PM on PM.ProjectNo = POM.ProjectNo 
			Where WBSRevisionNo = @WBSRevisionNo 
			
					
			Select @URLPara='#HostName/Authentication?q=',
				   @ForwardURL = REPLACE(@ForwardURL,'#SDSOJSON',masterdata.EncodeToBase64( 
																						   '{"FormCode":"' + @FormTag + '","Id":' + CONVERT(varchar,@WBSRevisionNo) + '}'			
																						   )),
				   @LoginDetail = 'LoginNo:'+Convert(varchar,@LoginNo)
			
			
			Select @URLPara = @URLPara+masterdata.EncodeToBase64(@LoginDetail+','+@ForwardURL)
			
		--	Select @LoginDetail,@ForwardURL,@URLPara
		
									
			Set @Subject = 'Authorization Request for Work BreakDown Structure for Project Code ' +  '-' + @ProjectDocumentNoYearly 
			Set @Body = '<p>Dear '+ @LoginName + ',</p>

				<p><b>Greetings from '+ISNULL(@CompanyName,'')+' </b></p>

				<p> Work Breakdown Structure for Project Code : <strong>'+ISNULL(@ProjectDocumentNoYearly,'')+'</strong>, Project Description : <strong>'+ ISNULL(@ProjectDescription,'')+ '</strong>, Dated : <strong>'+CONVERT(varchar,@ProjectDocumentDate,103)+'</strong> has been created. It is pending for your approval ' + IsNull('of Level ' + CONVERT(Varchar,@LevelNo),'') +  '.</p>

				<p>For more detail login at:</p>'
				
			If @LocalURL is not null	
			Set @Body= @Body+'
			         
			        <p>If you are <strong> connected to Local  </strong> then use below link </p>
					
					<p><a href="'+REPLACE(@URLPara,'#HostName',@LocalURL)+'" target="_blank">'+ REPLACE(@URLPara,'#HostName',@LocalURL)+ '</a></p>'
					
			
			If @StaticURL is not null
			Begin		
				Set @Body= @Body+'
						
						<p>If you are <strong>Not connected to Local </strong> then use below link </p>
					    
						<p><a href="'+REPLACE(@URLPara,'#HostName',@StaticURL)+'" target="_blank">'+ REPLACE(@URLPara,'#HostName',@StaticURL)+ '</a></p>'
				
			    Set @SmsSubject = @Subject + ','+ CHAR(10) + ' Please refer this link for more details : '+ REPLACE(@URLPara,'#HostName',@StaticURL)
			End
			Else						--When Static url is null ,assigning mail subject to sms subject
			 Begin
			    Set @SmsSubject = @Subject 
		     End
			  
				   
		   
		  IF @MailProfile IS NOT NULL
			exec [Utility].[SendEmailThroughProfile] @Email,@Body,@Subject,@MailProfile
		
		  if isnull(@PhoneNo,'') <>''
		   insert into @SMSTable
			exec  [Notifications].[SendSMS] @PhoneNo,@SmsSubject,@sResponse output	
				
		 End	
	
	else If @Email is not null and @IsRejected = 1
	 Begin
			Select @WBSNo  = WBSRevisionNo ,			   
			@ProjectDocumentDate = PM.DocumentDate	,
			@ProjectDocumentNoYearly  = PM.DocumentNoYearly
			From Project.PrjWBSRevisionMain  POM 
			inner join Project.ProjectMain PM on PM.ProjectNo = POM.ProjectNo 
			Where WBSRevisionNo = @WBSRevisionNo 

                Select @ForwardURL = @ForwardURL + '&IsRejection=#Rejection'
			    Select @ForwardURL = Replace(@ForwardURL,'#Rejection',masterdata.EncodeToBase64( CONVERT(varchar,1)))   
			      
				Select @URLPara='#HostName/Authentication?q=',
				   @ForwardURL = REPLACE(@ForwardURL,'#SDSOJSON',masterdata.EncodeToBase64(
																						   '{"FormCode":"' + @FormTag + '","Id":' + CONVERT(varchar,@WBSRevisionNo) + '}'			
				                                                                         --CONVERT(varchar,@WBSRevisionNo)
				                                                                                                                  )),
				   @LoginDetail = 'LoginNo:'+Convert(varchar,@LoginNo)
			
			
			Select @URLPara = @URLPara+masterdata.EncodeToBase64(@LoginDetail+','+@ForwardURL)
			
			
			
			
			select @Subject = ' Work BreakDown Structure for Projet Code -' + @ProjectDocumentNoYearly  + ',Dated - ' + CONVERT(varchar,@ProjectDocumentDate,103) + ' is rejected for Reason: ' + isnulL(LastComment ,'')
				from Notifications.TATAuthorizationDetail where FormCode ='WBSR' and InternalNo = @WBSRevisionNo  
				and  AuthDetailNo >= @AuthDetailNo and LastComment is not null
				
				 
			select @Body = '<p>Dear '+ @LoginName + ',</p>

				<p><b>Greetings from '+ISNULL(@CompanyName,'')+' </b></p>

				<p>We would like to inform you that ' + @FormName + ' No. ' + @ProjectDocumentNoYearly + ' has been rejected, due to Reason :"' + isnulL(LastComment,'') + '".'
				 from Notifications.TATAuthorizationDetail where 
				  AuthDetailNo >= @AuthDetailNo and LastComment is not null
				  and FormCode ='WBSR' and InternalNo = @WBSRevisionNo 
		
		  	Select @Body = @Body + '<p>For more detail login at:</p>'
				
			If @LocalURL is not null	
			Set @Body= @Body+'
			         
			        <p>If you are <strong> connected to Local  </strong> then use below link </p>
					
					<p><a href="'+REPLACE(@URLPara,'#HostName',@LocalURL)+'" target="_blank">'+ REPLACE(@URLPara,'#HostName',@LocalURL)+ '</a></p>'
			
			If @StaticURL is not null
			Begin		
				Set @Body= @Body+'
						
						<p>If you are <strong>Not connected to Local </strong> then use below link </p>
					    
						<p><a href="'+REPLACE(@URLPara,'#HostName',@StaticURL)+'" target="_blank">'+ REPLACE(@URLPara,'#HostName',@StaticURL)+ '</a></p>'
				
				Set @SmsSubject = @Subject + ',' + CHAR(10) + ' Please refer this link for more details : ' + REPLACE(@URLPara,'#HostName',@StaticURL)
			End
			Else							--When Static url is null ,assigning mail subject to sms subject
			 Begin
			  Set @SmsSubject = @Subject 
		     End
		
			IF @MailProfile IS NOT NULL
				exec [Utility].[SendEmailThroughProfile] @Email,@Body,@Subject,@MailProfile
			 
			if isnull(@PhoneNo,'') <>''
			 insert into @SMSTable
			  exec [Notifications].[SendSMS] @PhoneNo,@SmsSubject,@sResponse output		
		 End		 
   
END
