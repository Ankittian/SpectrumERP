CREATE PROCEDURE [Information].[StockDetail] 
	-- Add the parameters for the stored procedure here
	@DocumentDate smalldatetime,
	@ItemNo int ,
	@StockNatureNo tinyint = 1
AS
BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;

    -- Insert statements for procedure here
	declare @str varchar(max)
    select @str=masterdata.JSForInformation('MMPR')
    set @str=@str+masterdata.CSSForInformation('MMPR')
	set @str=@str+'<table class="altrowstable" id="alternatecolorStockInfo">'
	set @str=@str+'<Caption><Font color="Maroon"><h3 align="Left"><b>Stock Detail</b></h3></Font></Caption>'
    set @str=@str+'<tr><th>Sl. No.</th><th>Company Name</th><th>Division</th><th>Warehouse</th><th>Make</th><th>Iscapital</th><th>Stock Quantity</th></tr>'
	set @str=@str
				 +isnull((replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					replace((    
							 Select ROW_NUMBER() over (order by CM.CompanyName) as td1
							        ,CM.CompanyName as [td2]
								    ,isnull(DV.DivisionName,'') as [td3]
								    ,WH.WareHouseName as [td4]
								    ,isnull(MM.MakeName,'') as [td5]
								    ,ls.IsCapital as [td8] -- added on 29 aug 2019
								    ,sum(isnull(ld.ReceivedQty,0)) - sum(isnull(ld.IssuedQty,0))
								                                   + sum(isnull(ld.ReturnQty,0)) as [td7]
							 from
								  inventory.invlotSummary	ls
								  inner join inventory.invlotDetail ld on ls.InvlotSummaryNo  = ld.InvlotSummaryNo
								  Left Join payroll.tblDivisionMaster DV on ls.DivisionNo=DV.DivisionNo
								  Inner Join masterdata.mWarehouse WH on WH.WareHouseNo=ls.WarehouseNo
								  Inner Join masterdata.mCompanyMaster CM on CM.CompanyNo=ls.CompanyNo
								  Left Join masterdata.mMakeMaster MM on MM.MakeNo=ls.MakeNo
							Where ld.DocumentDate<=@DocumentDate
							And ls.ItemNo=@ItemNo
							And LS.IsRejectedLot=0		      
							and ls.StockNatureNo = @StockNatureNo And ls.IsRejectedLot=0 -- added on 29 aug 2019
							Group By CM.CompanyName,DV.DivisionName,WH.WareHouseName,MM.MakeName ,ls.IsCapital
							for xml path('tr')),'<td1>','<td style="text-align: Center;">')
							                   ,'<td2>','<td>')
											   ,'<td3>','<td>')
											   ,'<td4>','<td>')
											   ,'<td5>','<td>')
											   ,'<td6>','<td>')
											   ,'<td8>','<td>')											   
											   ,'<td7>','<td style="text-align: right;">')),'')
				   
	set @str=@str+'</table>'
	
	select @str
END
