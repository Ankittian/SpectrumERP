				,CAST(null as int) [ARDNo]
		from masterdata.mVendorLocationDetail VLD 
			left join Finance.ARDVendorLocationRule LR on LR.VendorLocationNo=VLD.VendorLocationNo
													and LR.AccountTypeNo=4
			left join Purchase.SCOrderAccountAssignmentDetail AD on AD.Id=@VendorLocationNo
															and AD.Flag='VL'
															and AD.AccountTypeNo=4
															and AD.SCOrderNo=@SCOrderNo
		where vld.VendorLocationNo=@VendorLocationNo
		union all
		select top 1 'Supplier Prepayable Account',VLD.VendorName+ISNULL(REPLACE(AD.Flag,AD.Flag,''),' *')
				,isnull(AD.AccountNo,LR.AccountNo)
				,5,@VendorLocationNo Id,'VL' [Flag]
				, 0 as [Editable]
				,CAST(null as int) [ARDNo]
		from masterdata.mVendorLocationDetail VLD 
			left join Finance.ARDVendorLocationRule LR on LR.VendorLocationNo=VLD.VendorLocationNo
													and LR.AccountTypeNo=5
			left join Purchase.SCOrderAccountAssignmentDetail AD on AD.Id=@VendorLocationNo
															and AD.Flag='VL'
															and AD.AccountTypeNo=5
															and AD.SCOrderNo=@SCOrderNo
		where vld.VendorLocationNo=@VendorLocationNo
			and exists (select 1 from @PaymentTermDetail where PaymentTypeNo<>3)
		
	END TRY
	BEGIN CATCH
		INSERT INTO [masterdata].[mErrorLog]
			   ([ErrorDate]
			   ,[ErrorNo]
			   ,[ErrorSeverity]
			   ,[ErrorState]
			   ,[ErrorProcedure]
			   ,[ErrorLine]
			   ,[ErrorDescription]
			   ,[UserName]
			   ,[FormCode])
		VALUES
			   (getdate()
			   ,ERROR_NUMBER()
			   ,ERROR_SEVERITY()
			   ,ERROR_STATE()
			   ,ERROR_PROCEDURE()
			   ,ERROR_LINE()
			   ,ERROR_MESSAGE()
			   ,SUSER_SNAME()
			   ,'SCSO')
	END CATCH
