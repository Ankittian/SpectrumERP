CREATE PROCEDURE [Information].[POGateEntryDetail] 
	-- Add the parameters for the stored procedure here
	@CompanyNo tinyint,
	@DocumentDate smalldatetime,
	@PONo int
	
AS
BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;

    -- Insert statements for procedure here
	

	declare @str varchar(max)
    select @str=masterdata.JSForInformation('PMPO')
    set @str=@str+masterdata.CSSForInformation('PMPO')
	set @str=@str+'<table class="altrowstable" id="alternatecolorPOGateEntryDetail">'
	set @str=@str+'<Caption><Font color="Maroon"><h3 align="Left"><b>Gate Entry Detail</b></h3></Font></Caption>'
    set @str=@str+'<tr><th>Sl. No.</th><th>Doc. No.</th><th>Doc. Date</th><th>Challan No.</th><th>Challan Date</th><th>Transporter</th><th>Vehicle No.</th><th>LR No.</th><th>LR Date</th><th>Ref No./Date</th><th>Item Name</th><th>Challan Qty.</th><th>Rate</th><th>Amount</th> <th>GRN No.</th><th>GRN Date</th><th>Route Id</th></tr>'
	set @str=@str
				 +( 
					replace( 
					replace(
					replace(
					replace( 
					replace(
					replace(
					replace( 
					replace(
					replace(
					replace(
					replace( 
					replace(
					replace(
					replace( 
					replace(
					replace(
					replace( 
							(
								Select   Row_number() over (order by GEM.DocumentDate desc) as td1
										,GEM.DocumentNoYearly td2
										,Convert(varchar(20), GEM.DocumentDate,103) td3
										,GEM.ChallanNo td4
										,Convert(varchar(20), GEM.ChallanDate,103) td5
										,ISNULL( VC.Name,'') td6
										,ISNULL( GEM.VehicleNo,'') td7
										,ISNULL( GEM.LRNo,'') td8
										,Isnull( Convert(varchar(20), GEM.LRDate,103),'') td9
										,Isnull( GEM.ReferenceNo,'') td10
										,IMM.ItemDescription td11
										,GEID.ChallanQty td12
										,Isnull( Cast( CONVERT(Decimal(15,2), GEID.Rate ) as varchar(20)),'') td13
										,ISNULL( Convert( Decimal(15,2),GEID.ChallanQty *GEID.Rate) ,0) td14
										,ISNULL( GRNM.DocumentNoYearly,'') td15
										,ISNULL( Convert(varchar(20),GRNM.DocumentDate,103),'') td16
										,ISNULL(CAST( GEID.RouteIdentity as varchar(50)) ,'') td17
								From GateEntry.InvGateEntryMain	GEM	
								INNER JOIN GateEntry.InvGateEntryItemDetail GEID ON GEID.GateEntryNo=GEM.GateEntryNo
																				  And GEM.GateEntrySourceNo=2
																				  And GEM.ModuleNo=2
								INNER JOIN Purchase.POAmendmentMain POAM ON POAM.POAmendmentNo=GEID.POAmendmentNo
								Inner Join Purchase.POMain POM on POM.PONo=POAM.PONo
								Left JOIN Inventory.GRNItemDetail GRID	 ON GEID.GateEntryNo=GRID.GateEntryNo
																			   AND IsNull(GEID.POAmendmentNo,0)=IsNull(GRID.POAmendmentNo,0)
																			   AND GEID.ItemNo=GRID.ItemNo
																			   AND IsNull(GEID.MakeNo,0)=IsNull(GRID.MakeNo,0)
																			   And IsNull(GEID.TechnicalGradeNo,0)=IsNull(GRID.TechnicalGradeNo,0)
								Left Join Inventory.GRNMain GRNM ON GRID.GRNNo=GRNM.GRNNo
								Inner Join masterdata.mItemMaster IMM on IMM.ItemNo=GEID.ItemNo
								LEFT JOIN masterdata.mVendorCard VC ON VC.VendorNo=GEM.TransporterNo And VC.VendorTypeNo=1


								Where POM.PONo = @PONo
										And POM.DocumentDate = @DocumentDate
										And POM.CompanyNo = @CompanyNo
							for xml path('tr')),'<td1>','<td style="text-align: Center;">')
							                   ,'<td2>','<td>')
		   									   ,'<td3>','<td>')				
											   ,'<td4>','<td>')
											   ,'<td5>','<td>')
		   									   ,'<td6>','<td>')
											   ,'<td7>','<td>')
											   ,'<td8>','<td>')
											   ,'<td9>','<td>')
											   ,'<td10>','<td>')
											   ,'<td11>','<td>')
											   ,'<td12>','<td style="text-align: Right;">')	
											   ,'<td13>','<td style="text-align: Right;">')
											   ,'<td14>','<td style="text-align: Right;">')
											   ,'<td15>','<td>')
											   ,'<td16>','<td>')
											   ,'<td17>','<td>')
											   
									    	)
								
				   
	set @str=@str+'</table>'
	
	select @str
END
