-- =============================================
--- Inventory].[IMSTFinancePosting] '<DocumentElement />' 
CREATE PROCEDURE [Finance].[ISMLFinancePosting]	
	@ItemDetail xml ,
	@CompanyNo tinyint ,
	@ItemNos varchar(max)
	
AS
BEGIN

	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;

begin try
declare @IsManagingExpenseAccount bit

Create table #ItemDetail (AccountNo Int,DiffAmount decimal(18,2))



create table #Posting ([Add to Basic] bit,Particular varchar(500),[Debit Amount] decimal(15,2) , 
		[Credit Amount] decimal(15,2),[Account Name] int, [Ref. Detail] varchar(50),[Cost Center] varchar(50),
		[Amount] decimal(15,2), [Type] tinyint,SlNo tinyint ,MiscChargeNo int , TaxCodeMiscChargesDetailNo int,
			ItemNos varchar(200))

;with cte as 
	(
		SELECT v.value('AccountNo[1]','Int') AS AccountNo,
		v.value('Difference[1]','money') [Difference]
		FROM @ItemDetail.nodes('/DocumentElement/Table') x(v)
	)
	Insert into #ItemDetail 
	select AccountNo ,sum([Difference])
	from cte group by AccountNo
		 
	 
	 Insert into #Posting 
		select cast(0 as bit) [Add to Basic]
				,'Stock Account' as Particular
				,convert(decimal(15,2), case when sum(Amount)>0 then sum(Amount) else null end ) [Debit Amount]
				,convert(decimal(15,2), case when sum(Amount) >0 then null else abs(sum(Amount)) end) [Credit Amount]
				,I.AccountNo [Account Name]
				,cast(null as varchar) [Ref. Detail]
				,cast(null as varchar) [Cost Center]
				,abs(SUM(Amount)) [Amount]
				,case when sum(Amount)>0 then 1 else 2 end [Type]
				,ROW_NUMBER()over(order by I.AccountNo ) + (select isnull(max(SlNo),0) from #Posting) SlNo
				,null MiscChargeNo
				,null TaxCodeMiscChargesDetailNo
				,'' ItemNos
			from (select  
				sum(DiffAmount) Amount ,AccountNo
							from #ItemDetail group by AccountNo  )I
		group by I.AccountNo 
		 
		Insert into #Posting 
		select cast(0 as bit) [Add to Basic]
				,'Stock Adjustment A/c' as Particular
				,convert(decimal(15,2), case when sum(Amount)>0 then null else ABS(sum(Amount)) end ) [Debit Amount]
				,convert(decimal(15,2), case when sum(Amount) >0 then ABS(sum(Amount)) else null end) [Credit Amount]
				,IR.AccountNo [Account Name]
				,cast(null as varchar) [Ref. Detail]
				,cast(null as varchar) [Cost Center]
				,abs(SUM(Amount)) [Amount]
				,case when sum(Amount)>0 then 2 else 1 end [Type]
				,ROW_NUMBER()over(order by IR.AccountNo ) + (select isnull(max(SlNo),0) from #Posting) SlNo
				,null MiscChargeNo
				,null TaxCodeMiscChargesDetailNo
				,'' ItemNos
			from (select  
				sum(DiffAmount) Amount ,AccountNo
							from #ItemDetail group by AccountNo  )I
				outer apply (select max(IR.AccountNo)AccountNo
							from 
							 Finance.ARDItemRule IR
							where AccountTypeNo= 19 and 
							ItemNo in (select ItemNo from Finance.ARDItemRule a where a.AccountNo = I.AccountNo and a.AccountTypeNo =14)
								) IR 
		group by IR.AccountNo 
		

 if exists(select 1 from #Posting having abs(SUM([Credit Amount])- SUM([Debit Amount]))<=.5)
 begin
	update PD set [Debit Amount] = [Debit Amount] + ISNULL(diff,0)
		from
		#Posting PD
		cross join (select SUM([Credit Amount])- SUM([Debit Amount]) Diff from #Posting) tt
		where PD.SlNo  = 1
	
	update PD set [Credit Amount] = [Credit Amount] + ISNULL(diff,0)
		from
		#Posting PD
		cross join (select SUM([Debit Amount])- SUM([Credit Amount]) Diff from #Posting) tt
		where PD.SlNo  = 1
			
 end
 
 
select * from #Posting
 where isnull([Debit Amount],0) >0 or isnull([Credit Amount],0) >0
   
   
   
END TRY
	
	BEGIN CATCH
		select ERROR_MESSAGE()
		INSERT INTO [masterdata].[mErrorLog]
			   ([ErrorDate]
			   ,[ErrorNo]
			   ,[ErrorSeverity]
			   ,[ErrorState]
			   ,[ErrorProcedure]
			   ,[ErrorLine]
			   ,[ErrorDescription]
			   ,[UserName]
			   ,[FormCode])
		VALUES
			   (getdate()
			   ,ERROR_NUMBER()
			   ,ERROR_SEVERITY()
			   ,ERROR_STATE()
			   ,ERROR_PROCEDURE()
			   ,ERROR_LINE()
			   ,ERROR_MESSAGE()
			   ,SUSER_SNAME()
			   ,'IGRN')
	END CATCH
END
