Stored procedure EAM.ChangeMRStatus: To Change Status OF Maintenance Request. Called from EAM.EARQAuthorize.
Writes: EAM.MaintenanceRequestMain, EAM.RequestFaultDetails
Reads: EAM.MaintenanceOrderMain, EAM.MaintenanceOrderRefDetail, EAM.MaintenanceReporting, EAM.MaintenanceRequestMain, EAM.RequestFaultDetails
Parameters: @RequestNo int, @FaultNo int, @RequestDescriptionNo int, @StatusNo tinyint, @OutputStatus bit OUTPUT, @Result varchar(100) OUTPUT

CREATE PROCEDURE [EAM].[ChangeMRStatus] 	
	 @RequestNo  int
	,@FaultNo  int
	,@RequestDescriptionNo  int
	,@StatusNo tinyint
	,@OutputStatus bit output
	,@Result varchar(100) output	
AS
BEGIN
Begin Try
	
If Exists (Select 1 From EAM.MaintenanceRequestMain  Where RequestNo =@RequestNo)
BEGIN			
	
	If @StatusNo = 2
	Begin 
		If Exists(	Select 1 
					From EAM.MaintenanceOrderMain MOM
					Inner Join EAM.MaintenanceOrderRefDetail MORF On MOM.MaintenanceOrderNo =  MORF.MaintenanceOrderNo
					Where MOM.SourceDocumentNo = 2
						  And ISNULL(MORF.RefDocNo ,0) = @RequestNo
						  And (MORF.FaultNo =ISNULL(@FaultNo,0) Or MORF.MaintenanceDescriptionNo = ISNULL(@RequestDescriptionNo,0))				
					Union All
					Select 1 
					From EAM.MaintenanceReporting RP
					Inner Join EAM.MaintenanceRequestMain RM on RM.RequestNo = RP.RequestNo
					Inner Join EAM.RequestFaultDetails RD on RD.RequestNo = RP.RequestNo
					Where ISNULL(RD.RequestNo ,0) = @RequestNo 
						 And (FaultNo =ISNULL(@FaultNo,0) Or RequestDescriptionNo = ISNULL(@RequestDescriptionNo,0))								
						 And RD.RCANo is not null
				)
		Begin
		
			UPDATE EAM.RequestFaultDetails
			SET StatusNo = 2 
			Where RequestNo = @RequestNo
				  And (FaultNo =ISNULL(@FaultNo,0) Or RequestDescriptionNo = ISNULL(@RequestDescriptionNo,0) )				
		End
		Else
		Begin
			UPDATE EAM.RequestFaultDetails
			SET StatusNo = 1 
			Where RequestNo = @RequestNo
				  And (FaultNo =ISNULL(@FaultNo,0) Or RequestDescriptionNo = ISNULL(@RequestDescriptionNo,0) )				
		End
	
	End
	
	If @StatusNo <> 2
	Begin 
		UPDATE EAM.RequestFaultDetails
		SET StatusNo =@StatusNo 
		Where RequestNo = @RequestNo
			  And (FaultNo =ISNULL(@FaultNo,0) Or RequestDescriptionNo = ISNULL(@RequestDescriptionNo,0) )				
	End
	
	 IF Exists(	select 1 From EAM.RequestFaultDetails where RequestNo = @RequestNo And statusNo IN (1,8))
	 BEGIN
		If Exists(	Select 1 
					From EAM.MaintenanceOrderMain MOM
					Inner Join EAM.MaintenanceOrderRefDetail MORF On MOM.MaintenanceOrderNo = MORF.MaintenanceOrderNo
					Where ISNULL(MORF.RefDocNo,0) = @RequestNo And MOM.SourceDocumentNo = 2
					Union All
					Select 1 
					From EAM.MaintenanceReporting RP
					Inner Join EAM.MaintenanceRequestMain RM on RM.RequestNo = RP.RequestNo
					Inner Join EAM.RequestFaultDetails RD on RD.RequestNo = RP.RequestNo
					Where ISNULL(RD.RequestNo ,0) = @RequestNo 
						 And (FaultNo =ISNULL(@FaultNo,0) Or RequestDescriptionNo = ISNULL(@RequestDescriptionNo,0))				
						 And RD.RCANo is not null
				)
		Begin
			update EAM.MaintenanceRequestMain set StatusNo = 8 where  RequestNo= @RequestNo
		End
		Else
		Begin
			update EAM.MaintenanceRequestMain set StatusNo = 1 where  RequestNo= @RequestNo
		End
	 END
	 Else if exists(select 1 from EAM.RequestFaultDetails where  RequestNo = @RequestNo and Statusno = 3 )-- Hold 
	 Begin
		update EAM.MaintenanceRequestMain set StatusNo = 3 where  RequestNo= @RequestNo
	 End
	 Else If exists(select 1 from EAM.RequestFaultDetails where  RequestNo = @RequestNo and Statusno = 4 )--cancel
	 Begin
		update EAM.MaintenanceRequestMain set StatusNo = 4 where  RequestNo= @RequestNo
	 End
	 Else if exists(select 1 from EAM.RequestFaultDetails where  RequestNo = @RequestNo and Statusno = 7)--shortclose
	 Begin
		update EAM.MaintenanceRequestMain set StatusNo = 7 where  RequestNo= @RequestNo
	 End				
	 Else if exists(select 1 from EAM.RequestFaultDetails where  RequestNo = @RequestNo and Statusno = 2)--completed
	 Begin
		update EAM.MaintenanceRequestMain set StatusNo = 2 where  RequestNo= @RequestNo
	 End
 End

 set @OutputStatus =1
 set @Result='Success'

End Try

Begin Catch
	set @OutputStatus =0
	set @Result=error_message()

	select 0, error_message()
End Catch

END
