CREATE PROCEDURE [Information].[PendingRFQForVendor] 
	-- Add the parameters for the stored procedure here
	@CompanyNo tinyint,
	@DocumentDate smalldatetime,
	@VendorLocationNo int
	
AS
BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;

    -- Insert statements for procedure here
	

	declare @str varchar(max)
    select @str=masterdata.JSForInformation('PRFQ')
    set @str=@str+masterdata.CSSForInformation('PRFQ')
	set @str=@str+'<table class="altrowstable" id="alternatecolorPendingRFQForVendorInfo">'
	set @str=@str+'<Caption><Font color="Maroon"><h3 align="Left"><b>Pending RFQ For Quotation</b></h3></Font></Caption>'
    set @str=@str+'<tr><th>Sl. No.</th><th>RFQ No.</th><th>RFQ Date</th><th>Submission Date</th><th>Validity Date</th><th>Created By</th><th>Authorized By</th></tr>'
	set @str=@str
				 +( replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					replace((    
					      Select ROW_NUMBER() over ( Order by RM.DocumentDate desc) as td1,
								 RM.DocumentNoYearly td2,
								 Convert(Varchar(20),RM.DocumentDate,103) td3, 
								 Convert(Varchar(20),RM.SubmissionDate,103) td4, 
								 Convert(Varchar(20),RM.ValidityDate,103) td5,
								 SL.Printingname as td6, 
								 Isnull(SLA.Printingname,'') td7
						From Purchase.RFQMain RM
						Inner Join Purchase.RFQVendorDetail VD ON VD.RFQNo=RM.RFQNo
						Inner JOIN Security.Login SL on SL.LoginNo =  RM.CreatedBy
						Left JOIN Security.Login SLA on SLA.LoginNo =  RM.AuthorizedBy
						Left Join 
						(Select QM.RFQNo From Purchase.RevisedQuotationMain QM 
						 Inner Join Purchase.POMain PO ON PO.RefDocumentNo= QM.RevisedQuotationNo
						)tt ON tt.RFQNo=RM.RFQNo---For Pending Quotation
						Where tt.RFQNo Is null And RM.StatusNo IN (1,8,6)
								And RM.CompanyNo=@CompanyNo
								And RM.DocumentDate<=@DocumentDate
								And VD.VendorLocationNo=@VendorLocationNo
						For xml path('tr')),'<td1>','<td style="text-align: Center;">')
						                     ,'<td2>','<td>')
											 ,'<td3>','<td style="text-align: Center;">')
											 ,'<td4>','<td style="text-align: Center;">')
											 ,'<td5>','<td style="text-align: Center;">')
											 ,'<td6>','<td>')
											 ,'<td7>','<td>')	
											)
	set @str=@str+'</table>'
	
	select @str
END
