CREATE PROCEDURE [Information].[PendingPurchaseRequestForItem] 
	-- Add the parameters for the stored procedure here
	@CompanyNo tinyint,
	@DocumentDate smalldatetime,
	@ItemNo int
AS
BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;

    -- Insert statements for procedure here
	

	declare @str varchar(max)
    select @str=masterdata.JSForInformation('PMPO')
    set @str=@str+masterdata.CSSForInformation('PMPO')
	set @str=@str+'<table class="altrowstable" id="alternatecolorPendingPurchaseRequestInfo">'
	set @str=@str+'<Caption><Font color="Maroon"><h3 align="Left"><b>Pending Purchase Request</b></h3></Font></Caption>'
    set @str=@str+'<tr><th>Sl. No.</th><th>Division</th><th>Department</th><th>Cost Center</th><th>Document No.</th><th>Document Date</th><th>Make</th><th>PR Quantity</th><th>PO Quantity</th><th>Balance Quantity</th><th>Created By </th><th>Authorized By</th></tr>'
	set @str=@str
				 +( REPLACE(
					REPLACE(
				    REPLACE(
				    REPLACE(
				    REPLACE(
				    REPLACE(
				    replace(
					replace(
					replace(
					replace(
					replace(
					replace((    
							Select   ROW_NUMBER() over ( Order by IDM.DocumentDate desc) as td1
							        ,Isnull(DM.DivisionName,'') td2
									,Isnull( TDM.DeptName ,'') td3
									,Isnull(CCM.CostCenterName,'') td4
									,IDM.DocumentNoYearly td5
									,convert(varchar(20),IDM.DocumentDate,103) td6
									,Isnull(MM.MakeName,'') td7
									,Sum( IDIT.IndentQty )td8
									,SUM(Isnull(IDIT.POQty,0)) td9
									,SUM((IDIT.IndentQty-ISNULL(IDIT.POQty,0))) td10
									,CRT.PrintingName td11
									,Isnull(AUT.PrintingName,'') td12
							From  Inventory.IndentMain IDM
							INNER JOIN  Inventory.IndentItemDetail IDIT ON IDIT.IndentNo=IDM.IndentNo
							INNER JOIN payroll.tblDeptMaster TDM ON TDM.DeptNo=IDM.DeptNo
							Inner JOIN payroll.tblDivisionMaster DM ON DM.DivisionNo=TDM.DivisionNo
							LEFT JOIN masterdata.mCostCenterMaster CCM ON CCM.CostCenterNo=IDIT.CostCenterNo
							LEFT JOIN Masterdata.mMakeMaster MM ON MM.MakeNo=IDIT.MakeNo
							Inner join Security.Login CRT On IDM.CreatedBy = CRT.LoginNo
							Left join Security.Login AUT On IDM.AuthorizedBy = AUT.LoginNo
							Where IDIT.ItemNo = @ItemNo
									And (IDIT.IndentQty-Isnull(IDIT.POQty,0)) >= 0 -----For Pending Purchase Request
									and IDM.CompanyNo=@CompanyNo
									And IDM.DocumentDate<=@DocumentDate
							Group By  DM.DivisionName
									,TDM.DeptName
									,CCM.CostCenterName
									,IDM.DocumentNoYearly 
									,IDM.DocumentDate 
									,MM.MakeName
									,CRT.PrintingName 
									,AUT.PrintingName	  
						 for xml path('tr')),'<td1>','<td style="text-align: center;">')
						                     ,'<td2>','<td>')
											 ,'<td3>','<td>')
											 ,'<td4>','<td>')
											 ,'<td5>','<td style="text-align: center;">')
											 ,'<td6>','<td>')
											 ,'<td7>','<td>')
											 ,'<td8>','<td style="text-align: right;">')
											 ,'<td9>','<td style="text-align: right;">')
											 ,'<td10>','<td style="text-align: right;">')
											 ,'<td11>','<td>')
											 ,'<td12>','<td>')
											)
								
				   
	set @str=@str+'</table>'
	
	select @str
END
