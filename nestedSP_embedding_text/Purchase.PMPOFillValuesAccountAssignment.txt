													and LR.AccountTypeNo = 7
			left join Purchase.POAccountAssignmentDetail AD on AD.Id=@VendorLocationNo
															and AD.Flag='VL'
															and AD.AccountTypeNo  = 7
															and AD.PONo=@PONo
		Left Join masterdata.mAccountMaster AM on AM.AccountNo=isnull(AD.AccountNo,LR.AccountNo) 															
		where vld.VendorLocationNo=@VendorLocationNo
		union all
		select top 1 'Supplier Prepayable Account',VLD.VendorName+ISNULL(REPLACE(AD.Flag,AD.Flag,''),' *')
				,isnull(AD.AccountNo,LR.AccountNo)
				,5,@VendorLocationNo Id,'VL' [Flag]
				,AM.AccountName
				,0 [Editable]
				,CAST(NULL as int) [ARDNo]
		from masterdata.mVendorLocationDetail VLD 
			left join Finance.ARDVendorLocationRule LR on LR.VendorLocationNo=VLD.VendorLocationNo
													and LR.AccountTypeNo=5
			left join Purchase.POAccountAssignmentDetail AD on AD.Id=@VendorLocationNo
															and AD.Flag='VL'
															and AD.AccountTypeNo=5
															and AD.PONo=@PONo
		Left Join masterdata.mAccountMaster AM on AM.AccountNo=isnull(AD.AccountNo,LR.AccountNo) 															
		where vld.VendorLocationNo=@VendorLocationNo
			and exists (select 1 from @PaymentTermDetail where PaymentTypeNo<>3)
		
	END TRY

			   ,[ErrorDescription]
			   ,[UserName]
			   ,[FormCode])

			   (getdate()
			   ,ERROR_NUMBER()
			   ,ERROR_SEVERITY()

			   ,ERROR_PROCEDURE()
			   ,ERROR_LINE()
			   ,ERROR_MESSAGE()

			   ,'PMPO')
	END CATCH
