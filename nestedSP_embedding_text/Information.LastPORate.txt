CREATE PROCEDURE [Information].[LastPORate]  
	-- Add the parameters for the stored procedure here
	@DocumentDate smalldatetime,
	@ItemNo int
AS
BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;
	

    -- Insert statements for procedure here
	declare @str varchar(max)
    select @str=masterdata.JSForInformation('PMPO')
    set @str=@str+masterdata.CSSForInformation('PMPO')
	set @str=@str+'<table class="altrowstable" id="alternatecolorPORateInfo">'
	set @str=@str+'<Caption><Font color="Maroon"><h3 align="Left"><b>Last 10 Purchase Rate</b></h3></Font></Caption>'
    set @str=@str+'<tr><th>Sl. No.</th><th>Company Name</th><th>Document No.</th><th>Document Date</th><th>Party Name</th><th>Make</th><th>Quantity</th><th>Unit</th><th>Rate</th><th>LRate</th><th>Discount</th><th>Discount(%)</th></tr>'
	set @str=@str
				 +( replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					replace((   
						Select  Row_number() over (order by PM.DocumentDate desc) td1
						       ,CM.AliasName As [td2]
					           ,PM.DocumentNoYearly As [td3]
					           ,convert(varchar,PM.DocumentDate,103) As [td4]
					           ,MVC.MailingName As [td5]
							   ,isnull(MM.MakeName,'') As [td6]
							   ,Sum(PO.Quantity) as [td7]
							   ,UM.Alias as [td8]
							   ,convert(decimal(12,2),sum(PO.BasicAmount-isnull(ttPOITD.TotalAmount,0))/SUM(Quantity)) As [td9]
							   ,convert(decimal(12,2),sum(ttLRate.LRate)) As [td10]
							   ,convert(decimal(12,2),isnull(Sum(ttPOITD.TotalAmount),0)) as [td11]
							   ,convert(decimal(12,2),isnull(Sum(ttPOITD.TotalAmount),0)/Sum(PO.BasicAmount)*100) as [td12]
							   From Purchase.POItemDetail PO
							   Inner Join(  Select POLR.PONo,POLR.ItemNo,POLR.MakeNo,POLR.TechnicalGradeNo,Sum(LRate - PO.Rate)+PO.Rate as LRate
											From Purchase.POItemLRate POLR
											Inner Join Purchase.POItemDetail PO On POLR.PONo = PO.PONo
																			AND POLR.ItemNo= PO.ItemNo 	
																			AND ISNULL(POLR.MakeNo,0)= ISNULL(PO.MakeNo,0)
																			And isnull(POLR.TechnicalGradeNo,0)=isnull(PO.TechnicalGradeNo,0)			
											Group by POLR.PONo,POLR.ItemNo,POLR.MakeNo,POLR.TechnicalGradeNo,PO.Rate							   
										 ) ttLRate    
										On ttLRate.PONo = PO.PONo
										AND ttLRate.ItemNo= PO.ItemNo 	
										AND ISNULL(ttLRate.MakeNo,0)= ISNULL(PO.MakeNo,0)
										And isnull(ttLRate.TechnicalGradeNo,0)=isnull(PO.TechnicalGradeNo,0)													                           
							   Inner Join Purchase.POMain PM on PO.PONo=PM.PONo
							   inner Join masterdata.mCompanyMaster CM on CM.CompanyNo=PM.CompanyNo
							   Inner Join masterdata.mItemMaster IM on IM.ItemNo=PO.ItemNo
							   inner Join masterdata.mUnitMaster UM on UM.UnitNo=po.UnitNo
							   Left Join masterdata.mMakeMaster MM on MM.MakeNo=PO.MakeNo
							   Inner Join masterdata.mVendorLocationDetail MVC on MVC.VendorLocationNo = PM.VendorLocationNo
							   Left Join (
										  SELECT PONo,ItemNo,MakeNo,TotalAmount,TechnicalGradeNo 
										  from Purchase.POItemTaxDetail POITD
										  INNER JOIN  masterdata.mMiscCharges MC ON MC.MiscChargeNo=POITD.MiscChargeNo 
										  Inner Join masterdata.mTaxGroup TXG On TXG.TaxGroupNo=MC.TaxGroupNo
										  AND TXG.TaxCategoryNo = 11 ---for discount
										 )ttPOITD ON isnull(ttPOITD.PONo,0)=isnull(PO.PONo,0)  
	                                                AND ttPOITD.ItemNo= PO.ItemNo
                                                    AND ISNULL(ttPOITD.MakeNo,0)= ISNULL(PO.MakeNo,0)
                                                    And isnull(ttPOITD.TechnicalGradeNo,0)=isnull(PO.TechnicalGradeNo,0)
                         Where PO.POItemDetailNo in (
                                                     Select top 10 (ID.POItemDetailNo)
													 From Purchase.POItemDetail ID
													 inner join Purchase.POMain IDM on IDM.PONo = ID.PONo  
													 Inner Join(  Select POLR.PONo,POLR.ItemNo,POLR.MakeNo,POLR.TechnicalGradeNo,Sum(LRate - PO.Rate)+PO.Rate as LRate
																  From Purchase.POItemLRate POLR
																  Inner Join Purchase.POItemDetail PO On POLR.PONo = PO.PONo
																									  AND POLR.ItemNo= PO.ItemNo 	
																									  AND ISNULL(POLR.MakeNo,0)= ISNULL(PO.MakeNo,0)
																									  And isnull(POLR.TechnicalGradeNo,0)=isnull(PO.TechnicalGradeNo,0)			
																  Group by POLR.PONo,POLR.ItemNo,POLR.MakeNo,POLR.TechnicalGradeNo,PO.Rate) ttLRate    
													 On ttLRate.PONo = ID.PONo
													 AND ttLRate.ItemNo= ID.ItemNo 	
													 AND ISNULL(ttLRate.MakeNo,0)= ISNULL(ID.MakeNo,0)
													 And isnull(ttLRate.TechnicalGradeNo,0)=isnull(ID.TechnicalGradeNo,0)										
													 	 Where PO.ItemNo=ID.ItemNo
														 And IDM.DocumentDate<=@DocumentDate
														 --And ID.ItemNo=@ItemNo
														 And IDM.AuthorizedBy is not NULL
														 order by IDM.DocumentDate desc
													)                               
					     And PM.DocumentDate<=@DocumentDate
					     And PO.ItemNo=@ItemNo
					     And PM.AuthorizedBy is not null
						 Group By CM.AliasName,PM.DocumentNoYearly,PM.DocumentDate,MVC.MailingName
							     ,PO.ItemNo,MM.MakeName,UM.Alias
						 order by PM.DocumentDate desc
					for xml path('tr')),'<td1>','<td style="text-align: center;">')
					                       ,'<td2>','<td style="text-align: center;">')
										   ,'<td3>','<td style="text-align: center;">')
										   ,'<td4>','<td>')
										   ,'<td5>','<td>')
										   ,'<td6>','<td>')
										   ,'<td7>','<td style="text-align: right;">')
										   ,'<td8>','<td>')
										   ,'<td9>','<td style="text-align: right;">')
										   ,'<td10>','<td style="text-align: right;">')
										   ,'<td11>','<td style="text-align: right;">')
										   ,'<td12>','<td style="text-align: right;">'))
				   
	set @str=@str+'</table>'
	
	select @str

END
