Writes: Notifications.TATAuthorizationDetail
Reads: EAM.MaintenanceOrderMain, EAM.MaintenanceOrderRefDetail, Notifications.MultiAuthorizationSetting, Notifications.TATAuthorizationDetail, Security.LoginRoleDetail, Security.LoginTempRoleDetail, Security.mRoleMasterFormDetail
Parameters: @MaintenanceOrderNo Int, @IsCallingForMultiLevelAuthorization tinyint = 0, @OutputStatusNo tinyint = null OUTPUT, @OutputStatusRemarks varchar(4000) = null OUTPUT
Calls procedures: EAM.EAMOSendAuthorizationMail, EAM.MaintenanceOccurrenceStatus

-- =============================================
-- Author:		Himanshu
-- Create date: 24 Aug 2022
-- Description: Authorize script for Maintenance Order
-- =============================================
CREATE PROCEDURE [EAM].[EAMOAuthorize]
(
 @MaintenanceOrderNo Int,
 @IsCallingForMultiLevelAuthorization tinyint = 0, 
 @OutputStatusNo tinyint = null output,
 @OutputStatusRemarks varchar(4000)  = null output
)
AS

BEGIN
	SET NOCOUNT ON;
	
	BEGIN TRY
		declare @DocumentDate smalldatetime
				
		select @DocumentDate=DocumentDate
				,@MaintenanceOrderNo=MaintenanceOrderNo
		from EAM.MaintenanceOrderMain where MaintenanceOrderNo=@MaintenanceOrderNo
       if @IsCallingForMultiLevelAuthorization = 1   ---------- When 1 then  Notification for Multi Level Authorization will call
	   Begin 
	   
		 declare @MAPValuationNo smallint,@AuthorizedBy int,@CompanyNo tinyint ,@DivisionNo int ,@DocumentStatusNo tinyint
	 ,@AuthDetailNo int ,@CreatedBy smallint
	
	 select @AuthorizedBy = AuthorizedBy,@CompanyNo =CompanyNo  ,@DivisionNo = DivisionNo ,@DocumentStatusNo = DocumentStatusNo
	,@CreatedBy = sam.CreatedBy  
	from EAM.MaintenanceOrderMain sam where MaintenanceOrderNo = @MaintenanceOrderNo 
	Declare  @UserCount int 
	Declare @NextUsers table(SNo smallint,AuthDetailNo int,LevelNo int,LoginNo int) 
	Declare @UserNO int,@LevelNo smallint,@UserCounter int =1
	   
	   If Exists (Select 1  From Notifications.MultiAuthorizationSetting where CompanyNo =@CompanyNo and ISNULL(DivisionNo,0) = ISNULL(@DivisionNo,0)
 					and FormCode ='EAMA') and @DocumentStatusNo = 20
 		BEGIN
 		
 			 if Exists (Select 1 From Notifications.TATAuthorizationDetail where FormCode ='EAMA' and InternalNo = @MaintenanceOrderNo  
 							and IsReady = 1 and IsEmailSent=0)
				BEGIN
						-- Send Email
					
					Insert into @NextUsers
					select ROW_NUMBER () over (order by LoginNo) SlNo,AuthDetailNo, LevelNo, LoginNo  from (
					Select LevelNo,LoginNo,AuthDetailNo
					 From Notifications.TATAuthorizationDetail where FormCode ='EAMA' and InternalNo = @MaintenanceOrderNo  
						and IsReady = 1 and IsEmailSent =0	and LoginNo is not null
					union 
					Select LevelNo,lrd.LoginNo,AuthDetailNo
					 From Notifications.TATAuthorizationDetail tad
					 inner join Security.mRoleMasterFormDetail rfd on rfd.RoleMasterNo = tad.RoleNo and rfd.FormNo in( 151,167,464)
					 inner join Security.LoginRoleDetail lrd on rfd.RoleMasterNo = lrd.RoleMasterNo 
					 where FormCode ='EAMA' and InternalNo = @MaintenanceOrderNo 
						and IsReady = 1 and IsEmailSent =0	and tad.RoleNo  is not null	
					union 
					Select LevelNo,lrd.LoginNo,AuthDetailNo
					 From Notifications.TATAuthorizationDetail tad
					 inner join Security.mRoleMasterFormDetail rfd on rfd.RoleMasterNo = tad.RoleNo and rfd.FormNo in( 151,167,464)
					 inner join Security.LoginTempRoleDetail lrd on rfd.RoleMasterNo = lrd.RoleMasterNo 
					 where FormCode ='EAMA' and InternalNo = @MaintenanceOrderNo and CONVERT(smalldatetime,convert(varchar(15), GETDATE(),106))
						 between lrd.FromDate and ISNULL( lrd.ToDate,CONVERT(smalldatetime,convert(varchar(15), GETDATE(),106)))
						and IsReady = 1 and IsEmailSent =0	and tad.RoleNo  is not null	) tt
						group by LevelNo, LoginNo 	,AuthDetailNo							
					
					
					Set @UserCount = @@ROWCOUNT	
					
					While @UserCounter <= @UserCount
					 Begin
					
						 Select @UserNO=LoginNo , @LevelNo = LevelNo,@AuthDetailNo=AuthDetailNo From @NextUsers Where SNo = @UserCounter
								
						 Exec [EAM].[EAMOSendAuthorizationMail] @MaintenanceOrderNo,@UserNo,@LevelNo ,@AuthDetailNo
						 
						Update Notifications.TATAuthorizationDetail  set IsEmailSent =1 where FormCode = 'EAMA' and InternalNo = @MaintenanceOrderNo 
								and LevelNo = @LevelNo and IsReady =1
						 Set @UserCounter=@UserCounter+1 
					 End
		 
				END							   
				
				 
			 					  	
 		 END
 	  else If Exists (Select 1  From Notifications.MultiAuthorizationSetting where CompanyNo =@CompanyNo and ISNULL(DivisionNo,0) = ISNULL(@DivisionNo,0)
 					and FormCode ='EAMA') and Exists (Select 1 From Notifications.TATAuthorizationDetail where FormCode ='EAMA' 
 						and InternalNo = @MaintenanceOrderNo and IsRejectedMailSend =1) -- rejection mail
 		BEGIN
 			 
 			 if Exists (Select 1 From Notifications.TATAuthorizationDetail where FormCode ='EAMA' and InternalNo = @MaintenanceOrderNo 
 							and IsRejectedMailSend =1)
				BEGIN
						-- Send Email
					 
					Insert into @NextUsers
					select ROW_NUMBER () over (order by LoginNo) SlNo,AuthDetailNo, LevelNo, LoginNo  from (
					Select LevelNo,LoginNo,AuthDetailNo
					 From Notifications.TATAuthorizationDetail where FormCode ='EAMA' and InternalNo = @MaintenanceOrderNo 
						and IsRejectedMailSend =1 and LoginNo is not null
					union 
					Select LevelNo,lrd.LoginNo,AuthDetailNo
					 From Notifications.TATAuthorizationDetail tad
					 inner join Security.mRoleMasterFormDetail rfd on rfd.RoleMasterNo = tad.RoleNo and rfd.FormNo in(151,167,464)
					 inner join Security.LoginRoleDetail lrd on rfd.RoleMasterNo = lrd.RoleMasterNo 
					 where FormCode ='EAMA' and InternalNo = @MaintenanceOrderNo 
						and IsRejectedMailSend =1 and tad.RoleNo  is not null	
					union 
					Select LevelNo,lrd.LoginNo,AuthDetailNo
					 From Notifications.TATAuthorizationDetail tad
					 inner join Security.mRoleMasterFormDetail rfd on rfd.RoleMasterNo = tad.RoleNo and rfd.FormNo in(151,167,464)
					 inner join Security.LoginTempRoleDetail lrd on rfd.RoleMasterNo = lrd.RoleMasterNo 
					 where FormCode ='EAMA' and InternalNo = @MaintenanceOrderNo and CONVERT(smalldatetime,convert(varchar(15), GETDATE(),106))
						 between lrd.FromDate and ISNULL( lrd.ToDate,CONVERT(smalldatetime,convert(varchar(15), GETDATE(),106)))
						and IsRejectedMailSend =1 and tad.RoleNo  is not null	) tt
						group by LevelNo, LoginNo 	,AuthDetailNo							
					
					
					Set @UserCount = @@ROWCOUNT	
					
					
					While @UserCounter <= @UserCount
					 Begin
					
						 Select @UserNO=LoginNo , @LevelNo = LevelNo,@AuthDetailNo=AuthDetailNo From @NextUsers Where SNo = @UserCounter
								
						 Exec [EAM].[EAMOSendAuthorizationMail] @MaintenanceOrderNo,@UserNo,@LevelNo ,@AuthDetailNo ,1
						 
						Update Notifications.TATAuthorizationDetail  set IsRejectedMailSend =0 where FormCode = 'EAMA' and InternalNo = @MaintenanceOrderNo 
								and LevelNo = @LevelNo and IsRejectedMailSend =1
						 Set @UserCounter=@UserCounter+1 
					 End
				END							   					  	
 		 END 
 		
		--MultiAuthorization Query Ends here 
		End 		
		Else if @IsCallingForMultiLevelAuthorization = 0 -- Code for Order Account Assignment
		Begin 
		  Select @OutputStatusNo = 1,@OutputStatusRemarks = 'Success'				  	  
		  Select 1,'Success'
	    End 
	    
	    ---- Maintenance Schedule occurrence change status
		DECLARE	@OutputStatus bit,@Result varchar(100)	,@Count int ,@i int	
		Declare @MSOccurrenceDetail as table (MaintenanceScheduleNo int,TriggerNo int,OccurrenceNo int)
				
		Insert into @MSOccurrenceDetail(MaintenanceScheduleNo,TriggerNo ,OccurrenceNo)
		Select  DT.RefMSNo,DT.TriggerNo,DT.OccurrenceNo
		From (
				Select MORD.RefDocNo RefMSNo,MORD.TriggerNo,MORD.OccurrenceNo
				From EAM.MaintenanceOrderRefDetail MORD
				Inner Join EAM.MaintenanceOrderMain MOM ON MOM.MaintenanceOrderNo = MORD.MaintenanceOrderNo
				Where MOM.SourceDocumentNo = 3 
					And MOM.MaintenanceOrderNo = @MaintenanceOrderNo
				Union All
				Select MOM.RefMaintenanceScheduleNo,MOM.TriggerNo,MOM.RefOccurrenceNo
				From EAM.MaintenanceOrderMain MOM
				Where MOM.MaintenanceOrderNo = @MaintenanceOrderNo
					And MOM.RefMaintenanceScheduleNo is not null
			)DT
			
		DECLARE	@MaintenanceScheduleNo int,@OccurrenceNo int,@TriggerNo int	
		Set @Count =  (Select COUNT(MaintenanceScheduleNo) From @MSOccurrenceDetail)
		Set  @i  =0
		While @i<@Count
		Begin
			Set @i=@i+1
			
			
			Select @MaintenanceScheduleNo =MaintenanceScheduleNo
					,@OccurrenceNo = OccurrenceNo 
					,@TriggerNo=TriggerNo  
			From (Select ROW_NUMBER() over (order by MaintenanceScheduleNo, TriggerNo,OccurrenceNo ) Rno
						,MaintenanceScheduleNo
						,TriggerNo
						,OccurrenceNo
				  From @MSOccurrenceDetail
				 ) tt where Rno=@i
			
			EXEC EAM.MaintenanceOccurrenceStatus
			@MaintenanceScheduleNo = @MaintenanceScheduleNo,
			@OccurrenceNo =@OccurrenceNo,
			@TriggerNo = @TriggerNo,
			@StatusNo=2,
			@OutputStatus =@OutputStatus  output,
			@Result =@Result output

			IF @OutputStatus = 0
				raiserror(@Result,16,-1)
		End
			
	    
	    
	END TRY
	
	BEGIN CATCH
		Select 0,Error_Message()
		Select @OutputStatusNo = 0,@OutputStatusRemarks = ERROR_MESSAGE()				  
	END CATCH
END
