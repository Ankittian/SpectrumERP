Reads: Customize.ErosAMCMain, Finance.ARDImprestRule, Finance.ARDItemRule, Finance.ARDVendorLocationRule, Inventory.IndentRequisitionMain, masterdata.mAccountMaster, masterdata.mItemMaster, masterdata.mVendorLocationDetail, Purchase.SCOrderAccountAssignmentDetail, Purchase.SCOrderAmendmentItemDetail, Purchase.SCOrderAmendmentMain, Purchase.SCOrderMain

--- Inventory].[IMSTFinancePosting] '<DocumentElement />' 
CREATE PROCEDURE [Inventory].[IMMIFinancePosting] 
	@OtherDataInXml xml ,
	@CompanyNo tinyint ,
	@ItemNos varchar(max)
	
as
BEGIN
begin try
declare @IsManagingExpenseAccount bit
 

create table #Posting ([Add to Basic] bit,Particular varchar(500),[Debit Amount] decimal(15,2) , 
		[Credit Amount] decimal(15,2),[Account Name] int, [Ref. Detail] varchar(50),[Cost Center] varchar(50),
		[Amount] decimal(15,2), [Type] tinyint,SlNo tinyint identity(1,1) ,MiscChargeNo int , TaxCodeMiscChargesDetailNo int,
			ItemNos varchar(200),GroupNo smallint, GroupTypeNo tinyint, NatureNo  tinyint, AccountTypeNo tinyint)

		declare @ItemTable table (ItemNo int,BasicAmount money,Id int,Qty decimal(12,3),Rate money)
		insert into @ItemTable
		select Col1, sum(convert(money,Col2)), Col3, sum(CONVERT(decimal(12,3),Col4)), sum(CONVERT(money,Col5))
		from masterdata.Split_RowColumn(@ItemNos,'`','~')
		group by Col1, Col3

declare @SourceDocumentNo tinyint , @RefDocNo int, @IsNonChargeable bit , @IssueTo tinyint ,@IssueName int , @IssueTypeNo tinyint
, @IsFree bit = 0 ,@IsCapitalLot bit = 0
	

		SELECT @SourceDocumentNo=v.value('SourceDocumentNo[1]','tinyint')
					,@RefDocNo=v.value('RefDocNo[1]','int')
					,@IsNonChargeable=v.value('IsNonChargeable[1]','bit')
					,@IssueTo=v.value('IssueTo[1]','tinyint')
					,@IssueName=v.value('IssueName[1]','int')
					,@IssueTypeNo=v.value('IssueTypeNo[1]','tinyint')
					,@IsCapitalLot=v.value('IsCapitalLot[1]','bit')
			FROM @OtherDataInXml.nodes('/DocumentElement/Table') x(v)
		if @IsCapitalLot is null
			set @IsCapitalLot = 0
			
		select @IsFree = amc.IsFree  from Inventory.IndentRequisitionMain irm 
			inner join Customize.ErosAMCMain amc on amc.ErosAMCNo = irm.EROSAMCNo 
			where irm.IndentRequisitionNo = @RefDocNo  and @SourceDocumentNo = 5
				
			select @IssueTypeNo=2 where @IssueTypeNo is null
			
			Insert into #Posting ([Add to Basic] ,Particular,[Debit Amount]  , [Credit Amount],[Account Name] , 
						[Amount], [Type])		
			select cast(0 as bit) [Add to Basic] ,'FM Consumption A/c' Particular 
				,sum(I.BasicAmount) [Debit Amount] ,CAST(null as money) [Credit Amount] 
				,IR.AccountNo [Account Name] 
				,sum(I.BasicAmount) Amount ,1 [Type] 
			from @ItemTable I
				outer apply (select IR.AccountNo--,AM.AccountName
							from masterdata.mAccountMaster  IR
							where IR.AccountTypeNo=67) IR 
			where  @IsNonChargeable=1 and @IssueTypeNo<>1
			and @IsFree=1 and @SourceDocumentNo = 5
			group by  IR.AccountNo
			
			Insert into #Posting ([Add to Basic] ,Particular,[Debit Amount]  , [Credit Amount],[Account Name] , 
						[Amount], [Type])		
			select cast(0 as bit) [Add to Basic] ,'Consumption A/C' Particular 
				,sum(I.BasicAmount) [Debit Amount] ,CAST(null as money) [Credit Amount] 
				,IR.AccountNo [Account Name] 
				,sum(I.BasicAmount) Amount ,1 [Type] 
			from @ItemTable I
				inner join masterdata.mItemMaster IM on IM.ItemNo=I.ItemNo
				left join (select ItemNo,IR.AccountNo--,AM.AccountName
							from Finance.ARDItemRule IR
							--inner join masterdata.mAccountMaster AM on AM.AccountNo=IR.AccountNo
							where IR.AccountTypeNo=15) IR on IR.ItemNo=I.ItemNo
			where @SourceDocumentNo in(1,4,5,6,7,9,10,11) and @IsNonChargeable=1 and @IssueTypeNo<>1
			and @IsFree<>1
			group by  IR.AccountNo
			Union All
			select cast(0 as bit) [Add to Basic] ,'Batch Production A/C' Particular 
				,sum(I.BasicAmount) [Debit Amount] ,CAST(null as money) [Credit Amount] 
				,IR.AccountNo [Account Name] 
				,sum(I.BasicAmount) Amount ,1 [Type] 
			from @ItemTable I
				inner join masterdata.mItemMaster IM on IM.ItemNo=I.ItemNo
				outer apply (select Max(IR.AccountNo) AccountNo
							from masterdata.mAccountMaster  IR
							where IR.AccountTypeNo=70) IR 
			where @SourceDocumentNo =8  and @IsNonChargeable=1 and @IssueTypeNo<>1 -- Batch Order
			group by  IR.AccountNo
			
			union all
			select cast(0 as bit) [Add to Basic] ,'Consumption A/C' Particular 
				,sum(I.BasicAmount) [Debit Amount] ,CAST(null as money) [Credit Amount] 
				,IR.AccountNo [Account Name] 
				,sum(I.BasicAmount) Amount ,1 [Type] 
			from @ItemTable I
				inner join masterdata.mItemMaster IM on IM.ItemNo=I.ItemNo
				inner join (select Id ItemNo,AAD.AccountNo--,AM.AccountName
							from Purchase.SCOrderAccountAssignmentDetail AAD
							--inner join masterdata.mAccountMaster AM on AM.AccountNo=AAD.AccountNo
							where Flag='IM'
								and AAD.AccountTypeNo=15
								and SCOrderNo=(select SCOrderNo from Purchase.SCOrderAmendmentMain where SCOrderAmendmentNo=@RefDocNo)
								and not exists (select 1 from Purchase.SCOrderAmendmentItemDetail
												where ItemFlowNo in (3,4)
												and SCOrderAmendmentNo=@RefDocNo)) IR on IR.ItemNo=I.ItemNo
			where (@SourceDocumentNo=3) and @IsNonChargeable=1
			group by  IR.AccountNo
			
			union all
			select cast(0 as bit) [Add to Basic] ,'Consumption A/C' Particular 
				,sum(I.BasicAmount) [Debit Amount] ,CAST(null as money) [Credit Amount] 
				,IR.AccountNo [Account Name] 
				,sum(I.BasicAmount) Amount ,1 [Type] 
			from @ItemTable I
				cross apply (select  AAD.AccountNo--,AM.AccountName
							from Purchase.SCOrderAccountAssignmentDetail AAD
							Inner join Purchase.SCOrderMain SM on SM.SCOrderNo = AAD.SCOrderNo --and SM.IsExpenseMapped = 0
							--inner join masterdata.mAccountMaster AM on AM.AccountNo=AAD.AccountNo
							where ((Flag='IM'and AAD.AccountTypeNo=16) or (Flag='CA'and AAD.AccountTypeNo=15))
								and aad.SCOrderNo=(select SCOrderNo from Purchase.SCOrderAmendmentMain where SCOrderAmendmentNo=@RefDocNo)
							--Union
							--select max(IR.AccountNo)AccountNo
							--from 
							--Finance.ARDIntermediatePostingAccountRule AR
							--inner join Finance.ARDItemRule IR on IR.ARDNo=AR.ARDNo and IR.AccountNo = AR.AccountNo
							--inner join masterdata.mAccountMaster AM on AM.AccountNo=IR.AccountNo
							--where AM.AccountTypeNo=54 and AR.IntermediateAccountTypeNo =18
							--and ItemNo = I.ItemNo
							--and exists (select 1 from Purchase.SCOrderAmendmentMain SM where IsExpenseMapped =1 and SCOrderAmendmentNo = @RefDocNo )
							--group by ir.ItemNo
								
								) IR
			where (@SourceDocumentNo=3) and @IsNonChargeable=1
			group by IR.AccountNo,IR.AccountNo
			
			
			Insert into #Posting ([Add to Basic] ,Particular,[Debit Amount]  , [Credit Amount],[Account Name] , 
						[Amount], [Type] , AccountTypeNo )	
			select cast(0 as bit) [Add to Basic] ,'Supplier Payable A/c' Particular 
				,(BA.Amount) [Debit Amount] ,CAST(null as money) [Credit Amount] 
				,LR.AccountNo [Account Name] 
				,(BA.Amount) Amount ,1 [Type] , 4 
			from (select SUM(BasicAmount) Amount from @ItemTable) BA
				Outer apply (select LR.AccountNo from Finance.ARDVendorLocationRule LR 
					inner join masterdata.mVendorLocationDetail LD on LD.VendorLocationNo=LR.VendorLocationNo
						where LD.VendorNo=@IssueName and @IssueTo in (2,3) and LR.AccountTypeNo = 4
					union all select AccountNo from Finance.ARDImprestRule LR where LR.EmpNo=@IssueName and @IssueTo=1)LR
			where @IsNonChargeable=0 
			--and (@SourceDocumentNo=1 or @SourceDocumentNo=4 or @SourceDocumentNo=5 or @SourceDocumentNo=3)
			
			Insert into #Posting ([Add to Basic] ,Particular,[Debit Amount]  , [Credit Amount],[Account Name] , 
						[Amount], [Type],AccountTypeNo )	
			select cast(0 as bit) [Add to Basic] ,'Stock A/C' Particular 
				,CAST(null as money) [Debit Amount] ,sum(I.BasicAmount) [Credit Amount] 
				,IR.AccountNo [Account Name] 
				,sum(I.BasicAmount) Amount ,2 [Type] ,14
			from @ItemTable I
				inner join masterdata.mItemMaster IM on IM.ItemNo=I.ItemNo
				left join (select ItemNo,IR.AccountNo--,AM.AccountName
							from Finance.ARDItemRule IR
							--inner join masterdata.mAccountMaster AM on AM.AccountNo=IR.AccountNo
							where IR.AccountTypeNo=14 ) IR on IR.ItemNo=I.ItemNo
			where  @IsCapitalLot = 0
			group by IR.AccountNo	
			
			
			Insert into #Posting ([Add to Basic] ,Particular,[Debit Amount]  , [Credit Amount],[Account Name] , 
						[Amount], [Type] , AccountTypeNo )	
			select cast(0 as bit) [Add to Basic] ,'Asset Clearing A/c' Particular 
				,CAST(null as money) [Debit Amount] ,sum(I.BasicAmount) [Credit Amount] 
				,null [Account Name] 
				,sum(I.BasicAmount) Amount ,2 [Type] ,54
			from @ItemTable I
				inner join masterdata.mItemMaster IM on IM.ItemNo=I.ItemNo
			where @IsNonChargeable = 0 and @IsCapitalLot = 1 and ItemTypeNo = 1
			
			
			Insert into #Posting ([Add to Basic] ,Particular,[Debit Amount]  , [Credit Amount],[Account Name] , 
						[Amount], [Type] , AccountTypeNo )	
			select cast(0 as bit) [Add to Basic] ,'WIP A/C' Particular 
				,CAST(null as money) [Debit Amount] ,sum(I.BasicAmount) [Credit Amount] 
				,null [Account Name] 
				,sum(I.BasicAmount) Amount ,2 [Type] ,16
			from @ItemTable I
				inner join masterdata.mItemMaster IM on IM.ItemNo=I.ItemNo
			where @IsNonChargeable = 0 and @IsCapitalLot = 1 and ItemTypeNo <> 1

			Insert into #Posting ([Add to Basic] ,Particular,[Debit Amount]  , [Credit Amount],[Account Name] , 
						[Amount], [Type] , AccountTypeNo )	
			select cast(0 as bit) [Add to Basic] ,'WIP A/C' Particular 
				,(I.Amount) [Debit Amount] ,CAST(null as money) [Credit Amount] 
				,null [Account Name] ,(I.Amount) Amount ,1 [Type] ,16
			from (select sum(BasicAmount) Amount from @ItemTable) I
			where (@IssueTypeNo=1) and @IsNonChargeable=1
				and (@SourceDocumentNo<>3)

 if exists(select 1 from #Posting having abs(SUM([Credit Amount])- SUM([Debit Amount]))<=.5)
 begin
	update PD set [Debit Amount] = [Debit Amount] + ISNULL(diff,0)
		from
		#Posting PD
		cross join (select SUM([Credit Amount])- SUM([Debit Amount]) Diff from #Posting) tt
		where PD.SlNo  = 1
	
	update PD set [Credit Amount] = [Credit Amount] + ISNULL(diff,0)
		from
		#Posting PD
		cross join (select SUM([Debit Amount])- SUM([Credit Amount]) Diff from #Posting) tt
		where PD.SlNo  = 1
			
 end
 
 
select * from #Posting
 where isnull([Debit Amount],0) >0 or isnull([Credit Amount],0) >0
   
  
  drop table #Posting 
   
END TRY
	
	BEGIN CATCH
		select ERROR_MESSAGE()
		INSERT INTO [masterdata].[mErrorLog]
			   ([ErrorDate]
			   ,[ErrorNo]
			   ,[ErrorSeverity]
			   ,[ErrorState]
			   ,[ErrorProcedure]
			   ,[ErrorLine]
			   ,[ErrorDescription]
			   ,[UserName]
			   ,[FormCode])
		VALUES
			   (getdate()
			   ,ERROR_NUMBER()
			   ,ERROR_SEVERITY()
			   ,ERROR_STATE()
			   ,ERROR_PROCEDURE()
			   ,ERROR_LINE()
			   ,ERROR_MESSAGE()
			   ,SUSER_SNAME()
			   ,'IMMI')
	END CATCH
END
