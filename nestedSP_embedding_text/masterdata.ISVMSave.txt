Stored procedure masterdata.ISVMSave: Save Item Attribute Standard Value. Called from masterdata.MMMMSaveAttributeStandardValue.
Writes: masterdata.mItemAttributeStandardValueMaster
Reads: masterdata.mItemAttributeStandardValueMaster
Parameters: @StandardValueNo smallint = null OUTPUT, @StandardValueName varchar(25), @LoginNo SmallInt
Author: Alok, created 29 Dec 2012

-- =============================================
-- Author:		Alok
-- Create date: 29 Dec 2012
-- Description: Save Item Attribute Standard Value 
-- =============================================
CREATE PROCEDURE [masterdata].[ISVMSave]
( 
 @StandardValueNo  smallint = null Output 
,@StandardValueName  varchar(25)
,@LoginNo   SmallInt
)
AS

BEGIN
	SET NOCOUNT ON;	
	BEGIN TRY
	
		IF IsNull(@StandardValueNo,0)=0
			BEGIN
					
			Select @StandardValueNo=StandardValueNo
			From masterdata.mItemAttributeStandardValueMaster 
			Where StandardValueName=LTRIM(RTrim(@StandardValueName))
			
				IF IsNull(@StandardValueNo,0)=0
				BEGIN
					Declare @Code char(6)

					SELECT @Code = 'SV'+ RIGHT(REPLICATE('0', 4) +cast(isnull(max(Convert(int,substring(Code,3,4))),0) + 1 as varchar(4)),4)  
					FROM  [masterdata].[mItemAttributeStandardValueMaster]


						INSERT INTO  [masterdata].[mItemAttributeStandardValueMaster]
						   (
							Code
						   ,StandardValueName
						   ,CreatedBy
						   ,CreatedDate
						   )
						VALUES
						   (
							@Code
						   ,@StandardValueName
						   ,@LoginNo
						   ,Getdate()
						   )

						   Select @StandardValueNo=IDENT_CURRENT('[masterdata].[mItemAttributeStandardValueMaster]')

				END			   

			END
		ELSE
			BEGIN

				UPDATE  [masterdata].[mItemAttributeStandardValueMaster]
				SET 				    
				    StandardValueName = @StandardValueName
				 WHERE StandardValueNo = @StandardValueNo
			END

		Select 1,@StandardValueNo
	END TRY
	BEGIN CATCH
		Select 0,Error_Message()
	END CATCH
END
