CREATE PROCEDURE [Inventory].[InvLotDelete]	 
(
	@TransactionFlag varchar(12), 
	@DocumentNo int, 
	@Result bit = 0 output,
	@ResultMsg varchar(2000) = null output
) 
AS

SET NOCOUNT ON;
BEGIN TRY

declare @Lot as table (invLotsummaryNo int,returnQty decimal(12,3)  )
	 Declare @MAPDetail as table (CompanyNo tinyint,DivisionNo int,WareHouseNo smallint, ItemNo int,MakeNo smallint,TechnicalGradeNo tinyint)
	declare @AllBillNo varchar(max)	 
		Declare @MAPRate decimal(18,6)  = 0

Create table #Lot (invLotSummaryNo int,ItemRate decimal(18,6))
Create table #LotDetail (invLotSummaryNo int,InvLotReserveNo int,Qty decimal(12,3),Pcs smallint)
Insert into #LotDetail select InvLotSummaryNo,null,BillQty , null  from Inventory.InvLotBillDetail 
		where BillNo =@DocumentNo and TransactionFlag =@TransactionFlag and transactionflag not in ('SDDC','SDEI','SDCI')

update pls set MAPRate=((dt.ClosingQty + bd.billqty) * bd.MAPRate - MAPAmt) / 
					(dt.ClosingQty + bd.Qty )
  	from 
	(select ls.ItemNo,ls.WarehouseNo ,ls.CompanyNo ,ls.MakeNo,ls.IsCapital ,ls.DivisionNo 
			,SUM(case when   bd.TransactionFlag='ISML' then 0 else  BillQty end) billqty
			,SUM(case when bd.TransactionFlag = ls.TransactionFlag  then LotBalanceQty * bd.VRate else bd.ClosingQty * bd.MAPDiffRate  end) MAPAmt 
			,sum(case when bd.TransactionFlag = ls.TransactionFlag or bd.TransactionFlag='ISML' then 0 else bd.BillQty end) Qty
			,ls.MAPRate  from
	Inventory.InvLotBillDetail bd
	inner join Inventory.InvLotSummary ls on ls.InvLotSummaryNo = bd.InvLotSummaryNo 
	inner join masterdata.mItemOtherInformation ioi on ioi.ItemNo = ls.ItemNo 
	where bd.BillNo = @DocumentNo and bd.TransactionFlag=   @TransactionFlag 
		and bd.TransactionFlag not in ('SDDC','SDEI','SDCI')
	and ioi.StockValueMethodNo = 4 and ls.IsCapital = 0
	group by ls.ItemNo,ls.WarehouseNo ,ls.CompanyNo ,ls.MakeNo,ls.IsCapital ,ls.DivisionNo,ls.MAPRate
	) bd
	inner join Inventory.InvLotSummary pls on pls.ItemNo = bd.ItemNo and ISNULL(pls.MakeNo,0) = ISNULL(bd.MakeNo,0)
		and pls.CompanyNo = bd.CompanyNo and pls.WarehouseNo = bd.WarehouseNo and pls.IsCapital = bd.IsCapital 
		and ISNULL(pls.DivisionNo,0) = ISNULL(bd.DivisionNo,0)
	left join (
			  select cls.CompanyNo , cls.WarehouseNo,cls.DivisionNo ,cls.ItemNo ,cls.MakeNo , isnull(sum(BalanceQty),0)ClosingQty   
			  from
				Inventory.InvLotSummary  cls
				where  cls.isCapital=0 and balanceQty>0 
				and exists(select 1 from Inventory.InvLotSummary ls,Inventory.InvLotBillDetail bd
							 where bd.InvLotSummaryNo = ls.InvLotSummaryNo and bd.TransactionFlag =  @TransactionFlag  and BillNo  = @DocumentNo
						and CompanyNo = cls.CompanyNo and WarehouseNo = cls.WarehouseNo   and ls.ItemNo = cls.ItemNo  
						and bd.TransactionFlag not in ('SDDC','SDEI','SDCI'))
				and not exists(select 1 from Inventory.InvLotBillDetail where TransactionFlag = @TransactionFlag  and BillNo  = @DocumentNo	
								and cls.InvLotSummaryNo = InvLotSummaryNo and @TransactionFlag <>'ISML' )	
				group by cls.CompanyNo ,cls.WarehouseNo,cls.DivisionNo ,cls.ItemNo ,cls.MakeNo
				) dt on dt.CompanyNo = pls.CompanyNo and dt.ItemNo = pls.ItemNo and dt.WarehouseNo = pls.WarehouseNo 
						and ISNULL(dt.DivisionNo,0) = ISNULL( pls.DivisionNo,0) and ISNULL( dt.MakeNo,0) = ISNULL( pls.MakeNo,0)
	 --where pls.DocumentNo <>@DocumentNo 
set @MAPRate =0
declare @invLotBillNo int =0 
 
  
update pbd set ClosingQty = pbd.ClosingQty - case when bd.TransactionFlag = ls.TransactionFlag then bd.BillQty else 0 end
		, MAPAmount = pbd.MAPAmount - case when bd.TransactionFlag = ls.TransactionFlag then (bd.LotBalanceQty * bd.VRate ) 
						else (bd.MAPDiffRate * bd.ClosingQty )end
		, ActualMAPRate = case when pbd.ClosingQty - bd.BillQty>0 then (pbd.MAPAmount - case when bd.TransactionFlag = ls.TransactionFlag then (bd.LotBalanceQty * bd.VRate ) 
						else (bd.MAPDiffRate * bd.ClosingQty )end )/(pbd.ClosingQty - bd.BillQty) end						
		--, MAPAmount = pbd.MAPAmount - (bd.MAPDiffRate * bd.ClosingQty )- isnull(dt.MAPAmount ,0)
		--, ActualMAPRate = case when pbd.ClosingQty - bd.BillQty>0 then (pbd.MAPAmount - (bd.MAPDiffRate * bd.ClosingQty )- isnull(dt.MAPAmount,0)  )/(pbd.ClosingQty - bd.BillQty) end
		--, MAPDiffRate = ls.maprate - ((pbd.MAPAmount - (bd.MAPDiffRate * bd.ClosingQty )- isnull(dt.MAPAmount,0)  )/(pbd.ClosingQty - bd.BillQty))
		--, MAPDiffRate = (pbd.MAPDiffRate * (pbd.ClosingQty ) )/(pbd.ClosingQty - bd.BillQty)
	from Inventory.InvLotBillDetail bd
	inner join Inventory.InvLotSummary ls on ls.InvLotSummaryNo = bd.InvLotSummaryNo 
	inner join masterdata.mItemOtherInformation ioi on ioi.ItemNo = ls.ItemNo 
	inner join Inventory.InvLotSummary pls on pls.ItemNo = ls.ItemNo and ISNULL(pls.MakeNo,0) = ISNULL(ls.MakeNo,0)
		and pls.CompanyNo = ls.CompanyNo and pls.WarehouseNo = ls.WarehouseNo and pls.IsCapital = ls.IsCapital 
		and ISNULL(pls.DivisionNo,0) = ISNULL(ls.DivisionNo,0) and pls.InvLotSummaryNo <> ls.InvLotSummaryNo 
	inner join Inventory.InvLotBillDetail pbd  on pbd.InvLotSummaryNo = pls.InvLotSummaryNo 
	--outer apply 
	--(select ROW_NUMBER () over (order by cbd.invlotbilldetailno desc) slNo , cbd.ClosingQty , cbd.MAPAmount  from
	--inventory.InvLotBillDetail cbd 
	--inner join Inventory.InvLotSummary  cls on cls.InvLotSummaryNo = cbd.InvLotSummaryNo 
	--where cls.itemno = pls.ItemNo  and cls.warehouseno = pls.warehouseno 
	--and cls.companyno = pls.CompanyNo and isnull(cls.Divisionno,0) = isnull(pls.DivisionNo,0)
	--and isnull(cls.MakeNo,0)=ISNULL(pls.MakeNo,0) and  cbd.MAPAmount<>0 and cbd.BillDate >@OpeningDate
	--and cls.isCapital=0 and cbd.CreatedDate < pbd.CreatedDate and cbd.InvLotSummaryNo<> bd.InvLotSummaryNo and 1=2  ) dt
	where bd.BillNo =@DocumentNo   and bd.TransactionFlag not in ('SDDC','SDEI','SDCI')
	--and (dt.slNo =1 or dt.slNo is null)
	and bd.TransactionFlag = @TransactionFlag  and pbd.CreatedDate > bd.CreatedDate 
	and ls.IsCapital =0 and ioi.StockValueMethodNo = 4  and bd.MAPDiffRate <>0

				Inventory.MaterialIssueMain im 
				inner join  Inventory.MaterialIssueLotDetail id on id.IssueNo = im.IssueNo 
						 where im.AuthorizedBy is not null and im.IssueNo = @DocumentNo 
else if @transactionflag='DRMI'
	Insert into #LotDetail 
		select InvLotSummaryNo,InvLotReserveNo,IssuedWt ,id.IssuedQty  
				from DiscreteRM.MaterialIssueMain im
		inner join  DiscreteRM.MaterialIssueLotDetail id on id.IssueNo = im.IssueNo 
		 where im.IssueNo = @DocumentNo  and im.AuthorizedBy is not null
else if @transactionflag='DRST'
	Insert into #LotDetail 
		select InvLotSummaryNo,InvLotReserveNo,sd.Weight, sd.Quantity   from DiscreteRM.StockTransferMain sm
			inner join  DiscreteRM.StockTransferIssueDetail sd on sd.StockTransferNo = sm.StockTransferNo 
				 where sm.StockTransferNo  = @DocumentNo  and sm.AuthorizedBy is not null
else if @transactionflag='IMST'
	Insert into #LotDetail 
		select InvLotSummaryNo,InvLotReserveNo, Quantity,null from Inventory.InvStockTransferMain im
				inner join  Inventory.InvStockTransferLotDetail sd on sd.InvStockTransferNo = im.InvStockTransferNo
				 where im.AuthorizedBy is not null and im.InvStockTransferNo = @DocumentNo 
else if @transactionflag='MGPO'
	Insert into #LotDetail 
	
		select InvLotSummaryNo,InvLotReserveNo, Quantity,null from Inventory.GatePassOutMain gm
			inner join  Inventory.GatePassOutLotDetail gd on gd.GatePassOutNo = gm.GatePassOutNo 
			 where gm.GatePassOutNo= @DocumentNo  and gm.AuthorizedBy is not null
else if @transactionflag='SDCI'
	Insert into #LotDetail 
		select InvLotSummaryNo,InvLotReserveNo,Qty,cd.Pcs from Sales.CommercialInvoiceMain cm
			inner join  Sales.CommercialInvoiceItemLotDetail cd on cd.CommercialInvoiceNo = cm.CommercialInvoiceNo 
			 where cm.CommercialInvoiceNo= @DocumentNo and cm.AuthorizedBy is null
else if @transactionflag='SDCS'
	Insert into #LotDetail 
		select InvLotSummaryNo,InvLotReserveNo,Quantity,null   from Sales.CashSaleMain cm
		inner join Sales.CashSaleItemLotDetail cd on cd.CashSaleNo = cm.CashSaleNo   where cm.CashSaleNo = @DocumentNo 
		and cm.AuthorizedBy is not null
else if @transactionflag='SDDC'
	Insert into #LotDetail 
		select InvLotSummaryNo,InvLotReserveNo,Qty ,dd.Pcs    from Sales.DispatchChallan dc
			inner join  Sales.DispatchChallanItemLotDetail dd on dd.DispatchChallanNo = dc.DispatchChallanNo 
				  where dc.DispatchChallanNo = @DocumentNo  and AuthorizedBy is not null
else if @transactionflag='DIPP'
	Insert into #LotDetail 
		select InvLotSummaryNo,InvLotReserveNo,Quantity ,null from Dispensary.PatientPrescriptionMain pm
				inner join  Dispensary.PatientPrescriptionMedicineLotDetail pd on pd.PatientPrescriptionNo = pm.PatientPrescriptionNo
					 where pm.PatientPrescriptionNo= @DocumentNo  and pm.AuthorizedBy is not null
else if @transactionflag='SDEI'
	Insert into #LotDetail 
		select InvLotSummaryNo,InvLotReserveNo,Qty,id.Pcs  from Sales.InvoiceMain im
			inner join Sales.InvoiceItemLotDetail id on id.InvoiceNo = im.InvoiceNo 
			 where im.InvoiceNo = @DocumentNo  and im.AuthorizedBy is not null
else if @transactionflag='MMSA'
	Insert into #LotDetail 
		select InvLotSummaryNo,InvLotReserveNo,Qty,sd.Pcs  from Inventory.StockAdjustmentMain sa
				inner join  Inventory.StockAdjustmentLotDetail sd on sd.StockAdjustmentNo = sa.StockAdjustmentNo 
				 where sa.StockAdjustmentNo = @DocumentNo  and sa.AuthorizedBy is not null
else if @transactionflag='SDPS'
begin
	delete from #LotDetail 
	Insert into #LotDetail 
		select InvLotSummaryNo,InvLotReserveNo,Qty ,pd.IssuedPcs from Sales.PickingMain pm 
			inner join  Sales.PickingItemLotDetail pd on pd.PickingNo = pm.PickingNo 
			 where pm.PickingNo = @DocumentNo  and pm.AuthorizedBy is not null
		union all
		select InvLotSummaryNo,InvLotReserveNo,Qty ,pd.IssuedPcs from Sales.PickingMain pm 
		inner join  Sales.PickingComponentLotDetail pd on pd.PickingNo = pm.PickingNo 
		 where pm.PickingNo = @DocumentNo and AuthorizedBy is not null
end
else if @transactionflag='SDPK'
	Insert into #LotDetail 
		select InvLotSummaryNo,InvLotReserveNo,Qty ,pd.IssuedPcs   from Sales.PackSlipMain pm 
				inner join  Sales.PackSlipItemLotDetail pd on pm.PackSlipNo = pm.PackSlipNo 
					 where pm.PackSlipNo = @DocumentNo  and pm.AuthorizedBy is not null
		union all
		select InvLotSummaryNo,InvLotReserveNo,Qty ,sd.IssuedPcs    from Sales.PackSlipMain pm
				inner join  Sales.PackSlipComponentLotDetail sd on sd.PackSlipNo = pm.PackSlipNo
				 where pm.PackSlipNo = @DocumentNo  and pm.AuthorizedBy is not null
else if @transactionflag='VTJI'
	Insert into #LotDetail 
		select InvLotSummaryNo,InvLotReserveNo,id.RejectedQty,null from VTPL.JointInspectionMain  im
			inner join VTPL.JointInspectionMarkLotDetail  id on id.JointInspectionNo = im.JointInspectionNo
			 where im.JointInspectionNo = @DocumentNo  and im.AuthorizedBy is not null
			 and id.RejectedQty>0

else if @transactionflag='DRSA'
	Insert into #LotDetail 
		select ls.InvLotSummaryNo,id.InvLotReserveNo,id.ReserveQty ,id.ReservePcs  from 
				DiscreteRM.DiscreteStockAdjustmentMain im 
				inner join  [DiscreteRM].[DiscreteStockAdjustmentReserveDetail] id on id.DiscreteStockAdjustmentNo = im.DiscreteStockAdjustmentNo
				inner join Inventory.InvLotSummaryReserveDetail ls on ls.InvLotReserveNo = id.InvLotReserveNo 
						 where im.AuthorizedBy is not null and im.DiscreteStockAdjustmentNo = @DocumentNo 
		union all select ld.InvLotSummaryNo,null,ld.IssuedQty,ld.IssuedPcs
		 from Inventory.InvLotSummary ls 
		 inner join  Inventory.InvLotDetail ld on ld.InvLotSummaryNo = ls.InvLotSummaryNo 
		  where ld.DocumentNo =@DocumentNo and ld.TransactionFlag =@TransactionFlag and ls.IsReserved =0
else
	Insert into #LotDetail select InvLotSummaryNo,null,IssuedQty,IssuedPcs
		 from Inventory.InvLotDetail where DocumentNo =@DocumentNo and TransactionFlag =@TransactionFlag 		
		
if  exists  (select 1 from masterdata.mItemOtherInformation ioi 
				inner join Inventory.InvLotSummary ls on ls.ItemNo =ioi.ItemNo 
				inner join #LotDetail ld on ld.invLotSummaryNo = ls.InvLotSummaryNo 
				where ioi.StockValueMethodNo =4 and ls.IsCapital =0) and @TransactionFlag <>'ISML'
begin
	if  exists (select 1 from Finance.StockMonthlyLockingMain slm 
				inner join  Finance.ARDItemRule ard on ard.AccountNo = slm.StockAccountNo 
				inner join Inventory.InvLotDetail ld on ld.ItemNo = ard.ItemNo 
				where ard.ItemNo=ld.ItemNo and slm.CompanyNo = ld.CompanyNo  and ld.DocumentDate <= slm.DocumentDate
				and ld.DocumentNo  = @DocumentNo and ld.TransactionFlag = @TransactionFlag  )
		raiserror('Month is locked via Stock Monthly Closing(ISML)',16,-1)		

	if exists (select 1 from Finance.StockMonthlyLockingMain slm 
				inner join  Finance.ARDItemRule ard on ard.AccountNo = slm.StockAccountNo 
				inner join Inventory.InvLotBillDetail bd on bd.ItemNo = ard.ItemNo
				inner join Inventory.InvLotSummary ls on ls.InvLotSummaryNo = bd.InvLotSummaryNo 
				where ard.ItemNo= bd.ItemNo and slm.CompanyNo = ls.CompanyNo  and bd.BillDate  <= slm.DocumentDate
				and bd.BillNo = @DocumentNo and bd.TransactionFlag = @TransactionFlag  )
		raiserror('Month is locked via Stock Monthly Closing(ISML)',16,-1)		
		
end 
  
 if  @TransactionFlag in('PBPB','FIFI','FASB','FIDC','QDAS','MSVA')
	 begin
		
		 set @AllBillNo = null
		 set @AllBillNo =STUFF((select  ',' + lld.DocumentNoYearly + ' Dt-' + CONVERT(Varchar,LLD.DocumentDate,103)  + '(' +  lld.TransactionFlag  +  ')' from 
				Inventory.InvLotBillDetail bd 
				inner join Inventory.InvLotDetail ld on ld.InvLotDetailNo = bd.InvLotDetailNo 
				inner join Inventory.InvLotSummary ls on ls.InvLotSummaryNo = ld.InvLotSummaryNo 
				inner join Inventory.InvLotDetail lld on lld.InvLotSummaryNo = ls.InvLotSummaryNo 
				inner join masterdata.mItemOtherInformation ioi on ioi.ItemNo = ls.ItemNo 
				where bd.TransactionFlag = @TransactionFlag  and bd.BillNo =  @DocumentNo 
						and (ioi.StockValueMethodNo <>4 or ls.IsCapital =1)
					--and round(ld.VRate ,2) = round(bd.VRate,2)
				--and (bd.BillDate > ld.DocumentDate ) --or bd.createdDate > ld.createdDate)
				and bd.createdDate < lld.createdDate
				and lld.IssuedQty >0
				--and BillQty <>ISNULL(LotBalanceQty,0)
				group By lld.DocumentNoYearly ,lld.TransactionFlag ,lld.DocumentDate  
				for xml path ('')),1,1,'')
		  
		  if @AllBillNo is not null
		  begin
			set @AllBillNo ='Material is Issue against this Lot.,So you Can''t Delete this Entry - Issue No ' + @AllBillNo 
			raiserror(@AllBillNo,16,-1)		
		  end	
		  
		   set @AllBillNo = null
		 set @AllBillNo =STUFF((select  ',' + bd.BillNoYearly  + ' Dt-' + CONVERT(Varchar,bd.BillDate,103) +  '(' +  bd.TransactionFlag  +  ')' from 
				Inventory.InvLotBillDetail bd 
				inner join Inventory.InvLotSummary ls on ls.InvLotSummaryNo = bd.InvLotSummaryNo 
				inner join masterdata.mItemOtherInformation ioi on ioi.ItemNo = ls.ItemNo 
				where 
				bd.TransactionFlag <> @TransactionFlag and bd.BillNo <> @DocumentNo  and 
						(ioi.StockValueMethodNo <>4  or ls.IsCapital=1)
				and exists (select 1 from inventory.invlotBilldetail 
						where  TransactionFlag = @TransactionFlag and BillNo = @DocumentNo 
						and InvLotSummaryNo = bd.InvLotSummaryNo 
						--and (bd.BillDate > BillDate )--
						and bd.CreatedDate > CreatedDate
						)
				group By bd.BillNoYearly  ,bd.TransactionFlag  ,bd.BillDate
				for xml path ('')),1,1,'')
		  
		  if @AllBillNo is not null
		  begin
			set @AllBillNo ='Bill ' + @AllBillNo  + ' is prepared.,So you Can''t Delete this Entry.' 
			raiserror(@AllBillNo,16,-1)		
		  end	
		  
	 end

if @TransactionFlag ='MGPO'
begin
	if exists(select 1 from Inventory.GatepassOutMain where gatePassOutNo  = @DocumentNo and 
			GatePassTypeNo =8)
	begin
		Delete from Inventory.InvLotDetail WHERE invLotsummaryNo in ( select invLotsummaryNo from @Lot )
	end

else if @TransactionFlag ='FIFI'
begin
	
	
		Update ls set VRate = ls.VRate - bd.DiffRate ,LRate = LRate - bd.DiffInLRate 
		from
		Inventory.InvLotSummary ls
		cross apply(select isnull(SUM(bd.DiffRate),0) DiffRate,isnull(SUM(bd.DiffInLRate),0) DiffInLRate
				from Inventory.InvLotBillDetail bd where bd.InvLotSummaryNo = ls.InvLotSummaryNo 
				and bd.TransactionFlag =@TransactionFlag	and bd.BillNo = @DocumentNo 
				)bd 
			where 	InvLotSummaryNo in(select InvLotSummaryNo from 
			Inventory.InvLotBillDetail where  TransactionFlag =@TransactionFlag	and BillNo = @DocumentNo )
		
		  
		Update ld set  LRate = ld.LRate - bd.DiffInLRate 
		from
		Inventory.InvLotDetail ld
		inner join Inventory.InvLotBillDetail bd on bd.InvLotDetailNo  = ld.InvLotDetailNo  
		inner join Inventory.InvLotSummary ls on ls.InvLotSummaryNo = ld.InvLotSummaryNo 
			--and ls.RouteIdentity is null
		where bd.TransactionFlag =@TransactionFlag	and bd.BillNo = @DocumentNo
		
		Update LSTD set  RatePerUnit  = RatePerUnit  - bd.DiffRate  
		,Amount = Amount - (bd.DiffRate * bd.BillQty )
		from
		Inventory.InvLotSummaryTaxDetail   LSTD 
		inner join Inventory.InvLotBillDetail bd on bd.InvLotSummaryNo  = LSTD.InvLotSummaryNo  
		where bd.TransactionFlag =@TransactionFlag	and bd.BillNo = @DocumentNo
		and LSTD.MiscChargeNo is null and LSTD.ExpenseHeadNo is null
		
		delete ls
		from Inventory.InvLotBillDetail ld 
		inner join Inventory.InvLotSummaryTaxDetail  ls on ls.InvLotSummaryNo = ld.InvLotSummaryNo 
		inner join Inventory.InvLotTaxDetail td on ld.InvLotBillDetailNo = td.InvLotBillDetailNo 
				and ls.MiscChargeNo = td.MiscChargeNo 
		and ld.TransactionFlag =@TransactionFlag	and ld.BillNo = @DocumentNo 
		
		Delete td
		from Inventory.InvLotBillDetail ld 
		inner join Inventory.InvLotTaxDetail td on ld.InvLotBillDetailNo = td.InvLotBillDetailNo 
		and ld.TransactionFlag =@TransactionFlag	and ld.BillNo = @DocumentNo 
		
		Delete ld
		from Inventory.InvLotBillDetail ld 
		where ld.TransactionFlag =@TransactionFlag	and ld.BillNo = @DocumentNo

else if @TransactionFlag in('FASB' ,'PBPB' ,'FIDC' ,'QDAS','MSVA','ISML')
begin
	if exists (select 1 from Finance.ServiceBillMain where ServiceBillNo =@DocumentNo  and 
				ServiceBillSourceNo =2 and @TransactionFlag ='FASB') -- service Bill
		or exists (select 1 from Finance.PurchaseBillMain where PurchaseBillNo =@DocumentNo  and 
				PBSourceDocumentNo =2 and @TransactionFlag='PBPB') -- Purchase Bill
		or exists (select 1 from Finance.DrCrNoteMain where DrCrNoteNo=@DocumentNo  and 
				 @TransactionFlag='FIDC') -- Purchase Bill		
		or exists (select 1 from Inventory.QDAStocktransferMain where QDANo=@DocumentNo and 
				 @TransactionFlag='QDAS')
		or exists (select 1 from Inventory.StockValueAdjustmentMain where ValueAdjustmentNo =@DocumentNo and 
				 @TransactionFlag='MSVA')
		or exists (select 1 from Finance.StockMonthlyLockingMain  where StockLockingNo =@DocumentNo and 
				 @TransactionFlag='ISML')		 
	begin
		
		insert into @MAPDetail 
		 select CompanyNo,DivisionNo ,WarehouseNo ,ls.ItemNo ,ls.MakeNo ,TechnicalGradeNo 
		 from Inventory.InvLotSummary  ls
		 inner join Inventory.InvLotBillDetail ld  on ld.InvLotSummaryNo = ls.InvLotSummaryNo 
		inner join Inventory.InvLotTaxDetail td on ld.InvLotBillDetailNo = td.InvLotBillDetailNo 
		where ld.TransactionFlag =@TransactionFlag	and ld.BillNo = @DocumentNo 	
		 
		 group by CompanyNo,DivisionNo ,WarehouseNo ,ls.ItemNo ,ls.MakeNo ,TechnicalGradeNo
		 
		Delete td
		from Inventory.InvLotBillDetail ld 
		inner join Inventory.InvLotTaxDetail td on ld.InvLotBillDetailNo = td.InvLotBillDetailNo 
		and ld.TransactionFlag =@TransactionFlag	and ld.BillNo = @DocumentNo 	
		
		delete td
		from
		Inventory.InvLotSummaryTaxDetail td
		inner join Inventory .InvLotDetail ld on ld.InvLotSummaryNo = td.InvLotSummaryNo

Create table #temp (InvLotSummaryNo int,SNo smallint, DiffRate decimal(18,6),DiffInLRate decimal(18,6)	, Lrate  decimal(18,6) ,VRate  decimal(18,6) , 
			InvLotBillDetailNo int  , ItemNo int	, transactionflag char(4), ItemRate decimal(18,6) primary key (InvLotSummaryNo,SNo) ) 
		Insert into #temp Select InvLotSummaryNo, DENSE_RANK () over (Partition By InvLotSummaryNo order by CreatedDate,invLotBillDetailNo) as SNo,
			DiffRate,DiffInLRate,ItemRate + ISNULL(DiffInLRate,0) Lrate, VRate,InvLotBillDetailNo,ItemNo,transactionflag,ItemRate 
			From Inventory.InvLotBillDetail 
            where  TransactionFlag not in ('SDDC','SDEI','SDCI') 
            and invlotsummaryNo in(select invlotsummaryNo from #LotDetail )
         
         declare @Vrate decimal(18,6), @Lrate decimal(18,6),@InvlotSummaryNo int,@DiffRate  decimal(18,6)    
         Update #temp Set 
          @Lrate= case when Sno=1 then Itemrate + DiffInLRate else @Lrate + DiffInLRate end,
       	  Lrate=case when Sno=1 then Itemrate + DiffInLRate else  @Lrate end ,
       	  @Vrate= case when Sno=1 then Itemrate + DiffRate else @Vrate + DiffRate end,
       	  Vrate=case when Sno=1 then Itemrate + DiffRate else  @Vrate end,
       	 @InvlotSummaryNo=InvlotSummaryNo
		From #temp
        update ls set LRate =t.Lrate,VRate =t.Vrate
        from #temp t,Inventory.InvLotBillDetail   bd,Inventory.InvLotSummary ls where(T.InvLotBillDetailNo = bd.InvLotBillDetailNo) 
       And (t.Lrate <> ls.LRate or t.Vrate <> ls.Vrate ) And ls.InvLotSummaryNo = bd.InvLotSummaryNo and sNo = (select MAX(SNo) from #temp where InvLotSummaryNo= ls.InvLotSummaryNo )

       and ls.VRate <>t.Vrate and ls.InvLotSummaryNo = bd.InvLotSummaryNo and sNo = (select MAX(SNo) from #temp where InvLotSummaryNo= ls.InvLotSummaryNo )
      update bd set VRate =t.Vrate  from #temp t,Inventory.InvLotBillDetail   bd where(T.InvLotBillDetailNo = bd.InvLotBillDetailNo) and t.Vrate <>bd.VRate
      update ld set VRate =ls.Vrate,LRate = ls.LRate from #temp t,Inventory.InvLotDetail ld , Inventory.InvLotSummary ls	 
      where ld.invlotsummaryNo = ls.invlotsummaryNo and T.invlotsummaryNo = ld.invlotsummaryNo and t.Vrate <>ld.VRate and ls.TransactionFlag = ld.TransactionFlag
