CREATE PROCEDURE [Information].[PendingMaterialRequestForCostCenter] 
	-- Add the parameters for the stored procedure here
	@CompanyNo tinyint,
	@DocumentDate smalldatetime,
	@CostCenterNo int
	,@ItemNo int
AS
BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;

    -- Insert statements for procedure here

    
	declare @str varchar(max)
    select @str=masterdata.JSForInformation('IDMR')
    set @str=@str+masterdata.CSSForInformation('IDMR')
	set @str=@str+'<table class="altrowstable" id="alternatecolorPMRInfoCost">'
	set @str=@str+'<Caption><Font color="Maroon"><h3 align="Left"><b>Pending Matarial Request</b></h3></Font></Caption>'
    set @str=@str+'<tr><th>Sl. No.</th><th>Division</th><th>Department</th><th>Document No.</th><th>Document Date</th><th>Make</th><th>Material Request Quantity</th><th>Purchase Request Quantity</th><th>Issue Quantity</th><th>Balance Quantity</th><th>Created By</th><th>Authorized By</th></tr>'
	set @str=@str
				 +(
				    REPLACE(
				    replace(
					replace(
					replace(
					replace(
					REPLACE(
				    REPLACE(
				    replace(
					replace(
					replace(
					replace(
					replace(
					  (    
						 select   ROW_NUMBER() over ( Order by irm.DocumentDate desc) as td1
						          ,isnull(DM.DivisionName,'') td2     
							      ,DT.DeptName td3
								  ,irm.DocumentNoYearly as   td4
								  , convert(varchar(20),irm.DocumentDate,103)td5
								  ,isnull(MM.MakeName,'') td6
								  ,cast( Sum(ird.Quantity) as decimal(15,3)) td7
								  ,sum(isnull(dtIndent.IndentQty,0)) td8
								  ,Sum(isnull(ird.IssueQty,0)) td9
								  ,Sum(isnull(ird.Quantity,0)-(isnull(dtIndent.IndentQty,0)+isnull(ird.IssueQty,0)))  td10
								  ,L1.PrintingName  td11
								  ,isnull(L2.PrintingName,'') td12
							
						from 
						Inventory.IndentRequisitionMain irm 
						Inner join Inventory.IndentRequisitionItemDetail ird on ird.IndentRequisitionNo = irm.IndentRequisitionNo 
						Inner Join payroll.tblDeptMaster DT ON DT.DeptNo=irm.DeptNo
						Left Join masterdata.mMakeMaster MM ON MM.MakeNo=ird.MakeNo
						Left Join payroll.tblDivisionMaster DM ON DM.DivisionNo=irm.DivisionNo
						Inner Join Security.Login L1 on L1.LoginNo=irm.CreatedBy
						Left Join Security.Login L2 On L2.LoginNo=irm.AuthorizedBy
						Left join  ( Select inm.RefDocumentNo,SUM(IND.IndentQty) as IndentQty,
									 IND.ItemNo,IND.MakeNo,IND.TechnicalGradeNo
						             from Inventory.IndentMain inm 
									 Left Join Inventory.IndentItemDetail IND ON IND.IndentNo=inm.IndentNo
									 Where inm.IndentAgainstNo = 3 
									 And IND.ItemNo = @ItemNo
									 GROUP by inm.RefDocumentNo,IND.ItemNo,IND.MakeNo,IND.TechnicalGradeNo
								   ) dtIndent	 
						            On dtIndent.RefDocumentNo = irm.IndentRequisitionNo
						            And dtIndent.ItemNo=ird.ItemNo
						            And isnull(dtIndent.MakeNo,0)=isnull(ird.MakeNo,0)
						            And ISNULL(dtIndent.TechnicalGradeNo,0) = ISNULL(ird.TechnicalGradeNo,0)
						where
							 irm.CompanyNo = @CompanyNo 
							 and irm.StatusNo in(1,8,5,6)   ---- For Pending
							 And ird.CostCenterNo  =@CostCenterNo 
							 and ird.ItemNo=@ItemNo 
							 and irm.DocumentDate <= @DocumentDate
						Group By DM.DivisionName,DT.DeptName,irm.DocumentNoYearly 
								 ,irm.DocumentDate,MM.MakeName
								  ,L1.PrintingName,L2.PrintingName
								   							 	
						
			  for xml path('tr')
			     ),'<td1>','<td style="text-align: center;">'),'<td2>','<td>')
								  ,'<td3>','<td>')
								  ,'<td4>','<td>')
								  ,'<td5>','<td  style="text-align: center;">')
								  ,'<td6>','<td>')
								  ,'<td7>','<td  style="text-align: Right;">')
								  ,'<td8>','<td  style="text-align: Right;">')
								  ,'<td9>','<td  style="text-align: Right;">')
								  ,'<td10>','<td  style="text-align: Right;">')
								  ,'<td11>','<td>')
								  ,'<td12>','<td>')
								)
								
				   
	set @str=@str+'</table>'
	
	select @str
END
