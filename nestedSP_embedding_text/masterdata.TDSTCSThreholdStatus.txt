Stored procedure masterdata.TDSTCSThreholdStatus: Update Threshold Status. Called from Finance.PBPBAuthorize.
Writes: masterdata.TDSTCSYearlySetupVendorDetail
Reads: Finance.DrCrNoteItemDetail, Finance.DrCrNoteMain, Finance.PurchaseBillItemDetail, Finance.PurchaseBillMain, masterdata.mCompanyMaster, masterdata.mFYear, masterdata.mItemMaster, masterdata.mVendorLocationDetail, masterdata.mVendorLocationInformation, masterdata.TDSTCSSetup, masterdata.TDSTCSSetupExcludedItemDetail, masterdata.TDSTCSYearlySetup, masterdata.TDSTCSYearlySetupCompanyDetail, masterdata.TDSTCSYearlySetupVendorDetail, Sales.CommercialInvoiceMain, Sales.InvoiceCancellationMain, Sales.InvoiceItemDetails, Sales.InvoiceMain, Sales.SalesReturnItemDetail, Sales.SalesReturnMain, Sales.SalesSupplementaryInvoiceItemDetails, Sales.SalesSupplementaryInvoiceMain
Parameters: @FormCode Varchar(4), @DocumentNo Int, @IsDelete Bit = 0, @VendorLocationNo Int = Null, @CompanyNo TinyInt = Null
Author: Alok, created 22 July 2021
Code:
-- =============================================
-- Author:		Alok
-- Create date: 22 July 2021
-- Description:	Update Threshold Status
-- =============================================
CREATE PROCEDURE [masterdata].[TDSTCSThreholdStatus]
@FormCode Varchar(4),
@DocumentNo Int,
@IsDelete Bit=0, --- Pass True when you want to update threhold when cancelling invoice or deleting invoices
@VendorLocationNo Int= Null,-- Pass Value only when deleting document
@CompanyNo TinyInt =Null -- Pass only when deleting 
AS
BEGIN
Begin try
		
		Declare @YearNo TinyInt,@DocumentDate SmallDateTime,@TDSSetupNo SmallInt,@TCSSetupNo SmallInt,@IsUpdateStatus Bit=0,@ValidateFor tinyint
		
		Declare @VendorDetail as Table(VendorLocationNo Int)
		Declare @CompanyTCSDetail as Table(CompanyNo TinyInt)
		Declare @CompanyTDSDetail as Table(CompanyNo TinyInt)
		
		
		If @FormCode='SDEI'
		BEGIN
				Select @YearNo=IM.YearNo,@CompanyNo=IM.CompanyNo,@DocumentDate=IM.DocumentDate,@VendorLocationNo=IM.CustomerLocationNo
				From Sales.InvoiceMain IM
				Where IM.InvoiceNo=@DocumentNo
		END
		Else If @FormCode='FIDC'
		BEGIN
				Select @YearNo=IM.YearNo,@CompanyNo=IM.CompanyNo,@DocumentDate=IM.DocumentDate,@VendorLocationNo=IM.VendorLocationNo
				From Finance.DrCrNoteMain IM
				Where IM.DrCrNoteNo=@DocumentNo
		END

		Else If @FormCode='PBPB'
		BEGIN
				Select @YearNo=IM.YearNo,@CompanyNo=IM.CompanyNo,@DocumentDate=IM.DocumentDate,@VendorLocationNo=IM.VendorLocationNo
				From Finance.PurchaseBillMain IM
				Where IM.PurchaseBillNo=@DocumentNo
		END			
		Else If @FormCode='SLTI'
		BEGIN
				Select @YearNo=IM.YearNo,@CompanyNo=IM.CompanyNo,@DocumentDate=IM.DocumentDate,@VendorLocationNo=IM.CustomerLocationNo
				From Sales.SalesSupplementaryInvoiceMain IM
				Where IM.SupplementaryInvoiceNo=@DocumentNo
		END
		Else If @FormCode='SDIC'
		BEGIN
				Select @YearNo=ICM.YearNo,@CompanyNo=ICM.CompanyNo,@DocumentDate=ICM.DocumentDate
				,@VendorLocationNo=IsNull(IM.CustomerLocationNo,CIM.CustomerLocationNo)
				From Sales.InvoiceCancellationMain ICM
				Left join Sales.InvoiceMain IM on IM.InvoiceNo=ICM.RefDocumentNo and ICM.InvoiceTypeNo=1
				Left Join Sales.CommercialInvoiceMain CIM on CIM.CommercialInvoiceNo=ICM.RefDocumentNo AND ICM.InvoiceTypeNo=2				
				Where ICM.InvoiceCancellationNo=@DocumentNo
		END		
		
		
		If @IsDelete=1 or @FormCode='SDIC'
		BEGIN
			Select @IsUpdateStatus=1
		END
		
		If @DocumentDate is Null
		BEGIN
			Select @DocumentDate = CONVERT(Date,Getdate())
			Select @YearNo=YearNo
			From masterdata.mFYear 
			Where @DocumentDate between StartDate and EndDate
		END
		

		-- TDS Setup
		Select @TDSSetupNo=TS.ITYSetupNo,@ValidateFor = TD.ValidateFor
		From masterdata.TDSTCSYearlySetup TS
		Inner Join masterdata.TDSTCSSetup TD on TD.TaxSetupNo=TS.TaxSetupNo
		Inner Join masterdata.TDSTCSYearlySetupCompanyDetail CD on CD.ITYSetupNo=TS.ITYSetupNo
		Where TS.YearNo=@YearNo And CD.CompanyNo=@CompanyNo
		And TD.TaxCategoryNo=5 -- TDS
		And TD.EffectiveFromDate<=@DocumentDate And TS.Inactive=0
		Order by TD.EffectiveFromDate asc,  TS.ITYSetupNo asc -- If getting more then one row then last row will be assigned to variable thats why sorted by date and no
		
		-- TCS Setup
		Select @TCSSetupNo=TS.ITYSetupNo,@ValidateFor = TD.ValidateFor
		From masterdata.TDSTCSYearlySetup TS
		Inner Join masterdata.TDSTCSSetup TD on TD.TaxSetupNo=TS.TaxSetupNo
		Inner Join masterdata.TDSTCSYearlySetupCompanyDetail CD on CD.ITYSetupNo=TS.ITYSetupNo
		Where TS.YearNo=@YearNo And CD.CompanyNo=@CompanyNo
		And TD.TaxCategoryNo=6 -- TCS		
		And TD.EffectiveFromDate<=@DocumentDate And TS.Inactive=0
		Order by TD.EffectiveFromDate asc,  TS.ITYSetupNo asc -- If getting more then one row then last row will be assigned to variable thats why sorted by date and no
		
		
		-- Company for TCS
		Insert Into @CompanyTCSDetail
		Select CD.CompanyNo
		From masterdata.TDSTCSYearlySetupCompanyDetail CD
		Where CD.ITYSetupNo=@TCSSetupNo
		
		-- Company for TDS
		Insert Into @CompanyTDSDetail
		Select CD.CompanyNo
		From masterdata.TDSTCSYearlySetupCompanyDetail CD
		Where CD.ITYSetupNo=@TDSSetupNo			
		
		
		--- Get All Vendor having Same PAN No
		Insert Into @VendorDetail
		Select VLI1.VendorLocationNo
		From masterdata.mVendorLocationDetail VLD
		Inner Join masterdata.mVendorLocationInformation VLI1 on VLI1.VendorLocationNo=VLD.VendorLocationNo
		Inner Join masterdata.mVendorLocationInformation VLI2 on (VLI1.PANNo=VLI2.PANNo and VLI2.VendorLocationNo=@VendorLocationNo) -- Vendor Having Same PAN
															  Or (VLI1.PANNo Is Null And VLI2.VendorLocationNo=@VendorLocationNo 
																  And VLI2.VendorLocationNo=VLI1.VendorLocationNo
																  )-- When PAN is null then take current Vendor location only
		Where VLD.CountryNo In (
									Select CM.CountryNo
									from masterdata.mCompanyMaster CM
									Inner join @CompanyTCSDetail CTCS On CM.CompanyNo = CTCS.CompanyNo
									Union all
									Select CM.CountryNo
									from masterdata.mCompanyMaster CM
									Inner join @CompanyTDSDetail CTDS On CM.CompanyNo = CTDS.CompanyNo
								)
		
		--- TCS
		BEGIN
			If Not Exists (
							Select 1
							From masterdata.TDSTCSYearlySetupVendorDetail TVD
							Inner Join @VendorDetail VD on VD.VendorLocationNo=TVD.VendorLocationNo
							Where TVD.ITYSetupNo=@TCSSetupNo
							And (TVD.IsExcludeFromValidation=1 or TVD.IsAlwaysValidate=1)
						  )--- If Always validate for TCS or Excluded from validation then don't check for threshold
			BEGIN
				--If Not Exists (
				--				Select 1
				--				From @VendorDetail VD
				--				Inner Join masterdata.TDSTCSYearlySetupVendorDetail TVD on TVD.VendorLocationNo=VD.VendorLocationNo 
				--																		And TVD.ITYSetupNo=@TCSSetupNo
				--				Where TVD.IsThresholdReached=0
				--			  ) -- All Vendor reached threshold then don't do any thing
				--BEGIN
				--	 Select 1
				--END
				--Else 
				If Exists (
								Select 1
								From @VendorDetail VD
								Inner Join masterdata.TDSTCSYearlySetupVendorDetail TVD on TVD.VendorLocationNo=VD.VendorLocationNo 
																				And TVD.ITYSetupNo=@TCSSetupNo
								Where TVD.IsThresholdReached=1
								) -- If Partial Reached threshold then update to others too
					And Exists (
								Select 1
								From @VendorDetail VD
								Left Join masterdata.TDSTCSYearlySetupVendorDetail TVD on TVD.VendorLocationNo=VD.VendorLocationNo 
																				And TVD.ITYSetupNo=@TCSSetupNo
								Where IsNull(TVD.IsThresholdReached,0)=0
								) -- If Partial Reached threshold then update to others too
				BEGIN
						Update TVD set TVD.IsThresholdReached=1
						From masterdata.TDSTCSYearlySetupVendorDetail TVD
						Inner Join @VendorDetail VD on VD.VendorLocationNo=TVD.VendorLocationNo
						Where TVD.ITYSetupNo=@TCSSetupNo
						
						Insert Into masterdata.TDSTCSYearlySetupVendorDetail 
						Select @TCSSetupNo,VD.VendorLocationNo,1,0,0
						From @VendorDetail VD
						Left Join masterdata.TDSTCSYearlySetupVendorDetail TVD on TVD.VendorLocationNo=VD.VendorLocationNo 
																		And TVD.ITYSetupNo=@TCSSetupNo
						Where IsNull(TVD.IsThresholdReached,0)=0
						And Not Exists (Select 1 From masterdata.TDSTCSYearlySetupVendorDetail  TVD1 Where TVD1.ITYSetupNo=@TCSSetupNo and TVD1.VendorLocationNo=VD.VendorLocationNo)
				END
				Else If Exists(
								Select 1
								From @VendorDetail VD
								Left Join masterdata.TDSTCSYearlySetupVendorDetail TVD on TVD.VendorLocationNo=VD.VendorLocationNo 
																						And TVD.ITYSetupNo=@TCSSetupNo
								Where IsNull(TVD.IsThresholdReached,0)=0
							  ) -- Check for threshold
				And  Exists ( --- Check wheather threshold reached or not
							Select 1
							From (
								Select IID.NetAmount
								From Sales.InvoiceMain INVM
								Inner Join masterdata.TDSTCSYearlySetupCompanyDetail CD on CD.CompanyNo=INVM.CompanyNo
								Inner Join @VendorDetail VLD on VLD.VendorLocationNo=INVM.CustomerLocationNo
								Inner Join @CompanyTCSDetail CCD on CCD.CompanyNo=INVM.CompanyNo
								Inner Join Sales.InvoiceItemDetails IID on IID.InvoiceNo=INVM.InvoiceNo
								Inner Join masterdata.mItemMaster im on im.ItemNo=IID.ItemNo and im.ItemTypeNo<>11
								Where CD.ITYSetupNo=@TCSSetupNo
								And INVM.YearNo=@YearNo
								And INVM.StatusNo<>4
								And IID.ItemNo Not in (Select EID.ItemNo
													   From masterdata.TDSTCSSetupExcludedItemDetail EID 
													   Inner Join masterdata.TDSTCSYearlySetup TY on TY.TaxSetupNo=EID.TaxSetupNo
													   Where TY.ITYSetupNo=@TCSSetupNo
													   And EID.ItemNo is Not Null
													   )
								And INVM.DocumentStatusNo >= @ValidateFor		   
								Union All
								Select IID.NetAmount
								From Sales.SalesSupplementaryInvoiceMain INVM
								Inner Join masterdata.TDSTCSYearlySetupCompanyDetail CD on CD.CompanyNo=INVM.CompanyNo
								Inner Join @VendorDetail VLD on VLD.VendorLocationNo=INVM.CustomerLocationNo
								Inner Join @CompanyTCSDetail CCD on CCD.CompanyNo=INVM.CompanyNo
								Inner Join Sales.SalesSupplementaryInvoiceItemDetails IID on IID.SupplementaryInvoiceNo=INVM.SupplementaryInvoiceNo								
								Inner Join masterdata.mItemMaster im on im.ItemNo=IID.ItemNo and im.ItemTypeNo<>11
								Where CD.ITYSetupNo=@TCSSetupNo
								And INVM.YearNo=@YearNo
								And INVM.StatusNo<>4
								And IID.ItemNo Not in (Select EID.ItemNo
													   From masterdata.TDSTCSSetupExcludedItemDetail EID 
													   Inner Join masterdata.TDSTCSYearlySetup TY on TY.TaxSetupNo=EID.TaxSetupNo
													   Where TY.ITYSetupNo=@TCSSetupNo
													   And EID.ItemNo is Not Null
													   )								
								And INVM.DocumentStatusNo  >= @ValidateFor				   					   
								Union All								
								Select (-1) * IID.NetAmount
								From Sales.SalesReturnMain INVM
								Inner Join masterdata.TDSTCSYearlySetupCompanyDetail CD on CD.CompanyNo=INVM.CompanyNo
								Inner Join @VendorDetail VLD on VLD.VendorLocationNo=INVM.CustomerLocationNo
								Inner Join @CompanyTCSDetail CCD on CCD.CompanyNo=INVM.CompanyNo
								Inner Join Sales.SalesReturnItemDetail IID on IID.SalesReturnNo=INVM.SalesReturnNo								
								Inner Join masterdata.mItemMaster im on im.ItemNo=IID.ItemNo and im.ItemTypeNo<>11
								Where CD.ITYSetupNo=@TCSSetupNo
								And INVM.YearNo=@YearNo								
								And IID.ItemNo Not in (Select EID.ItemNo
													   From masterdata.TDSTCSSetupExcludedItemDetail EID 
													   Inner Join masterdata.TDSTCSYearlySetup TY on TY.TaxSetupNo=EID.TaxSetupNo
													   Where TY.ITYSetupNo=@TCSSetupNo
													   And EID.ItemNo is Not Null
													   )
								And INVM.DocumentStatusNo >= @ValidateFor						   					   
								Union all
								Select DID.NetAmount * MF.MultiFactor BasicAmount								
								From Finance.DrCrNoteMain DR
								Inner Join Finance.DrCrNoteItemDetail DID on DID.DrCrNoteNo=DR.DrCrNoteNo								
								Inner Join @VendorDetail VLD on VLD.VendorLocationNo=DR.VendorLocationNo
								Inner Join @CompanyTCSDetail CCD on CCD.CompanyNo=DR.CompanyNo								
								Inner Join masterdata.mItemMaster im on im.ItemNo=DID.ItemNo and im.ItemTypeNo<>11
								Inner Join (
											Select CONVERT(TinyInt,1) Id,1 MultiFactor Union All
											Select CONVERT(TinyInt,2) Id,-1 MultiFactor
		  									)MF on MF.Id=DR.DrCrTypeNo
								Where DR.SourceDocumentNo=2 -- Tax Invoice		
								And dr.DocumentDate<=@DocumentDate								
								And DR.YearNo=@YearNo								
								And DR.DocumentStatusNo  >= @ValidateFor						   
								)tt								
							Having IsNull(Sum(NetAmount),0)>=(Select ThresholdAmount 
													From masterdata.TDSTCSYearlySetup YS 
													Inner Join Masterdata.TDSTCSSetup TS on TS.TaxSetupNo=YS.TaxSetupNo 
													Where YS.ITYSetupNo=@TCSSetupNo													
													)
						)
				BEGIN	
						Update TVD set TVD.IsThresholdReached=1
						From masterdata.TDSTCSYearlySetupVendorDetail TVD
						Inner Join @VendorDetail VD on VD.VendorLocationNo=TVD.VendorLocationNo
						Where TVD.ITYSetupNo=@TCSSetupNo	
						
						Insert Into masterdata.TDSTCSYearlySetupVendorDetail 
						Select @TCSSetupNo,VD.VendorLocationNo,1,0,0
						From @VendorDetail VD
						Left Join masterdata.TDSTCSYearlySetupVendorDetail TVD on TVD.VendorLocationNo=VD.VendorLocationNo 
																		And TVD.ITYSetupNo=@TCSSetupNo
						Where Not Exists (Select 1 From masterdata.TDSTCSYearlySetupVendorDetail  TVD1 where TVD1.ITYSetupNo=@TCSSetupNo and TVD1.VendorLocationNo=VD.VendorLocationNo)
						
				END
			 END

				If @IsUpdateStatus=1 --- Update only for delete case
				And Not Exists(
								Select 1
								From @VendorDetail VD
								Inner Join masterdata.TDSTCSYearlySetupVendorDetail TVD on TVD.VendorLocationNo=VD.VendorLocationNo 
																						And TVD.ITYSetupNo=@TCSSetupNo
								Where TVD.IsAlwaysValidate=1
							  )				
				And Exists ( --- Check wheather threshold reached or not
							Select 1
							From (
								Select IID.NetAmount
								From Sales.InvoiceMain INVM
								Inner Join masterdata.TDSTCSYearlySetupCompanyDetail CD on CD.CompanyNo=INVM.CompanyNo
								Inner Join @VendorDetail VLD on VLD.VendorLocationNo=INVM.CustomerLocationNo
								Inner Join @CompanyTCSDetail CCD on CCD.CompanyNo=INVM.CompanyNo
								Inner Join Sales.InvoiceItemDetails IID on IID.InvoiceNo=INVM.InvoiceNo
								Inner Join masterdata.mItemMaster im on im.ItemNo=IID.ItemNo and im.ItemTypeNo<>11
								Where CD.ITYSetupNo=@TCSSetupNo
								And INVM.YearNo=@YearNo								
								And INVM.StatusNo<>4
								And INVM.DocumentStatusNo  >= @ValidateFor	
								And IID.ItemNo Not in (Select EID.ItemNo
													   From masterdata.TDSTCSSetupExcludedItemDetail EID 
													   Inner Join masterdata.TDSTCSYearlySetup TY on TY.TaxSetupNo=EID.TaxSetupNo
													   Where TY.ITYSetupNo=@TCSSetupNo
													   And EID.ItemNo is Not Null
													   )
								Union All
								Select IID.NetAmount
								From Sales.SalesSupplementaryInvoiceMain INVM
								Inner Join masterdata.TDSTCSYearlySetupCompanyDetail CD on CD.CompanyNo=INVM.CompanyNo
								Inner Join @VendorDetail VLD on VLD.VendorLocationNo=INVM.CustomerLocationNo
								Inner Join @CompanyTCSDetail CCD on CCD.CompanyNo=INVM.CompanyNo
								Inner Join Sales.SalesSupplementaryInvoiceItemDetails IID on IID.SupplementaryInvoiceNo=INVM.SupplementaryInvoiceNo								
								Inner Join masterdata.mItemMaster im on im.ItemNo=IID.ItemNo and im.ItemTypeNo<>11
								Where CD.ITYSetupNo=@TCSSetupNo
								And INVM.YearNo=@YearNo								
								And INVM.StatusNo<>4
								And  INVM.DocumentStatusNo  >= @ValidateFor
								And IID.ItemNo Not in (Select EID.ItemNo
													   From masterdata.TDSTCSSetupExcludedItemDetail EID 
													   Inner Join masterdata.TDSTCSYearlySetup TY on TY.TaxSetupNo=EID.TaxSetupNo
													   Where TY.ITYSetupNo=@TCSSetupNo
													   And EID.ItemNo is Not Null
													   )								
								Union All								
								Select (-1) * IID.NetAmount
								From Sales.SalesReturnMain INVM
								Inner Join masterdata.TDSTCSYearlySetupCompanyDetail CD on CD.CompanyNo=INVM.CompanyNo
								Inner Join @VendorDetail VLD on VLD.VendorLocationNo=INVM.CustomerLocationNo
								Inner Join @CompanyTCSDetail CCD on CCD.CompanyNo=INVM.CompanyNo
								Inner Join Sales.SalesReturnItemDetail IID on IID.SalesReturnNo=INVM.SalesReturnNo								
								Inner Join masterdata.mItemMaster im on im.ItemNo=IID.ItemNo and im.ItemTypeNo<>11
								Where CD.ITYSetupNo=@TCSSetupNo
								And INVM.YearNo=@YearNo								
								And  INVM.DocumentStatusNo  >= @ValidateFor
								And IID.ItemNo Not in (Select EID.ItemNo
													   From masterdata.TDSTCSSetupExcludedItemDetail EID 
													   Inner Join masterdata.TDSTCSYearlySetup TY on TY.TaxSetupNo=EID.TaxSetupNo
													   Where TY.ITYSetupNo=@TCSSetupNo
													   And EID.ItemNo is Not Null
													   )
								Union all
								Select DID.NetAmount * MF.MultiFactor BasicAmount								
								From Finance.DrCrNoteMain DR
								Inner Join Finance.DrCrNoteItemDetail DID on DID.DrCrNoteNo=DR.DrCrNoteNo								
								Inner Join @VendorDetail VLD on VLD.VendorLocationNo=DR.VendorLocationNo
								Inner Join @CompanyTCSDetail CCD on CCD.CompanyNo=DR.CompanyNo								
								Inner Join masterdata.mItemMaster im on im.ItemNo=DID.ItemNo and im.ItemTypeNo<>11
								Inner Join (
											Select CONVERT(TinyInt,1) Id,1 MultiFactor Union All
											Select CONVERT(TinyInt,2) Id,-1 MultiFactor
		  									)MF on MF.Id=DR.DrCrTypeNo
								Where DR.SourceDocumentNo=2 -- Tax Invoice										
								And DR.YearNo=@YearNo								
								And  DR.DocumentStatusNo  >= @ValidateFor
								)tt								
							Having IsNull(Sum(NetAmount),0)<(Select ThresholdAmount 
													From masterdata.TDSTCSYearlySetup YS 
													Inner Join Masterdata.TDSTCSSetup TS on TS.TaxSetupNo=YS.TaxSetupNo 
													Where YS.ITYSetupNo=@TCSSetupNo
													)
						)
				BEGIN
						Update TVD set TVD.IsThresholdReached=0
						From masterdata.TDSTCSYearlySetupVendorDetail TVD
						Inner Join @VendorDetail VD on VD.VendorLocationNo=TVD.VendorLocationNo
						Where TVD.ITYSetupNo=@TCSSetupNo
				END			 
		END
		
		--- TDS
		BEGIN
			If Not Exists (
							Select 1
							From masterdata.TDSTCSYearlySetupVendorDetail TVD
							Inner Join @VendorDetail VD on VD.VendorLocationNo=TVD.VendorLocationNo
							Where TVD.ITYSetupNo=@TDSSetupNo
							And (TVD.IsExcludeFromValidation=1 or TVD.IsAlwaysValidate=1)
						  )--- If Always validate for TDS or Excluded from validation then don't check for threshold
			BEGIN
				If Exists (
								Select 1
								From @VendorDetail VD
								Inner Join masterdata.TDSTCSYearlySetupVendorDetail TVD on TVD.VendorLocationNo=VD.VendorLocationNo 
																				And TVD.ITYSetupNo=@TDSSetupNo
								Where TVD.IsThresholdReached=1
								) -- If Partial Reached threshold then update to others too
					And Exists (
								Select 1
								From @VendorDetail VD
								Inner Join masterdata.TDSTCSYearlySetupVendorDetail TVD on TVD.VendorLocationNo=VD.VendorLocationNo 
																				And TVD.ITYSetupNo=@TDSSetupNo
								Where TVD.IsThresholdReached=0
								) -- If Partial Reached threshold then update to others too
				BEGIN
						Update TVD set TVD.IsThresholdReached=1
						From masterdata.TDSTCSYearlySetupVendorDetail TVD
						Inner Join @VendorDetail VD on VD.VendorLocationNo=TVD.VendorLocationNo
						Where TVD.ITYSetupNo=@TDSSetupNo

						Insert Into masterdata.TDSTCSYearlySetupVendorDetail 
						Select @TDSSetupNo,VD.VendorLocationNo,1,0,0
						From @VendorDetail VD
						Left Join masterdata.TDSTCSYearlySetupVendorDetail TVD on TVD.VendorLocationNo=VD.VendorLocationNo 
																		And TVD.ITYSetupNo=@TDSSetupNo
						Where Not Exists (Select 1 From masterdata.TDSTCSYearlySetupVendorDetail  TVD1 where TVD1.ITYSetupNo=@TDSSetupNo and TVD1.VendorLocationNo=VD.VendorLocationNo)
						
						
				END
				Else If Exists(
								Select 1
								From @VendorDetail VD
								Left Join masterdata.TDSTCSYearlySetupVendorDetail TVD on TVD.VendorLocationNo=VD.VendorLocationNo 
																						And TVD.ITYSetupNo=@TDSSetupNo
								Where IsNull(TVD.IsThresholdReached,0)=0
							  ) -- Check for threshold
				And  Exists ( --- Check wheather threshold reached or not
							Select 1
							From (
								Select PBM.PurchaseBillNo,PBID.BasicAmount--,PBM.NetAmount
								From Finance.PurchaseBillMain PBM
								Inner Join Finance.PurchaseBillItemDetail PBID on PBID.PurchaseBillNo=PBM.PurchaseBillNo
								Inner Join masterdata.TDSTCSYearlySetupCompanyDetail CD on CD.CompanyNo=PBM.CompanyNo
								Inner Join @VendorDetail VLD on VLD.VendorLocationNo=PBM.VendorLocationNo
								Inner Join @CompanyTDSDetail CCD on CCD.CompanyNo=PBM.CompanyNo
								Inner Join masterdata.mItemMaster im on im.ItemNo=PBID.ItemNo and im.ItemTypeNo<>11
								Where CD.ITYSetupNo=@TDSSetupNo								
								And PBM.YearNo=@YearNo
								and PBM.DocumentDate<=@DocumentDate
								And ((@FormCode='PBPB') Or @FormCode<>'PBPB')
								and  PBM.DocumentStatusNo  >= @ValidateFor
								And PBID.ItemNo Not in (Select EID.ItemNo
														From masterdata.TDSTCSSetupExcludedItemDetail EID 
														Inner Join masterdata.TDSTCSYearlySetup TY on TY.TaxSetupNo=EID.TaxSetupNo
														Where TY.ITYSetupNo=@TdSSetupNo
														)
								Union all
								Select DR.DrCrNoteNo
								,DID.Amount * MF.MultiFactor BasicAmount
								--,ID.NetAmount* MF.MultiFactor
								From Finance.DrCrNoteMain DR
								Inner Join Finance.DrCrNoteItemDetail DID on DID.DrCrNoteNo=DR.DrCrNoteNo
								Inner Join masterdata.mItemMaster im on im.ItemNo=DID.ItemNo and im.ItemTypeNo<>11
								Inner Join @VendorDetail VLD on VLD.VendorLocationNo=DR.VendorLocationNo
								Inner Join @CompanyTDSDetail CCD on CCD.CompanyNo=DR.CompanyNo								
								Inner Join (
											Select CONVERT(TinyInt,2) Id,1 MultiFactor Union All -- Cr Note
											Select CONVERT(TinyInt,1) Id,-1 MultiFactor -- Dr Note
		  									)MF on MF.Id=DR.DrCrTypeNo
								Where DR.SourceDocumentNo=1 -- PUrchaseBill
								And dr.DocumentDate<=@DocumentDate
								And DR.YearNo=@YearNo
								and  DR.DocumentStatusNo  >= @ValidateFor
								And (( @FormCode='FIDC') Or @FormCode<>'FIDC')
								And DID.ItemNo Not in (Select EID.ItemNo
														From masterdata.TDSTCSSetupExcludedItemDetail EID 
														Inner Join masterdata.TDSTCSYearlySetup TY on TY.TaxSetupNo=EID.TaxSetupNo
														Where TY.ITYSetupNo=@TDSSetupNo
														)								
								)tt								
							Having IsNull(Sum(BasicAmount),0)>=(Select ThresholdAmount 
													From masterdata.TDSTCSYearlySetup YS 
													Inner Join Masterdata.TDSTCSSetup TS on TS.TaxSetupNo=YS.TaxSetupNo 
													Where YS.ITYSetupNo=@TDSSetupNo													
													)
						)
