CREATE PROCEDURE [Sales].[SDSOFillValuesAccountAssignment]
(
	@ItemNos xml
	,@MiscChargeNos xml
	,@FreightTypeNos xml
	,@PaymentTermNos xml
	,@CustomerLocationNo int
	,@DocumentDate smalldatetime
	,@SalesOrderNo int
	,@SalesCategoryNo tinyint
	,@SalesOrderTypeNo tinyint
	,@CostHeadNos xml
	,@IsFreightRateExists bit = 0
)
AS

BEGIN
	SET NOCOUNT ON;
	
	BEGIN TRY
		declare @FreightTypeDetail table(FreightTypeNo int)
		insert into @FreightTypeDetail
		SELECT v.value('Freight_x0020_Type[1]','int') AS FreightTypeNo
		FROM @FreightTypeNos.nodes('/DocumentElement/Table') x(v)
		where v.value('Route_x0020_Id[1]','varchar(10)')<>''
		
		declare @ItemDetail table(ItemNo int,ProcessingTypeNo tinyint)
		insert into @ItemDetail
		SELECT v.value('Item_x0020_Code[1]','int') AS ItemNo
				,v.value('Processing_x0020_Type[1]','int') AS ProcessingTypeNo
		FROM @ItemNos.nodes('/DocumentElement/Table') x(v)
		where v.value('Item_x0020_Code[1]','int')>0
		
		/*
		declare @MiscChargeDetail table(MiscChargeNo smallint,TaxCodeMiscChargesDetailNo int,AccountType varchar(50))
		insert into @MiscChargeDetail
		SELECT v.value('Charge_x0020_Name[1]','int') AS MiscChargeNo
				,v.value('TDC_x0020_Account_x0020_Rule[1]','int') AS TaxCodeMiscChargesDetailNo
				,'Charge Account'
		FROM @MiscChargeNos.nodes('/DocumentElement/Table') x(v)
		where ISNUMERIC(v.value('Charge_x0020_Name[1]','varchar(50)'))=1
		--union all
		--select MiscChargeNo,null,'Freight Account'
		--from masterdata.mMiscCharges MC
		--inner join masterdata.mTaxGroup TG on TG.TaxGroupNo=MC.TaxGroupNo
		--where TG.TaxCategoryNo=10
		--	and exists (select 1 from @FreightTypeDetail where FreightTypeNo=1)
		*/
		
		declare @PaymentTermDetail table (PaymentTypeNo tinyint)
		insert into @PaymentTermDetail
		SELECT v.value('Payment_x0020_Type[1]','int') AS PaymentTypeNo
		FROM @PaymentTermNos.nodes('/DocumentElement/Table') x(v)
		where v.value('Payment_x0020_Type[1]','int')>0

		declare @CostHeadDetail table (CostHeadNo int)
		insert into @CostHeadDetail
		SELECT v.value('Cost_x0020_Head[1]','int') AS CostHeadNo
		FROM @CostHeadNos.nodes('/DocumentElement/Table') x(v)
		where v.value('Cost_x0020_Head[1]','int')>0
		
	--0
		/*
		select MC.AccountType [Account Type],MCM.MiscChargeName+'('+tc.Description+') - '+tt.ItemDescription [Particular]
				,isnull(AD.AccountNo,tt.AccountNo) [Account Name]
				,tt.AccountTypeNo,mcd.MiscChargeNo Id,'MC'+convert(varchar(8),tt.ItemNo) [Flag]
		from @MiscChargeDetail MC
			inner join masterdata.mMiscCharges MCM on MCM.MiscChargeNo=MC.MiscChargeNo
			inner join masterdata.mTaxCodeMiscChargesDetail mcd on mcd.MiscChargeNo=MC.MiscChargeNo
															and mcd.TaxCodeMiscChargesDetailNo=MC.TaxCodeMiscChargesDetailNo
			inner join masterdata.mTaxCode tc on tc.taxcodeno=mcd.taxcodeno
			outer apply (select Am.AccountNo,am.AccountTypeNo,tt.ItemDescription,tt.ItemNo
					    from Finance.ARDTDCAccountRule ar 
							inner join Finance.ARDMain ARDM on ARDM.ARDNo=ar.ARDNo
							cross apply (select MAX(ARDMIn.EffectiveFrom) EffectiveFrom
										 from Finance.ARDMain ARDMIn
										 where ARDMIn.ARDNo=ar.ARDNo
											and ARDMIn.EffectiveFrom<=@DocumentDate
										 having MAX(ARDMIn.EffectiveFrom)=ARDM.EffectiveFrom) MaxARD
							cross apply(select s.ItemNo,IM.ItemDescription,ir.AccountNo
										from @ItemDetail s 
										inner join Finance.ARDItemRule ir on ir.ItemNo=s.ItemNo
																		and ir.ARDNo=ar.ARDNo
										inner join masterdata.mItemMaster IM on IM.ItemNo=ir.ItemNo
										where tc.TaxAccountBasisNo=1
										union all
										select 0 ItemNo,'' ItemDescription,CAST(null as int) AccountNo
										where tc.TaxAccountBasisNo=2)tt
							left join masterdata.mAccountMaster am on ((am.AccountNo=tt.AccountNo and tc.TaxAccountBasisNo=1)
																		or(am.AccountNo=ar.AccountNo and tc.TaxAccountBasisNo=2))
						where mcd.TaxCodeMiscChargesDetailNo=ar.TaxCodeMiscChargesDetailNo)tt
			left join Sales.SalesOrderAccountAssignmentDetail AD on AD.Id=mcd.MiscChargeNo
															and AD.Flag='MC'+convert(varchar(8),tt.ItemNo)
															and AD.AccountTypeNo=tt.AccountTypeNo
															and AD.SalesOrderNo=@SalesOrderNo
		union all
		*/

		Declare @IsEditable bit
		Select @IsEditable = IsLockOrderAccountAssignment
		from Finance.Setting
		Where CompanyNo = (Select CompanyNo from Sales.SalesOrderMain where SalesOrderNo = @SalesOrderNo)
		

		select 'Stock Account' [Account Type],IM.ItemDescription+ISNULL(REPLACE(AD.Flag,AD.Flag,''),' *') [Particular]
				,isnull(AD.AccountNo,tt.AccountNo)[Account Name]
				,14 AccountTypeNo,ID.ItemNo Id,'IM' [Flag]
				,@IsEditable [Editable]
				,tt.ARDNo 
		from @ItemDetail ID 
			inner join masterdata.mItemMaster IM on IM.ItemNo=ID.ItemNo
			left join (select AM.AccountNo,IR.ItemNo,IR.ARDNo 
					   from Finance.ARDItemRule IR 
						inner join Finance.ARDMain ARDM on ARDM.ARDNo=IR.ARDNo
						cross apply (select MAX(ARDMIn.EffectiveFrom) EffectiveFrom
									 from Finance.ARDMain ARDMIn
									 where ARDMIn.ARDNo=IR.ARDNo
										and ARDMIn.EffectiveFrom<=@DocumentDate
									having MAX(ARDMIn.EffectiveFrom)=ARDM.EffectiveFrom) MaxARD
						inner join masterdata.mAccountMaster AM on AM.AccountNo=IR.AccountNo
															and AM.AccountTypeNo=14)tt on tt.ItemNo=IM.ItemNo
			left join Sales.SalesOrderAccountAssignmentDetail AD on AD.Id=id.ItemNo
															and AD.Flag='IM'
															and AD.AccountTypeNo=14
															and AD.SalesOrderNo=@SalesOrderNo
		where IM.ItemTypeNo not in (11,13,14,15) --As discussed with Shyam on 13 Oct
		union all
		select 'Sales Revenue A/c',IM.ItemDescription+ISNULL(REPLACE(AD.Flag,AD.Flag,''),' *')
				,isnull(AD.AccountNo,tt.AccountNo)
				,11,ID.ItemNo Id,'IM' [Flag]
				,0 [Editable]
				,CAST(NULL as int) [ARDNo]
		from @ItemDetail ID 
			inner join masterdata.mItemMaster IM on IM.ItemNo=ID.ItemNo
			outer apply (select sum(AM.AccountNo)AccountNo,sum(IR.ItemNo)ItemNo
					     from Finance.ARDItemRule IR 
						  inner join Finance.ARDMain ARDM on ARDM.ARDNo=IR.ARDNo
						  cross apply (select MAX(ARDMIn.EffectiveFrom) EffectiveFrom
									 from Finance.ARDMain ARDMIn
									 where ARDMIn.ARDNo=IR.ARDNo
										and ARDMIn.EffectiveFrom<=@DocumentDate
									having MAX(ARDMIn.EffectiveFrom)=ARDM.EffectiveFrom) MaxARD
						  inner join masterdata.mAccountMaster AM on AM.AccountNo=IR.AccountNo
															and AM.AccountTypeNo=11
						  inner join Finance.ARDSaleAccountRule SAR on SAR.ARDNo=IR.ARDNo
						where SAR.SaleAccPostingRuleNo in (select PRD.SaleAccPostingRuleNo
															from Finance.SaleAccPostingRuleDetails PRD
															where isnull(PRD.SalesCategoryNo,@SalesCategoryNo)=@SalesCategoryNo
																and isnull(PRD.SalesTypeNo,@SalesOrderTypeNo)=@SalesOrderTypeNo
																and isnull(PRD.SalesProcessingNo,ID.ProcessingTypeNo)=ID.ProcessingTypeNo)
							and IR.ItemNo=IM.ItemNo
						having sum(IR.ItemNo)=IM.ItemNo
						
						union --union all
						select sum(AM.AccountNo)AccountNo,sum(IR.ItemNo)ItemNo
					     from Finance.ARDItemRule IR 
						  inner join Finance.ARDMain ARDM on ARDM.ARDNo=IR.ARDNo
						  cross apply (select MAX(ARDMIn.EffectiveFrom) EffectiveFrom
									 from Finance.ARDMain ARDMIn
									 where ARDMIn.ARDNo=IR.ARDNo
										and ARDMIn.EffectiveFrom<=@DocumentDate
									having MAX(ARDMIn.EffectiveFrom)=ARDM.EffectiveFrom) MaxARD
						  inner join masterdata.mAccountMaster AM on AM.AccountNo=IR.AccountNo
															and AM.AccountTypeNo=11
						where ARDM.ARDNo not in (select ARDNo
													from Finance.ARDSaleAccountRule)
							and IR.ItemNo=IM.ItemNo
						having sum(IR.ItemNo)=IM.ItemNo)tt 			
			left join Sales.SalesOrderAccountAssignmentDetail AD on AD.Id=id.ItemNo
															and AD.Flag='IM'
															and AD.AccountTypeNo=11
															and AD.SalesOrderNo=@SalesOrderNo
		union all
		select top 1 'Customer Receivable Account',VLD.VendorName+ISNULL(REPLACE(AD.Flag,AD.Flag,''),' *')
				,isnull(AD.AccountNo,LR.AccountNo)
				,7,@CustomerLocationNo Id,'CRA' [Flag]
				,0 [Editable]
				,CAST(NULL as int) [ARDNo]
		from masterdata.mVendorLocationDetail VLD 
			left join Finance.ARDVendorLocationRule LR on LR.VendorLocationNo=VLD.VendorLocationNo
													and LR.AccountTypeNo=7
			left join Sales.SalesOrderAccountAssignmentDetail AD on AD.Id=@CustomerLocationNo
															and AD.Flag='CRA'
															and AD.AccountTypeNo=7
															and AD.SalesOrderNo=@SalesOrderNo
		where vld.VendorLocationNo=@CustomerLocationNo
		union all
		select top 1 'Customer Prepayable Account',VLD.VendorName+ISNULL(REPLACE(AD.Flag,AD.Flag,''),' *')
				,isnull(AD.AccountNo,LR.AccountNo)
				,8,@CustomerLocationNo Id,'CPA' [Flag]
				,0 [Editable]
				,CAST(NULL as int) [ARDNo]
		from masterdata.mVendorLocationDetail VLD 
			left join Finance.ARDVendorLocationRule LR on LR.VendorLocationNo=VLD.VendorLocationNo
													and LR.AccountTypeNo=8
			left join Sales.SalesOrderAccountAssignmentDetail AD on AD.Id=@CustomerLocationNo
															and AD.Flag='CPA'
															and AD.AccountTypeNo=8
															and AD.SalesOrderNo=@SalesOrderNo
		where vld.VendorLocationNo=@CustomerLocationNo
			and exists (select 1 from @PaymentTermDetail where PaymentTypeNo=2)
		Union all
		select 'Cost Head Expense' [Account Type],IM.HeadName +ISNULL(REPLACE(AD.Flag,AD.Flag,''),' *') [Particular]
				,isnull(AD.AccountNo,tt.AccountNo)[Account Name]
				,39 AccountTypeNo,ID.CostHeadNo Id,'CHE' [Flag]
				,1 [Editable]
				,tt.ARDNo 
		from @CostHeadDetail ID 
			inner join masterdata.mCostHeadEROSMain  IM on IM.CostHeadNo=ID.CostHeadNo
			left join (select AM.AccountNo,IR.CostHeadNo,IR.ARDNo 
					   from Finance.ARDIntermediatePostingAccountRule IR 
						inner join Finance.ARDMain ARDM on ARDM.ARDNo=IR.ARDNo
						cross apply (select MAX(ARDMIn.EffectiveFrom) EffectiveFrom
									 from Finance.ARDMain ARDMIn
									 where ARDMIn.ARDNo=IR.ARDNo
										and ARDMIn.EffectiveFrom<=@DocumentDate
									having MAX(ARDMIn.EffectiveFrom)=ARDM.EffectiveFrom) MaxARD
						inner join masterdata.mAccountMaster AM on AM.AccountNo=IR.AccountNo
						Where AM.AccountTypeNo = 39
						Group by AM.AccountNo,IR.CostHeadNo,IR.ARDNo
									)tt on tt.CostHeadNo=IM.CostHeadNo
			left join Sales.SalesOrderAccountAssignmentDetail AD on AD.Id=id.CostHeadNo
															and AD.Flag='CHE'
															and AD.AccountTypeNo=39
															and AD.SalesOrderNo=@SalesOrderNo
		Union all
		select 'Cost Head Provision' [Account Type],IM.HeadName +ISNULL(REPLACE(AD.Flag,AD.Flag,''),' *') [Particular]
				,isnull(AD.AccountNo,tt.AccountNo)[Account Name]
				,54 AccountTypeNo,ID.CostHeadNo Id,'CHP' [Flag]
				,1 [Editable]
				,tt.ARDNo 
		from @CostHeadDetail ID 
			inner join masterdata.mCostHeadEROSMain  IM on IM.CostHeadNo=ID.CostHeadNo
			left join (select AM.AccountNo,IR.CostHeadNo,IR.ARDNo 
					   from Finance.ARDIntermediatePostingAccountRule IR 
						inner join Finance.ARDMain ARDM on ARDM.ARDNo=IR.ARDNo
						cross apply (select MAX(ARDMIn.EffectiveFrom) EffectiveFrom
									 from Finance.ARDMain ARDMIn
									 where ARDMIn.ARDNo=IR.ARDNo
										and ARDMIn.EffectiveFrom<=@DocumentDate
									having MAX(ARDMIn.EffectiveFrom)=ARDM.EffectiveFrom) MaxARD
						inner join masterdata.mAccountMaster AM on AM.AccountNo=IR.AccountNo
						Where AM.AccountTypeNo = 54
									)tt on tt.CostHeadNo=IM.CostHeadNo
			left join Sales.SalesOrderAccountAssignmentDetail AD on AD.Id=id.CostHeadNo
															and AD.Flag='CHP'
															and AD.AccountTypeNo=54
															and AD.SalesOrderNo=@SalesOrderNo
		Union all
		select 'Freight Expense Account' [Account Type],IM.ItemDescription +ISNULL(REPLACE(AD.Flag,AD.Flag,''),' *') [Particular]
				,isnull(AD.AccountNo,tt.AccountNo)[Account Name]
				,28 AccountTypeNo,ID.ItemNo Id,'FRE' [Flag]
				,1 [Editable]
				,tt.ARDNo 
		from @ItemDetail ID 
		inner join masterdata.mItemMaster IM on IM.ItemNo = ID.ItemNo 
			Outer Apply (				
					Select top 1 AR.AccountNo,AR.ARDNo 
					From Finance.ARDTDCAccountRule AR
					inner join masterdata.mTaxCodeMiscChargesDetail MCD on MCD.TaxCodeMiscChargesDetailNo=AR.TaxCodeMiscChargesDetailNo
					inner join masterdata.mTaxCode TC on TC.TaxCodeNo=MCD.TaxCodeNo
					Where TC.TaxCategoryNo=10
					and TC.TaxAccountTypeNo=28
					and TC.EffectiveFrom <= @DocumentDate
					and TaxAccountBasisNo =1 -- add by akhil on 13 apr 15 (for item wise)
					and exists (select AccountNo from Finance.ARDItemRule I
									where AR.AccountNo=I.AccountNo
										and AR.ARDNo=I.ARDNo
										and I.ItemNo = ID.ItemNo 
								)
					Group by AR.AccountNo,AR.ARDNo 			
					) tt
			left join Sales.SalesOrderAccountAssignmentDetail AD on AD.Id=id.ItemNo 
															and AD.Flag='FRE'
															and AD.AccountTypeNo=28
															and AD.SalesOrderNo=@SalesOrderNo
			where IsNULL(@IsFreightRateExists,0) = 1 
		Union all
		select 'Freight Provision Account' [Account Type],IM.ItemDescription +ISNULL(REPLACE(AD.Flag,AD.Flag,''),' *') [Particular]
				,isnull(AD.AccountNo,tt.AccountNo)[Account Name]
				,54 AccountTypeNo,ID.ItemNo Id,'FRP' [Flag]
				,1 [Editable]
				,tt.ARDNo 
		from @ItemDetail ID 
		inner join masterdata.mItemMaster IM on IM.ItemNo = ID.ItemNo 
			Outer Apply (				
					Select Max(AR.AccountNo)AccountNo,AR.ARDNo 
					From Finance.ARDIntermediatePostingAccountRule AR
					Inner Join Finance.ARDItemRule AI on AI.ARDNo=AR.ARDNo And AI.ItemNo=ID.ItemNo					
					Inner Join masterdata.mAccountMaster AM on AM.AccountNo=AR.AccountNo
					Where AM.AccountTypeNo=54 And AR.IntermediateAccountTypeNo=4	
					And AI.ItemNo=ID.ItemNo
					Group by AR.AccountNo,AR.ARDNo 			
					) tt
			left join Sales.SalesOrderAccountAssignmentDetail AD on AD.Id=id.ItemNo 
															and AD.Flag='FRP'
															and AD.AccountTypeNo=54
															and AD.SalesOrderNo=@SalesOrderNo
			where IsNULL(@IsFreightRateExists,0) = 1 
		
	
	END TRY

			   ,SUSER_SNAME()
			   ,'PMPO')
	END CATCH
