CREATE PROCEDURE  [Quality].[QMIRFinancePosting] 
	@OtherDataInXml xml ,
	@CompanyNo tinyint  
	
as
BEGIN
begin try
declare @IsManagingExpenseAccount bit
 

create table #Posting ([Add to Basic] bit,Particular varchar(500),[Debit Amount] decimal(15,2) , 
		[Credit Amount] decimal(15,2),[Account Name] int, [Ref. Detail] varchar(50),[Cost Center] varchar(50),
		[Amount] decimal(15,2), [Type] tinyint,SlNo tinyint identity(1,1) ,MiscChargeNo int , TaxCodeMiscChargesDetailNo int,
			ItemNos varchar(200),GroupNo smallint, GroupTypeNo tinyint, NatureNo  tinyint, AccountTypeNo tinyint)
		
		declare @ItemTable table (ItemNo int,BasicAmount money)
		
		declare @InspectionOfferNo int
		SELECT @InspectionOfferNo=v.value('InspectionOfferNo[1]','int')
			FROM @OtherDataInXml.nodes('/DocumentElement/Table') x(v)
	if exists (select 1 from Finance.Setting where CompanyNo = @CompanyNo and ProcessProductionEntryPostingInApp =0)
	begin	
		insert into @ItemTable
		select bod.OutputItemNo ,sum(ls.BalanceQty * ls.Vrate)
				from Quality.InspectionOrderMain iom 
				inner join Quality.InspectionOrderRefDocDetail ord on ord.InspectionOrderNo = iom.InspectionOrderNo 
				inner join ProcessProduction.BatchOrderAmendmentOutputDetail bod on bod.BatchOrderAmendmentNo = ord.RefDocNo and bod.SLNo = ord.ActivitySLNo 
				inner join Production.IntermediateLotSummary ls on ls.BatchOrderAmendmentNo = bod.BatchOrderAmendmentNo and bod.ActivityNo = ls.ActivityNo 
							and ls.SequenceNo = bod.SLNo 
				inner join masterdata.mActivityMaster am on am.ActivityNo = ls.ActivityNo   and am.ActivityTypeNo <>2
				where iom.InspectionOrderNo = @InspectionOfferNo  and iom.SourceDocumentNo  = 10
				group by bod.OutputItemNo 
				union
				select bod.OutputItemNo ,sum(ls.BalanceQty *ls.Vrate)
				from 
				Quality.InspectionOrderMain iom 
				inner join Quality.InspectionOrderRefDocDetail ord on ord.InspectionOrderNo = iom.InspectionOrderNo 
				inner join ProcessProduction.BatchOrderAmendmentProcessDetail bpd on bpd.BatchOrderAmendmentNo = ord.RefDocNo and bpd.SLNo = ord.ActivitySLNo 
				inner join ProcessProduction.BatchOrderAmendmentInputDetail bid on bid.BatchOrderAmendmentNo = bpd.BatchOrderAmendmentNo and bid.ActivityNo = bpd.ActivityNo
								and bpd.SLNo = bid.SLNo 
				inner join ProcessProduction.BatchOrderAmendmentOutputDetail bod on bod.BatchOrderAmendmentNo = bid.BatchOrderAmendmentNo and bod.SLNo = bid.OutPutSlNo 
							and bod.OutputItemNo = bid.InputItemNo
				inner join Production.IntermediateLotSummary ls on ls.BatchOrderAmendmentNo = bod.BatchOrderAmendmentNo and bod.ActivityNo = ls.ActivityNo 
							and ls.SequenceNo = bod.SLNo 
				inner join masterdata.mActivityMaster am on am.ActivityNo = bid.ActivityNo and am.ActivityTypeNo =2
				where iom.InspectionOrderNo  = @InspectionOfferNo  and iom.SourceDocumentNo  = 10
				group by bod.OutputItemNo
			
		 Insert into #Posting ([Add to Basic] ,Particular,[Debit Amount]  , [Credit Amount],[Account Name] , 
						[Amount], [Type])		
			select cast(0 as bit) [Add to Basic] ,'Batch Production A/C' Particular 
				,CAST(null as money) [Debit Amount] ,sum(I.BasicAmount)  [Credit Amount] 
				,IR.AccountNo [Account Name] 
				,sum(I.BasicAmount) Amount ,2 [Type] 
			from @ItemTable I
				inner join masterdata.mItemMaster IM on IM.ItemNo=I.ItemNo
				outer apply (select Max(IR.AccountNo) AccountNo
							from masterdata.mAccountMaster  IR
							where IR.AccountTypeNo=70) IR 
			group by  IR.AccountNo
			Union All
			select cast(0 as bit) [Add to Basic] ,'Stock A/C' Particular 
				,sum(I.BasicAmount) [Debit Amount] , CAST(null as money) [Credit Amount] 
				,IR.AccountNo [Account Name] 
				,sum(I.BasicAmount) Amount ,1 [Type] 
			from @ItemTable I
				left join (select ItemNo,IR.AccountNo--,AM.AccountName
							from Finance.ARDItemRule IR
							--inner join masterdata.mAccountMaster AM on AM.AccountNo=IR.AccountNo
							where IR.AccountTypeNo=14) IR on IR.ItemNo=I.ItemNo
			group by IR.AccountNo		 
	  
	end
	 

 if exists(select 1 from #Posting having abs(SUM([Credit Amount])- SUM([Debit Amount]))<=.5)
 begin
	update PD set [Debit Amount] = [Debit Amount] + ISNULL(diff,0)
		from
		#Posting PD
		cross join (select SUM([Credit Amount])- SUM([Debit Amount]) Diff from #Posting) tt
		where PD.SlNo  = 1
	
	update PD set [Credit Amount] = [Credit Amount] + ISNULL(diff,0)
		from
		#Posting PD
		cross join (select SUM([Debit Amount])- SUM([Credit Amount]) Diff from #Posting) tt
		where PD.SlNo  = 1
			
 end
 
 
select * from #Posting
 where isnull([Debit Amount],0) >0 or isnull([Credit Amount],0) >0
   
  
  drop table #Posting 
   
END TRY
	
	BEGIN CATCH
		select ERROR_MESSAGE()
		INSERT INTO [masterdata].[mErrorLog]
			   ([ErrorDate]
			   ,[ErrorNo]
			   ,[ErrorSeverity]
			   ,[ErrorState]
			   ,[ErrorProcedure]
			   ,[ErrorLine]
			   ,[ErrorDescription]
			   ,[UserName]
			   ,[FormCode])
		VALUES
			   (getdate()
			   ,ERROR_NUMBER()
			   ,ERROR_SEVERITY()
			   ,ERROR_STATE()
			   ,ERROR_PROCEDURE()
			   ,ERROR_LINE()
			   ,ERROR_MESSAGE()
			   ,SUSER_SNAME()
			   ,'IMMI')
	END CATCH
END
