	@POAmendmentNo int,

BEGIN

	Select @CurrentDB =DB_NAME()

	If @PortalDBName is null
	BEGIN
		Raiserror('Portal Database not defined found.',16,-1)
		return

		Raiserror('Portal Database not defined found.',16,-1)

						Group by MT.MaterialNatureNo,MT.MaterialNature

		-- Validate

				If Exists (
							Select 1

										Inner Join ' + @PortalDBName + '.masterdata.mDBMaster DB on DB.DBNo=CM.DBNo
