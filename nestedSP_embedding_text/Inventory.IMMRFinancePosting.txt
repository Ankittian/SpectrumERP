CREATE PROCEDURE [Inventory].[IMMRFinancePosting] 
	@OtherDataInXml xml ,
	@CompanyNo tinyint ,
	@ItemNos varchar(max),
	@FormCode Varchar(4)=Null-- Added by Alok Ticket No 011021011
as

BEGIN
begin try
declare @IsManagingExpenseAccount bit
 

create table #Posting ([Add to Basic] bit,Particular varchar(500),[Debit Amount] decimal(15,2) , 
		[Credit Amount] decimal(15,2),[Account Name] int, [Ref. Detail] varchar(50),[Cost Center] varchar(50),
		[Amount] decimal(15,2), [Type] tinyint,SlNo tinyint identity(1,1) ,MiscChargeNo int , TaxCodeMiscChargesDetailNo int,
			ItemNos varchar(200),GroupNo smallint, GroupTypeNo tinyint, NatureNo  tinyint, AccountTypeNo tinyint)

		declare @ItemTable table (ItemNo int,BasicAmount money,Id int,Qty decimal(12,3),Rate money)
		insert into @ItemTable
		select Col1, sum(convert(money,Col2)), Col3, sum(CONVERT(decimal(12,3),Col4)), sum(CONVERT(money,Col5))
		from masterdata.Split_RowColumn(@ItemNos,'`','~')
		group by Col1, Col3
 

declare @IMMR_SourceDocumentNo tinyint
					,@IMMR_IssueNos varchar(1000)
					,@IMMR_IsScrapReturn bit
			
			SELECT @IMMR_SourceDocumentNo=v.value('SourceDocumentNo[1]','tinyint')
					,@IMMR_IssueNos=v.value('IssueNo[1]','varchar(1000)')
					,@IMMR_IsScrapReturn=v.value('IsScrapReturn[1]','bit')
					
			FROM @OtherDataInXml.nodes('/DocumentElement/Table') x(v)
			
			declare @IssueDetail as table (IssueNo int,Amount decimal(18,2),IssueTypeNo tinyint,IsNonChargeable bit)
			insert into @IssueDetail (IssueNo ,Amount ,IssueTypeNo ,IsNonChargeable)
			Select IssueNo , Col2,IndentTypeNo ,IsNonChargeable 
			From Inventory.MaterialIssueMain im 
			inner join masterdata.Split_RowColumn(@IMMR_IssueNos,'`','~') INs on INs.Col1=IM.IssueNo
			Where @FormCode is Null or @FormCode='IMMR'
			Union All
			Select IssueNo , Col2,IndentTypeNo ,IsNonChargeable 
			From DiscreteRM.MaterialIssueMain im
			inner join masterdata.Split_RowColumn(@IMMR_IssueNos,'`','~') INs on INs.Col1=IM.IssueNo
			Where @FormCode='DRMR'			

				Declare @IsLotCapital bit = 0
		
				If exists(Select 1 from @IssueDetail Id 
						  --inner join Inventory.InvLotSummary ILS on ILS.TransactionFlag like 'IMMI' and ILS.DocumentNo = Id.IssueNo 
						  --and ILS.IsCapital = 1
						  inner join Inventory.InvLotDetail ILD on ILD.TransactionFlag like 'IMMI' and ILD.DocumentNo = Id.IssueNo
						  inner join Inventory.InvLotSummary ILS on ILS.InvLotSummaryNo = ILD.InvLotSummaryNo 
						  and ILS.IsCapital = 1
						  )
				Begin 
				   Select @IsLotCapital = 1
				End 
				Else 
				Begin 
				   Select @IsLotCapital = 0
				End			
			
			insert into #Posting ([Add to Basic] ,Particular ,[Debit Amount] ,
								[Credit Amount] ,[Account Name] ,Amount ,[Type] ,
								MiscChargeNo ,TaxCodeMiscChargesDetailNo ,ItemNos)
			select cast(0 as bit) [Add to Basic] ,'Stock A/C' Particular 
				,sum(I.BasicAmount) [Debit Amount] ,CAST(null as money) [Credit Amount] 
				,IR.AccountNo [Account Name] 
				,sum(I.BasicAmount) Amount ,1 [Type] 
				,CAST(null as smallint) MiscChargeNo,null TaxCodeMiscChargesDetailNo,CAST(null as varchar(5000))ItemNos
			from @ItemTable I
				inner join masterdata.mItemMaster IM on IM.ItemNo=I.ItemNo
				left join (select ItemNo,IR.AccountNo--,AM.AccountName
							from Finance.ARDItemRule IR
							--inner join masterdata.mAccountMaster AM on AM.AccountNo=IR.AccountNo
							where IR.AccountTypeNo=14) IR on IR.ItemNo=I.ItemNo
			where isnull(@IMMR_IsScrapReturn,0)=0
			and not exists (
							Select 1 
							From @IssueDetail 
							Where IssueTypeNo =1 
							And IsNonChargeable =0
							And @IsLotCapital=1
							)
			group by IR.AccountNo,isnull(IR.AccountNo,I.ItemNo)
			
			 
			union all
			select cast(0 as bit) [Add to Basic] ,'Stock A/C' Particular 
				,sum(I.BasicAmount) [Debit Amount] ,CAST(null as money) [Credit Amount] 
				,IR.AccountNo [Account Name] 
				,sum(I.BasicAmount) Amount ,1 [Type] 
				,CAST(null as smallint) MiscChargeNo,null TaxCodeMiscChargesDetailNo,CAST(null as varchar(5000))ItemNos
			from @ItemTable I
				inner join Inventory.ScrapItemSetting sis on sis.ItemNo=I.ItemNo
				inner join masterdata.mItemMaster IM on IM.ItemNo=sis.ScrapItemNo
				left join (select ItemNo,IR.AccountNo--,AM.AccountName
							from Finance.ARDItemRule IR
							--inner join masterdata.mAccountMaster AM on AM.AccountNo=IR.AccountNo
							where IR.AccountTypeNo=14) IR on IR.ItemNo=sis.ScrapItemNo
			where @IMMR_IsScrapReturn=1
			group by  IR.AccountNo,IR.AccountNo 
			union all
			select cast(0 as bit) [Add to Basic] ,IM.ItemDescription+' Change In Inventory A/C' Particular 
				,CAST(null as money) [Debit Amount] ,sum(I.BasicAmount) [Credit Amount] 
				,IR.AccountNo [Account Name] 
				,sum(I.BasicAmount) Amount ,2 [Type] 
				,CAST(null as smallint) MiscChargeNo,null TaxCodeMiscChargesDetailNo,CAST(null as varchar(5000))ItemNos
			from @ItemTable I
				inner join masterdata.mItemMaster IM on IM.ItemNo=I.ItemNo
				left join (select ItemNo,IR.AccountNo--,AM.AccountName
							from Finance.ARDItemRule IR
							--inner join masterdata.mAccountMaster AM on AM.AccountNo=IR.AccountNo
							where IR.AccountTypeNo=55) IR on IR.ItemNo=I.ItemNo
			where @IMMR_SourceDocumentNo =1
			group by IM.ItemDescription+' Change In Inventory A/C',IR.AccountNo,isnull(IR.AccountNo,I.ItemNo)
			
			Union All

			select cast(0 as bit) [Add to Basic] , 'Consumption A/C' Particular 
				,CAST(null as money) [Debit Amount] 
				,sum(coalesce(VIR.BasicAmount,IR.BasicAmount,I.BasicAmount)) [Credit Amount] 
				,isnull(VIR.AccountNo,IR.AccountNo) [Account Name] 
				,sum(coalesce(VIR.BasicAmount,IR.BasicAmount,I.BasicAmount)) Amount ,2 [Type] 
				,CAST(null as smallint) MiscChargeNo,null TaxCodeMiscChargesDetailNo,CAST(null as varchar(5000))ItemNos
			from @ItemTable I
				inner join masterdata.mItemMaster IM on IM.ItemNo=I.ItemNo
				left join (select AAD.Id ItemNo,AAD.AccountNo--,AM.AccountName
											,sum(convert(money,INs.Amount)) BasicAmount
							from Purchase.SCOrderAmendmentMain sm
							inner join Inventory.MaterialIssueMain IM on IM.RefDocumentNo = sm.SCOrderAmendmentNo 
							inner join  @IssueDetail INs on INs.IssueNo =IM.IssueNo
							inner join Purchase.SCOrderAccountAssignmentDetail AAD on AAD.SCOrderNo=sm.SCOrderNo
							--inner join masterdata.mAccountMaster AM on AM.AccountNo=AAD.AccountNo
							where IM.IssueSourceNo =3
								and AAD.Flag='IM'
								and AAD.AccountTypeNo=14
								And ((@FormCode is Null or @FormCode='IMMR'))
							group by AAD.Id ,AAD.AccountNo--,AM.AccountName
								) IR on IR.ItemNo=I.ItemNo
				outer apply (select ItemNo,IR.AccountNo--,AM.AccountName
											,I.BasicAmount
								from Finance.ARDItemRule IR
								--inner join masterdata.mAccountMaster AM on AM.AccountNo=IR.AccountNo
								where IR.AccountTypeNo=15
									and IR.ItemNo=I.ItemNo
							 ) VIR 
						
			where @IMMR_SourceDocumentNo=2
			and exists (select 1 from @IssueDetail where IssueTypeNo <>1  and IsNonChargeable =1)
			group by  IR.AccountNo,VIR.AccountNo,I.ItemNo
			
			Insert into #Posting ([Add to Basic] ,Particular,[Debit Amount]  , [Credit Amount],[Account Name] , 
						[Amount], [Type] , AccountTypeNo )	
			select cast(0 as bit) [Add to Basic] ,'Supplier Payable A/c' Particular 
				,CAST(null as money) [Debit Amount] ,sum(I.BasicAmount) [Credit Amount] 
				,IR.AccountNo [Account Name] 
				,sum(I.BasicAmount) Amount ,2 [Type] ,4
			from @ItemTable I
				,(Select max(AccountNo) AccountNo 
				  From (				  
						  Select AccountNo 
						  From @IssueDetail id ,Inventory.MaterialIssueMain im,Finance.accVoucherDetail vd
						  Where vd.VoucherNo = im.VoucherNo and id.IssueNo = im.IssueNo and BaseDrAmount >0
						  And (@FormCode is Null or @FormCode='IMMR')
						  Union All
						  Select AccountNo 
						  From @IssueDetail id ,DiscreteRM.MaterialIssueMain im,Finance.accVoucherDetail vd
						  Where vd.VoucherNo = im.VoucherNo and id.IssueNo = im.IssueNo and BaseDrAmount >0
						  And (@FormCode='DRMR')
				      )tt
				 ) IR
			where isnull(@IMMR_IsScrapReturn,0)=0
			and exists (select 1 from @IssueDetail where IssueTypeNo <>1  and IsNonChargeable =0)
			group by IR.AccountNo
			
			Insert into #Posting ([Add to Basic] ,Particular,[Debit Amount]  , [Credit Amount],[Account Name] , 
						[Amount], [Type] , AccountTypeNo )	
			select cast(0 as bit) [Add to Basic] ,'Supplier Payable A/c' Particular 
				,CAST(null as money) [Debit Amount] ,sum(I.BasicAmount) [Credit Amount] 
				,IR.AccountNo [Account Name] 
				,sum(I.BasicAmount) Amount ,2 [Type] ,4
			from @ItemTable I
				,(Select max(AccountNo) AccountNo 
				  From (				  
						  Select AccountNo 
						  From @IssueDetail id ,Inventory.MaterialIssueMain im,Finance.accVoucherDetail vd
						  Where vd.VoucherNo = im.VoucherNo and id.IssueNo = im.IssueNo and BaseDrAmount >0
						  And (@FormCode is Null or @FormCode='IMMR')
						  Union All
						  Select AccountNo 
						  From @IssueDetail id ,DiscreteRM.MaterialIssueMain im,Finance.accVoucherDetail vd
						  Where vd.VoucherNo = im.VoucherNo and id.IssueNo = im.IssueNo and BaseDrAmount >0
						  And (@FormCode='DRMR')
				      )tt
				 ) IR
			where isnull(@IMMR_IsScrapReturn,0)=0
			and exists (select 1 from @IssueDetail where IssueTypeNo =1 and IsNonChargeable =0)
			group by IR.AccountNo
			Union All
			select cast(0 as bit) [Add to Basic] ,'Asset Clearing A/c' Particular 
				,sum(I.BasicAmount) [Debit Amount] 
				,CAST(null as money) [Credit Amount] 
				,IR.AccountNo [Account Name] 
				,sum(I.BasicAmount) Amount ,1 [Type] ,54
			from @ItemTable I
			Cross Apply(Select max(AccountNo) AccountNo 
				  From (				  
						  Select AccountNo 
						  From @IssueDetail id ,Inventory.MaterialIssueMain im,Finance.accVoucherDetail vd
						  Where vd.VoucherNo = im.VoucherNo and id.IssueNo = im.IssueNo and BaseCrAmount >0
						  And (/*@FormCode is Null and */@FormCode='IMMR')
						  Union All
						  Select AccountNo 
						  From @IssueDetail id ,DiscreteRM.MaterialIssueMain im,Finance.accVoucherDetail vd
						  Where vd.VoucherNo = im.VoucherNo and id.IssueNo = im.IssueNo and BaseCrAmount >0
						  And (@FormCode='DRMR')
				      )tt
				 ) IR
			inner join masterdata.mItemMaster IM on IM.ItemNo = I.ItemNo and IM.ItemTypeNo = 1
			where @IMMR_SourceDocumentNo=2
			and exists (select 1 from @IssueDetail where IssueTypeNo =1 and IsNonChargeable =0)
			group by  IR.AccountNo

			if exists(Select 1 from @IssueDetail where IssueTypeNo = 1)
			and exists(Select 1 from @ItemTable IT
						inner join masterdata.mItemMaster IM on IM.ItemNo = IT.itemNo and IM.ItemTypeNo <> 1) 
			Begin 


									insert into #Posting ([Add to Basic] ,Particular ,[Debit Amount] ,
													[Credit Amount] ,[Account Name] ,Amount ,[Type] ,
													MiscChargeNo ,TaxCodeMiscChargesDetailNo ,ItemNos)
									select cast(0 as bit) [Add to Basic] ,'WORK IN PROGRESS' Particular 
										,Case when IsNULL(FT.IsNonChargeable,0) = 0 then sum(I.BasicAmount) Else CAST(null as money) End [Debit Amount] 
										,Case when IsNULL(FT.IsNonChargeable,0) = 0 then CAST(null as money)Else sum(I.BasicAmount) End [Credit Amount] 
										,FT.AccountNo [Account Name] 
										,sum(I.BasicAmount) Amount 
										,Case when IsNULL(FT.IsNonChargeable,0) = 0 then 1 else 2 End [Type] 
										,CAST(null as smallint) MiscChargeNo,null TaxCodeMiscChargesDetailNo,CAST(null as varchar(5000))ItemNos
									from @ItemTable I
										inner join masterdata.mItemMaster IM on IM.ItemNo=I.ItemNo
										Cross Apply (Select AM.AccountNo,ID.IsNonChargeable from @IssueDetail ID 
													inner join Inventory.MaterialIssueMain MIM on MIM.IssueNo = ID.IssueNo 
													inner join Inventory.MaterialIssueItemDetail MIID on MIID.IssueNo = MIM.IssueNo and MIID.ItemNo = I.ItemNo 
													inner join Finance.accVoucherMain avm on avm. VoucherNo = MIM.VoucherNo 
													inner join Finance.accVoucherDetail avd on avd.VoucherNo = avm.VoucherNo 
													inner join masterdata.mAccountMaster AM on AM.AccountNo = avd.AccountNo and AM.AccountTypeNo = 16
													Group by AM.AccountNo,Id.IsNonChargeable 
													) FT
									Where @FormCode = 'IMMR' and @IMMR_SourceDocumentNo =2 and IsNULL(@IMMR_IsScrapReturn,0) <> 1 
									group by FT.AccountNo,FT.IsNonChargeable  
			End 


			 

 if exists(select 1 from #Posting having abs(SUM([Credit Amount])- SUM([Debit Amount]))<=.5)
 begin
	update PD set [Debit Amount] = [Debit Amount] + ISNULL(diff,0)
		from
		#Posting PD
		cross join (select SUM([Credit Amount])- SUM([Debit Amount]) Diff from #Posting) tt
		where PD.SlNo  = 1
	
	update PD set [Credit Amount] = [Credit Amount] + ISNULL(diff,0)
		from
		#Posting PD
		cross join (select SUM([Debit Amount])- SUM([Credit Amount]) Diff from #Posting) tt
		where PD.SlNo  = 1
			
 end
 
 
select * from #Posting
 where isnull([Debit Amount],0) >0 or isnull([Credit Amount],0) >0
   
  
  drop table #Posting 
   
END TRY
	
	BEGIN CATCH
		select ERROR_MESSAGE()
		INSERT INTO [masterdata].[mErrorLog]
			   ([ErrorDate]
			   ,[ErrorNo]
			   ,[ErrorSeverity]
			   ,[ErrorState]
			   ,[ErrorProcedure]
			   ,[ErrorLine]
			   ,[ErrorDescription]
			   ,[UserName]
			   ,[FormCode])
		VALUES
			   (getdate()
			   ,ERROR_NUMBER()
			   ,ERROR_SEVERITY()
			   ,ERROR_STATE()
			   ,ERROR_PROCEDURE()
			   ,ERROR_LINE()
			   ,ERROR_MESSAGE()
			   ,SUSER_SNAME()
			   ,'IMMI')
	END CATCH
END
