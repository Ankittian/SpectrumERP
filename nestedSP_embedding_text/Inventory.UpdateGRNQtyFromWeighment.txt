Stored procedure Inventory.UpdateGRNQtyFromWeighment: Update GRN Qty by from weighment. Called from Inventory.IMIEAuthorize.
Writes: Inventory.GRNItemDetail, Inventory.GRNItemIndentDetail, Inventory.GRNMain, Inventory.InspectionItemDetail
Reads: GateEntry.InvGateEntryItemDetail, GateEntry.InvGateEntryMain, Inventory.GRNIndentReserveDetail, Inventory.GRNItemDetail, Inventory.GRNItemIndentDetail, Inventory.GRNItemLotAdditionalDetail, Inventory.GRNMain, Inventory.IndentItemDetail, Inventory.InspectionItemDetail, Inventory.InspectionMain, masterdata.mItemMaster, masterdata.mItemOtherInformation, masterdata.mUnitConversionDetail, Purchase.POAmendmentMain, Purchase.POIndentDetail, WeighBridge.mWeighmentMachine, WeighBridge.WeighmentSlip
Parameters: @WeighmentNo int, @InspectionNo int, @OutputStatus bit OUTPUT, @Result varchar(4000) OUTPUT
Author: K M Disha, created 05 Oct 2026
Code:
-- =============================================
-- Author:		K M Disha
-- Create date: 05 Oct 2026
-- Modified:    12 Oct 2026 - DEF-06: cap indent Rejected Qty in Inspection Item Detail to allocated Qty
-- Description: Update GRN Qty by from weighment
-- =============================================
CREATE PROCEDURE [Inventory].[UpdateGRNQtyFromWeighment]
(
	 @WeighmentNo  int
	,@InspectionNo int
	,@OutputStatus bit output
	,@Result varchar(4000) output 
)
AS

BEGIN
	SET NOCOUNT ON;
	
	BEGIN TRY

	Declare @GRNNo int
	,@Netweight  decimal(12,3)
	,@MachineUnitNo smallint
	,@GRNUnitNo smallint
		
		If ISNULL(@WeighmentNo,0) <> 0
		Begin
			Select @GRNNo = GRID.GRNNo
			from [WeighBridge].[WeighmentSlip] WS
			Inner Join GateEntry.InvGateEntryMain GEM On WS.RefDocumentNo = GEM.GateEntryNo
			Inner Join Inventory.GRNItemdetail GRID On GRID.GateEntryNo = GEM.GateEntryNo
			Where WeighmentNo = ISNULL(@WeighmentNo,0)
			And IsFinal = 1 -- Final Weighment
			And SourceDocumentNo = 2 ---GE
		End
		Else If ISNULL(@InspectionNo,0) <> 0
		Begin
			Select @GRNNo = GRNI.GRNNo
			from Inventory.GRNItemdetail GRNI
			Inner Join Inventory.InspectionItemDetail IID On IID.GRNItemDetailNo = GRNI.GRNItemDetailNo
			Inner Join Inventory.InspectionMain IM On IM.InspectionNo = IID.InspectionNo
			Where IID.GRNItemLotDetailNo Is Null
			And IID.InspectionNo = ISNULL(@InspectionNo,0)
			And IM.AuthorizedBy is not null
		End

		

		If Exists(
					Select 1
					from Inventory.GRNItemdetail GRNI
					Inner Join Inventory.GRNMain GRN On GRN.GRNNo = GRNI.GRNNo
					Inner Join masterdata.mItemMaster IM On IM.ItemNo = GRNI.ItemNo
					Inner Join masterdata.mItemOtherInformation IOI On IOI.ItemNo = IM.ItemNo
					Left Join Inventory.GRNItemLotAdditionalDetail GRLD On GRLD.GRNItemDetailNo = GRNI.GRNItemDetailNo
					Left Join Inventory.GRNIndentReserveDetail GRD On GRD.GRNItemDetailNo = GRNI.GRNItemDetailNo
					Where GRNI.GRNNo = @GRNNo
					And IOI.IsMaintainSerialNo = 0
					And IM.ItemTypeNo = 2 --Raw Material
					And GRLD.GRNItemLotDetailNo Is null
					And GRD.GRNItemDetailNo Is Null
					And GRN.AuthorizedBy Is Null -- GRN is authorized
					And GRNI.RouteIdentity Is Null -- Not a case of Route
				) And
		Exists(
				Select 1
				from Inventory.GRNItemdetail GRNI
				Inner Join masterdata.mItemMaster IM On IM.ItemNo = GRNI.ItemNo
				Where GRNI.GRNNo = @GRNNo
				Group by GRNI.GRNNo
				Having Count(GRNI.GRNItemDetailNo) = 1
			) 
			And
		Exists(
				Select 1
				from Inventory.GRNItemdetail GRNI
				Inner Join masterdata.mItemMaster IM On IM.ItemNo = GRNI.ItemNo
				Where GRNI.GRNNo = @GRNNo
				Group by GRNI.GateEntryNo
				Having Count(GRNI.GRNItemDetailNo) = 1
			)And
		Exists(
				Select 1
				from [WeighBridge].[WeighmentSlip] WS
				Inner Join GateEntry.InvGateEntryMain GEM On WS.RefDocumentNo = GEM.GateEntryNo
				Inner Join Inventory.GRNItemdetail GRID On GRID.GateEntryNo = GEM.GateEntryNo
				Where GRID.GRNNo = @GRNNo
				And IsFinal = 1 -- Final Weighment
				And SourceDocumentNo = 2 ---GE
			)And
		Exists(
					Select 1
					from Inventory.GRNItemdetail GRNI
					Inner Join Inventory.InspectionItemDetail IID On IID.GRNItemDetailNo = GRNI.GRNItemDetailNo
					Inner Join Inventory.InspectionMain IM On IM.InspectionNo = IID.InspectionNo
					Where IID.GRNItemLotDetailNo Is Null
					And IID.GRNNo = @GRNNo
					And IM.AuthorizedBy is not null
				)
		Begin

			Select GRNItemDetailNo,GRNIndentDetailNo,RejectedQty
			into #tblGRNIndentDetail
			from Inventory.InspectionItemDetail IID 
			Where IID.GRNNo = @GRNNo

			Select @MachineUnitNo = WM.UnitNo
			,@Netweight = NetWeight
			From WeighBridge.WeighmentSlip WS
			Inner Join WeighBridge.mWeighmentMachine WM On WM.WeighmentMachineNo = WS.WeighmentMachineNo
			Inner Join GateEntry.InvGateEntryMain GEM On WS.RefDocumentNo = GEM.GateEntryNo
													And WS.SourceDocumentNo = 2
			Inner Join Inventory.GRNItemdetail GRID On GRID.GateEntryNo = GEM.GateEntryNo
			Where GRID.GRNNo = @GRNNo
			And IsFinal = 1


			Select @GRNUnitNo = UnitNo
			From Inventory.GRNItemDetail
			Where GRNNo = @GRNNo

			If @MachineUnitNo <> @GRNUnitNo
			Begin
				If Exists (
							Select 1 
							From masterdata.mUnitConversionDetail
							Where UnitNo = @MachineUnitNo 
							And ConversionUnitNo = @GRNUnitNo 
							And FirstCF > 0 And SecondCF > 0
						)
				Begin
					Select @Netweight = (@Netweight * SecondCF) / FirstCF
					From masterdata.mUnitConversionDetail
					Where UnitNo = @MachineUnitNo And ConversionUnitNo = @GRNUnitNo
				End
				Else If Exists (
								Select 1 
								From masterdata.mUnitConversionDetail
								Where UnitNo = @GRNUnitNo 
								And ConversionUnitNo = @MachineUnitNo 
								And FirstCF > 0 And SecondCF > 0
							)
				Begin
					Select @Netweight = (@Netweight * FirstCF) / SecondCF
					From masterdata.mUnitConversionDetail
					Where UnitNo = @GRNUnitNo And ConversionUnitNo = @MachineUnitNo
				End
			Else
				Raiserror('Unit conversion is not defined between weighment machine unit and GRN unit.',16,1)
			End

			--If @TotalRejectedQty > @Netweight
			--	Raiserror('Rejected quantity in Inspection cannot be more than the weighed quantity.',16,1)

			--Update Received Qty of GRNItemDetail from weighment
			Update Inventory.GRNItemDetail
			Set ReceivedQty = @Netweight
			,AcceptedQty = @Netweight
			,StockQty = @Netweight  * FirstCF /SecondCF
			Where GRNNo = @GRNNo

			--Update Accepted Qty of GRNIndentDetail from weighment
			SELECT GRII.GRNIndentDetailNo,
			ISNULL(POID.POQty, IID.IndentQty) AS IndentQty,
			SUM(ISNULL(POID.POQty, IID.IndentQty)) OVER
			(
				ORDER BY GRII.GRNIndentDetailNo
				ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
			) AS CumulativeIndentQty,
			SUM(ISNULL(POID.POQty, IID.IndentQty)) OVER () AS TotalIndentQty
			INTO #GRNIndentQty
			FROM Inventory.GRNItemIndentDetail GRII 
			INNER JOIN Inventory.GRNItemDetail GID ON GID.GRNItemDetailNo = GRII.GRNItemDetailNo
			INNER JOIN GateEntry.InvGateEntryItemDetail GEID ON GEID.GateEntryNo = GID.GateEntryNo
											AND GEID.ItemNo = GID.ItemNo
											AND ISNULL(GEID.MakeNo, 0) = ISNULL(GID.MakeNo, 0)
											AND ISNULL(GEID.TechnicalGradeNo, 0) = ISNULL(GID.TechnicalGradeNo, 0)
											AND ISNULL(GEID.POAmendmentNo, 0) = ISNULL(GID.POAmendmentNo, 0)
			INNER JOIN Inventory.IndentItemDetail IID ON IID.IndentNo = GRII.IndentNo
										AND IID.ItemNo = GEID.ItemNo
										AND ISNULL(IID.MakeNo, 0) = ISNULL(GRII.IndentMakeNo, 0)
										AND ISNULL(IID.TechnicalGradeNo, 0) = ISNULL(GEID.TechnicalGradeNo, 0)
			LEFT JOIN Purchase.POAmendmentMain PAM ON PAM.POAmendmentNo = GEID.POAmendmentNo
			LEFT JOIN Purchase.POIndentDetail POID ON POID.PONo = PAM.PONo
										AND POID.ItemNo = GEID.ItemNo
										AND ISNULL(POID.MakeNo, 0) = ISNULL(GEID.MakeNo, 0)
										AND ISNULL(POID.IndentMakeNo, 0) = ISNULL(IID.MakeNo, 0)
										AND ISNULL(POID.IndentTechnicalGradeNo, 0) = ISNULL(IID.TechnicalGradeNo, 0)
										AND ISNULL(POID.IndentNo, 0) = ISNULL(IID.IndentNo, 0)
			WHERE GRII.GRNNo = @GRNNo;

			UPDATE GRII
			SET AcceptedQty =
			CASE
			-- Net Weight exceeds total Indent Qty:
			-- add the excess to the first GRNIndentDetailNo
			WHEN GRII.GRNIndentDetailNo =
				(SELECT MIN(GRNIndentDetailNo) FROM #GRNIndentQty)
				AND @NetWeight > Q.TotalIndentQty
			THEN Q.IndentQty + (@NetWeight - Q.TotalIndentQty)

			-- Net Weight falls within this row
			WHEN @NetWeight > Q.CumulativeIndentQty - Q.IndentQty
			THEN
			CASE
				WHEN @NetWeight - (Q.CumulativeIndentQty - Q.IndentQty) > Q.IndentQty
				THEN Q.IndentQty
				WHEN @NetWeight - (Q.CumulativeIndentQty - Q.IndentQty) < 0
				THEN 0
				ELSE @NetWeight - (Q.CumulativeIndentQty - Q.IndentQty)
			END

			ELSE 0 END
			FROM Inventory.GRNItemIndentDetail GRII
			INNER JOIN #GRNIndentQty Q ON Q.GRNIndentDetailNo = GRII.GRNIndentDetailNo
			WHERE GRII.GRNNo = @GRNNo;

			DROP TABLE #GRNIndentQty;

			--Rejected Qty of an indent can not be more than the Qty allocated to that indent:
			--set Rejected Qty in Inspection Item Detail equal to the allocated (accepted) Qty of the indent
			Update IID
			Set RejectedQty = ISNULL(GRII.AcceptedQty,GRID.AcceptedQty)
			From Inventory.InspectionItemDetail IID
			Inner Join Inventory.GRNItemDetail GRID On GRID.GRNItemDetailNo = IID.GRNItemDetailNo
			Left Join Inventory.GRNItemIndentDetail GRII On GRII.GRNIndentDetailNo = IID.GRNIndentDetailNo
														And GRII.GRNItemDetailNo = IID.GRNItemDetailNo
			Where IID.GRNNo = @GRNNo
			And IID.GRNItemLotDetailNo Is Null
			And IID.RejectedQty > ISNULL(GRII.AcceptedQty,GRID.AcceptedQty)

			Update TGRI
			Set RejectedQty = ISNULL(GRII.AcceptedQty,GRID.AcceptedQty)
			From #tblGRNIndentDetail TGRI
			Inner Join Inventory.GRNItemDetail GRID On GRID.GRNItemDetailNo = TGRI.GRNItemDetailNo
			Left Join Inventory.GRNItemIndentDetail GRII On GRII.GRNIndentDetailNo = TGRI.GRNIndentDetailNo
														And GRII.GRNItemDetailNo = TGRI.GRNItemDetailNo
			Where GRID.GRNNo = @GRNNo
			And TGRI.RejectedQty > ISNULL(GRII.AcceptedQty,GRID.AcceptedQty)

			--Update Accepted Qty of GRNIndentDetail from Inspection
			update GRID
			set RejectedQty = TGRI.RejectedQty, 
			AcceptedQty = GRID.AcceptedQty - TGRI.RejectedQty
			from Inventory.GRNItemIndentDetail GRID
			Inner Join #tblGRNIndentDetail TGRI On TGRI.GRNIndentDetailNo = GRID.GRNIndentDetailNo
			where GRNNo = @GRNNo 

			--Update Accepted Qty of GRNItemDetail for direct GRN			
			Update 	GRID
			set RejectedQty = Case When GRID.AcceptedQty - TGRI.RejectedQty < 0 Then GRID.AcceptedQty Else TGRI.RejectedQty End, 
			AcceptedQty = Case When GRID.AcceptedQty - TGRI.RejectedQty < 0 Then 0 Else GRID.AcceptedQty - TGRI.RejectedQty End
			,StockQty = (Case When GRID.AcceptedQty - TGRI.RejectedQty < 0 Then 0 Else GRID.AcceptedQty - TGRI.RejectedQty End) * FirstCF /SecondCF
			from Inventory.GRNItemDetail GRID
			Inner Join #tblGRNIndentDetail TGRI On TGRI.GRNItemDetailNo = GRID.GRNItemDetailNo
			where TGRI.GRNIndentDetailNo is null 
			And GRID.GRNNo = @GRNNo 
			
			--Update Accepted Qty of GRNItemDetail for GRN against Purchase Request
			update gd 
			set rejectedQty = tt.rejectedQty 
			,AcceptedQty = tt.AcceptedQty 
			,StockQty = StockQty - isnull((tt.RejectedQty * FirstCF )/SecondCF,0)
			from Inventory.GRNItemDetail gd 
			inner join(
						select GRID.GRNNo ,GRID.GRNItemDetailNo 
						,sum(GRID.RejectedQty)rejectedQty,sum(GRID.AcceptedQty) AcceptedQty
						from Inventory.GRNItemIndentDetail GRID
						Inner Join #tblGRNIndentDetail TGRI On TGRI.GRNIndentDetailNo = GRID.GRNIndentDetailNo
						group by GRID.GRNNo ,GRID.GRNItemDetailNo 
					) tt on tt.GRnNo = gd.GRnNo 
			and tt.GRNItemDetailNo= gd.GRNItemDetailNo
			where gd.grnno = @GRNNo	

			--Update LasteModifiedDate of GRN
			Update Inventory.GRNMain
			Set LastModifiedDate = GetDate()
			Where GRNNo = @GRNNo

			end				

		set @OutputStatus =1
		set @Result='Success'
	END TRY
	BEGIN CATCH
		set @OutputStatus =0
		set @Result=error_message()
	END CATCH
END
