CREATE PROCEDURE [Inventory].[MMSAFinancePosting]  
	@CompanyNo tinyint ,
	@ItemNos varchar(max)
	
AS
BEGIN

	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;

begin try
declare @IsManagingExpenseAccount bit

 create table #Posting ([Add to Basic] bit,Particular varchar(500),[Debit Amount] decimal(15,2) , 
		[Credit Amount] decimal(15,2),[Account Name] int, [Ref. Detail] varchar(50),[Cost Center] varchar(50),
		[Amount] decimal(15,2), [Type] tinyint,SlNo tinyint ,MiscChargeNo int , TaxCodeMiscChargesDetailNo int,
			ItemNos varchar(200))

declare @ItemTable table (ItemNo int,BasicAmount money,Id int)

		insert into @ItemTable
		select Col1, sum(convert(money,Col2)), Col3 from masterdata.Split_RowColumn(@ItemNos,'`','~')
		group by Col1, Col3

	insert into #Posting		
	select cast(0 as bit) [Add to Basic] , 'Stock Account' Particular 
					,case when sum(I.BasicAmount)>0 then sum(I.BasicAmount) else null end [Debit Amount] 
					,case when sum(I.BasicAmount) >0 then null else ABS(sum(I.BasicAmount)) end  [Credit Amount] 
					,IR.AccountNo [Account Name] ,'',''
					,abs(sum(I.BasicAmount)) Amount 
					,case when sum(I.BasicAmount)>0 then 1 else 2 end [Type] 
					, ROW_NUMBER () over (Order by  IR.AccountNo) SlNo
					,null MiscChargeNo
					,null TaxCodeMiscChargesDetailNo
					,'' ItemNos
	from @ItemTable I
		left join (select ItemNo,max(IR.AccountNo) AccountNo
					from Finance.ARDItemRule IR
					inner join masterdata.mAccountMaster AM on AM.AccountNo=IR.AccountNo
							and AM.AccountTypeNo=14 group by ItemNo) IR on IR.ItemNo=I.ItemNo
			group by  IR.AccountNo

	insert into #Posting		
	select cast(0 as bit) [Add to Basic] , 'Stock Adjustment' Particular 
					,case when sum(I.BasicAmount)>0 then null else abs(sum(I.BasicAmount)) end [Debit Amount] 
					,case when sum(I.BasicAmount) >0 then ABS(sum(I.BasicAmount)) else null end  [Credit Amount] 
					,IR.AccountNo [Account Name] ,'',''
					,abs(sum(I.BasicAmount)) Amount 
					,case when sum(I.BasicAmount)>0 then 2 else 1 end [Type] 
					, ROW_NUMBER () over (Order by  IR.AccountNo) SlNo
					,null MiscChargeNo
					,null TaxCodeMiscChargesDetailNo
					,'' ItemNos
	from @ItemTable I
		left join (select ItemNo,max(IR.AccountNo) AccountNo
					from Finance.ARDItemRule IR
							where IR.AccountTypeNo=19  group by ItemNo) IR on IR.ItemNo=I.ItemNo
			group by  IR.AccountNo				  

	 

 if exists(select 1 from #Posting having abs(SUM([Credit Amount])- SUM([Debit Amount]))<=.5)
 begin
	update PD set [Debit Amount] = [Debit Amount] + ISNULL(diff,0)
		from
		#Posting PD
		cross join (select SUM([Credit Amount])- SUM([Debit Amount]) Diff from #Posting) tt
		where PD.SlNo  = 1
	
	update PD set [Credit Amount] = [Credit Amount] + ISNULL(diff,0)
		from
		#Posting PD
		cross join (select SUM([Debit Amount])- SUM([Credit Amount]) Diff from #Posting) tt
		where PD.SlNo  = 1
			
 end
 
 
select * from #Posting
 where isnull([Debit Amount],0) >0 or isnull([Credit Amount],0) >0
   
   
   
END TRY
	
	BEGIN CATCH
		select ERROR_MESSAGE()
		INSERT INTO [masterdata].[mErrorLog]
			   ([ErrorDate]
			   ,[ErrorNo]
			   ,[ErrorSeverity]
			   ,[ErrorState]
			   ,[ErrorProcedure]
			   ,[ErrorLine]
			   ,[ErrorDescription]
			   ,[UserName]
			   ,[FormCode])
		VALUES
			   (getdate()
			   ,ERROR_NUMBER()
			   ,ERROR_SEVERITY()
			   ,ERROR_STATE()
			   ,ERROR_PROCEDURE()
			   ,ERROR_LINE()
			   ,ERROR_MESSAGE()
			   ,SUSER_SNAME()
			   ,'MMSA')
	END CATCH
END
