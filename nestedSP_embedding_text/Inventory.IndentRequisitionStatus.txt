Stored procedure Inventory.IndentRequisitionStatus: to Chage Status OF Indent requisition. Called from Inventory.IMMIDelete, Inventory.MMPRDelete.
Writes: Inventory.IndentRequisitionItemDetail, Inventory.IndentRequisitionMain
Reads: Inventory.IndentItemDetail, Inventory.IndentMain, Inventory.IndentRequisitionItemDetail, Inventory.MaterialIssueItemDetail, Inventory.MaterialIssueMain
Parameters: @IndentRequisitionNo int, @ItemNo int, @MakeNo smallint, @TechnicalGradeNo tinyint, @OutputStatus bit OUTPUT, @Result varchar(100) OUTPUT, @StatusNo tinyint = 2
Author: akhil, created 26 oct 2012
Code:
-- =============================================
-- Author:		akhil
-- Create date: 26 oct 2012
-- Description:	to Chage Status OF Indent requisition
-- =============================================
CREATE PROCEDURE [Inventory].[IndentRequisitionStatus] 
	-- Add the parameters for the stored procedure here
	@IndentRequisitionNo int,
	@ItemNo int,
	@MakeNo smallint ,
	@TechnicalGradeNo tinyint,
	@OutputStatus bit output,
	@Result varchar(100) output,
	@StatusNo tinyint = 2 
	
AS
BEGIN
begin try
		Declare @IssueQty decimal(10,3), @IndentQty decimal(10,3)

if (@StatusNo = 2 or @StatusNo = 4 or @StatusNo =7) -- Cancel or complete status
	begin
		
	select  @IssueQty= sum(mid.IssuedQty) from Inventory.MaterialIssueMain mim 
		inner join Inventory.MaterialIssueItemDetail mid on mid.IssueNo = mim.IssueNo 
		where mim.IssueSourceNo =5 and RefDocumentNo = @IndentRequisitionNo
		and mid.ItemNo = @ItemNo  and ISNULL(MakeNo,0) = ISNULL(@MakeNo,0)
		and isnull(TechnicalGradeNo,0)=isnull(@TechnicalGradeNo,0)
	
		select  @IndentQty= sum(IID.IndentQty) from Inventory.IndentMain IM
		inner join Inventory.IndentItemDetail IID on IM.IndentNo = IID.IndentNo 
		where IM.IndentAgainstNo =3 and RefDocumentNo = @IndentRequisitionNo
		and IID.ItemNo = @ItemNo  and ISNULL(MakeNo,0) = ISNULL(@MakeNo,0)
		and isnull(TechnicalGradeNo,0)=isnull(@TechnicalGradeNo,0)
		

		Update Inventory.IndentRequisitionItemDetail  set IssueQty  = @IssueQty 
					where IndentRequisitionNo =@IndentRequisitionNo and 
					ItemNo = @ItemNo and isnull(MakeNo,0) = isnull(@MakeNo,0)
					and isnull(TechnicalGradeNo,0)=isnull(@TechnicalGradeNo,0)
					
		Update Inventory.IndentRequisitionItemDetail  set StatusNo = 2
					where IndentRequisitionNo =@IndentRequisitionNo and 
					ItemNo = @ItemNo and isnull(MakeNo,0) = isnull(@MakeNo,0)
					and isnull(TechnicalGradeNo,0)=isnull(@TechnicalGradeNo,0)
					and Quantity  <= isnull(@IssueQty ,0)
					
		Update Inventory.IndentRequisitionItemDetail set StatusNo = 8
					where IndentRequisitionNo = @IndentRequisitionNo and 
					ItemNo = @ItemNo and isnull(MakeNo,0) = isnull(@MakeNo,0)
					and isnull(TechnicalGradeNo,0)=isnull(@TechnicalGradeNo,0)
					and Quantity > isnull(IssueQty ,0) and isnull(IssueQty ,0) >0

 
		Update Inventory.IndentRequisitionItemDetail set StatusNo = 1
					where IndentRequisitionNo = @IndentRequisitionNo and 
					ItemNo = @ItemNo and isnull(MakeNo,0) = isnull(@MakeNo,0)
					and isnull(TechnicalGradeNo,0)=isnull(@TechnicalGradeNo,0)
					and Quantity  > ISNULL(@IssueQty,0) and StatusNo =2
	end
	 if @StatusNo <> 2
	 begin
		Update Inventory.IndentRequisitionItemDetail set  StatusNo = @StatusNo
					where IndentRequisitionNo = @IndentRequisitionNo and ItemNo = @ItemNo and isnull(MakeNo,0) = isnull(@MakeNo,0)
					and isnull(TechnicalGradeNo,0) = isnull(@TechnicalGradeNo,0)
	 end 
	 -- Preferences list Authpoized/Hold/Cancel/Short Close/Completed
	 
	 if exists(select 1 from Inventory.IndentRequisitionItemDetail where IndentRequisitionNo = @IndentRequisitionNo and statusNo =1)
	 begin
	   if exists(select 1 from Inventory.IndentRequisitionItemDetail where	
						IndentRequisitionNo = @IndentRequisitionNo and statusNo =8)
		update Inventory.IndentRequisitionMain set StatusNo = 8 where IndentRequisitionNo= @IndentRequisitionNo
	   else
	     update Inventory.IndentRequisitionMain set StatusNo = 1 where IndentRequisitionNo= @IndentRequisitionNo
	      
		
	 end	
	 else if exists(select 1 from Inventory.IndentRequisitionItemDetail where 
				IndentRequisitionNo = @IndentRequisitionNo and statusNo =8)
	 begin
	    update Inventory.IndentRequisitionMain set StatusNo = 8 where IndentRequisitionNo= @IndentRequisitionNo
	 end	
	
	 else if exists(select 1 from Inventory.IndentRequisitionItemDetail where IndentRequisitionNo = @IndentRequisitionNo and Statusno = 3 )-- Hold 
		update Inventory.IndentRequisitionMain set StatusNo =3 where IndentRequisitionNo = @IndentRequisitionNo
	 else if exists(select 1 from Inventory.IndentRequisitionItemDetail where IndentRequisitionNo = @IndentRequisitionNo and Statusno = 4 )--cancel
	 	update Inventory.IndentRequisitionMain set StatusNo =4 where IndentRequisitionNo = @IndentRequisitionNo
	else if exists(select 1 from Inventory.IndentRequisitionItemDetail where IndentRequisitionNo = @IndentRequisitionNo and Statusno=7)--shortclose
	 	update Inventory.IndentRequisitionMain set StatusNo =7 where IndentRequisitionNo = @IndentRequisitionNo						
	else if exists(select 1 from Inventory.IndentRequisitionItemDetail where IndentRequisitionNo = @IndentRequisitionNo and Statusno=2)--completed
		update Inventory.IndentRequisitionMain set StatusNo =2 where IndentRequisitionNo = @IndentRequisitionNo
		  					
	 
set @OutputStatus =1
set @Result='Success'
end try
begin catch
set @OutputStatus =0
set @Result=error_message()

		select 0, error_message()
end Catch


END
