-- Description:	Get SO wise qty. for purchase request 
-- =============================================
CREATE PROCEDURE [Information].[PRSODetail] 
	-- Add the parameters for the stored procedure here
	@MaterialPlanningNo int,
	@ItemNo int,
	@MakeNo smallint
AS
BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;

    -- Insert statements for procedure here
	declare @str varchar(max)
    select @str=masterdata.JSForInformation('MMPR')
    set @str=@str+masterdata.CSSForInformation('MMPR')
	set @str=@str+'<table class="altrowstable" id="alternatecolorPRSOInfo">'
	set @str=@str+'<Caption><Font color="Maroon"><h3 align="Left"><b>Sales Order Detail</b></h3></Font></Caption>'
    set @str=@str+'<tr><th>Sales Order No.</th><th>Qty.</th></tr>'
	set @str=@str
				 +(	replace(
					replace(
					(    
						Select DocumentNoYearly [td1],CONVERT(Decimal(12,3),SUM(SOQty)) as [td2]
						From 
						(
						Select SOAM.DocumentNoYearly,BOMQty.OrderQty *BMD.TotalProductionQty SOQty
						From [Inventory].[MaterialPlanningMain] MPM
						Inner Join [Inventory].[MaterialPlanningBOMDetail] MPB On MPB.MaterialPlanningNo = MPM.MaterialPlanningNo
						Inner Join [Inventory].[MaterialPlanningDetail] MPD On MPD.MaterialPlanningNo = MPM.MaterialPlanningNo
						Inner Join Production.ProductionPlanningItemDetail PID On PID.ProductionPlanningNo = MPM.ProductionPlanningNo
																			   And PID.BOMRevisionNo = MPB.BOMRevisionNo
						Inner Join Production.BOMRevisionMain BRM On BRM.BOMRevisionNo = PID.BOMRevisionNo
						Inner Join Sales.SalesOrderAmendmentMain SOAM On SOAM.SalesOrderAmendmentNo = BRM.SalesOrderAmendmentNo
						Inner Join Production.BOMMarkDetail BMD On BMD.BOMNo = BRM.BOMNo
																And BMD.BOMSerialNo = PID.BOMRevisionSerialNo
																And BMD.ItemNo = MPD.ItemNo
																And ISNULL (BMD.MakeNo,0) = ISNULL(MPD.MakeNo,0)
						Cross apply [Production].[FuncGetBOMQty](BRM.BOMNo) BOMQty											
						Where MPM.ProductionPlanningNo is not null
						And MPM.MaterialPlanningNo = @MaterialPlanningNo
						And MPD.ItemNo = @ItemNo
						And ISNULL(MPD.MakeNo,0) = ISNULL(@MakeNo,0)
						Union All
						Select SOAM.DocumentNoYearly,BOMQty.OrderQty *BMD.TotalProductionQty
						From [Inventory].[MaterialPlanningMain] MPM
						Inner Join [Inventory].[MaterialPlanningBOMDetail] MPB On MPB.MaterialPlanningNo = MPM.MaterialPlanningNo
						Inner Join [Inventory].[MaterialPlanningDetail] MPD On MPD.MaterialPlanningNo = MPM.MaterialPlanningNo
						Inner Join Production.BOMRevisionMain BRM On BRM.BOMRevisionNo = MPB.BOMRevisionNo
						Inner Join Sales.SalesOrderAmendmentMain SOAM On SOAM.SalesOrderAmendmentNo = BRM.SalesOrderAmendmentNo
						Inner Join Production.BOMMarkDetail BMD On BMD.BOMNo = BRM.BOMNo
																And BMD.ItemNo = MPD.ItemNo
																And ISNULL (BMD.MakeNo,0) = ISNULL(MPD.MakeNo,0) 										
						Cross apply [Production].[FuncGetBOMQty](BRM.BOMNo) BOMQty											
						Where MPM.MaterialPlanningNo = @MaterialPlanningNo
						And MPD.ItemNo = @ItemNo
						And ISNULL(MPD.MakeNo,0) = ISNULL(@MakeNo,0)
						And MPM.ProductionPlanningNo is null
						)dt
						Group By DocumentNoYearly
						for xml path('tr')),'<td1>','<td style="text-align: center;">')						             
										   ,'<td2>','<td style="text-align: right;">')
					)				   
	set @str=@str+'</table>'	
	select @str
END
