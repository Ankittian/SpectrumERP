CREATE PROCEDURE [Finance].[VoucherSaveBeforeDelete]
(
	@VoucherNo  int
)
AS

BEGIN
	SET NOCOUNT ON;
	
	BEGIN TRY
		
		
		UPDATE rm
		   SET [SpecificInprocessDrAmount] = [SpecificInprocessDrAmount]-tt.SpecificDrAmt
			  ,[SpecificInprocessCrAmount] = [SpecificInprocessCrAmount]-tt.SpecificCrAmt
			  ,[BaseInprocessDrAmount] = BaseInprocessDrAmount-tt.BaseDrAmt
			  ,[BaseInprocessCrAmount] = BaseInprocessCrAmount-tt.BaseCrAmt
			  ,[SpecificBalanceDrAmount] = SpecificBalanceDrAmount-tt.SpecificAuthorizedDrAmt
			  ,[SpecificBalanceCrAmount] = SpecificBalanceCrAmount-tt.SpecificAuthorizedCrAmt
			  ,[BaseBalanceDrAmount] = BaseBalanceDrAmount-tt.BaseAuthorizedDrAmt
			  ,[BaseBalanceCrAmount] = BaseBalanceCrAmount-tt.BaseAuthorizedCrAmt
		from (select rdIn.ReferenceNo
					,sum(case when vmIn.AuthorizedBy is null then rdIn.BaseDrAmount else 0 end) BaseDrAmt
					,sum(case when vmIn.AuthorizedBy is null then rdIn.BaseCrAmount else 0 end) BaseCrAmt
					,sum(case when vmIn.AuthorizedBy is not null then rdIn.BaseDrAmount else 0 end) BaseAuthorizedDrAmt
					,sum(case when vmIn.AuthorizedBy is not null then rdIn.BaseCrAmount else 0 end) BaseAuthorizedCrAmt
					
					,sum(case when vmIn.AuthorizedBy is null then rdIn.SpecificDrAmount else 0 end) SpecificDrAmt
					,sum(case when vmIn.AuthorizedBy is null then rdIn.SpecificCrAmount else 0 end) SpecificCrAmt
					,sum(case when vmIn.AuthorizedBy is not null then rdIn.SpecificDrAmount else 0 end) SpecificAuthorizedDrAmt
					,sum(case when vmIn.AuthorizedBy is not null then rdIn.SpecificCrAmount else 0 end) SpecificAuthorizedCrAmt

				from [Finance].[accVoucherReferenceDetail] rdIn
					inner join finance.accvoucherdetail vdIn on vdIn.voucherdetailno=rdIn.voucherdetailno
					inner join finance.accvouchermain vmIn on vmIn.voucherno=vdIn.voucherno
				where vmIn.VoucherNo=@VoucherNo
				group by rdIn.ReferenceNo) tt
			inner join [Finance].[ReferenceMain] rm on tt.ReferenceNo=rm.ReferenceNo

			If Exists(Select 1 from (select rdIn.ReferenceNo
									 from [Finance].[accVoucherReferenceDetail] rdIn
									 inner join finance.accvoucherdetail vdIn on vdIn.voucherdetailno=rdIn.voucherdetailno
									 inner join finance.accvouchermain vmIn on vmIn.voucherno=vdIn.voucherno
									 where vmIn.VoucherNo=@VoucherNo
									group by rdIn.ReferenceNo) tt
					  inner join [Finance].[ReferenceMain] rm on tt.ReferenceNo=rm.ReferenceNo
					  where ((rm.ByToNo = 1 and IsNULL(rm.BalanceAmount,0) <0) or (rm.ByToNo = 2 and IsNULL(rm.BalanceAmount,0) >0)))
			Begin 
				Raiserror('Contact your vendor,Reference getting -ve',16,-1)
			End


		return(1)

	END TRY
	BEGIN CATCH
		return(@@Error)
	END CATCH
END
