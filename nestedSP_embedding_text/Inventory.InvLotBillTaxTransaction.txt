				inner join Sales.InvoiceItemDetails iid on idcd.InvoiceNo = iid.InvoiceNo 
					and iid.ItemNo = ld.ItemNo and isnull(iid.MakeNo,0) = isnull(ld.MakeNo,0) 
					and isnull(iid.SalesOrderAmendmentNo,0) = isnull(ld.SalesOrderAmendmentNo ,0)
			inner join Inventory.InvLotSummary ls on ls.InvLotSummaryNo = ld.InvLotSummaryNo 		
			 where idcd.InvoiceNo = @DocumentNo And ld.InvLotSummaryNo = @InvLotSummaryNo 
			 
			end
			else
			begin
			
				select @ItemRate = iid.Rate , @taxAcceptedQty =ld.Qty from

				inner join Sales.InvoiceItemDetails iid on ld.InvoiceNo = iid.InvoiceNo 
					and iid.ItemNo = ld.ItemNo and isnull(iid.MakeNo,0) = isnull(ld.MakeNo,0) 
					and isnull(iid.CombinationNo,0) = isnull(ld.CombinationNo,0)

			inner join Inventory.InvLotSummary ls on ls.InvLotSummaryNo = ld.InvLotSummaryNo 		
			 where iid.InvoiceNo = @DocumentNo And ld.InvLotSummaryNo = @InvLotSummaryNo 
			 
			end
			
			select  @DiffRate=  
					 isnull(sum((case when mc.CalculationNature =1 then iitd.TotalAmount
					when mc.CalculationNature =2 then 0 -iitd.TotalAmount  else 0 end) / iid.Quantity),0)
				from 
				Sales.InvoiceItemDetails  iid
				inner join Sales.InvoiceItemTaxDetail iitd on iitd.InvoiceNo = iid.InvoiceNo 
					and iid.ItemNo = iitd.ItemNo and isnull(iid.MakeNo,0) = isnull(iitd.MakeNo,0) 
					and isnull(iid.CombinationNo,0) = isnull(iitd.CombinationNo,0) 
					and isnull(iid.SalesOrderAmendmentNo,0) = isnull(iitd.SalesOrderAmendmentNo ,0)
				inner join sales.InvoiceTaxDetail td on td.InvoiceNo = iitd.InvoiceNo and td.MiscChargeNo = iitd.MiscChargeNo

			DiffRate,DiffInLRate,ItemRate + ISNULL(DiffInLRate,0) Lrate, VRate,InvLotBillDetailNo,ItemNo,transactionflag,ItemRate 
			From Inventory.InvLotBillDetail 
            where  TransactionFlag not in ('SDDC','SDEI','SDCI') 
            and invlotsummaryNo = @invlotsummaryno
            
         Update #temp Set 
          @Lrate= case when Sno=1 then Itemrate + DiffInLRate else @Lrate + DiffInLRate end,
       	  Lrate=case when Sno=1 then Itemrate + DiffInLRate else  @Lrate end ,
       	  @Vrate= case when Sno=1 then Itemrate + DiffRate else @Vrate + DiffRate end,
       	  Vrate=case when Sno=1 then Itemrate + DiffRate else  @Vrate end,
       	 @InvlotSummaryNo=InvlotSummaryNo
		From #temp
        update ls set LRate =t.Lrate,VRate =Case When  t.Vrate <=-0.000001 Then 0 Else t.Vrate End
        from #temp t,Inventory.InvLotBillDetail   bd,Inventory.InvLotSummary ls where(T.InvLotBillDetailNo = bd.InvLotBillDetailNo) 
       And (t.Lrate <> ls.LRate or t.Vrate <> ls.Vrate ) And ls.InvLotSummaryNo = bd.InvLotSummaryNo and sNo = (select MAX(SNo) from #temp where InvLotSummaryNo= ls.InvLotSummaryNo )
       update ls set VRate =Case When  t.Vrate <=-0.000001 Then 0 Else t.Vrate End  from #temp t,Inventory.InvLotBillDetail   bd,Inventory.InvLotSummary ls where(T.InvLotBillDetailNo = bd.InvLotBillDetailNo)
       and ls.VRate <>t.Vrate and ls.InvLotSummaryNo = bd.InvLotSummaryNo and sNo = (select MAX(SNo) from #temp where InvLotSummaryNo= ls.InvLotSummaryNo )
      update bd set VRate =Case When  t.Vrate <=-0.000001 Then 0 Else t.Vrate End  from #temp t,Inventory.InvLotBillDetail   bd where(T.InvLotBillDetailNo = bd.InvLotBillDetailNo) and t.Vrate <>bd.VRate
      update ld set VRate =ls.Vrate,LRate = ls.LRate from #temp t,Inventory.InvLotDetail ld , Inventory.InvLotSummary ls	 
      where ld.invlotsummaryNo = ls.invlotsummaryNo and T.invlotsummaryNo = ld.invlotsummaryNo and t.Vrate <>ld.VRate and ls.TransactionFlag = ld.TransactionFlag
      drop table #temp
      
      if exists (select 1 from Inventory.InvLotSummary where InvLotSummaryNo = @InvLotSummaryNo and VRate <0 )
		raiserror('Vrate can''t be Negative.',16,-1)
	end
 
		
 set @Status =1
 set @result ='Success'

    BEGIN CATCH
		 set @Status = 0
		 set @result =Error_Message()
		
	END CATCH
