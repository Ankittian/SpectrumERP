CREATE PROCEDURE [Information].[PurchaseRequestDetail] 
	-- Add the parameters for the stored procedure here
	@CompanyNo tinyint,
	@DocumentDate smalldatetime,
	@ItemNo int
AS
BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;

    -- Insert statements for procedure here
	declare @str varchar(max)
    select @str=masterdata.JSForInformation('MMPR')
    set @str=@str+masterdata.CSSForInformation('MMPR')
	set @str=@str+'<table class="altrowstable" id="alternatecolorPRInfo">'
	set @str=@str+'<Caption><Font color="Maroon"><h3 align="Left"><b>Pending Purchase Request</b></h3></Font></Caption>'
    set @str=@str+'<tr><th>Sl. No.</th><th>Division</th><th>Department</th><th>Make</th><th>Document No.</th><th>Document Date</th><th>Quantity</th><th>Balance Quantity</th><th>Created By</th><th>Authorized By</th></tr>'
	set @str=@str
				 +(	replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					replace((    
						Select  Row_number() over (order by IM.DocumentDate desc) td1
						        ,DV.DivisionName As [td2]
								,DM.DeptName As [td3]
								,isnull(MM.MakeName,'') As [td4]
								,IM.DocumentNoYearly As [td5]
								,convert(varchar,IM.DocumentDate,103) As [td6]
								,ID.IndentQty As [td7]
								,(isnull(ID.IndentQty,0)-isnull(ID.POQty,0)) As [td8]
								,L.PrintingName [td9]
                                ,Isnull(L1.PrintingName,'') [td10]
						From Inventory.IndentMain IM 
							Inner Join Inventory.IndentItemDetail ID on ID.IndentNo=IM.IndentNo
							Inner Join masterdata.mCompanyMaster CM on CM.CompanyNo=IM.CompanyNo
							Inner Join payroll.tblDeptMaster DM on DM.DeptNo=IM.DeptNo
							Inner Join payroll.tblDivisionMaster DV on DM.DivisionNo=DV.DivisionNo
							Left Join masterdata.mMakeMaster MM on MM.MakeNo=ID.MakeNo
							Left Join Security.Login L ON L.LoginNo=IM.CreatedBy
                            Left Join Security.Login L1 ON L1.LoginNo=IM.AuthorizedBy
						Where (isnull(ID.IndentQty,0)-isnull(ID.POQty,0))>0
							And IM.CompanyNo=@CompanyNo
							And IM.DocumentDate<=@DocumentDate
							And ID.ItemNo=@ItemNo and ID.StatusNo in(1,8)
						Order By IM.DocumentDate,IM.DocumentNoYearly
						for xml path('tr')),'<td1>','<td style="text-align: center;">')
						                   ,'<td2>','<td>')
										   ,'<td3>','<td>')
										   ,'<td4>','<td>')
										   ,'<td5>','<td>')
										   ,'<td6>','<td style="text-align: center;">')
										   ,'<td7>','<td style="text-align: right;">')
										   ,'<td8>','<td style="text-align: right;">')
										   ,'<td9>','<td>')
										   ,'<td10>','<td>'))
				   
	set @str=@str+'</table>'
	
	select @str
END
