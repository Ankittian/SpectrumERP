Stored procedure Purchase.POQCDetail: Fetching PO Amendment QC Detail. Called from Inventory.IMIEGetPOQCDetail.
Reads: masterdata.mQCParameters, masterdata.mUnitMaster, Purchase.POAmendmentItemDetail, Purchase.POAmendmentItemQCParameterDetail
Parameters: @POAmendmentNo int, @ItemNo int, @MakeNo smallint
Author: Kirti, created 16-November-2023
Code:
-- =============================================
-- Author:		Kirti
-- Create date: 16-November-2023
-- Description: Fetching PO Amendment QC Detail
-- =============================================
--  [Purchase].[PMPOLoadData]  45, 0,'PMPO'
CREATE PROCEDURE [Purchase].[POQCDetail] 	

@POAmendmentNo int,
@ItemNo int,
@MakeNo smallint


AS
BEGIN
		
	SET NOCOUNT ON;

Begin try	

	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;
	

    -- Insert statements for procedure here
	declare @str varchar(max)
    select @str=masterdata.JSForInformation('PMPO')
    set @str=@str+masterdata.CSSForInformation('PMPO')
	set @str=@str+'<table class="altrowstable" id="alternatecolorPOQCInfo">'
	set @str=@str+'<Caption><Font color="Maroon"><h3 align="Left"><b>PO QC Detail</b></h3></Font></Caption>'
    set @str=@str+'<tr><th>Sl. No.</th><th>Parameter Name</th><th>Unit</th><th>Standard Value 1</th><th>Standard Value 2</th></tr>'
	set @str=@str
				 +( 
		    		replace(
					replace(
					replace(
					replace(
					replace(
				 
				    replace(
					replace(
					replace(
					replace(
					replace((   
						Select Row_number() over (order by POQCD.QCParameterNo ) td1
						,QCP.QCParameterName  as td2
						,UM.Alias as td3
						,POQCD.StandardValue as	td4
						,POQCD.StandardValue2 as td5
							From Purchase.POAmendmentItemQCParameterDetail POQCD
							INNER JOIN Purchase.POAmendmentItemDetail pid on pid.POAmendmentItemDetailNo = POQCD .POAmendmentItemDetailNo 
							INNER JOIN  Masterdata.mQCParameters QCP On QCP.QCParameterNo = POQCD.QCParameterNo --And QCP.Inactive=0 
							Inner join masterdata.mUnitMaster UM on UM.UnitNo = QCP.UnitNo 
							Where pid.POAmendmentNo = @POAmendmentNo 	
							and PID.ItemNo = @ItemNo 
							and IsNULL(PID.MakeNo,0) =  IsNULL(@MakeNo,0)
					for xml path('tr')),'<td1>','<td style="text-align: center;">')
					                       ,'<td2>','<td style="text-align: center;">')
										   ,'<td3>','<td style="text-align: center;">')
										   ,'<td4>','<td>')
										   ,'<td5>','<td>')
										   
										   ,'</td1>','</td>')
										   ,'</td2>','</td>')
										   ,'</td3>','</td>')
										   ,'</td4>','</td>')
										   ,'</td5>','</td>')
										   
										   )
				   
	set @str=@str+'</table>'
	
	select @str
 
End try
 
Begin Catch
		INSERT INTO [masterdata].[mErrorLog]
				   ([ErrorDate]
				   ,[ErrorNo]
				   ,[ErrorSeverity]
				   ,[ErrorState]
				   ,[ErrorProcedure]
				   ,[ErrorLine]
				   ,[ErrorDescription]
				   ,[UserName]
				   ,[FormCode])
			 VALUES
				   (getdate()
				   ,ERROR_NUMBER()
				   ,ERROR_SEVERITY()
				   ,ERROR_STATE()
				   ,ERROR_PROCEDURE()
				   ,ERROR_LINE()
				   ,ERROR_MESSAGE()
				   ,SUSER_SNAME()
				   ,'POQC')
end catch
    
END
