CREATE PROCEDURE [Inventory].[SCMRFinancePosting] 
	@ItemDetail xml ,
	@CompanyNo tinyint ,
	@VendorLocationNo int  
	
AS
 
BEGIN

	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;

begin try

declare @IsManagingExpenseAccount bit ,@ItemNos as varchar(max)

select @IsManagingExpenseAccount = IsManagingExpenseAccount from Finance.Setting where CompanyNo = @CompanyNo 
 
Create table #IncludedTax(IncludedAmount decimal(18,6),ItemNo int,MakeNo smallint,TechnicalGradeNo tinyint
				,ServiceItemNo int,ActivityNo tinyint,ExpenseHeadNo smallint,SequenceNo tinyint)

Create table #Freight(FreightAmount decimal(18,6),ItemNo int,MakeNo smallint,TechnicalGradeNo tinyint,IsPaid bit,SCOrderNo int)

Create table #Expenses(ExpenseAmount decimal(18,6),AccountNo int,ItemNo int,
					MakeNo smallint,TechnicalGradeNo tinyint, ExpenseAccountNo int)
 
create table #Posting ([Add to Basic] bit,Particular varchar(500),[Debit Amount] decimal(15,2) , 
		[Credit Amount] decimal(15,2),[Account Name] int, [Ref. Detail] varchar(50),[Cost Center] varchar(50),
		[Amount] decimal(15,2), [Type] tinyint,SlNo tinyint ,MiscChargeNo int , TaxCodeMiscChargesDetailNo int,
			ItemNos varchar(200))
			
Create table #SCOrderDetail (SCOrderAmendmentNo int,ServiceItemNo int,ActivityNo smallint,SequenceNo tinyint , 
		ExpenseHeadNo smallint, ReceivedQty decimal(15,3),Rate decimal(18,6),SCOrderNo int, IncludeAmt money)

Create table #ItemDetail (ReferenceNo int,ReferenceType varchar(4),ItemNo int,MakeNo smallint,
	 TechnicalGradeNo tinyint,AcceptedQty decimal(15,3),Rate decimal(24,10),GateEntryNo int,RouteIdentity decimal(10,8)
	 ,IncludedTax decimal(18,6) , ChallanQty decimal(15,3),ReceivedQty decimal(15,3),SCOrderNo int
	 ,StockRate decimal(24,10),[Select] bit)

	 ;with cte as 
	(
		SELECT v.value('ReferenceNo[1]','int') AS ReferenceNo,
		v.value('ReferenceType[1]','varchar(4)') ReferenceType,
		v.value('ItemNo[1]','int') ItemNo,
		v.value('MakeNo[1]','smallint') MakeNo,
		v.value('TechnicalGradeNo[1]','tinyint') TechnicalGradeNo,
		v.value('AcceptedQty[1]','decimal(15,3)') AcceptedQty,
		v.value('Rate[1]','decimal(18,6)') Rate,
		v.value('GateEntryNo[1]','int') GateEntryNo,
		v.value('RouteIdentity[1]','decimal(10,8)') RouteIdentity,
		v.value('IncludedTax[1]','decimal(18,6)') IncludedTax,
		v.value('ChallanQty[1]','decimal(15,3)') ChallanQty,
		v.value('ReceivedQty[1]','decimal(15,3)') ReceivedQty,
		v.value('SCOrderAmendmentNo[1]','int') SCOrderAmendmentNo
		FROM @ItemDetail.nodes('/DocumentElement/Table') x(v)
	)
	Insert into #ItemDetail select cte.ReferenceNo,cte.ReferenceType,cte.ItemNo,cte.MakeNo,cte.TechnicalGradeNo,
	--cte.AcceptedQty
	case when s.GRNPassOnQty = 2 and im.ItemTypeNo = 2 then ChallanQty else AcceptedQty end
	,cte.Rate,cte.GateEntryNo,cte.RouteIdentity,cte.IncludedTax,cte.ChallanQty ,cte.ReceivedQty,
	SCOrderNo,cte.Rate ,1 from cte
	left join Purchase.SCOrderAmendmentMain scm on scm.SCOrderAmendmentNo = cte.SCOrderAmendmentNo 
	inner join Inventory.Setting s on s.CompanyNo = @CompanyNo	
	inner join masterdata.mItemMaster im on im.ItemNo = cte.ItemNo
	 where cte.ItemNo is not null -- and AcceptedQty>0
	and not exists(select 1 from purchase.SCOrderItemDetail where scorderNo= scm.scorderNo
					and ItemFlowNo =5 and ItemNo =cte.ItemNo and
					isnull(MakeNo,0) = isnull(cte.MakeNo,0) 
					and isnull(TechnicalGradeNo,0) =isnull(cte.TechnicalGradeNo ,0))
	union
	select cte.ReferenceNo,cte.ReferenceType,cte.ItemNo,cte.MakeNo,cte.TechnicalGradeNo,
	cte.AcceptedQty,0.00000001 Rate,cte.GateEntryNo,cte.RouteIdentity,cte.IncludedTax,cte.ChallanQty ,cte.ReceivedQty,
	SCOrderNo,null Rate ,0 from cte
	left join Purchase.SCOrderAmendmentMain scm on scm.SCOrderAmendmentNo = cte.SCOrderAmendmentNo 
	 where ItemNo is not null
	and exists(select 1 from purchase.SCOrderItemDetail where scorderNo= scm.scorderNo
					and ItemFlowNo =5 and ItemNo =cte.ItemNo and
					isnull(MakeNo,0) = isnull(cte.MakeNo,0) 
					and isnull(TechnicalGradeNo,0) =isnull(cte.TechnicalGradeNo ,0))				
	 
	
	
	; with cte as
	 (
		SELECT v.value('SCOrderAmendmentNo[1]','int') AS SCOrderAmendmentNo,
		v.value('ServiceItemNo[1]','int') AS ServiceItemNo,
		v.value('ActivityNo[1]','smallint') ActivityNo,
		v.value('SequenceNo[1]','Tinyint') SequenceNo,
		v.value('ExpenseHeadNo[1]','smallint') ExpenseHeadNo,
		v.value('ReceivedQty[1]','decimal(15,3)') ReceivedQty,
		v.value('Rate[1]','decimal(18,6)') Rate,
		v.value('ItemNo[1]','int') ItemNo
		FROM @ItemDetail.nodes('/DocumentElement/Table') x(v)
	)
	Insert into #SCOrderDetail(SCOrderAmendmentNo ,ServiceItemNo ,ActivityNo ,SequenceNo,ExpenseHeadNo,
				 ReceivedQty ,Rate,SCOrderNo ,IncludeAmt  )
	select cte.SCOrderAmendmentNo ,ServiceItemNo ,cte.ActivityNo ,cte.SequenceNo ,cte. ExpenseHeadNo,
			ReceivedQty ,cte.Rate ,scm.SCOrderNo ,
			sum(case when MC.CalculationNature=1 then   (TotalAmount  /sad.Quantity ) * cte.ReceivedQty
				when calculationnature =2 then 0-(TotalAmount  /sad.Quantity ) * cte.ReceivedQty else 0 end)
			
			 from cte 
			left join Purchase.SCOrderAmendmentMain scm on scm.SCOrderAmendmentNo = cte.SCOrderAmendmentNo 
		left join Purchase.SCOrderAmendmentDetail sad on sad .SCOrderAmendmentNo = scm.SCOrderAmendmentNo 
				and ((   Coalesce(sad.ActivityNo ,sad.ExpenseHeadNo,sad.ItemNo ) =
					Coalesce(cte.ActivityNo , cte.ExpenseHeadNo,cte.ServiceItemNo) )
		or (cte.ActivityNo is null and cte.ExpenseHeadNo is null and cte.ServiceItemNo is null))
		left join Purchase.SCOrderAmendmentTaxDetail td on td.SCOrderAmendmentNo = scm.SCOrderAmendmentNo 
				and td.IncludeInBasic =1 
		left join Purchase.SCOrderAmendmentItemTaxDetail std on std.SCOrderAmendmentNo = cte.SCOrderAmendmentNo
			and ((Coalesce(std.ActivityNo , std.ExpenseHeadNo,std.ItemNo) =
				Coalesce(cte.ActivityNo , cte.ExpenseHeadNo,cte.ServiceItemNo ) )
		or (cte.ActivityNo is null and cte.ExpenseHeadNo is null and cte.ServiceItemNo is null))
				and td.MiscChargeNo = std.MiscChargeNo 
		left join masterdata.mMiscCharges mc on mc.MiscChargeNo = std.MiscChargeNo 
			where cte.ItemNo is null and (td.IncludeInBasic =1 or td.MiscChargeNo is null)
		Group by cte.SCOrderAmendmentNo ,ServiceItemNo ,cte.ActivityNo ,cte.SequenceNo ,cte. ExpenseHeadNo,
			ReceivedQty ,cte.Rate ,scm.SCOrderNo	

Insert  into #ItemDetail
	select sid.SCOrderAmendmentNo ReferenceNo,'SCSO',sid.ItemNo,sid.MakeNo,sid.TechnicalGradeNo,
	Null ,sid.Rate,null,null RouteIdentity,null IncludedTax,null ChallanQty ,null ReceivedQty,
	sam.SCOrderNo,sid.Rate ,0
	from Purchase.SCOrderAmendmentItemDetail sid
	inner join Purchase.SCOrderAmendmentMain sam on sam.SCOrderAmendmentNo = sid.SCOrderAmendmentNo 
	inner join #SCOrderDetail sd on sd.SCOrderAmendmentNo = sid.SCOrderAmendmentNo 
		
	where ((Coalesce(sd.ActivityNo , sd.ExpenseHeadNo,sd.ServiceItemNo ) =
				Coalesce(sid.ActivityNo , sid.ExpenseHeadNo,sid.ServiceItemNo) )
		or (sid.ActivityNo is null and sid.ExpenseHeadNo is null and sid.ServiceItemNo is null))
		and ItemFlowNo in(2,3,4)
		and not exists (select 1 from #ItemDetail id where id.SCOrderNo = sam.SCOrderNo 
					and ItemNo = sid.ItemNo and isnull(MakeNo,0)= isnull(sid.MakeNo,0) 
					and isnull(TechnicalGradeNo,0) = isnull(sid.TechnicalGradeNo,0) )
					
if exists(select 1 from #SCOrderDetail where SCOrderAmendmentNo is not null)
begin
	
	update id set StockRate = ROUND(sid.Rate ,6)
	from
	Purchase.SCOrderAmendmentItemDetail sid
	inner join #ItemDetail id on id.ReferenceNo = sid.SCOrderAmendmentNo 
	and id.ItemNo = sid.ItemNo and isnull(id.MakeNo,0) = isnull(sid.MakeNo,0) and 
	isnull(id.TechnicalGradeNo,0) = isnull(sid.TechnicalGradeNo ,0)
	inner join #SCOrderDetail sd on sd.SCOrderAmendmentNo = sid.SCOrderAmendmentNo 
		and (coalesce(sd.ActivityNo,sd.ExpenseHeadNo ,sd.ServiceItemNo ) =
			coalesce(sid.ActivityNo,sid.ExpenseHeadNo ,sid.ServiceItemNo )
			or (sid.ActivityNo is null and sid.ExpenseHeadNo is null and sid.ServiceItemNo is null)
			)
				and ISNULL(sd.SequenceNo,0)= ISNULL(sid.SequenceNo,0)
	where sid.ItemFlowNo in(3,4,2)   
	
	select @IsManagingExpenseAccount = case when SCNatureNo in(5,6,7)  then 0 else @IsManagingExpenseAccount end
		 from Purchase.SCOrderAmendmentMain sm
	inner join #SCOrderDetail sd on sd.SCOrderAmendmentNo = sm.SCOrderAmendmentNo 

end	
   
 
 set @ItemNos = STUFF((select '`' + convert(varchar(7),ItemNo) +'~' + convert(varchar(25),SUM(AcceptedQty * Rate))		  from #ItemDetail group by ItemNo for xml path ('')),1,1,'')

  
---- expense Heads
begin
	 
	Insert into #Expenses 
	select  sum((((gsd.rate * gsd.ReceivedQty)+isnull(IncludeAmt,0)) * ISNULL(said.BifercationPercentage*0.01,1) ) )  ,tt.AccountNo,
					gid.ItemNo ,gid.MakeNo ,gid.TechnicalGradeNo ,ExAcc.AccountNo 
					
				from #ItemDetail  gid
			inner join #SCOrderDetail gsd on gsd.scorderno = gid.scorderno
			inner join Purchase.SCOrderAmendmentMain sam on sam.SCOrderAmendmentNo = gsd.SCOrderAmendmentNo
			inner join Purchase.SCOrderAmendmentDetail sad on sad.SCOrderAmendmentNo = gsd.SCOrderAmendmentNo 
				and coalesce(sad.ExpenseHeadNo ,sad.ActivityNo,sad.ItemNo)= 
					coalesce(gsd.ExpenseHeadNo ,gsd.ActivityNo,gsd.ServiceItemNo)
					and isnull(sad.SequenceNo,0) = isnull(gsd.SequenceNo,0)
			inner join Purchase.SCOrderAmendmentItemDetail said on said.SCOrderAmendmentNo=sad.SCOrderAmendmentNo
			 and ( coalesce(sad.ExpenseHeadNo,sad.ActivityNo,sad.ItemNo) =
					coalesce(said.ExpenseHeadNo,said.ActivityNo,said.ServiceItemNo)
					or (said.ExpenseHeadNo is null and said.ActivityNo is null and said.ServiceItemNo is null)
					)
						and isnull(sad.SequenceNo,0)=isnull(said.SequenceNo,0)
						and ItemFlowNo in(3,4)
						and said.ItemNo = gid.ItemNo 
						and isnull(said.Makeno,0) =isnull( gid.Makeno ,0)
						and isnull(said.TechnicalGradeNo,0) = isnull(gid.TechnicalGradeNo ,0)
			Outer apply (SELECT pad.AccountNo FROM Purchase.SCOrderAccountAssignmentDetail pad
						where pad.Flag ='IM'
						and pad.SCOrderNo = gid.SCOrderNo and pad.AccountTypeNo =16
						and sam.SCNatureNo in(5,6,7) -- Capital
						group by pad.AccountNo 
						union
						SELECT pad.AccountNo FROM Purchase.SCOrderAccountAssignmentDetail pad
						where pad.SCOrderNo = gid.SCOrderNo and pad.AccountTypeNo = 39
						and pad.Id =coalesce(gsd.ExpenseHeadNo ,gsd.ActivityNo,gsd.ServiceItemNo)
						and sam.SCNatureNo  not in(5,6,7) -- revenue
						and pad.Flag in('SIM','EH','ECA')
						group by pad.AccountNo 
					 ) tt 
			Outer apply  (SELECT pad.AccountNo FROM Purchase.SCOrderAccountAssignmentDetail pad
						where  pad.Flag ='IM'
						and pad.SCOrderNo = gid.SCOrderNo and pad.AccountTypeNo =16
						and sam.SCNatureNo in(5,6,7) -- Capital
						group by pad.AccountNo 
						union
						SELECT pad.AccountNo FROM Purchase.SCOrderAccountAssignmentDetail pad
						where  pad.SCOrderNo = gid.SCOrderNo and pad.AccountTypeNo = 54
						and pad.Id =coalesce(gsd.ExpenseHeadNo ,gsd.ActivityNo,gsd.ServiceItemNo)
						and sam.SCNatureNo  not in(5,6,7) -- revenue
						and pad.Flag in('SIM','EH')
						group by pad.AccountNo 
					 )	ExAcc  
					 
			group by gid.ItemNo ,gid.MakeNo ,gid.TechnicalGradeNo ,ExAcc.AccountNo ,tt.AccountNo 
	union 
	
select  sum((((gsd.rate * gsd.ReceivedQty)+isnull(IncludeAmt,0)) * ISNULL(said.BifercationPercentage*0.01,1) ) )  ,tt.AccountNo,
					gid.ItemNo ,gid.MakeNo ,gid.TechnicalGradeNo ,ExAcc.AccountNo 
		 	
				from #ItemDetail  gid
			inner join #SCOrderDetail gsd on gsd.scorderno = gid.scorderno
			inner join Purchase.SCOrderAmendmentMain sam on sam.SCOrderAmendmentNo = gsd.SCOrderAmendmentNo
			inner join Purchase.SCOrderAmendmentBillDetail sad on sad.SCOrderAmendmentNo = gsd.SCOrderAmendmentNo 
				and sad.BillingItemNo = gsd.ServiceItemNo 
			inner join Purchase.SCOrderAmendmentBillActivityDetail sbd on sbd.SCOrderAmendmentNo = sad.SCOrderAmendmentNo and
							sbd.SerialNo = sad.SerialNo 	
			inner join Purchase.SCOrderAmendmentItemDetail said on said.SCOrderAmendmentNo=sad.SCOrderAmendmentNo
			 and  sbd.ActivityNo  = said.ActivityNo
						and isnull(sbd.SequenceNo,0)=isnull(said.SequenceNo,0)
						and ItemFlowNo in(3,4)
						and said.ItemNo = gid.ItemNo 
						and isnull(said.Makeno,0) =isnull( gid.Makeno ,0)
						and isnull(said.TechnicalGradeNo,0) = isnull(gid.TechnicalGradeNo ,0)
			Outer apply (SELECT pad.AccountNo FROM Purchase.SCOrderAccountAssignmentDetail pad
						where pad.Flag ='IM'
						and pad.SCOrderNo = gid.SCOrderNo and pad.AccountTypeNo =16
						and sam.SCNatureNo in(5,6,7) -- Capital
						group by pad.AccountNo 
						union
						SELECT pad.AccountNo FROM Purchase.SCOrderAccountAssignmentDetail pad
						where pad.SCOrderNo = gid.SCOrderNo and pad.AccountTypeNo = 39
						and pad.Id =coalesce(gsd.ExpenseHeadNo ,gsd.ActivityNo,gsd.ServiceItemNo)
						and sam.SCNatureNo  not in(5,6,7) -- revenue
						and pad.Flag in('SIM','EH','ECA')
						group by pad.AccountNo 
					 ) tt 
			Outer apply  (SELECT pad.AccountNo FROM Purchase.SCOrderAccountAssignmentDetail pad
						where  pad.Flag ='IM'
						and pad.SCOrderNo = gid.SCOrderNo and pad.AccountTypeNo =16
						and sam.SCNatureNo in(5,6,7) -- Capital
						group by pad.AccountNo 
						union
						SELECT pad.AccountNo FROM Purchase.SCOrderAccountAssignmentDetail pad
						where  pad.SCOrderNo = gid.SCOrderNo and pad.AccountTypeNo = 54
						and pad.Id =coalesce(gsd.ExpenseHeadNo ,gsd.ActivityNo,gsd.ServiceItemNo)
						and sam.SCNatureNo  not in(5,6,7) -- revenue
						and pad.Flag in('SIM','EH')
						group by pad.AccountNo 
					 )	ExAcc  
			group by gid.ItemNo ,gid.MakeNo ,gid.TechnicalGradeNo ,ExAcc.AccountNo ,tt.AccountNo 		
	
	
end

insert into #Freight 
 
	select sum(FreightAmount),  ID.ItemNo ,ID.MakeNo , ID.TechnicalGradeNo ,1,SCOrderNo	 from (
    select  sum(fd.PayableAmount + ISNULL(fd.AdvanceAmount,0)) * (ID.AcceptedQty * ID.Rate)/
				(select SUM(AcceptedQty* rate) from #ItemDetail where GateEntryNo = id.GateEntryNo and Rate>0) FreightAmount
		 , ID.ItemNo ,ID.MakeNo , ID.TechnicalGradeNo ,id.SCOrderNo
		 from
	Finance.FreightInvoiceMain fm 
	inner join Finance.FreightInvoiceDocumentDetail fd on fd.FreightInvoiceNo = fm.FreightInvoiceNo 
	inner join Finance.FreightInvoiceDocumentDetails fds on fds.FreightInvoiceNo = fd.FreightInvoiceNo 
					and (fds.GateEntryNo = fd.RefDocNo or fds.GRNNo = fd.RefDocNo )
	inner join #ItemDetail ID on ID.GateEntryNo = fds.GateEntryNo 				
	where fm.AuthorizedBy is not null
	group by ID.ItemNo ,ID.MakeNo , ID.TechnicalGradeNo ,id.GateEntryNo,ID.AcceptedQty , ID.Rate,id.SCOrderNo) ID
	group by   ID.ItemNo ,ID.MakeNo , ID.TechnicalGradeNo ,SCOrderNo

begin

insert into #Freight 
select SUM(FreightAmt), ItemNo , MakeNo , TechnicalGradeNo ,0 ,SCorderNo from (
select  (FreightAmount- isnull(DeductionAmount ,0)) * ID.AcceptedQty * ID.Rate 
			/(select SUM(AcceptedQty*rate) from #ItemDetail where ReferenceNo = ID.ReferenceNo and Rate<>0 ) FreightAmt
			, ID.ItemNo ,ID.MakeNo ,ID.TechnicalGradeNo ,id.SCOrderNo
		from [Purchase].[funcFreightOrderAmount] (@ItemDetail) foa
		inner join #ItemDetail ID on ID.ReferenceNo = isnull(foa.SCOrderAmendmentNo,foa.POAmendmentNo)  and ID.scOrderNo is not null
		where   not exists(select 1 from Purchase.FreightOrderOrderDetail fd where fd.SCOrderNo = id.SCOrderNo)
		 ) tt group by  ItemNo , MakeNo , TechnicalGradeNo ,SCorderNo
		
 
  if exists(select 1 from #ItemDetail where scOrderNo is not null)
  begin
		insert into #Freight 
		select isnull(sum(FreightAmount * id.AcceptedQty),0),id.ItemNo ,id.MakeNo ,id.TechnicalGradeNo ,0,sam.SCorderNo  from 
	Purchase.SCOrderMain  SAM
	inner join #ItemDetail id on id.SCOrderNo= SAM.SCOrderNo
	where 
	SAM.FreightTypeNo =1 -- To Pay only
	and SAM.FreightRateTypeNo  =1 -- per unit
	--and id.RouteIdentity is null  commented on 20 sep 2018 RRI case
	and not exists(select 1 from Finance.FreightInvoiceMain fm 
						inner join Finance.FreightInvoiceDocumentDetails  fd on fd.FreightInvoiceNo = fm.FreightInvoiceNo 
					 where fd.GateEntryNo = id.GateEntryNo and fm.AuthorizedBy is Not null )
	and not exists(select 1 from Purchase.FreightOrderOrderDetail fd where fd.SCOrderNo = SAM.SCOrderNo)
	group by id.ItemNo ,id.MakeNo ,id.TechnicalGradeNo ,sam.scorderNo
	
	insert into #Freight 
	select isnull(sum(FreightAmount),0),ItemNo ,MakeNo ,TechnicalGradeNo ,0 ,SCOrderNo from
	(
	select   gid.AcceptedQty * ((FreightAmount * said.Quantity * said.Rate)/
				(select SUM(pd.Quantity * pd.Rate) 
						from Purchase.SCOrderAmendmentItemDetail pd where  ItemFlowNo in(3,4) and 
								pd.SCOrderAmendmentNo = sam.SCOrderAmendmentNo and  pd.Rate<>0)
				  )/ said.Quantity FreightAmount
	
	   ,gid.ItemNo,gid.MakeNo ,gid.TechnicalGradeNo  ,gid.SCorderNo
	   from 
			#ItemDetail gid
			inner join #SCOrderDetail gsd on gsd.SCOrderNo = gid.SCOrderNo
			inner join Purchase.SCOrderAmendmentDetail sad on sad.SCOrderAmendmentNo = gsd.SCOrderAmendmentNo 
				and coalesce(sad.ExpenseHeadNo ,sad.ActivityNo,sad.ItemNo,0)= 
					coalesce(gsd.ExpenseHeadNo ,gsd.ActivityNo,gsd.ServiceItemNo,0)
					and isnull(sad.SequenceNo,0) = isnull(gsd.SequenceNo,0)
			inner join Purchase.SCOrderAmendmentItemDetail said on 
						said.SCOrderAmendmentNo = sad.SCOrderAmendmentNo
					and (coalesce(sad.ExpenseHeadNo ,sad.ActivityNo,sad.ItemNo)= 
					coalesce(said.ExpenseHeadNo ,said.ActivityNo,said.ServiceItemNo) or 
					(said.ExpenseHeadNo is null and said.ActivityNo is null and said.ServiceItemNo is null))
					and isnull(sad.SequenceNo,0) = isnull(said.SequenceNo,0)
					and ItemFlowNo in(3,4)
					and said.ItemNo = gid.ItemNo 
					and isnull(said.MakeNo,0) =isnull( gid.MakeNo ,0)
					and isnull(said.TechnicalGradeNo,0) = isnull(gid.TechnicalGradeNo ,0)
			inner join Purchase.SCOrderAmendmentMain SAM on SAM.SCOrderAmendmentNo = sad.SCOrderAmendmentNo
			where 
				SAM.FreightTypeNo =1 -- To Pay only
				and FreightRateTypeNo  =2 -- Fixed
			--	and RouteIdentity is null  commented on 20 sep 2018 RRI case
	and not exists(select 1 from Finance.FreightInvoiceMain fm 
						inner join Finance.FreightInvoiceDocumentDetails  fd on fd.FreightInvoiceNo = fm.FreightInvoiceNo 
					 where fd.GateEntryNo = gid.GateEntryNo and fm.AuthorizedBy is Not null  )
	and not exists(select 1 from Purchase.FreightOrderOrderDetail fd where fd.SCOrderNo = SAM.SCOrderNo)
	) tt group by ItemNo ,MakeNo ,TechnicalGradeNo ,scorderNo
	
	insert into #Freight 
	select  sum(FreightAmount),ItemNo ,MakeNo ,TechnicalGradeNo,0,SCorderNo

	inner join Finance.ARDTDCAccountRule ARAR on ARAR.TaxCodeMiscChargesDetailNo = TCD.TaxCodeMiscChargesDetailNo
		inner join Finance.ARDItemRule IR on IR.ARDNo = ARAR.ARDNo and IR.ItemNo = I.ItemNo 
					where tc.TaxCategoryNo =10 and TaxAccountTypeNo =27
					and tc.TaxAccountBasisNo=1
					union all
					select AccountNo, mcd.TaxCodeMiscChargesDetailNo
					from
					masterdata.mTaxCode tc 
					inner join masterdata.mTaxCodeMiscChargesDetail mcd on tc.taxcodeno=mcd.taxcodeno
					inner join Finance.ARDTDCAccountRule ar on 
						mcd.TaxCodeMiscChargesDetailNo=ar.TaxCodeMiscChargesDetailNo
					where tc.TaxAccountBasisNo=2
					 and tc.TaxCategoryNo =10 and TaxAccountTypeNo =27
					) tt) IR 
			where @IsManagingExpenseAccount=1
			and IsPaid = 0
				group by IR.AccountNo
			 having sum(I.FreightAmount)<>0	
			 
insert into #Posting
select cast(0 as bit) [Add to Basic]
				,'Purchase Offsetting Account' as Particular
				,convert(decimal(15,2),Null) [Debit Amount]
				,convert(decimal(15,2),sum(Amount + ISNULL(IncludedAmount,0) 
								+ ISNULL(FreightAmount,0) + isnull(ExpenseAmount,0) )) [Credit Amount]
				,  ItemAC.AccountNo [Account Name]
				,cast(null as varchar) [Ref. Detail]
				,cast(null as varchar) [Cost Center]
				,SUM(Amount + ISNULL(IncludedAmount,0)+ ISNULL(FreightAmount,0) + isnull(ExpenseAmount,0)) 
								[Amount]
				,2 [Type]
				,ROW_NUMBER()over(order by ItemAC.AccountNo  ) + (select isnull(max(SlNo),0) from #Posting) SlNo
				,null MiscChargeNo
				,null TaxCodeMiscChargesDetailNo
				,'' ItemNos
			from (select  0 Amount ,ItemNo ,MakeNo,TechnicalGradeNo  
							from #ItemDetail group by ItemNo,MakeNo,TechnicalGradeNo
							)I
 			outer apply( select ad.AccountNo from  Purchase.SCOrderAccountAssignmentDetail AD 
 						inner join #SCOrderDetail gsd on gsd.SCOrderNo = AD.SCOrderNo 
 						where  AD.Flag in('IM17','EH17') and AD.AccountTypeNo=54
 						 and AD.Id =coalesce(gsd.ExpenseHeadNo ,gsd.Activityno ,gsd.ServiceItemNo)
 						  group by ad.AccountNo
 						) ItemAC 
			left join #IncludedTax tt on I.ItemNo = tt.ItemNo 
							and ISNULL(I.MakeNo,0) = ISNULL(tt.MakeNo,0)	
							and ISNULL(I.TechnicalGradeNo,0) = isnull(tt.TechnicalGradeNo ,0)
			left join (select SUM(FreightAmount)FreightAmount ,ItemNo ,MakeNo ,TechnicalGradeNo 
							 from  #Freight where IsPaid =0
							  group by ItemNo ,MakeNo ,TechnicalGradeNo) FT on I.ItemNo = FT.ItemNo 
							and ISNULL(I.MakeNo,0) = ISNULL(FT.MakeNo,0)	
							and ISNULL(I.TechnicalGradeNo,0) = isnull(FT.TechnicalGradeNo ,0)	
										
			left join(select SUM( ExpenseAmount) ExpenseAmount,ItemNo ,MakeNo,TechnicalGradeNo  
							from #Expenses group by ItemNo,MakeNo,TechnicalGradeNo 
							) EH on I.ItemNo = EH.ItemNo 
							and ISNULL(I.MakeNo,0) = ISNULL(EH.MakeNo,0)	
							and ISNULL(I.TechnicalGradeNo,0) = isnull(EH.TechnicalGradeNo ,0)					
			where @IsManagingExpenseAccount=1
		group by  ItemAC .AccountNo 
		
			

if exists(select 1 from #Posting having abs(SUM([Credit Amount])- SUM([Debit Amount]))<=.5)
 begin
		update PD set [Debit Amount] = [Debit Amount] + ISNULL(diff,0)
		from
		#Posting PD
		cross join (select SUM([Credit Amount])- SUM([Debit Amount]) Diff from #Posting) tt
		where PD.SlNo  = 1
	
	update PD set [Credit Amount] = [Credit Amount] + ISNULL(diff,0)
		from
		#Posting PD
		cross join (select SUM([Debit Amount])- SUM([Credit Amount]) Diff from #Posting) tt
		where PD.SlNo  = 1
	
	update #Posting set [Debit Amount]=abs([Credit Amount]) ,[Credit Amount]=null,Type =1 ,Amount=abs([Credit Amount])
			where SlNo = 1 and [Credit Amount]<0
	
	update #Posting set [Credit Amount]=abs([Debit Amount]) ,Type =2 ,Amount=abs([Debit Amount]),[Debit Amount]=null
			where SlNo = 1 and [Debit Amount]<0
			
 end
