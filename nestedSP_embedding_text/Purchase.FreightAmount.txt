Stored procedure Purchase.FreightAmount: Get Freight Amount for Valuation. Called from Inventory.InvLotBillTaxTransaction, Inventory.IGRNBillDetailInsert.
Reads: GateEntry.InvGateEntryItemDetail, GateEntry.InvGateEntryMain, Inventory.GRNItemDetail, Inventory.GRNMain, Inventory.InvLotSummary, masterdata.mVendorLocationDetail, Purchase.FreightOrderAmendmentFreightBasisDetail, Purchase.FreightOrderAmendmentMain, Purchase.FreightOrderAmendmentToleranceRange, Purchase.FreightOrderFreightBasisDetail, Purchase.FreightOrderItemDetail, Purchase.FreightOrderMain, Purchase.FreightOrderOrderDetail, Purchase.FreightOrderTransporterDetail, Purchase.POAmendmentItemLRate, Purchase.POAmendmentMain, Purchase.POTransportationRoute, Purchase.SCOrderAmendmentMain
Parameters: @GRNNo int
Author: Yusuf, created 29-Mar-13

-- Create date: 29-Mar-13
-- Description:	Get Freight Amount for Valuation
-- =============================================
-- [Purchase].[FreightAmount]  10687
CREATE PROCEDURE [Purchase].[FreightAmount] 
(
	-- Add the parameters for the function here
	@GRNNo int
)
AS
BEGIN

	declare @LQ1 table(GateEntryNo int,Qty decimal(12,3))
	insert into @LQ1
	select LS.GateEntryNo,sum(LS.ReceivedQty*ls.itemRate)/sum(LS.ReceivedQty) Qty
	from Inventory.InvLotSummary LS 
		inner join (select GateEntryNo from inventory.grnitemdetail 
					where GRNNo=@GRNNo
					group by GateEntryNo) GRNID on GRNID.GateEntryNo=LS.GateEntryNo
	group by LS.GateEntryNo

	declare @LQ2 table(GateEntryNo int,Qty decimal(12,3))
	insert into @LQ2
	select ls.GateEntryNo,sum(LS.ReceivedQty) Qty
	from GateEntry.InvGateEntryItemDetail LS
		inner join (select GateEntryNo from inventory.grnitemdetail 
					where GRNNo=@GRNNo
					group by GateEntryNo) GRNID on GRNID.GateEntryNo=LS.GateEntryNo
	where ReceivedQty>0
	group by LS.GateEntryNo

	declare @InvGateEntryItemDetail table(GateEntryNo int,ChallanQty decimal(12,3),ReceivedQty  decimal(12,3)
											,FreightRate money,FreightOrderNo int,FreightRateTypeNo tinyint
											,LAmount money)
	insert into @InvGateEntryItemDetail
	select GEID.GateEntryNo,sum(ChallanQty)ChallanQty,sum(ReceivedQty)ReceivedQty
			,isnull(AVG(isnull(FOID.FreightAmount,POAM.FreightAmount)),0)FreightRate
			,MaxFreightOrderNo FreightOrderNo
			,max(isnull(FOID.FreightRateTypeNo,POAM.FreightRateTypeNo))FreightRateTypeNo
			--,COUNT(distinct POAM.PONo)GECount,COUNT(distinct FOOD.PONo)FOCount
			,sum( isnull( ILR.LRate,GEID.Rate) * GEID.ReceivedQty) LAmount
	from GateEntry.InvGateEntryItemDetail GEID
		inner join GateEntry.InvGateEntryMain GEM on GEM.GateEntryNo=GEID.GateEntryNo
		inner join (select GateEntryNo from inventory.grnitemdetail 
						where GRNNo=@GRNNo
						group by GateEntryNo) GRNID on GRNID.GateEntryNo=GEID.GateEntryNo
		Left join Purchase.POAmendmentItemLRate ILR on ILR.POAmendmentNo=GEID.POAmendmentNo
													and ILR.ItemNo = GEID.ItemNo 
													and isnull(ILR.MakeNo,0) = isnull(GEID.MakeNo ,0)
													and isnull(ILR.TechnicalGradeNo,0) = isnull(GEID.TechnicalGradeNo,0)
													and isnull(ILR.RouteIdentity,0) = isnull(GEID.RouteIdentity,0)
		
		inner join Purchase.POAmendmentMain POAM on POAM.POAmendmentNo=GEID.POAmendmentNo
		outer apply (select COUNT(distinct FOID.ItemNo*100+isnull(FOOD.RouteIdentity,0)) ICount,COUNT(distinct FOOD.PONo*100+isnull(FOOD.RouteIdentity,0)) PCount
								,max(FOOD.FreightOrderNo) MaxFreightOrderNo
								,max(TR.FreightRateTypeNo) FreightRateTypeNo
								,avg(isnull(TR.FreightAmount,bd.FreightRate)) FreightAmount
					from Purchase.FreightOrderOrderDetail FOOD 
						inner join Purchase.FreightOrderMain FOM on FOM.FreightOrderNo=FOOD.FreightOrderNo And FOM.AuthorizedBy is Not Null
						inner join Purchase.FreightOrderTransporterDetail TD on TD.FreightOrderNo=FOM.FreightOrderNo
																			and TD.TransporterNo=GEM.TransporterNo
						inner join Purchase.FreightOrderFreightBasisDetail bd on bd.FreightOrderNo=TD.FreightOrderNo
													and bd.FromLocationNo=gem.fromLocationNo and bd.ToLocationNo= gem.ToLocationNo
						left join Purchase.FreightOrderItemDetail FOID on FOOD.FreightOrderNo=FOID.FreightOrderNo
																	and FOID.ItemNo=GEID.ItemNo
						left join Purchase.POTransportationRoute TR on TR.PONo=FOOD.PONo
																and TR.RouteIdentity=FOOD.RouteIdentity
					where FOOD.PONo=POAM.PONo
						and isnull(FOOD.RouteIdentity,0) = isnull(GEID.RouteIdentity,0)
					) FOID 
	group by GEID.GateEntryNo,ICount,PCount,MaxFreightOrderNo
	union
	select GEID.GateEntryNo,sum(ChallanQty)ChallanQty,sum(ReceivedQty)ReceivedQty
			,isnull(AVG(SCOAM.FreightAmount),0)FreightRate
			,MAX(MaxFreightOrderNo)FreightOrderNo
			,SCOAM.FreightRateTypeNo
			--,COUNT(distinct POAM.PONo)GECount,COUNT(distinct FOOD.PONo)FOCount
			,cast(null as money)LAmount
	from GateEntry.InvGateEntryItemDetail GEID
		inner join GateEntry.InvGateEntryMain GEM on GEM.GateEntryNo=GEID.GateEntryNo
		inner join (select GateEntryNo from inventory.grnitemdetail 
						where GRNNo=@GRNNo
						group by GateEntryNo) GRNID on GRNID.GateEntryNo=GEID.GateEntryNo
		inner join Purchase.SCOrderAmendmentMain SCOAM on SCOAM.SCOrderAmendmentNo=GEID.SCOrderAmendmentNo
		cross apply (select COUNT(distinct FOID.ItemNo) ICount,COUNT(distinct FOOD.SCOrderNo) PCount
								,max(FOOD.FreightOrderNo) MaxFreightOrderNo
					from Purchase.FreightOrderOrderDetail FOOD 
						inner join Purchase.FreightOrderMain FOM on FOM.FreightOrderNo=FOOD.FreightOrderNo And FOM.AuthorizedBy is Not Null
						inner join Purchase.FreightOrderTransporterDetail TD on TD.FreightOrderNo=FOM.FreightOrderNo
																		and TD.TransporterNo=GEM.TransporterNo
						left join Purchase.FreightOrderItemDetail FOID on FOOD.FreightOrderNo=FOID.FreightOrderNo
																	and FOID.ItemNo=GEID.ItemNo
					where FOOD.SCOrderNo=SCOAM.SCOrderNo) FOID 
	group by GEID.GateEntryNo,ICount,PCount,MaxFreightOrderNo,SCOAM.FreightRateTypeNo
						
	-- Fill the table variable with the rows for your result set
	declare @Return TABLE 
		(
			-- Add the column definitions for the TABLE variable here
			GateEntryNo int,
			GRNNo int,
			FreightAmount money, 
			DeductionAmount money
		)	
	
	insert into @Return
	Select
		GEM.GateEntryNo [RefDocNo],
		null,
		cast(case(isnull(FOT.FreightRateTypeNo,GEID.FreightRateTypeNo)) 
				when 1 then coalesce(FOAFBD.FreightRate,FOAFBDD.FreightRate,FOT.FreightRate,GEID.FreightRate)*C.CalculationOnChange
				else coalesce(FOAFBD.FreightRate,FOAFBDD.FreightRate,FOT.FreightRate,GEID.FreightRate) end as decimal(15,2)) [Freight Amount],
		cast(
			isnull(GEID.LAmount/LQ2.Qty,LQ1.Qty)	
			  * (case 
					when GEID.ChallanQty<isnull(GEID.ReceivedQty,0)
					then 0 
					else
					 case (FOT.DeductionOn) 
							when 2 then case when GEID.ChallanQty-isnull(GEID.ReceivedQty,0)-isnull(TQ.ToleranceQty,0) < 0
												then 0 else GEID.ChallanQty-isnull(GEID.ReceivedQty,0)-isnull(TQ.ToleranceQty,0) end
							 when 1 then case when GEID.ChallanQty-isnull(GEID.ReceivedQty,0)-isnull(TQ.ToleranceQty,0) < 0
												then 0 else GEID.ChallanQty-isnull(GEID.ReceivedQty,0) end end
				end) as decimal(15,2)) [Deduction Amount]
		from GateEntry.InvGateEntryMain GEM
			inner join (select GateEntryNo from inventory.grnitemdetail 
						where GRNNo=@GRNNo
						group by GateEntryNo) GRNID on GRNID.GateEntryNo=GEM.GateEntryNo
			left join @LQ1 LQ1 on LQ1.GateEntryNo=GEM.GateEntryNo
			left join @LQ2 LQ2 on LQ2.GateEntryNo=GEM.GateEntryNo
			inner join masterdata.mVendorLocationDetail VLD on VLD.VendorLocationNo=GEM.VendorLocationNo
			inner join @InvGateEntryItemDetail GEID on GEID.GateEntryNo=GEM.GateEntryNo
			inner join (Select FOAM.FreightOrderAmendmentNo
							,FOAM.FreightOrderNo,FOAM.ToleranceUnitNo,FOAM.FreightRate,FOAM.DeductionOn,FOAM.RateTypeNo FreightRateTypeNo
							,FOAM.CalculationOn
						From Purchase.FreightOrderAmendmentMain as FOAM
							inner join Purchase.FreightOrderMain FOM on FOM.FreightOrderNo=FOAM.FreightOrderNo
																and FOM.AmendmentNo=FOAM.AmendmentNo) FOT on FOT.FreightOrderNo=GEID.FreightOrderNo
			left join Purchase.FreightOrderAmendmentFreightBasisDetail FOAFBD on FOAFBD.FreightOrderAmendmentNo=FOT.FreightOrderAmendmentNo
																			and FOAFBD.FromLocationNo=GEM.FromLocationNo
																			and FOAFBD.ToLocationNo=GEM.ToLocationNo
			left join Purchase.FreightOrderAmendmentFreightBasisDetail FOAFBDD on FOAFBDD.FreightOrderAmendmentNo=FOT.FreightOrderAmendmentNo
																			and GEM.Distance between FOAFBDD.DistanceFrom and FOAFBDD.DistanceTo
			outer apply (select cast(case when FOT.ToleranceUnitNo=2 then GEID.ChallanQty*Tolerance/100
									else Tolerance end as decimal(12,3)) ToleranceQty
							from Purchase.FreightOrderAmendmentToleranceRange FOATR
							where FOATR.FreightOrderAmendmentNo=FOT.FreightOrderAmendmentNo
								and GEID.ChallanQty between FOATR.RangeFrom and FOATR.RangeTo) TQ
			outer apply (select Case (FOT.CalculationOn)
							when 1 then GEID.ChallanQty
							when 2 then GEID.ReceivedQty
							when 3 then
								case when GEID.ChallanQty < GEID.ReceivedQty then GEID.ChallanQty
								Else GEID.ReceivedQty End
							when 4 then 1
							when 5 then GEID.ChallanQty
							Else 0 End CalculationOnChange) C
			where GEM.FreightTypeNo=1
	union all
	Select
		null,
		GRN.GRNNo [RefDocNo],
		cast(case(isnull(FOT.FreightRateTypeNo,GRNID.FreightRateTypeNo)) 
				when 1 then coalesce(FOAFBD.FreightRate,FOAFBDD.FreightRate,FOT.FreightRate,GRNID.FreightRate)*C.CalculationOnChange
				else coalesce(FOAFBD.FreightRate,FOAFBDD.FreightRate,FOT.FreightRate,GRNID.FreightRate) end as decimal(15,2)) [Freight Amount],
		cast((select sum(LS.ReceivedQty*LS.ItemRate)/sum(LS.ReceivedQty) 
				from Inventory.InvLotSummary LS
				where LS.DocumentNo=GRN.GRNNo
					and LS.TransactionFlag='IGRN')
			  * (case 
					when GRNID.ChallanQty<isnull(GRNID.ReceivedQty,0)
					then 0 
					else
					 case (FOT.DeductionOn) 
							when 2 then case when GRNID.ChallanQty-isnull(GRNID.ReceivedQty,0)-isnull(TQ.ToleranceQty,0) < 0
												then 0 else GRNID.ChallanQty-isnull(GRNID.ReceivedQty,0)-isnull(TQ.ToleranceQty,0) end
							when 1 then case when GRNID.ChallanQty-isnull(GRNID.ReceivedQty,0)-isnull(TQ.ToleranceQty,0) < 0
												then 0 else GRNID.ChallanQty-isnull(GRNID.ReceivedQty,0) end end
							
				end) as decimal(15,2)) [Deduction Amount]
		from Inventory.GRNMain GRN
			inner join masterdata.mVendorLocationDetail VLD on VLD.VendorLocationNo=GRN.VendorLocationNo
			inner join (select GRNID.GRNNo,sum(ChallanQty)ChallanQty,sum(ReceivedQty)ReceivedQty
								,isnull(AVG(isnull(FOID.FreightAmount,POAM.FreightAmount)),0)FreightRate
								,MaxFreightOrderNo FreightOrderNo
								,max(isnull(FOID.FreightRateTypeNo,POAM.FreightRateTypeNo))FreightRateTypeNo
								--,COUNT(distinct POAM.PONo)GECount,COUNT(distinct FOOD.PONo)FOCount
						from Inventory.GRNItemDetail GRNID
							inner join Inventory.GRNMain GRN on GRN.GRNNo=GRNID.GRNNo
							inner join Purchase.POAmendmentMain POAM on POAM.POAmendmentNo=GRNID.POAmendmentNo
							cross apply (select COUNT(distinct FOID.ItemNo*100+isnull(FOOD.RouteIdentity,0)) ICount,COUNT(distinct FOOD.PONo*100+isnull(FOOD.RouteIdentity,0)) PCount
													,max(FOOD.FreightOrderNo) MaxFreightOrderNo
													,MAX(TR.FreightRateTypeNo) FreightRateTypeNo
													,avg(ISNULL(bd.FreightRate, TR.FreightAmount)) FreightAmount
										from Purchase.FreightOrderOrderDetail FOOD 
											inner join Purchase.FreightOrderMain FOM on FOM.FreightOrderNo=FOOD.FreightOrderNo And FOM.AuthorizedBy is Not Null
											inner join Purchase.FreightOrderTransporterDetail TD on TD.FreightOrderNo=FOM.FreightOrderNo
																							and TD.TransporterNo=GRN.TransporterNo
											left join Purchase.FreightOrderItemDetail FOID on FOOD.FreightOrderNo=FOID.FreightOrderNo
																						and FOID.ItemNo=GRNID.ItemNo
											left join Purchase.POTransportationRoute TR on TR.PONo=FOOD.PONo
																					and TR.RouteIdentity=FOOD.RouteIdentity
											left join Purchase.FreightOrderFreightBasisDetail bd on bd.FreightOrderNo = FOOD.FreightOrderNo 
											and bd.FromLocationNo = TR.FromLocationNo and bd.ToLocationNo= TR.ToLocationNo 

										where FOOD.PONo=POAM.PONo
											and isnull(FOOD.RouteIdentity,0) = isnull(GRNID.RouteIdentity,0)) FOID 
						group by GRNID.GRNNo,ICount,PCount,MaxFreightOrderNo) GRNID on GRNID.GRNNo=GRN.GRNNo
			inner join (Select FOAM.FreightOrderAmendmentNo
							,FOAM.FreightOrderNo,FOAM.ToleranceUnitNo,FOAM.FreightRate,FOAM.DeductionOn,FOAM.RateTypeNo FreightRateTypeNo
							,FOAM.CalculationOn
						From Purchase.FreightOrderAmendmentMain as FOAM
							inner join Purchase.FreightOrderMain FOM on FOM.FreightOrderNo=FOAM.FreightOrderNo
																and FOM.AmendmentNo=FOAM.AmendmentNo) FOT on FOT.FreightOrderNo=GRNID.FreightOrderNo
			left join Purchase.FreightOrderAmendmentFreightBasisDetail FOAFBD on FOAFBD.FreightOrderAmendmentNo=FOT.FreightOrderAmendmentNo
																			and FOAFBD.FromLocationNo=GRN.FromLocationNo
																			and FOAFBD.ToLocationNo=GRN.ToLocationNo
			left join Purchase.FreightOrderAmendmentFreightBasisDetail FOAFBDD on FOAFBDD.FreightOrderAmendmentNo=FOT.FreightOrderAmendmentNo
																			and GRN.Distance between FOAFBDD.DistanceFrom and FOAFBDD.DistanceTo
			outer apply (select cast(case when FOT.ToleranceUnitNo=2 then GRNID.ChallanQty*Tolerance/100
									else Tolerance end as decimal(12,3)) ToleranceQty
							from Purchase.FreightOrderAmendmentToleranceRange FOATR
							where FOATR.FreightOrderAmendmentNo=FOT.FreightOrderAmendmentNo
								and GRNID.ChallanQty between FOATR.RangeFrom and FOATR.RangeTo) TQ
			outer apply (select Case (FOT.CalculationOn)
							when 1 then GRNID.ChallanQty
							when 2 then GRNID.ReceivedQty
							when 3 then
								case when GRNID.ChallanQty < GRNID.ReceivedQty then GRNID.ChallanQty
								Else GRNID.ReceivedQty End
							when 4 then 1
							when 5 then GRNID.ChallanQty
							Else 0 End CalculationOnChange) C
			where ((GRN.FormCode='IGRN' and GRN.GRNSourceNo=1)
					or (GRN.FormCode='SCMR' and GRN.GRNSourceNo in(1,6))
					or (GRN.GRNTypeNo=2))
				and GRN.FreightTypeNo=1
				and GRN.GRNNo=@GRNNo
				
	select*from @Return
END
