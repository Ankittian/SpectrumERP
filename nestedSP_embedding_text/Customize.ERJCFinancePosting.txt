CREATE  procedure [Customize].[ERJCFinancePosting]

(
 @OtherDataInXml xml
,@VoucherDate smalldatetime
,@CompanyNo tinyint
) as

Begin
begin try

declare @Return table ([Add to Basic] bit,Particular varchar(500),[Debit Amount] decimal(15,2),
								[Credit Amount] decimal(15,2),[Account Name] Int,Amount decimal(15,2),[Type] tinyint,
								MiscChargeNo smallint,TaxCodeMiscChargesDetailNo int,ItemNos varchar(5000),
								SlNo smallint identity(1,1))
Declare @SOTable as table (SalesOrderAmendmentNo int,SalesOrderNo int)
declare @ItemTable table (ItemNo int,BasicAmount money,Id int,Qty decimal(12,3),Rate money)
declare @FormNo smallint
select @FormNo=FormNo from GlobalData.FormMaster where FormCode='ERJC'

insert into @SOTable
SELECT v.value('SalesOrderAmendmentNo[1]','int') AS SalesOrderAmendmentNo , sam.SalesOrderNo 
			 FROM @OtherDataInXml.nodes('/DocumentElement/Table') x(v)
			 inner join Sales.SalesOrderAmendmentMain sam on sam.SalesOrderAmendmentNo = v.value('SalesOrderAmendmentNo[1]','int')
			
  
  
		insert into @Return([Add to Basic] ,Particular ,[Debit Amount] ,
						[Credit Amount] ,[Account Name] ,Amount ,[Type] ,
						MiscChargeNo ,TaxCodeMiscChargesDetailNo ,ItemNos)
		  select 0, 'COGS' , SUM(vd.BaseDrAmount+ BaseCrAmount), null, IR.AccountNo ,SUM(vd.BaseDrAmount+ BaseCrAmount),
					1 ,null, null, null
		  from Sales.DispatchChallan dc
		  inner join Finance.accVoucherDetail vd on vd.VoucherNo = dc.VoucherNo 
		  Outer apply (select max(IR.AccountNo) AccountNo--,AM.AccountName
								from Finance.ARDItemRule IR
								inner join Sales.DispatchChallanItemDetail id on id.ItemNo = IR.ItemNo and id.DispatchChallanNo = dc.DispatchChallanNo 
								inner join Finance.ARDIntermediatePostingAccountRule IPAR on IPAR.AccountNo=IR.AccountNo
																						and IPAR.IntermediateAccountTypeNo=1
								where IR.AccountTypeNo=54 ) IR
		  where 
			dc.CompanyNo = @CompanyNo and dc.DocumentDate <= @VoucherDate and 
		   exists (select 1 from Sales.DispatchChallanItemDetail did ,Sales.SalesOrderAmendmentMain sam ,@SOTable st
							where did.SalesOrderAmendmentNo = sam.SalesOrderAmendmentNo and st.SalesOrderNo = sam.SalesOrderNo
							and did.DispatchChallanNo = dc.DispatchChallanNo )
			and  vd.AccountNo in(select AM.AccountNo from Finance.ARDIntermediatePostingAccountRule IPAR 
									inner join masterdata.mAccountMaster AM on AM.AccountNo=IPAR.AccountNo
									where IntermediateAccountTypeNo = 31 and AM.AccountTypeNo=54) -- COGS Dispatch Suspense A/c
			Group By IR.AccountNo 
			union
			select 0, 'COGS Dispatch Suspense A/c' , null, SUM(vd.BaseDrAmount+ BaseCrAmount), vd.AccountNo ,SUM(vd.BaseDrAmount+ BaseCrAmount),
					2 ,null, null, null
		  from Sales.DispatchChallan dc
		  inner join Finance.accVoucherDetail vd on vd.VoucherNo = dc.VoucherNo 
		  where 
			dc.CompanyNo = @CompanyNo and dc.DocumentDate <= @VoucherDate and 
		   exists (select 1 from Sales.DispatchChallanItemDetail did ,Sales.SalesOrderAmendmentMain sam ,@SOTable st
							where did.SalesOrderAmendmentNo = sam.SalesOrderAmendmentNo and st.SalesOrderNo = sam.SalesOrderNo
							and did.DispatchChallanNo = dc.DispatchChallanNo )
			and  vd.AccountNo in(select AM.AccountNo from Finance.ARDIntermediatePostingAccountRule IPAR 
									inner join masterdata.mAccountMaster AM on AM.AccountNo=IPAR.AccountNo
									where IntermediateAccountTypeNo = 31 and AM.AccountTypeNo=54) -- COGS Dispatch Suspense A/c
			Group By vd.AccountNo 
					
			insert into @Return([Add to Basic] ,Particular ,[Debit Amount] ,
							[Credit Amount] ,[Account Name] ,Amount ,[Type] ,
							MiscChargeNo ,TaxCodeMiscChargesDetailNo ,ItemNos)
			select 0, 'Bill Till Final Stage (Revenue)' ,null, SUM(vd.BaseDrAmount+ BaseCrAmount), tt.AccountNo ,SUM(vd.BaseDrAmount+ BaseCrAmount),
						2 ,null, null, null 
			from 
			Sales.CommercialInvoiceMain cm
			inner join Finance.accVoucherDetail vd on vd.VoucherNo = cm.VoucherNo 
			inner join masterdata.mAccountMaster am on am.AccountNo = vd.AccountNo 
			outer apply(select top 1 AccountNo from masterdata.mAccountMaster where AccountTypeNo =62) tt
			where 
			cm.CompanyNo = @CompanyNo and cm.DocumentDate <= @VoucherDate and 
			exists (select 1 from Sales.CommercialInvoiceItemDetails cid, Sales.SalesOrderAmendmentMain sam, @SOTable st
							where cid.CommercialInvoiceNo = cm.CommercialInvoiceNo and cid.SalesOrderAmendmentNo = sam.SalesOrderAmendmentNo 
									and st.SalesOrderNo = sam.SalesOrderNo )
			and am.AccountTypeNo = 63 Group by tt.AccountNo 
			
				insert into @Return([Add to Basic] ,Particular ,[Debit Amount] ,
							[Credit Amount] ,[Account Name] ,Amount ,[Type] ,
							MiscChargeNo ,TaxCodeMiscChargesDetailNo ,ItemNos)
			select 0, 'Bill Till Final Stage (Liability)' ,SUM(vd.BaseDrAmount+ BaseCrAmount), null, vd.AccountNo ,SUM(vd.BaseDrAmount+ BaseCrAmount),
						1 ,null, null, null 
			from 
			Sales.CommercialInvoiceMain cm
			inner join Finance.accVoucherDetail vd on vd.VoucherNo = cm.VoucherNo 
			inner join masterdata.mAccountMaster am on am.AccountNo = vd.AccountNo 
			where 
			cm.CompanyNo = @CompanyNo and cm.DocumentDate <= @VoucherDate and 
			exists (select 1 from Sales.CommercialInvoiceItemDetails cid, Sales.SalesOrderAmendmentMain sam, @SOTable st
							where cid.CommercialInvoiceNo = cm.CommercialInvoiceNo and cid.SalesOrderAmendmentNo = sam.SalesOrderAmendmentNo 
									and st.SalesOrderNo = sam.SalesOrderNo )
			and am.AccountTypeNo = 63 Group by vd.AccountNo 
  
  				
	select [Add to Basic] ,Particular 
					,[Debit Amount] ,[Credit Amount] 
					,[Account Name] 
					,cast(null as varchar) [Ref. Detail] ,cast(null as varchar) [Cost Center] 
					,Amount ,[Type] 
					,SlNo
					,MiscChargeNo,TaxCodeMiscChargesDetailNo,ItemNos
			from @Return
			where (isnull([Debit Amount],0)+isnull([Credit Amount],0)>0
				or [Add to Basic]=1)




end try
begin catch
	select ERROR_MESSAGE ()
end catch
end
