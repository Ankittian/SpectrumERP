CREATE PROCEDURE [Information].[PurchaseRequestDetailForDocument] 
	-- Add the parameters for the stored procedure here
	@CompanyNo tinyint,
	@IndentNo int
AS
BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;

    -- Insert statements for procedure here
	
    
    
	declare @str varchar(max)
    select @str=masterdata.JSForInformation('MMPR')
    set @str=@str+masterdata.CSSForInformation('MMPR')
	set @str=@str+'<table class="altrowstable" id="alternatecolorPurchaseRequestDetailForDocument">'
	set @str=@str+'<Caption><Font color="Maroon"><h3 align="Left"><b>Purchase Request Detail</h3></Font></Caption>'
    set @str=@str+'<tr><th>Sl. No.</th><th>Item Description</th><th>Make</th><th>Technical Grade</th><th>Unit</th><th>Indent Qty.</th><th>Schedule Date/Qty.</th><th>CostCenter</th><th>PO Qty.</th><th>GRN Qty.</th><th>Purchase Bill Qty.</th><th>Freight Bill Qty.</th></tr>'
	set @str=@str
				 +(	REPLACE( 
				    REPLACE(
				    REPLACE(
					REPLACE( 
				    REPLACE(
				    REPLACE(
				    REPLACE(
				    replace(
					replace(
					replace(
					replace(
					replace(
					replace((    
								 Select  Row_number() over (order by ITM.ItemDescription) td1
										 ,ITM.ItemDescription td2
										,Isnull(MM.MakeName,'') td3
										,Isnull(TG.TechnicalGradeName,'') td4
										,UM.Alias td5
										,CONVERT(Decimal(10,3),ID.IndentQty) td6
										,Stuff((Select ','+ Convert(Varchar(20),ISD.ScheduleDate,103)+ ' - ' +Convert(Varchar(10),ISD.ScheduleQty )
										  From Inventory.IndentItemScheduleDetail ISD
										  Where ISD.IndentItemDetailNo=ID.IndentItemDetailNo
										  For XML path('')
										  ),1,1,'') td7
										,Isnull(CC.CostCenterName,'') td8
										,Isnull(CONVERT(Decimal(10,3),PO.POQty-Isnull(POR.POCancelQty,0)),0) td9
										,Isnull(Isnull(CONVERT(Decimal(10,3),PAM.[GRN Qty.]),CONVERT(Decimal(10,3),PAM1.[GRN Qty.])),0) td10
										,Isnull(CONVERT(Decimal(10,3),Bill.PBPB),0) as td11
									    ,Isnull(CONVERT(Decimal(10,3),Bill.FIFI),0) as td12   
								From Inventory.IndentMain IM
								Inner Join Inventory.IndentItemDetail ID ON ID.IndentNo=IM.IndentNo
								Inner Join masterdata.mItemMaster ITM ON ITM.ItemNo=ID.ItemNo
								Inner Join masterdata.mUnitMaster UM ON UM.UnitNo=ITM.UnitNo
								LEFT Join masterdata.mMakeMaster MM ON MM.MakeNo=ID.MakeNo
								LEFT Join masterdata.mCostCenterMaster CC ON CC.CostCenterNo=ID.CostCenterNo
								LEFT Join GlobalData.TechnicalGrade TG ON TG.TechnicalGradeNo=ID.TechnicalGradeNo
								-------------PO Qty--------------------------------------------------------------------
								Left Join (Select POID.IndentNo,POID.ItemNo,POID.MakeNo,POID.TechnicalGradeNo,sum(POID.POQty*(POID.FirstCF/POID.SecondCF)) POQty
										   From Purchase.POIndentDetail POID
										   Group by POID.ItemNo,POID.MakeNo,POID.TechnicalGradeNo,POID.IndentNo
										  )PO on PO.IndentNo=IM.IndentNo
											  And PO.ItemNo=ID.ItemNo
											  And isNull(PO.MakeNo,0)=isNull(ID.MakeNo,0)
											  And isNull(PO.TechnicalGradeNo,0)=isNUll(ID.TechnicalGradeNo,0)
								-------------PO Cancel Qty--------------------------------------------------------------------
								Left Join (Select POID.IndentNo,POID.ItemNo,POID.MakeNo,POID.TechnicalGradeNo,(SUM(POID.POQty*(POID.FirstCF/POID.SecondCF))-sum(GRNIND.AcceptedQty*(GRNI.FirstCF/GRNI.SecondCF))) POCancelQty
										   From Purchase.POIndentDetail POID
										   Inner Join Purchase.POItemDetail POI ON POID.ItemNo=POI.ItemNo
																				  And POI.StatusNo=4
																				  And POID.PONo=POI.PONo
																				  And isNull(POID.MakeNo,0)=isNull(POI.MakeNo,0)
																				  And isNull(POID.TechnicalGradeNo,0)=isNUll(POI.TechnicalGradeNo,0)
																				  
										   Inner Join Inventory.GRNItemIndentDetail GRNIND ON GRNIND.IndentNo=POID.IndentNo
										   Inner Join Inventory.GRNItemDetail GRNI ON GRNI.GRNItemDetailNo=GRNIND.GRNItemDetailNo
										    Inner Join Purchase.POAmendmentMain POA ON POA.POAmendmentNo=GRNI.POAmendmentNo
																				   And POA.PONo=POI.PONo
										   Group by POID.ItemNo,POID.MakeNo,POID.TechnicalGradeNo,POID.IndentNo--,POID.PONo,POID.POQty,POI.PONo
										  )POR on POR.IndentNo=PO.IndentNo
											   And POR.ItemNo=PO.ItemNo
											   And isNull(POR.MakeNo,0)=isNull(PO.MakeNo,0)
											   And isNull(POR.TechnicalGradeNo,0)=isNull(PO.TechnicalGradeNo,0)
								------------GRN Qty---------------------------------------------------------------------------
								Left Join (
										  Select SUM(GRNIND.AcceptedQty*(GRNI.FirstCF/GRNI.SecondCF)) [GRN Qty.],GRNIND.IndentNo,GRNI.ItemNo,GRNI.MakeNo,GRNI.TechnicalGradeNo
										  From Inventory.GRNItemDetail GRNI
										  Inner Join Inventory.GRNItemIndentDetail GRNIND ON GRNI.GRNItemDetailNo=GRNIND.GRNItemDetailNo 
										  Inner Join Purchase.POAmendmentMain PAM ON PAM.POAmendmentNo =GRNI.POAmendmentNo
										  Where PAM.IsTransportationRouteApplicable=0
										  Group By GRNIND.IndentNo,GRNI.ItemNo,GRNI.MakeNo,GRNI.TechnicalGradeNo
										  
										  Union All
										  
										  Select SUM(GRNI.AcceptedQty) [GRN Qty.],SLOD.IndentNo,GRNI.ItemNo,GRNI.MakeNo,null TechnicalGradeNo
										  From DiscreteRM.SINGateEntryItemDetail GRNI
										  Inner Join DiscreteRM.SINOrderDetail OD on OD.SINNo=GRNI.SINNo
										  Inner Join DiscreteRM.SINLotOrderDetail SLOD ON SLOD.ItemNo=GRNI.ItemNo
																					   And SLOD.SINNo=GRNI.SINNo
																					   And Isnull(SLOD.MakeNo,0)=Isnull(GRNI.MakeNo,0)
										  Inner Join Purchase.POAmendmentMain PAM ON PAM.POAmendmentNo =OD.POAmendmentNo
										  Where PAM.IsTransportationRouteApplicable=0
										  Group By SLOD.IndentNo,GRNI.ItemNo,GRNI.MakeNo
										  )	PAM on 	PAM.IndentNo =IM.IndentNo								
													And PAM.ItemNo =ID.ItemNo 
													AND (ISNULL(PAM.MakeNo,0)= ISNULL(ID.MakeNo,0) or ID.MakeNo is null)
													And ISnull(PAM.TechnicalGradeNo,0)=Isnull(ID.TechnicalGradeNo,0)
								--For Route Case					
								Left Join (
										  Select SUM(GRNIND.AcceptedQty*(GRNI.FirstCF/GRNI.SecondCF)) [GRN Qty.],GRNIND.IndentNo,GRNI.ItemNo,GRNI.MakeNo,GRNI.TechnicalGradeNo
										  From Inventory.GRNItemDetail GRNI
										  Inner Join Inventory.GRNItemIndentDetail GRNIND ON GRNI.GRNItemDetailNo=GRNIND.GRNItemDetailNo 
										  Inner Join Purchase.POAmendmentMain PAM ON PAM.POAmendmentNo =GRNI.POAmendmentNo
										  Where PAM.IsTransportationRouteApplicable=1
										  And GRNI.RouteIdentity% 0.1 = 0
										   Group By GRNIND.IndentNo,GRNI.ItemNo,GRNI.MakeNo,GRNI.TechnicalGradeNo
										  )	PAM1 on  PAM1.IndentNo =IM.IndentNo								
													 And PAM1.ItemNo =ID.ItemNo 
													 AND ISNULL(PAM1.MakeNo,0)= ISNULL(ID.MakeNo,0)
													 And ISnull(PAM1.TechnicalGradeNo,0)=Isnull(ID.TechnicalGradeNo,0)											
								---Calculate Bill Qty. (Service Invoice Left)
								Left Join (Select IndentNo,ItemNo,MakeNo,PBPB,FIFI
											 From(
											Select IM.IndentNo,InBD.ItemNo,InBD.MakeNo,InBD.BillQty
												  ,InBD.TransactionFlag
											From Inventory.IndentMain IM
											Inner Join Inventory.InvlotSummaryRefDetail  InVS ON InVS.IndentNo=IM.IndentNo
											Inner Join Inventory.InvLotBillDetail InBD on InVS.InvLotSummaryNo=InBD.InvLotSummaryNo
																					   And InBD.TransactionFlag In ('FIFI','PBPB'))t
											Pivot(sum(BillQty)
											For TransactionFlag in ([PBPB],[FIFI]))
												  As [Purchase Bill Qty.])Bill on Bill.IndentNo=IM.IndentNo
																			  And Bill.ItemNo=ID.ItemNo
																			  And isNull(Bill.MakeNo,0)=isNull(ID.MakeNo,0)
																			  
																			  
												   
								Where  IM.IndentNo=@IndentNo
								For xml path('tr')
			              ),'<td1>','<td style="text-align: Center;">')
			               ,'<td2>','<td>')
						   ,'<td3>','<td>')
						   ,'<td4>','<td>')
						   ,'<td5>','<td>')
						   ,'<td6>','<td  style="text-align: Right;">')
						   ,'<td7>','<td>')
						   ,'<td8>','<td>')
						   ,'<td9>','<td style="text-align: Right;">')
						   ,'<td10>','<td style="text-align: Right;">')
						   ,'<td11>','<td style="text-align: Right;">')
						   ,'<td12>','<td style="text-align: Right;">')
						   ,'<td13>','<td style="text-align: Right;">')
						)
								
				   
	set @str=@str+'</table>'
	
	select @str
END
