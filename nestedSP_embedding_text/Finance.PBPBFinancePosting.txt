CREATE PROCEDURE [Finance].[PBPBFinancePosting]
	@OtherDataInXml xml
	,@VendorLocationNo int
	,@BasicAmount Decimal(18,6)
	,@VoucherDate smalldatetime
	,@MiscChargeNos xml
	,@NetAmount Decimal(18,6)
	,@CompanyNo tinyint = null
	,@ItemNos varchar(max)
	,@ServiceTaxAbatementRuleNo int = null
	,@FBNo smallint = null
AS

BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;
begin try	

Declare @IsInterUnit bit
if exists (select 1 from masterdata.mVendorCard vc
			inner join masterdata.mVendorLocationDetail vld on vld.VendorNo = vc.VendorNo 
			where vld.VendorLocationNo =@VendorLocationNo and vc.VendorOfCompanyNo is not null)
begin
	select @IsInterUnit=1
end


declare @IsManagingExpenseAccount bit =1

declare @FormNo smallint ,@IncludedAmount Decimal(18,6),@ExchangeRate Decimal(18,6)
select @FormNo = FormNo from GlobalData.FormMaster where FormCode= 'PBPB'

 Create table #IncludedTax(IncludedAmount Decimal(18,6),ItemNo int,MakeNo smallint,TechnicalGradeNo tinyint)
 
 Create table #GRNGRAccount(GRAmount Decimal(18,6),GRNNo int,AccountNo int, ItemNo int,MakeNo smallint,
				TechnicalGradeNo tinyint,SINNo int)
				
 Create table #StockAccount(GRNNo int,ItemNo int,MakeNo smallint,TechnicalGradeNo tinyint ,
		 BalanceAmount decimal(18,6),IssuedAmount decimal(18,6),SINNo int , NextBalanceAmount decimal(18,6),
		 NextIssuedAmount decimal(18,6) , IsRouteCase bit,RefDocNo int,CompanyNo Tinyint,DivisionNo int
		,WarehouseNo smallint,MAPRate decimal(18,6),ClosingQty decimal(15,3),MaxMAPRate decimal(18,6),MinMAPRate decimal(18,6),
		IsCapital bit,NEWMAPRate decimal(18,6))
		  
declare @MiscChargeTable table(MiscChargeNo [smallint],Amount Decimal(18,6),[CalculationNature] [tinyint]
						,TaxCodeMiscChargesDetailNo int,IncludeInBasic tinyint,ChargeType tinyint)
declare @MiscFinancePosting table(Particular varchar(500),[Debit Amount] Decimal(18,6),
		[Credit Amount] Decimal(18,6),[Account Name] Int,[Amount] Decimal(18,6),[Type] tinyint,
		SlNo smallint identity(1,1),MiscChargeNo int,TaxCodeMiscChargesDetailNo int,ItemNos varchar(5000),
		IncludeInBasic tinyint)

Create table #StockTransfer(DocumentNo int,parentinvLotSummaryNo int,GRNNo int,InvLotSummaryNo int)


 declare @idoc int
 Create table #ItemDetailRaw (GRNNo int,POAmendmentNo int, ItemNo int,MakeNo smallint,TechnicalGradeNo tinyint,
		Qty decimal(18,6), Rate Decimal(18,6),IncludedTax Decimal(18,6),SINNo int,ExcisePostingNo int
		,Amount decimal(18,6),IsUpdateStock bit,IsDropShipment bit,GSTTaxRuleNo smallint,ItemTax varchar(500),TaxableValue decimal(15,2),ExchangeRate decimal(18,6))

 Create table #ItemDetail (GRNNo int,POAmendmentNo int, ItemNo int,MakeNo smallint,TechnicalGradeNo tinyint,
		Qty decimal(18,6), Rate Decimal(18,6),IncludedTax Decimal(18,6),SINNo int,ExcisePostingNo int
		,Amount decimal(18,6),IsUpdateStock bit,IsDropShipment bit,GSTTaxRuleNo smallint)

		
EXEC sp_xml_preparedocument @idoc OUTPUT, @OtherDataInXml
 
	Insert into #ItemDetailRaw 
	select * from OPENXML (@idoc, '/DocumentElement/Table',2) 
	with(GRNNo int,POAmendmentNo int , ItemNo int,MakeNo smallint , TechnicalGradeNo tinyint ,Qty decimal(18,6),Rate decimal(18,6),
	IncludedTax Decimal(18,6),SINNo int , ExcisePostingNo int ,Amount Decimal(18,6),IsUpdateStock bit,IsDropShipment bit,[GST Tax Rule] smallint,
	ItemTax varchar(1000) ,[Taxable Value] decimal(15,2),ExchangeRate decimal(18,6))
EXEC sp_xml_removedocument @idoc	 
if exists(select 1 from #ItemDetailRaw where TaxableValue >0)
 update #ItemDetailRaw set TaxableValue = 0 where TaxableValue is null

 Select @ExchangeRate = ExchangeRate from #ItemDetailRaw where IsNULL(ExchangeRate,0) <> 0

 Update #ItemDetailRaw 
 set Rate = Rate * @ExchangeRate,
 Amount = Amount * @ExchangeRate,
 IncludedTax = IncludedTax * @ExchangeRate,
 TaxableValue = TaxableValue * @ExchangeRate

 

Insert into #ItemDetail
	select GRNNo ,POAmendmentNo , ItemNo ,MakeNo ,TechnicalGradeNo ,sum(Qty), sum(Rate * Qty) /sum(Qty) Rate,sum(IncludedTax) ,SINNo ,
			ExcisePostingNo ,sum(Amount) ,IsUpdateStock ,IsDropShipment ,GSTTaxRuleNo 
	from 
	#ItemDetailRaw		
	group by GRNNo ,POAmendmentNo , ItemNo ,MakeNo ,TechnicalGradeNo ,SINNo ,ExcisePostingNo ,IsUpdateStock ,IsDropShipment ,GSTTaxRuleNo 
	 
EXEC sp_xml_preparedocument @idoc OUTPUT, @MiscChargeNos 

insert into @MiscChargeTable(MiscChargeNo,Amount,[CalculationNature],TaxCodeMiscChargesDetailNo,
				IncludeInBasic,ChargeType)
select MiscChargeNo,Amount,CalculationNature,[TDC Account Rule],isnull(IncludeInBasic,0)
			,[Charge Type]
From OPENXML (@idoc, '/DocumentElement/Table',2)  
	with (MiscChargeNo varchar(100),Amount Decimal(18,6) , CalculationNature tinyint,[TDC Account Rule] int,
			IncludeInBasic bit,[Charge Type] Tinyint) cte
		where ISNUMERIC(MiscChargeNo)=1	

EXEC sp_xml_removedocument @idoc

Update @MiscChargeTable 
set Amount = Amount * @ExchangeRate

Select @NetAmount = @NetAmount * @ExchangeRate


  			declare @ServiceTaxAbatementBasicAmount money
			select @ServiceTaxAbatementBasicAmount=SUM(Amount)
			from @MiscChargeTable	
			where MiscChargeNo in (select MiscChargeNo from Finance.ServiceTaxAbatementRuleChargeDetail
									where ServiceTaxAbatementRuleNo=@ServiceTaxAbatementRuleNo)
		
			select @ServiceTaxAbatementBasicAmount=@BasicAmount 
			where @ServiceTaxAbatementBasicAmount is null
			
			--set @ServiceTaxAbatementBasicAmount = @ServiceTaxAbatementBasicAmount + 
			--		isnull((select sum(case when dt.CalculationNature =1 then Amount else 0-Amount end) 
			--					from @MiscChargeTable dt
			--	inner join masterdata.mMiscCharges mc on mc.MiscChargeNo = dt.MiscChargeNo
				--where TaxGroupNo in(18,16)),0)
			
			delete from @MiscChargeTable where isnull(IncludeInBasic,0)=1
			
	insert into @MiscFinancePosting
					(Particular ,[Debit Amount] ,[Credit Amount] ,[Account Name] 
					,[Amount] ,[Type]
					,MiscChargeNo ,TaxCodeMiscChargesDetailNo ,ItemNos,IncludeInBasic)
			select 
				 MC.MiscChargeName+isnull(' ('+AppendAccountName+')','')
				,convert(decimal(15,2),case(s.CalculationNature) when 1 then isnull(tt.Amount,s.Amount)
																else null end)
				,convert(decimal(15,2),case(s.CalculationNature) when 1 then null
																else isnull(tt.Amount,s.Amount) end)
				,tt.AccountNo 
				,isnull(tt.Amount,s.Amount)
				,s.CalculationNature
				,s.MiscChargeNo
				,tcmcd.TaxCodeMiscChargesDetailNo
				,tt.ItemNos
				,s.IncludeInBasic
			from 
				@MiscChargeTable s
				inner join masterdata.mMiscCharges MC on MC.MiscChargeNo=s.MiscChargeNo -- and s.ChargeType =2
				left join masterdata.mTaxCodeMiscChargesDetail tcmcd on tcmcd.TaxCodeMiscChargesDetailNo=s.TaxCodeMiscChargesDetailNo
				outer apply (select AccountNo,Amount,AppendAccountName,ItemNos
							 from Finance.funcTaxesPosting(tcmcd.TaxCodeMiscChargesDetailNo, 
															s.MiscChargeNo,
															s.Amount,
															0,
															@BasicAmount,
															@ItemNos,
															@FormNo,
															@VoucherDate))tt
			Union All
			select AP.Particular Particular 
				,case(AP.Type) when 1 then AP.Amount else null end [Debit Amount]
				,case(AP.Type) when 2 then AP.Amount else null end [Credit Amount] 
				,AP.AccountNo [Account Name] 
				,AP.Amount [Amount]
				,AP.Type [Type] 
				,MiscChargeNo ,TaxCodeMiscChargesDetailNo ,CAST(null as varchar(5000)) ItemNos,0
			from 
			#ItemDetailRaw id
			cross apply ( select sum(convert(money,col2)) * @ExchangeRate TaxAmount from masterdata.Split_RowColumn( ItemTax,'~',',') dt
							where exists(select 1 from  masterdata.mMiscCharges mc 
							inner join masterdata.mTaxGroupDependOn dn on dn.DependOnTaxGroupNo = mc.TaxGroupNo
							inner join masterdata.mTaxGroup tg on tg.TaxGroupNo = dn.TaxGroupNo
							where tg.TaxCategoryNo = 14 and mc.MiscChargeNo = dt.Col1 ) ) it
			cross apply [Finance].[funcServiceTaxAbatementPosting] (id.GSTTaxRuleNo, isnull(TaxableValue,Amount) + isnull(TaxAmount,0),@FBNo) AP
																

	 
;with CTE
			as
			(select ld.DocumentNo, ls.InvlotSummaryNo parentinvLotSummaryNo ,id.GRNNo, ls.InvLotSummaryNo 
					from Inventory.InvLotDetail ld 
				inner join inventory.invlotSUmmary ls on ls.InvlotsummaryNo = ld.invlotsummaryNo
				inner join #ItemDetail id on id.GRNNo = ls.DocumentNo and ls.transactionFlag='IGRN' 
				 where 
					ld.TransactionFlag ='IMST' group by ld.DocumentNo,ls.InvlotSummaryNo,id.GRNNo
			union all
			select ld.DocumentNo ,cte.parentinvLotSummaryNo ,CTE.GRNNo  , ls.InvLotSummaryNo 
					from Inventory.InvLotSummary ls 
					Inner Join Inventory.InvLotDetail ld on ls.InvLotSummaryNo = ld.InvLotSummaryNo 
					Inner Join Inventory.InvlotSummaryRefDetail RD on RD.InvLotSummaryNo=ls.InvLotSummaryNo
					Inner Join CTE  CTE  on (ls.invStockTransferNo = CTE.DocumentNo  ----- Lot created against Stock transfer		
										Or
										(LD.IssuedQty>0 
										 And LS.DocumentNo=CTE.DocumentNo 
										 And LS.TransactionFlag='IMSt' 
										 ) ----- Issued against stock transfer
										 )
										 And RD.ParentInvLotSummaryNo =CTE.InvLotSummaryNo 
					where ld.TransactionFlag ='IMST'
		    union all
		    select lld.DocumentNo ,cte.parentinvLotSummaryNo ,CTE.GRNNo  , ls.InvLotSummaryNo 
					from Inventory.InvLotSummary ls , Inventory.InvLotDetail ld, CTE , Inventory.GRNMain grn, Inventory.InvLotSummary lls, Inventory.InvLotDetail lld
					,Inventory.InvlotSummaryRefDetail ILSR
					where ls.InvLotSummaryNo = ld.InvLotSummaryNo and ld.TransactionFlag ='IMMI'
					and (ls.invStockTransferNo = CTE.DocumentNo 
						 Or
						(LD.IssuedQty>0 
						And LS.DocumentNo=CTE.DocumentNo 
						And LS.TransactionFlag='IMST' 

						) ----- Issued against stock transfer
						)
					and ILSR.InvLotSummaryNo = ls.InvLotSummaryNo 
					and ILSR.ParentInvLotSummaryNo = CTE.InvLotSummaryNo 
					and grn.IssueNo = ld.DocumentNo and grn.GRNNo = lls.DocumentNo and lls.TransactionFlag ='IGRN'
							and lld.TransactionFlag ='IMST' and lld.InvLotSummaryNo = lls.InvLotSummaryNo 
)
insert into #StockTransfer  select DocumentNo,parentinvLotSummaryNo,GRNNo,InvLotSummaryNo from CTE 
					GROUP BY DocumentNo,parentinvLotSummaryNo,GRNNo,InvLotSummaryNo 

if exists(select 1 from Purchase.POAmendmentMain pam 
			inner join #ItemDetail id on id.POAmendmentNo = pam.POAmendmentNo and IndentTypeNo =1)-- capital
	set @IsManagingExpenseAccount=0
else			
	select @IsManagingExpenseAccount = IsManagingExpenseAccount from Finance.Setting where CompanyNo = @CompanyNo 
 
if exists(select 1 from #ItemDetail where SINNo is not null)
begin

 insert into #StockAccount( SINNo , ItemNo ,MakeNo ,TechnicalGradeNo ,BalanceAmount ,IssuedAmount,RefDocNo
	, CompanyNo ,DivisionNo ,WarehouseNo ,IsCapital)
  select id.SINNo  , id.ItemNo ,ID.MakeNo ,ID.TechnicalGradeNo ,
	((((id.Amount) +  ISNULL(sum(ID.IncludedTax),0)) -(inv.Rate* inv.ReceivedQty))* inv.BalanceQty)/inv.ReceivedQty ,
	((((id.Amount) +  ISNULL(sum(ID.IncludedTax),0)) -(inv.Rate* inv.ReceivedQty))* inv.IssuedQty)/inv.ReceivedQty
	, ID.POAmendmentNo,inv.CompanyNo,inv.DivisionNo ,inv.WarehouseNo ,inv.IsCapital 
	 from
 (select  ls.CompanyNo,ls.DivisionNo ,ls.WarehouseNo ,ls.IsCapital,PID.SINNo DocumentNo, PID.ItemNo ,PID.MakeNo,PID.TechnicalGradeNo ,ls.POAmendmentNo ,
			 SUM(ls.ReceivedQty * SecondCF/FirstCF) ReceivedQty,
			SUM(case when pam.IndentTypeNo =1 then ls.ReceivedQty else BalanceQty end*SecondCF/FirstCF) BalanceQty ,
			SUM(case when pam.IndentTypeNo =1 then 0 else ls.IssuedQty end* SecondCF/FirstCF) IssuedQty, 
			convert(decimal(18,6),SUM( FirstCF*ls.ReceivedQty*(ld.VRate - ISNULL(RatePerUnit,0)) /SecondCF)/SUM(ls.ReceivedQty)) Rate
				 from Inventory.InvLotSummary ls
				 left join Purchase.POAmendmentMain pam on pam.POAmendmentNo = ls.POAmendmentNo 
				 inner join Inventory.InvLotBillDetail ld on ld.invlotSummaryNo = ls.invlotSummaryNo 
						and ld.TransactionFlag in('DRSN','RSLE') and ld.isincluded=1
				left join ( select SUM( RatePerUnit  ) RatePerUnit ,itd.InvLotSummaryNo
						from  Inventory.InvLotTaxDetail itd ,Inventory.invLotBillDetail bd
							 
						 where itd.InvLotBillDetailNo = bd.InvLotBillDetailNo 
								and itd.InvLotDetailNo =  bd.InvLotDetailNo and TransactionFlag in('DRSN','RSLE')
							and itd.MiscChargeNo is null group by itd.InvLotSummaryNo )
							lst  on lst.InvLotSummaryNo = ls.InvLotSummaryNo 
				inner join #ItemDetail PID on 
						PID.ItemNo = ls.ItemNo and 
						(isnull(PID.MakeNo,0) = isnull(ls.MakeNo,0) or ls.MakeNo is null 
						)
						and isnull(PID.TechnicalGradeNo,0) = isnull(ls.TechnicalGradeNo,0) 
						and isnull(ls.POAmendmentNo,0) = isnull(PID.POAmendmentNo ,0)
				 left join DiscreteRM.SINLotEntryMain slm on slm.SINNo = PID.SINNo and slm.ItemNo = PID.ItemNo 
								and isnull(slm.MakeNo,0) = isnull(PID.MakeNo ,0)
				 where
				  ((PID.SINNo=ls.DocumentNo and ls.TransactionFlag ='DRSN')
						or (slm.SINLotEntryNo=ls.DocumentNo and ls.TransactionFlag ='RSLE'))
				group by  PID.ItemNo ,PID.MakeNo,PID.TechnicalGradeNo ,ls.POAmendmentNo ,PID.SINNo
				, ls.CompanyNo,ls.DivisionNo ,ls.WarehouseNo ,ls.IsCapital
				) inv 
  inner join #ItemDetail ID on ID.SINNo = inv.DocumentNo and id.itemno = inv.ItemNo 
		and ISNULL(inv.POAmendmentNo,0) = isnull(ID.POAmendmentNo ,0)
		and isnull(inv.MakeNo,0) = isnull(ID.MakeNo,0) 
		and isnull(inv.TechnicalGradeNo,0) = isnull(ID.TechnicalGradeNo ,0)
  group by  id.ItemNo ,ID.MakeNo ,ID.TechnicalGradeNo ,id.SINNo ,ID.POAmendmentNo 
	,inv.BalanceQty,inv.IssuedQty  ,inv.Rate ,id.Rate , id.Qty,inv.receivedqty,id.Amount
	,inv.CompanyNo,inv.DivisionNo ,inv.WarehouseNo ,inv.IsCapital
	

insert into #GRNGRAccount(GRAmount ,SINNo,AccountNo , ItemNo ,MakeNo ,TechnicalGradeNo )
 select vd.BaseCrAmount * SUM(gid.Qty  * gid.Rate)/TotalAmt
	,grn.SINNo , IR.AccountNo  AccountNo,id.ItemNo ,ID.MakeNo ,ID.TechnicalGradeNo 
	 from
	DiscreteRM.SINMain  grn
  inner join DiscreteRM.SINLotOrderDetail  gid on gid.SINNo = grn.SINNo
  left join DiscreteRM.SINLotEntryMain slem on slem.SINNo = gid.SINNo and slem.ItemNo = gid.ItemNo 
				and isnull(slem.MakeNo ,0)= isnull(gid.MakeNo,0)
  inner join Finance.accVoucherDetail vd on vd.VoucherNo = isnull(grn.VoucherNo ,slem.VoucherNo )
  inner join #ItemDetail ID on ID.SINNo = grn.SINNo and id.itemno = gid.itemno 
		and ISNULL(gid.POAmendmentNo,0) = isnull(ID.POAmendmentNo ,0)
		and isnull(gid.MakeNo,0) = isnull(ID.MakeNo,0) 
		and ID.SINNo = gid.SINNo
  inner join (select sum(pgd.Qty * pgd.Rate) TotalAmt ,pgd.SINNo ,slem.sinLotEntryNo
	 from DiscreteRM.SINLotOrderDetail pgd 
	  left join DiscreteRM.SINLotEntryMain slem on slem.SINNo = pgd.SINNo and slem.ItemNo = pgd.ItemNo 
				and isnull(slem.MakeNo ,0)= isnull(pgd.MakeNo,0)
	 group by pgd.SINNo ,slem.sinLotEntryNo) grdt on grdt.SINNo = grn.SINNo	
			and isnull(grdt.sinLotEntryNo,0)= isnull(slem.sinLotEntryNo,0)
    --outer apply (
				--select AccountNo from Purchase.POAccountAssignmentDetail ad 
				--inner join Purchase.POAmendmentMain pam on pam.PONo = ad.PONo 
				--where ad.Flag ='IM11' and ad.AccountTypeNo =54 and ad.Id = Id.ItemNo 
				--	and pam.POAmendmentNo = Id.POAmendmentNo 
				--group by AccountNo	
				--union all
				--select max(IR.AccountNo) AccountNo
				--from Finance.ARDItemRule IR
				--Inner join Finance.ARDIntermediatePostingAccountRule IPR on IPR.ARDNo = IR.ARDNo 
				--and IPR.AccountNo = IR.AccountNo and IPR.IntermediateAccountTypeNo =11 -- GR SUSPENSE 
				--inner join masterdata.mAccountMaster AM on AM.AccountNo=IR.AccountNo
				--												and AM.AccountTypeNo=54-- intermeidate
				--where IR.ItemNo = ID.ItemNo and ID.POAmendmentNo is null
				--)  IR
				 Left Join (
				select AccountNo,ad.Id ItemNo,PAM.POAmendmentNo from Purchase.POAccountAssignmentDetail ad 
				inner join Purchase.POAmendmentMain pam on pam.PONo = ad.PONo 
				where ad.Flag ='IM11' and ad.AccountTypeNo =54 --and ad.Id = Id.ItemNo 
					--and pam.POAmendmentNo = Id.POAmendmentNo 
				group by AccountNo,ad.Id,PAM.POAmendmentNo 	
				union 
				select max(IR.AccountNo) AccountNo,IR.ItemNo,NULL POAmendmentNo
				from Finance.ARDItemRule IR
				Inner join Finance.ARDIntermediatePostingAccountRule IPR on IPR.ARDNo = IR.ARDNo 
				and IPR.AccountNo = IR.AccountNo and IPR.IntermediateAccountTypeNo =11 -- GR SUSPENSE 
				inner join masterdata.mAccountMaster AM on AM.AccountNo=IR.AccountNo
																and AM.AccountTypeNo=54-- intermeidate
				--where IR.ItemNo = ID.ItemNo and ID.POAmendmentNo is null
				Group by IR.ItemNo
				)  IR on IR.ItemNo = Id.ItemNo and (IR.POAmendmentNo = ID.POAmendmentNo or ID.POAmendmentNo is NULL)
	where IR.AccountNo = vd.AccountNo 		
	group by grn.SINNo ,vd.BaseCrAmount, IR.AccountNo  ,id.ItemNo ,ID.MakeNo ,
	ID.TechnicalGradeNo ,TotalAmt
	
end
else
begin
 
  if exists (select 1 from #ItemDetail id inner join Purchase.POAmendmentMain pam on 
		pam.POAmendmentNo = id.POAmendmentNo and pam.IndentTypeNo =5) -- drop shipment
 begin
	set @IsManagingExpenseAccount =0
	insert into #StockAccount( GRNNo , ItemNo ,MakeNo ,TechnicalGradeNo ,BalanceAmount ,
			IssuedAmount,RefDocNo ,CompanyNo ,DivisionNo ,WarehouseNo  ,IsCapital )
 select id.GRNNo , id.ItemNo ,id.MakeNo ,id.TechnicalGradeNo ,
	 id.Amount + ISNULL( id.IncludedTax,0)  - vd.BaseCrAmount * SUM(case when grn.IsReturnThroughDC =1 then gid.AcceptedQty + ISNULL(gid.RejectedQty ,0) else gid.AcceptedQty  end * gid.Rate)/TotalAmt 
	 , null
	 , gid.POAmendmentNo ,grn.CompanyNo ,grn.DivisionNo ,isnull(grn.WarehouseNo , gid.WareHouseNo ),0
	 from
	Inventory.GRNMain  grn
  inner join Inventory.GRNItemDetail   gid on gid.GRNNo = grn.GRNNo
  inner join Finance.accVoucherDetail vd on vd.VoucherNo = grn.VoucherNo 
   inner join (select sum( case when pgm.IsReturnThroughDC=1 then pgd.AcceptedQty + ISNULL(pgd.RejectedQty ,0) else pgd.AcceptedQty end  * pgd.Rate) TotalAmt ,pa.AccountNo,pgd.GRNNo 
	 from 
	 Inventory.GRNMain pgm 
	 inner join Inventory.GRNItemDetail pgd on pgd.GRNNo = pgm.GRNNo 
	 inner join Purchase.POAmendmentMain ppam on ppam.POAmendmentno = pgd.POAmendmentno
	 inner join Purchase.POAccountAssignmentDetail pa on pa.PONo= ppam.PONo and 
					pgd.itemno = pa.Id and pa.flag='IM11' and pa.AccountTypeNo =54
	 where exists(select 1 from #ItemDetail where grnno = pgd.GRNNo)
	  group by pgd.GRNNo ,pa.AccountNo  ) grdt on grdt.GRNNo = grn.GRNNo
				and grdt.accountno=vd.AccountNo
  inner join #ItemDetail ID on ID.GRNNo = grn.GRNNo and id.itemno = gid.itemno 
		and ISNULL(gid.POAmendmentNo,0) = isnull(ID.POAmendmentNo ,0)
		and isnull(gid.MakeNo,0) = isnull(ID.MakeNo,0) 
		and ID.GRNNo = gid.GRNNo
  group by id.GRNNo ,grdt.AccountNo,id.ItemNo ,id.MakeNo,id.TechnicalGradeNo,id.Amount,vd.BaseCrAmount,TotalAmt
	, id.IncludedTax, gid.POAmendmentNo , gid.POAmendmentNo ,grn.CompanyNo ,grn.DivisionNo ,isnull(grn.WarehouseNo , gid.WareHouseNo )

 end	 
 else if exists (select 1 from Inventory.GRNItemDetail gd
			inner join #ItemDetail id on id.GRNNo =gd.GRNNo and gd.RouteIdentity is not null ) -- route case
		and not exists (select 1 from #ItemDetail ID , Purchase.POAmendmentMain pam , Purchase.POExpenseHead ped ,Purchase.POExpenseHeadOutputDetail pod
							  where ID.POAmendmentNo = pam.POAmendmentNo and pam.PONo = ped.PONo  and pod.POExpenseDetailNo = ped.POExpenseDetailNo 
						)
 begin	
 insert into #StockAccount( GRNNo , ItemNo ,MakeNo ,TechnicalGradeNo ,NextBalanceAmount ,
			NextIssuedAmount,BalanceAmount ,IssuedAmount ,IsRouteCase,refDocNo ,CompanyNo,DivisionNo,WarehouseNo ,IsCapital )
select  id.GRNNo , id.ItemNo ,ID.MakeNo ,ID.TechnicalGradeNo , 
			 inv.BalanceQty * ((((id.Amount) +  ISNULL(sum(ID.IncludedTax),0)) -(inv.Rate* inv.receivedqty))/inv.receivedqty)NextBalanceAmount,
			 inv.issuedQty * ((((id.Amount) +  ISNULL(sum(ID.IncludedTax),0)) -(inv.Rate* inv.receivedqty))/inv.receivedqty) NextIssuedAmount,
			 inv.NextbalanceQty * ((((id.Amount) +  ISNULL(sum(ID.IncludedTax),0)) -(inv.Rate* inv.receivedqty))/inv.receivedqty) BalanceAmount,
			 inv.nextIssuedQty * ((((id.Amount) +  ISNULL(sum(ID.IncludedTax),0)) -(inv.Rate* inv.receivedqty))/inv.receivedqty) IssuedAmount ,1
			 ,ID.POAmendmentNo,CompanyNo,DivisionNo,WarehouseNo ,IsCapital
				 from
			 (select  ls.DocumentNo, PID.ItemNo ,PID.MakeNo,PID.TechnicalGradeNo ,ls.POAmendmentNo ,
					SUM(ls.ReceivedQty) ReceivedQty, SUM(lls.ReceivedQty) NextReceivedQty,
					--SUM(case when ls.IsCapital=1 then ls.ReceivedQty else ls.BalanceQty end)BalanceQty ,
