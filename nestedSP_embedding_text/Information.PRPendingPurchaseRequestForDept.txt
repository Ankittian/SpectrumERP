CREATE PROCEDURE [Information].[PRPendingPurchaseRequestForDept] 
	-- Add the parameters for the stored procedure here
	@CompanyNo tinyint,
	@DocumentDate smalldatetime,
	@DeptNo int
AS
BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;

    -- Insert statements for procedure here
	
    
	declare @str varchar(max)
    select @str=masterdata.JSForInformation('MMPR')
    set @str=@str+masterdata.CSSForInformation('MMPR')
	set @str=@str+'<table class="altrowstable" id="alternatecolorPRPUPRInfo">'
	set @str=@str+'<Caption><Font color="Maroon"><h3 align="Left"><b>Pending Purchase Request</b></h3></Font></Caption>'
    set @str=@str+'<tr><th>Sl. No.</th><th>Division</th><th>Document No.</th><th>Document Date</th><th>Source Document</th><th>Ref. No.</th><th>Ref. Date</th><th>PR Type</th><th>Requested By</th><th>Created By</th><th>Authorized By</th></tr>'
	set @str=@str
				 +( REPLACE(
				    REPLACE( 
				    REPLACE(
				    REPLACE(
				    REPLACE(
				    REPLACE(
				    replace(
					replace(
					replace(
					replace(
					replace((    
							Select Row_number() over (order by IM.DocumentDate desc) as td1, 
								   isnull(DivM.DivisionName,'') td2, 
								   IM.DocumentNoYearly td3,
								   convert(varchar(20),IM.DocumentDate,103) td4, 
								   IA.IndentAgainst td5,
								   isNull(isNull(isNull(sam.DocumentNoYearly,IRM.DocumentNoYearly) 
								                       , mpm.DocumentNoYearly),'') td6 , 
									convert(varchar(20),isNull( isNull(isNull(sam.DocumentDate
									                    ,IRM.DocumentDate) , mpm.DocumentDate),''),103) td7, 
								   IndentType td8,
								   IM.RequestedBy td9,
								   SL.Printingname td10, 
								   isnull(SLA.Printingname,'') td11
				 From Inventory.IndentMain IM
				INNER JOIN payroll.tblDeptMaster DM on IM.DeptNo = DM.DeptNo
				Left JOIN payroll.tblDivisionMaster DivM on DivM.DivisionNo = DM.DivisionNo 
				INNER JOIN GlobalData.IndentAgainst IA on IA.IndentAgainstNo = IM.IndentAgainstNo
				Inner Join Inventory.IndentItemDetail ID ON ID.IndentNo=IM.IndentNo
				left Join Inventory.MaterialPlanningMain  mpm on mpm.MaterialPlanningNo=IM.RefDocumentNo 
								and IM.IndentAgainstNo =2
				left Join Inventory.IndentRequisitionMain  IRM on IRM.IndentRequisitionNo =IM.RefDocumentNo 
								and IM.IndentAgainstNo =3
				left join Purchase.SCOrderAmendmentMain sam on sam.SCOrderAmendmentNo = IM.RefDocumentNo and 
								  IM.IndentAgainstNo =4
				left join globaldata.IndentType it on it.indentTypeNO  = im.IndenttypeNO
				Inner JOIN Security.Login SL on SL.LoginNo =  IM.CreatedBy
				Left JOIN Security.Login SLA on SLA.LoginNo =  IM.AuthorizedBy
												
				Where (ID.IndentQty-Isnull(ID.POQty,0)) > 0	
				and IM.DeptNo=@DeptNo and IM.CompanyNo=@CompanyNo and IM.DocumentDate<=@DocumentDate	
				Group by     DivM.DivisionName, sam.DocumentNoYearly,IRM.DocumentNoYearly,mpm.DocumentNoYearly,IndentType,
						   IM.DocumentNoYearly ,sam.DocumentDate,IRM.DocumentDate,mpm.DocumentDate,IM.DocumentDate ,IA.IndentAgainst,SL.Printingname, 
						   SLA.Printingname ,IM.RequestedBy
			        for xml path('tr')),'<td1>','<td style="text-align: center;">')
			                     ,'<td2>','<td>')
								 ,'<td3>','<td>')
								 ,'<td4>','<td style="text-align: center;">')
								 ,'<td5>','<td>')
								 ,'<td6>','<td>')
								 ,'<td7>','<td style="text-align: center;">')
								 ,'<td8>','<td>')
								 ,'<td9>','<td>')
								 ,'<td10>','<td>')
								 ,'<td11>','<td>'))
	set @str=@str+'</table>'
	
	select @str
END
