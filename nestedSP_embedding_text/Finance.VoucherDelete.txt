CREATE PROCEDURE [Finance].[VoucherDelete]
(
 @VoucherNo  int
 ,@LoginNo SmallInt
 
,@OutputStatusNo tinyint = null output
,@OutputStatusRemarks varchar(4000) = null output

)
AS

BEGIN
	SET NOCOUNT ON;
	
	BEGIN TRY
	
		Declare @DocumentDate smalldatetime,@FBNo smallint

		Select @DocumentDate = FormattedDocumentDate,@FBNo = FBNo
		from Finance.accVoucherMain
		where VoucherNo = @VoucherNo

		if exists (select 1 from masterdata.mFBPeriodDetail 
					where @DocumentDate between StartDate and EndDate
					and FBNo=@FBNo and StatusNo=2
				   )		
		begin
			select 0,'Finance Book is closed for this Document Date.'
			return
		end

		BEGIN Transaction 
		
				-----------------------------------------------
				-------------- Audit Trail Start---------------
				-----------------------------------------------
				----- Here we are managing the log for deledted voucher 
				----- Don't remove this line 				
				-----------------------------------------------
				DECLARE	@ReturnValue int ------------------------------------------------------------------------------------------------------------------------

				EXEC @ReturnValue = [AuditTrail].[SaveVoucherAuditLog]  @LoginNo=@LoginNo,@VoucherNo=@VoucherNo--------------------------------------------------

				IF (@ReturnValue<>1)
					raiserror('Problem arised.',16,-1)

				-----------------------------------------------
				-------------- Audit Trail End---------------
				-----------------------------------------------
		

				--if exists (select 1 from finance.accvouchermain 
				--				where voucherno=@VoucherNo
				--					and VoucherTypeNo=6)
				--	or exists (select 1 from finance.accvouchermain 
				--				where voucherno=@VoucherNo
				--					and RegularInterFBNo=2)
				--begin
				
				--Deleting VoucherGSTDetail when gst Voucher not exists in advance posting but vouchegstposting is done
				-- Deleting VoucherGSTDetail for missing GSTVoucherNo Starts here
				Declare @VoucherFormNo as smallint,@FormCode varchar(10), @SourceDocumentNo int

				Select @VoucherFormNo = SourceDocumentFormNo
				,@SourceDocumentNo = SourceDocumentNo
				from Finance.accVoucherMain avm where avm.VoucherNo = @VoucherNo 

				Select @FormCode = FM.FormCode 
				from GlobalData.FormMaster FM
				Where FM.FormNo = @VoucherFormNo 

				Delete from Finance.VoucherGSTDetailItemWise 
								where VoucherGSTDetailNo in (
															Select GD.VoucherGSTDetailNo
															From Finance.VoucherGSTDetail GD
															Where FormCode = @FormCode
															And GD.DocumentNo in (@SourceDocumentNo)
															and GD.VoucherNo is Null
															) 

				Delete from Finance.VoucherGSTDetail 
								where VoucherGSTDetailNo in (
															Select GD.VoucherGSTDetailNo
															From Finance.VoucherGSTDetail GD
															Where FormCode= @FormCode
															And GD.DocumentNo in (@SourceDocumentNo)
															and GD.VoucherNo is Null
															) 
				-- Deleting VoucherGSTDetail for missing GSTVoucherNo Ends here
				

				Delete from Finance.VoucherGSTDetailItemWise 
				where VoucherGSTDetailNo in (Select VoucherGSTDetailNo 
												   from Finance.VoucherGSTDetail VGD
												   where VGD.VoucherNo = @VoucherNo) 
				                  
				Delete from Finance.VoucherGSTDetail
				where VoucherNo = @VoucherNo



					declare @Status table(Status tinyint,Remarks varchar(max))
					declare @Remarks varchar(max)
					declare @ChildVoucherNo int
					declare @Table table (ChildVoucherNo int,SNo int identity(1,1) PRIMARY KEY)

					;with cte (ChildVoucherNo,ParentVoucherNo)
					as
					(select VCD.ChildVoucherNo,VCD.ParentVoucherNo
						from Finance.accVoucherChildDetail VCD where ParentVoucherNo=@VoucherNo
						union all
						select VCD.ChildVoucherNo,VCD.ParentVoucherNo
							from Finance.accVoucherChildDetail VCD 
							inner join cte on cte.ChildVoucherNo=VCD.ParentVoucherNo
					)
					insert into @Table
					select ChildVoucherNo from cte
					
					declare @MaxCount int
					set @MaxCount = @@RowCount

					declare @Count int
					set @Count=1

					delete from Finance.accVoucherChildDetail 
					where ChildVoucherNo in (select ChildVoucherNo from @Table)

					while @Count<=@MaxCount
					begin
						select @ChildVoucherNo=ChildVoucherNo from @Table where SNo=@MaxCount
						
						insert into @Status
						exec [Finance].[VoucherDelete] 
							@VoucherNo = @ChildVoucherNo,
							@LoginNo=@LoginNo

						if exists(select 1 from @Status where Status=0)
						begin
							select @Remarks=Remarks from @Status 
							raiserror(@Remarks,16,-1)
						end

						select @MaxCount=@MaxCount-1
					end

				--end

				DECLARE	@return_value int

				EXEC	@return_value = [Finance].[VoucherSaveBeforeDelete]
						@VoucherNo = @VoucherNo

				if (@return_value<>1)
					raiserror('Problem arised.',16,-1)

-- if account is inactive and voucher is getting deleted
				update AID
					set InActive=0
						,EffectiveUpto=null
				from finance.accVoucherDetail vd
					inner join masterdata.COAAccountGroupDetail COAAGD on COAAGD.PostingAccountNo=vd.AccountNo
					inner join masterdata.COAMain COAM on COAM.COANo=COAAGD.COANo
													and COAM.COATypeNo=2
					--inner join masterdata.maccountmaster am on am.AccountNo=COAAGD.PostingAccountNo
					inner join masterdata.mAccountGroup ag on ag.GroupNo=COAAGD.PostingGroupNo
					inner join Finance.accVoucherMain VM on VM.VoucherNo=vd.VoucherNo
					inner join masterdata.mAccountInactiveDetail AID on AID.AccountNo=COAAGD.PostingAccountNo
																and AID.FBNo=VM.FBNo
				where vd.VoucherNo=@VoucherNo
					and AID.InActive=1
					and ag.GroupTypeNo not in (15,16,17,18)

				declare @VoucherDetail table(VoucherDetailNo int)

				insert into @VoucherDetail 
				select VoucherDetailNo
				from Finance.accVoucherDetail
				Where VoucherNo=@VoucherNo
	
				Delete From [FixedAsset].[ExcludeVoucherInAssetCapitalization] Where VoucherNo=@VoucherNo
				Delete From Finance.accVoucherCostCenterDetail Where VoucherDetailNo in (select VoucherDetailNo from @VoucherDetail)
			    Delete From Finance.accVoucherQtyDetail Where VoucherDetailNo in (select VoucherDetailNo from @VoucherDetail)

				Delete From Finance.accVoucherReferenceDetail Where VoucherDetailNo in (select VoucherDetailNo from @VoucherDetail)
			    
			    Update Finance.RecurringVoucherDetail set VoucherNo=null where VoucherNo=@VoucherNo 
			    
			    --Update Finance.SundryPaymentMain set VoucherNo=null where VoucherNo=@VoucherNo 
			    --Update Finance.SundryReceiptMain set VoucherNo=null where VoucherNo=@VoucherNo 
			    
			    --Update Finance.PurchaseBillMain set VoucherNo=null where VoucherNo=@VoucherNo 
			    --Update Finance.DrCrNoteMain set VoucherNo=null where VoucherNo=@VoucherNo 
			    
			    UPDATE rm
				   SET 
					   [SpecificInprocessDrAmount] = [SpecificInprocessDrAmount]+tt.Specific
					  ,[BaseInprocessDrAmount] = BaseInprocessDrAmount+Base
				from [Finance].[ReferenceMain] rm 
					inner join (select sum(pppd.PassingAmount) Specific
									,sum(round(pppd.PassingAmount*vm.MultiplicativeRate/vm.DivisibilityRate,6)) Base
									,pprd.ReferenceNo
								from Finance.PaymentPlanPaymentDetail pppd
								inner join Finance.BankPaymentDetail bpd on bpd.PaymentPlanPaymentDetailNo=pppd.PaymentPlanPaymentDetailNo
																		and bpd.PaymentModeNo=pppd.PaymentModeNo
								inner join Finance.PaymentPlanReferenceDetail pprd on pprd.PaymentPlanReferenceDetailNo=pppd.PaymentPlanReferenceDetailNo
								inner join Finance.accVoucherMain vm on vm.VoucherNo=bpd.voucherNo
								Where bpd.VoucherNo=@VoucherNo and pprd.ReferenceNo is not null
								group by pprd.ReferenceNo
								union all
								select sum(pppd.PassingAmount) Specific
									,sum(round(pppd.PassingAmount*vm.MultiplicativeRate/vm.DivisibilityRate,6)) Base
									,pprd.ReferenceNo
								from Finance.CashJournalMain cjm 
								inner join Finance.CashCounterMain ccm on ccm.CashCounterNo=cjm.CashCounterNo
								inner join Finance.PaymentPlanReferenceDetail pprd on pprd.PaymentPlanNo=ccm.DocumentNo
								inner join Finance.PaymentPlanPaymentDetail pppd on pppd.PaymentPlanReferenceDetailNo=pprd.PaymentPlanReferenceDetailNo
																				and pppd.PaymentModeNo=1
								inner join Finance.accVoucherMain vm on vm.VoucherNo=cjm.voucherNo
								Where cjm.VoucherNo=@VoucherNo and pprd.ReferenceNo is not null
								group by pprd.ReferenceNo)tt on rm.ReferenceNo=tt.ReferenceNo

				IF Exists(Select 1 from [Finance].[ReferenceMain] rm 
									inner join (select sum(pppd.PassingAmount) Specific
													,sum(round(pppd.PassingAmount*vm.MultiplicativeRate/vm.DivisibilityRate,6)) Base
													,pprd.ReferenceNo
												from Finance.PaymentPlanPaymentDetail pppd
												inner join Finance.BankPaymentDetail bpd on bpd.PaymentPlanPaymentDetailNo=pppd.PaymentPlanPaymentDetailNo
																						and bpd.PaymentModeNo=pppd.PaymentModeNo
												inner join Finance.PaymentPlanReferenceDetail pprd on pprd.PaymentPlanReferenceDetailNo=pppd.PaymentPlanReferenceDetailNo
												inner join Finance.accVoucherMain vm on vm.VoucherNo=bpd.voucherNo
												Where bpd.VoucherNo=@VoucherNo and pprd.ReferenceNo is not null
												group by pprd.ReferenceNo
												union all
												select sum(pppd.PassingAmount) Specific
													,sum(round(pppd.PassingAmount*vm.MultiplicativeRate/vm.DivisibilityRate,6)) Base
													,pprd.ReferenceNo
												from Finance.CashJournalMain cjm 
												inner join Finance.CashCounterMain ccm on ccm.CashCounterNo=cjm.CashCounterNo
												inner join Finance.PaymentPlanReferenceDetail pprd on pprd.PaymentPlanNo=ccm.DocumentNo
												inner join Finance.PaymentPlanPaymentDetail pppd on pppd.PaymentPlanReferenceDetailNo=pprd.PaymentPlanReferenceDetailNo
																								and pppd.PaymentModeNo=1
												inner join Finance.accVoucherMain vm on vm.VoucherNo=cjm.voucherNo
												Where cjm.VoucherNo=@VoucherNo and pprd.ReferenceNo is not null
												group by pprd.ReferenceNo)tt 
						on rm.ReferenceNo=tt.ReferenceNo
						And ((rm.ByToNo = 1 and IsNULL(rm.BalanceAmount,0) <0) or (rm.ByToNo = 2 and IsNULL(rm.BalanceAmount,0) >0))) 
				Begin
				        Raiserror('Contact your vendor,Reference getting -ve',16,-1)	
				End

				UPDATE rd
				   SET 
					   [SpecificDrAmount] = [SpecificDrAmount]+pppd.PassingAmount
					  ,[BaseDrAmount] = BaseDrAmount+round(pppd.PassingAmount*pprd.MultiplicativeRate/pprd.DivisibilityRate,6)
				from Finance.PaymentPlanPaymentDetail pppd
					inner join Finance.BankPaymentDetail bpd on bpd.PaymentPlanPaymentDetailNo=pppd.PaymentPlanPaymentDetailNo
															and bpd.PaymentModeNo=pppd.PaymentModeNo
					inner join Finance.PaymentPlanReferenceDetail pprd on pprd.PaymentPlanReferenceDetailNo=pppd.PaymentPlanReferenceDetailNo
					inner join [Finance].[ReferenceDetail] rd on rd.PaymentPlanPaymentDetailNo=pppd.PaymentPlanPaymentDetailNo
				Where bpd.VoucherNo=@VoucherNo
			
				UPDATE rd
				   SET 
					   [SpecificDrAmount] = [SpecificDrAmount]+pppd.PassingAmount
					  ,[BaseDrAmount] = BaseDrAmount+round(pppd.PassingAmount*pprd.MultiplicativeRate/pprd.DivisibilityRate,6)
				from Finance.CashJournalMain cjm
					inner join Finance.CashCounterMain ccm on ccm.CashCounterNo=cjm.CashCounterNo
					inner join Finance.PaymentPlanReferenceDetail pprd on pprd.PaymentPlanNo=ccm.DocumentNo
					inner join Finance.PaymentPlanPaymentDetail pppd on pppd.PaymentPlanReferenceDetailNo=pprd.PaymentPlanReferenceDetailNo
					inner join [Finance].[ReferenceDetail] rd on rd.PaymentPlanPaymentDetailNo=pppd.PaymentPlanPaymentDetailNo
				Where cjm.VoucherNo=@VoucherNo
					and pppd.PaymentModeNo=1
					and cjm.AccountNo=rd.AccountNo

				Delete From Finance.CashJournalMain Where VoucherNo=@VoucherNo 
			    
			    delete from Finance.BankPaymentDetail where VoucherNo=@VoucherNo 
			    
			    --** case of contra voucher from dd posting start yusuf 21-jun-12 
			    update Finance.BankPaymentMain
					set VoucherNo=null
			    where VoucherNo=@VoucherNo 
			    --** case of contra voucher from dd posting end
			    		    
			    Delete From Finance.ReferenceDetail Where VoucherDetailNo in (select VoucherDetailNo from @VoucherDetail)
			    Delete From Finance.ReferenceMain Where VoucherDetailNo in (select VoucherDetailNo from @VoucherDetail)

			    Delete From Finance.accVoucherDetail Where VoucherNo=@VoucherNo
			    Delete From Finance.accVoucherPaymentDetail Where VoucherNo=@VoucherNo
			    
			    Delete From Finance.accVoucherItemDetail Where VoucherNo=@VoucherNo
			    
			    Delete From Finance.accVoucherTCSDetail Where VoucherNo=@VoucherNo
			    Delete From Finance.accVoucherTDSDetail Where VoucherNo=@VoucherNo
			    Delete From Finance.accVoucherSTDetail Where VoucherNo=@VoucherNo
			    
			    Delete From Finance.VoucherGSTDetailItemWise Where VoucherGSTDetailNo in(
			    select VoucherGSTDetailNo From Finance.VoucherGSTDetail Where VoucherNo=@VoucherNo)
			    Delete From Finance.VoucherGSTDetail Where VoucherNo=@VoucherNo
			    
			    delete from Finance.ChangeVoucherStatusMain where VoucherNo=@VoucherNo
			    Delete From Finance.accVoucherMain Where VoucherNo=@VoucherNo

			select @OutputStatusNo=1,@OutputStatusRemarks='Success'
			Select 1,@OutputStatusRemarks
		COMMIT Transaction 
	END TRY
	BEGIN CATCH
		Rollback Transaction
			select @OutputStatusNo=0,@OutputStatusRemarks=Error_Message() + ' ' + CONVERT(nvarchar(10), ERROR_LINE())
			Select 0,@OutputStatusRemarks
	END CATCH
END
