-- =============================================

CREATE Procedure [masterdata].[MDLMSave] 
	-- Add the parameters for the stored procedure here
	@Name varchar(50)
   ,@LoginNo smallint
	,@Id smallint
	,@Code char(6)
	,@DocumentStatusNo tinyint = 30
	,@CityNo int = null
AS
BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;
	begin try

    -- Insert statements for procedure here

	if isnull(@Id,0)=0
	begin

		if @Code is null
		begin
			SELECT @Code='LM'+ RIGHT(REPLICATE('0', 4) +cast(isnull(max(Convert(int,substring(Code,3,4))),0) + 1 as varchar(4)),4) 
			FROM [masterdata].[mLocationMaster]
		end

		INSERT INTO [masterdata].[mLocationMaster]
			   (LocationName
			   ,[CreatedBy]
			   ,[CreatedDate]
			   ,[Code]
				,AuthorizedBy 
			   ,AuthorizedDate 
			   ,DocumentStatusNo
			   ,CityNo)
		 values( 
			   @Name
			   ,@LoginNo
			   ,getdate()
			   ,@Code
				,case when @DocumentStatusNo=30 then @LoginNo else null end
			,case when @DocumentStatusNo=30 then GETDATE ()else null end
			,@DocumentStatusNo
			,@CityNo)

		select @Id=ident_current('[masterdata].[mLocationMaster]')
		insert into [masterdata].mLocationMasterCompanyDetail (LocationNo ,CompanyNo , Inactive )
				select @Id,companyno ,0
					from masterdata.mCompanyMaster 
					
	end
	else
	begin
	
	--New Code
	
	 
		UPDATE [masterdata].[mLocationMaster]
		   SET LocationName = @Name
			  ,[ModifiedBy] = @LoginNo
			  ,[ModifiedDate] = getdate()
			  , AuthorizedBy = case when @DocumentStatusNo=30 then @LoginNo else null end
			  , AuthorizedDate = case when @DocumentStatusNo=30 then GETDATE ()else null end
			  , DocumentStatusNo= @DocumentStatusNo
		WHERE LocationNo =@Id  
		
	end

	select 1,@Id

	end try 
	begin catch	
		select 0,Error_message()

	end catch
END
