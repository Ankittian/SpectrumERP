CREATE PROCEDURE [Finance].[FASBFinancePosting] 
	@OtherDataInXml xml ,
	@CompanyNo tinyint ,
	@VendorLocationNo int  ,
	@NetAmount Decimal(18,6)
	,@ServiceTaxAbatementRuleNo smallint
	,@BasicAmount money
	,@FBNo smallint
	, @MiscChargeNos xml
	,@VoucherDate smalldatetime
	,@ItemNos varchar(max)
AS


BEGIN

begin try
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;

 

Declare @IsInterUnit bit,@IsManagingExpenseAccount bit  ,@FormNo smallint ,@IncludedAmount money
	,@ServiceTaxAbatementBasicAmount money ,@GSTApplicableDate smalldatetime


	
declare @MiscChargeTable table(MiscChargeNo [smallint],Amount Decimal(18,6),[CalculationNature] [tinyint]
						,TaxCodeMiscChargesDetailNo int,IncludeInBasic tinyint,ChargeType tinyint)

declare @MiscFinancePosting table(Particular varchar(500),[Debit Amount] Decimal(18,6),
		[Credit Amount] Decimal(18,6),[Account Name] Int,[Amount] Decimal(18,6),[Type] tinyint,
		SlNo smallint identity(1,1),MiscChargeNo int,TaxCodeMiscChargesDetailNo int,ItemNos varchar(5000),
		IncludeInBasic tinyint)


Create table #IncludedTax(IncludedAmount decimal(18,6),ItemNo int,MakeNo smallint,TechnicalGradeNo tinyint
							,Intransitper decimal(18,8), StockPer decimal(18,8) ,StockIssuePer decimal(18,8)
							,IsRouteCase bit , IsPOExpenseCase bit ,ReferenceNo int
							)
Create Table #LotDetail  (LotAmount decimal(18,6),ReferenceNo int, ItemNo int,MakeNo smallint,TechnicalGradeNo tinyint
				, IsRouteCase bit , IsPOExpenseCase bit,[Select] bit,InvLotsummaryNo int)

Create table #SCOrderDetail (SCOrderAmendmentNo int,POAmendmentNo int,JobCardNo int, GRNNo int,
		ServiceItemNo int,ActivityNo smallint,SequenceNo tinyint , 
		ExpenseHeadNo smallint,  Amount decimal(18,6),PONo int,IncludedTax decimal(15,2), SCOrderNo int,GSTTaxRuleNo smallint,ItemTax varchar(500))

Create table #ItemDetail (ReferenceNo int,GRNNo int, ItemNo int,MakeNo smallint,
		TechnicalGradeNo tinyint,AcceptedQty decimal(15,3), Rate decimal(15,2),GateEntryNo int 
		,RouteIdentity decimal(10,8) , IncludedTax decimal(18,6) , ChallanQty decimal(15,3),ReceivedQty decimal(15,3)
		,POAmendmentNo int,PONo int,SCMRGRNNo int,[Select] bit,ItemFlowNo tinyint)
	  
Create table #StockAccount(
		Id Int Identity(1,1),
		StockAmount Decimal(18,6),  ItemNo int,MakeNo smallint,
		TechnicalGradeNo tinyint, BalanceQty decimal(15,3),IssuedQty decimal(15,3), VRate Decimal(18,6)
		,  LotBalanceQty decimal(15,3),LotIssuedQty decimal(15,3), IsRouteCase bit , IsPOExpenseCase bit 
		,[Select] bit , ReferenceNo int,  InvLotSummaryNo int
		,NextIntransitNegativeQty Decimal(12,3)
		)

Create table #GRNMain (GRNNo int,MRANo int,VoucherNo int)		  

create table #Posting ([Add to Basic] bit,Particular varchar(500),[Debit Amount] decimal(15,2) , 
		[Credit Amount] decimal(15,2),[Account Name] int, [Ref. Detail] varchar(50),[Cost Center] varchar(50),
		[Amount] decimal(15,2), [Type] tinyint,SlNo tinyint ,MiscChargeNo int , TaxCodeMiscChargesDetailNo int,
			ItemNos varchar(max))
Create table #StockTransfer(DocumentNo int,parentinvLotSummaryNo int,GRNNo int)

select @GSTApplicableDate =GSTApplicableDate from finance.setting where CompanyNo = @CompanyNo
select @FormNo = FormNo from GlobalData.FormMaster where FormCode= 'FASB'

;with cte as 
			(
				SELECT v.value('MiscChargeNo[1]','varchar(100)') AS MiscChargeNo
					,v.value('Amount[1]','Decimal(18,6)') AS Amount
					,v.value('CalculationNature[1]','tinyint') AS CalculationNature
					,v.value('TDC_x0020_Account_x0020_Rule[1]','int') AS TaxCodeMiscChargesDetailNo
					,v.value('IncludeInBasic[1]','bit') AS IncludeInBasic
					,v.value('Charge_x0020_Type[1]','tinyint') AS ChargeType
				FROM @MiscChargeNos.nodes('/DocumentElement/Table') x(v)
			)
			insert into @MiscChargeTable(MiscChargeNo,Amount,[CalculationNature],TaxCodeMiscChargesDetailNo,
				IncludeInBasic,ChargeType)
			select MiscChargeNo,Amount,CalculationNature,TaxCodeMiscChargesDetailNo,isnull(IncludeInBasic,0)
			,ChargeType
			from cte
			where ISNUMERIC(MiscChargeNo)=1 --and isnull(IncludeInBasic,0)<>1 -- Commented by Yusuf 27-May-13
			
			

			select @ServiceTaxAbatementBasicAmount=SUM(Amount)
			from @MiscChargeTable	
			where MiscChargeNo in (select MiscChargeNo from Finance.ServiceTaxAbatementRuleChargeDetail
									where ServiceTaxAbatementRuleNo=@ServiceTaxAbatementRuleNo)
		
			select @ServiceTaxAbatementBasicAmount=@BasicAmount 
			where @ServiceTaxAbatementBasicAmount is null
			set @ServiceTaxAbatementBasicAmount = @ServiceTaxAbatementBasicAmount + isnull((select sum(case when dt.CalculationNature =1 then Amount else 0-Amount end) from @MiscChargeTable dt
				inner join masterdata.mMiscCharges mc on mc.MiscChargeNo = dt.MiscChargeNo
				where TaxGroupNo in(18,16)),0)

			
if exists (select 1 from masterdata.mVendorCard vc
			inner join masterdata.mVendorLocationDetail vld on vld.VendorNo = vc.VendorNo 
			where vld.VendorLocationNo =@VendorLocationNo and vc.VendorOfCompanyNo is not null)
begin
	select @IsInterUnit=1
end

	; with cte as
	 (
		SELECT 
		v.value('SCOrderAmendmentNo[1]','int') AS SCOrderAmendmentNo,
		v.value('POAmendmentNo[1]','int') AS POAmendmentNo,
		v.value('JobCardNo[1]','int') AS JobCardNo,
		v.value('GRNNo[1]','int') AS GRNNo,
		v.value('ServiceItemNo[1]','int') AS ServiceItemNo,
		v.value('ActivityNo[1]','smallint') ActivityNo,
		v.value('SequenceNo[1]','Tinyint') SequenceNo,
		v.value('ExpenseHeadNo[1]','smallint') ExpenseHeadNo,
		v.value('Amount[1]','decimal(18,6)') Amount,
		v.value('ItemNo[1]','int') ItemNo,
		v.value('IncludedTax[1]','decimal(18,6)') IncludedTax,
		v.value('GSTTaxRule[1]','smallint') GSTTaxRule,
		v.value('ItemTax[1]','varchar(500)') ItemTax
		FROM @OtherDataInXml.nodes('/DocumentElement/Table') x(v)
	)
	Insert into #SCOrderDetail(SCOrderAmendmentNo ,POAmendmentNo ,JobCardNo , GRNNo ,
				ServiceItemNo ,ActivityNo ,SequenceNo,ExpenseHeadNo,Amount,PONo,IncludedTax, SCOrderNo ,GSTTaxRuleNo , ItemTax)
	select cte.SCOrderAmendmentNo ,cte.POAmendmentNo ,cte.JobCardNo , cte.GRNNo ,
			cte.ServiceItemNo ,cte.ActivityNo ,cte.SequenceNo,cte.ExpenseHeadNo,cte.Amount,pam.PONo ,cte.IncludedTax ,scam.SCOrderNo , cte.GSTTaxRule
				,cte.Itemtax
				from cte 
			left join inventory.GRNSCOrderDetail gd on gd.grnno = cte.grnno
					and coalesce(cte.ExpenseHeadNo,cte.Activityno,	cte.ServiceItemNo)=
						coalesce(gd.ExpenseHeadNo,gd.Activityno,gd.ServiceItemNo)
			left join Inventory.GRNSCOrderRequisitionDetail	gsd on gsd.GRNNo = cte.GRNNo and 
					gsd.grnscorderDetailno = gd.grnscorderDetailno
			 left join purchase.POAmendmentExpenseHead peh on peh.poAmendmentExpenseDetailno= gsd.poAmendmentExpenseDetailno		
			left join Purchase.POAmendmentMain pam on pam.POAmendmentNo =isnull(cte.POAmendmentNo ,peh.POAmendmentNo)
			left join Purchase.SCOrderAmendmentMain scam on scam.SCOrderAmendmentNo = cte.SCOrderAmendmentNo 
				where cte.ItemNo is null
			group by cte.SCOrderAmendmentNo ,cte.POAmendmentNo ,cte.JobCardNo , cte.GRNNo ,scam.SCOrderNo ,
		cte.ServiceItemNo ,cte.ActivityNo,cte.SequenceNo,cte.ExpenseHeadNo,cte.Amount,pam.PONo ,cte.IncludedTax	,cte.GSTTaxRule ,cte.Itemtax
				
    insert into #GRNMain(GRNNo ,MRANo,VoucherNo)
		select sd.GRNNo ,md.MRANo ,gm.VoucherNo from
		#SCOrderDetail sd 
		left join Inventory.GRNMRADetail md on md.GRNNo = sd.GRNNo 		left join Inventory.GRNMain GM on GM.GRNNo = isnull(md.MRANo,sd.GRNNo)
		group by sd.GRNNo ,md.MRANo,GM.VOucherNo


Update MT set Amount  =MT.Amount + tt.Amount 
		from @MiscChargeTable MT ,
		(select sid.MiscChargeNo,sum(sid.TotalValue) Amount , mc.CalculationNature, std.TaxCodeMiscChargesDetailNo,std.IncludeInBasic , sid.ChargeType 
			from finance.SupplementaryInvoiceMain sim 
			inner join Finance.SupplementaryInvoiceItemTaxDetail sid on sid.SupplementaryInvoiceNo = sim.SupplementaryInvoiceNo 
			inner join finance.SupplementaryInvoiceTaxDetail std on std.SupplementaryInvoiceNo = sid.SupplementaryInvoiceNo 
					and std.MiscChargeNo = sid.MiscChargeNo
			inner join masterdata.mMiscCharges mc on mc.MiscChargeNo = sid.MiscChargeNo
			inner join #GRNMain    gm on gm.GRNNo = sim.GRNNo
			where sim.DocumentDate >=  @GSTApplicableDate
			group by 
			sid.MiscChargeNo,  mc.CalculationNature, std.TaxCodeMiscChargesDetailNo,std.IncludeInBasic , sid.ChargeType 
			) tt where tt.MiscChargeNo = MT.MiscChargeNo 
insert into @MiscChargeTable(MiscChargeNo,Amount,[CalculationNature],TaxCodeMiscChargesDetailNo,
				IncludeInBasic,ChargeType)

	select sid.MiscChargeNo,sum(sid.TotalValue)  , mc.CalculationNature, std.TaxCodeMiscChargesDetailNo,std.IncludeInBasic , sid.ChargeType 
			from finance.SupplementaryInvoiceMain sim 
			inner join Finance.SupplementaryInvoiceItemTaxDetail sid on sid.SupplementaryInvoiceNo = sim.SupplementaryInvoiceNo 
			inner join finance.SupplementaryInvoiceTaxDetail std on std.SupplementaryInvoiceNo = sid.SupplementaryInvoiceNo 
					and std.MiscChargeNo = sid.MiscChargeNo
			inner join masterdata.mMiscCharges mc on mc.MiscChargeNo = sid.MiscChargeNo
			inner join #GRNMain    gm on gm.GRNNo = sim.GRNNo
			where sim.DocumentDate >=  @GSTApplicableDate
			and not exists (select 1 from @MiscChargeTable where MiscChargeNo = sid.MiscChargeNo )
			group by 
			sid.MiscChargeNo,  mc.CalculationNature, std.TaxCodeMiscChargesDetailNo,std.IncludeInBasic , sid.ChargeType 

		
			
IF exists (select 1 from Purchase.POAmendmentMain pam 
				inner join #SCOrderDetail sd on sd.POAmendmentNo = pam.POAmendmentNo and IndentTypeNo =1)-- capital
	set @IsManagingExpenseAccount =0
else	
	select @IsManagingExpenseAccount = IsManagingExpenseAccount  from Finance.Setting where CompanyNo = @CompanyNo 



if exists(select 1 from finance.SupplementaryInvoiceMain sim ,#GRNMain    gm where gm.GRNNo = sim.GRNNo)
	set @ItemNos  = isnull((select convert(varchar(10),sid.ItemNo) +'~' + convert(varchar(25),sid.BasicAmount) +'~' + convert(varchar(25),sid.Qty) +'~' + CONVERT(varchar(25),sid.BasicAmount/Qty) +'`' 
	from 
	Finance.SupplementaryInvoiceMain sim 
	inner join Finance.SupplementaryInvoiceItemDetail sid on sid.SupplementaryInvoiceNo=sim.SupplementaryInvoiceNo 
	inner join #GRNMain gm on gm.GRNNo = sim.GRNNo where sim.IsExcludeFromValuation=0 for xml path('')),'')
							

insert into @MiscFinancePosting
					(Particular ,[Debit Amount] ,[Credit Amount] ,[Account Name] 
					,[Amount] ,[Type]
					,MiscChargeNo ,TaxCodeMiscChargesDetailNo ,ItemNos,IncludeInBasic)
			select 
				 MC.MiscChargeName+isnull(' ('+AppendAccountName+')','')
				,convert(decimal(15,2),case(s.CalculationNature) when 1 then isnull(tt.Amount,s.Amount)
																else null end)
				,convert(decimal(15,2),case(s.CalculationNature) when 1 then null
																else isnull(tt.Amount,s.Amount) end)
				,tt.AccountNo 
				,isnull(tt.Amount,s.Amount)
				,s.CalculationNature
				,s.MiscChargeNo
				,tcmcd.TaxCodeMiscChargesDetailNo
				,tt.ItemNos
				,s.IncludeInBasic
			from 
				@MiscChargeTable s
				inner join masterdata.mMiscCharges MC on MC.MiscChargeNo=s.MiscChargeNo -- and s.ChargeType =2
				left join masterdata.mTaxCodeMiscChargesDetail tcmcd on tcmcd.TaxCodeMiscChargesDetailNo=s.TaxCodeMiscChargesDetailNo
				outer apply (select AccountNo,Amount,AppendAccountName,ItemNos
							 from Finance.funcTaxesPosting(tcmcd.TaxCodeMiscChargesDetailNo, 
															s.MiscChargeNo,
															s.Amount,
															0,
															@BasicAmount,
															@ItemNos,
															@FormNo,
															@VoucherDate))tt
			Union All
			select AP.Particular Particular 
				,case(AP.Type) when 1 then AP.Amount else null end [Debit Amount]
				,case(AP.Type) when 2 then AP.Amount else null end [Credit Amount] 
				,AP.AccountNo [Account Name] 
				,AP.Amount [Amount]
				,AP.Type [Type] 
				,MiscChargeNo ,TaxCodeMiscChargesDetailNo ,CAST(null as varchar(5000)) ItemNos,0
			from 
			#SCOrderDetail id
			cross apply (select sum(convert(money,col2)) TaxAmount from masterdata.Split_RowColumn( ItemTax,'~',',') dt
							where exists(select 1 from  masterdata.mMiscCharges mc 
							inner join masterdata.mTaxGroupDependOn dn on dn.DependOnTaxGroupNo = mc.TaxGroupNo
							inner join masterdata.mTaxGroup tg on tg.TaxGroupNo = dn.TaxGroupNo
							where tg.TaxCategoryNo = 14 and mc.MiscChargeNo = dt.Col1 ) ) it
			cross apply [Finance].[funcServiceTaxAbatementPosting] (id.GSTTaxRuleNo,Amount + isnull(TaxAmount,0),@FBNo) AP
				
 
insert into #ItemDetail 
 
select gid.SCOrderAmendmentNo,gid.GRNNo , gid.ItemNo  ,gid.MakeNo  ,gid.TechnicalGradeNo  ,AcceptedQty  , 
		isnull(bd.VRate , gid.Rate)  ,gid.GateEntryNo   ,gid.RouteIdentity   ,
		-- gsd.IncludedTax * isnull(said.BifercationPercentage * 0.01,1)
		(gsd.IncludedTax * AcceptedQty * ls.Vrate )/ 
				(select sum(AcceptedQty* gdls.Vrate)  from Inventory.GRNItemDetail gd
				inner join Inventory.InvLotSummary gdls on gdls.DocumentNo = gd.GRNNo and gd.ItemNo = gdls.ItemNo 
				and ISNULL(gdls.MakeNo,0) = ISNULL(gd.MakeNo,0)
				and ISNULL(gdls.TechnicalGradeNo,0) = ISNULL(gd.TechnicalGradeNo,0)
				and ISNULL(gdls.GateEntryNo,0) = ISNULL(gd.GateEntryNo,0)
				and ISNULL(gdls.POAmendmentNo,0) = ISNULL(gd.POAmendmentNo,0)
				and ISNULL(gdls.invStockTransferNo,0) = ISNULL(gd.invStockTransferNo,0)
				and ISNULL(gdls.DispatchChallanNo,0) = ISNULL(gd.DispatchChallanNo,0)
				where gd.GRNNo = gid.GRNNo and gd.SCOrderAmendmentNo = gid.SCOrderAmendmentNo and 
				AcceptedQty* gdls.Vrate>0 --and ItemNo = gid.ItemNo commented on  01 dec 2014
				and gd.ItemFlowNo in(2,3,4) )  IncludedTax ,
		gid.ChallanQty  ,gid.ReceivedQty ,gid.POAmendmentNo ,pam.PONo ,gsd.GRNNo ,1,said.ItemFlowNo
		from Inventory.GRNItemDetail gid 
			left join purchase.POAmendmentMain pam on pam.POAmendmentNo = gid.POAmendmentNo 
			inner join Inventory.InvLotSummary ls on ls.DocumentNo = gid.GRNNo and gid.ItemNo = ls.ItemNo 
			and ISNULL(ls.MakeNo,0) = ISNULL(gid.MakeNo,0)
			and ISNULL(ls.TechnicalGradeNo,0) = ISNULL(gid.TechnicalGradeNo,0)
			and ISNULL(ls.GateEntryNo,0) = ISNULL(gid.GateEntryNo,0)
			and ISNULL(ls.POAmendmentNo,0) = ISNULL(gid.POAmendmentNo,0)
			and ISNULL(ls.invStockTransferNo,0) = ISNULL(gid.invStockTransferNo,0)
			and ISNULL(ls.DispatchChallanNo,0) = ISNULL(gid.DispatchChallanNo,0)
			inner join Inventory.InvLotBillDetail bd on bd.InvLotSummaryNo = ls.InvLotSummaryNo 
					and bd.TransactionFlag = ls.TransactionFlag 
					
			inner join #SCOrderDetail gsd on gsd.SCOrderAmendmentNo = gid.SCOrderAmendmentNo
					and gsd.GRnNo = gid.grnno
			inner join Purchase.SCOrderAmendmentDetail sad on sad.SCOrderAmendmentNo = gsd.SCOrderAmendmentNo 
				and 
				coalesce(sad.ActivityNo ,sad.ExpenseHeadNo,sad.ItemNo)=
					coalesce(gsd.ActivityNo,gsd.ExpenseHeadNo,gsd.ServiceItemNo) 
				 and isnull(sad.SequenceNo,0) = isnull(gsd.SequenceNo,0)
				
			inner join Purchase.SCOrderAmendmentItemDetail said on said.SCOrderAmendmentNo=sad.SCOrderAmendmentNo 
			and
				(coalesce(sad.ActivityNo ,sad.ExpenseHeadNo,sad.ItemNo)=
					coalesce(said.ActivityNo,said.ExpenseHeadNo,said.ServiceItemNo) 
				 or (said.ActivityNo is null and said.ExpenseHeadNo is null and said.ServiceItemNo is null))
				 and isnull(sad.SequenceNo,0) = isnull(said.SequenceNo,0)
					and said.ItemFlowNo in(2,3,4)
					and said.ItemNo = gid.ItemNo 
					and isnull(said.Makeno,0) =isnull( gid.Makeno ,0)
					and isnull(said.technicalgradeNo,0) = isnull(gid.technicalgradeNo ,0)
Union
 select SCOrderAmendmentNo , GRNNo,ItemNo,MakeNo , TechnicalGradeNo,AcceptedQty , Rate , GateEntryNo , RouteIdentity ,sum(IncludedTax)
	, ChallanQty  ,ReceivedQty,POAmendmentNo ,PONo  ,  SCMRGRNNo ,[Select],[ItemFlowNo]
	from(
 	select gid.SCOrderAmendmentNo,gid.GRNNo , gid.ItemNo  ,gid.MakeNo  ,gid.TechnicalGradeNo,AcceptedQty  ,gid.Rate  
		,GateEntryNo,gid.RouteIdentity ,gsd.IncludedTax  *  case when grn.IsReturnThroughDC =1 then AcceptedQty + RejectedQty else AcceptedQty end * case when peh.Rate >0 then peh.rate else 1 end
			 / (select sum(case when Agrn.IsReturnThroughDC =1 then Agid.AcceptedQty + Agid.RejectedQty else Agid.AcceptedQty end * case when Apeh.Rate >0 then Apeh.Rate else 1 end  ) Amount  from 
				Inventory.GRNMRADetail AMRA 
				inner join Purchase.POAmendmentExpenseHead Apeh on Apeh.POAmendmentExpenseDetailNo = AMRA.POAmendmentExpenseDetailNo 
				inner join Inventory.GRNItemDetail Agid on AMRA.MRANo = Agid.GRNNo and AMRA.ItemNo = Agid.ItemNo
					and isnull(Agid.MakeNo,0) = isnull(AMRA.MakeNo,0) 
					and ISNULL(Agid.TechnicalGradeNo,0) = isnull(AMRA.TechnicalGradeNo,0) 
					and isnull(AMRA.SCOrderAmendmentNo,0) = isnull(Agid.SCOrderAmendmentNo ,0)
				Inner join Inventory.GRNMain Agrn on Agrn.GRNNo = Agid.GRNNo 
				inner join Purchase.POAmendmentMain Apom on Apom.POAmendmentNo = AMRA.POAmendmentNo 
				inner join Purchase.POAmendmentMain Agsdpam on Agsdpam.POAmendmentNo = gsd.POAmendmentNo 
				inner join Purchase.POAmendmentMain Apam on Apam.POAmendmentNo = Agid.POAmendmentNo 
				where Apam.PONo = Apom.PONo and isnull(Agsdpam.PONo ,0) = isnull(Apom.PONo,0) 
				and ISNULL(Apeh.ExpenseheadNo,0) = isnull(gsd.ExpenseHeadNo ,0)
				 and AMRA.GRNNo= MRA.GRNNo 
				--and case when Agrn.IsReturnThroughDC =1 then AcceptedQty + RejectedQty else AcceptedQty end * Apeh.Rate  >0
				--inner join #SCOrderDetail gsd on gsd.GRNNo = MRA.GRNNo and gsd.ExpenseheadNo = 12
				
			) IncludedTax ,gid.ChallanQty  ,gid.ReceivedQty,gid.POAmendmentNo ,pom.PONo 
		,gsd.GRNNo SCMRGRNNo,1 [Select],null [ItemFlowNo]
		 
				from 
				Inventory.GRNMRADetail MRA 
				Left join Purchase.POAmendmentExpenseHead peh on peh.POAmendmentExpenseDetailNo = MRA.POAmendmentExpenseDetailNo 
				inner join Inventory.GRNItemDetail gid on MRA.MRANo = gid.GRNNo and MRA.ItemNo = gid.ItemNo
					and isnull(gid.MakeNo,0) = isnull(MRA.MakeNo,0) 
					and ISNULL(gid.TechnicalGradeNo,0) = isnull(MRA.TechnicalGradeNo,0) 
					-- and isnull(gid.POAmendmentNo,0) = isnull(MRA.POAmendmentNo,0) 
					and isnull(MRA.SCOrderAmendmentNo,0) = isnull(gid.SCOrderAmendmentNo ,0)
				Inner join Inventory.GRNMain grn on grn.GRNNo = gid.GRNNo 
				inner join Purchase.POAmendmentMain pom on pom.POAmendmentNo = MRA.POAmendmentNo 
				inner join #SCOrderDetail gsd on gsd.GRNNo = MRA.GRNNo 
				inner join Purchase.POAmendmentMain gsdpam on gsdpam.POAmendmentNo = gsd.POAmendmentNo 
				inner join Purchase.POAmendmentMain pam on pam.POAmendmentNo = gid.POAmendmentNo 
				where pam.PONo = pom.PONo and isnull(gsdpam.PONo ,0) = isnull(pom.PONo,0) 
				and ISNULL(peh.ExpenseheadNo,0) = isnull(gsd.ExpenseHeadNo ,0)
				and case when grn.IsReturnThroughDC =1 then AcceptedQty + RejectedQty else AcceptedQty end >0
		 ) tt
		 group by SCOrderAmendmentNo , GRNNo,ItemNo,MakeNo , TechnicalGradeNo,AcceptedQty , Rate , GateEntryNo , RouteIdentity  
	, ChallanQty  ,ReceivedQty,POAmendmentNo ,PONo  ,SCMRGRNNo ,[Select],[ItemFlowNo]

insert into #ItemDetail (ReferenceNo ,GRNNo , ItemNo ,MakeNo ,TechnicalGradeNo , Rate ,SCMRGRNNo ,[Select],itemflowNo)
	select sid.SCOrderAmendmentNo ReferenceNo,sd.GRNNo ,sid.ItemNo,sid.MakeNo,sid.TechnicalGradeNo,
		sid.Rate,sd.GRNNo ,0 ,sid.ItemflowNo
	from Purchase.SCOrderAmendmentItemDetail sid
	inner join Purchase.SCOrderAmendmentMain sam on sam.SCOrderAmendmentNo = sid.SCOrderAmendmentNo 
	inner join #SCOrderDetail sd on sd.SCOrderAmendmentNo = sid.SCOrderAmendmentNo 
		
	where ((coalesce(sd.ActivityNo , sd.ExpenseHeadNo,sd.ServiceItemNo) =
				coalesce(sid.ActivityNo , sid.ExpenseHeadNo,sid.ServiceItemNo ) )
		or (sid.ActivityNo is null and sid.ExpenseHeadNo is null and sid.ServiceItemNo is null))
		and sid.ItemFlowNo in(2,3,4)
		and not exists (select 1 from #ItemDetail id where id.ReferenceNo = sid.SCOrderAmendmentNo 
					and ItemNo = sid.ItemNo and isnull(MakeNo,0)= isnull(sid.MakeNo,0) 
					and grnno = sd.grnno 
					and isnull(TechnicalGradeNo,0) = isnull(sid.TechnicalGradeNo,0) )

;with CTE
			as

																					inner join Purchase.SCOrderMain smain on smain.SCOrderNo = sodsm.SCOrderNo and smain.AmendmentNo = sodsm.AmendmentNo 
																					inner join Purchase.SCOrderAmendmentExpenseHead sed on sed.SCOrderAmendmentNo = sod.SCOrderAmendmentNo 
																					inner join Purchase.POAmendmentExpenseHead ped on ped.POAmendmentExpenseDetailNo = sed.POAmendmentExpenseDetailNo 
																					inner join Purchase.POAmendmentMain sopm on sopm.POAmendmentNo= ped.POAmendmentNo 
																					inner join Purchase.POAmendmentExpenseHeadOutputDetail peod on peod.POAmendmentExpenseDetailNo = ped.POAmendmentExpenseDetailNo 
																					outer apply (select sum(sd.Quantity )/ sum(sid.AcceptedQty) Ratio  from Inventory.GRNSCOrderDetail sd , Purchase.SCOrderAmendmentMain sm , Inventory.GRNItemDetail sid
																									where sd.SCOrderAmendmentNo = sm.SCOrderAmendmentNo and sm.SCOrderNo = smain.SCOrderNo

																									and sd.ExpenseheadNo = ped.ExpenseheadNo and sid.ItemNo = peod.ItemNo

																						) eod

												) NxtL
