CREATE PROCEDURE [Information].[PODrCrDetail] 
	-- Add the parameters for the stored procedure here
	@CompanyNo tinyint,
	@DocumentDate smalldatetime,
	@PONo int
	
AS
BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;

    -- Insert statements for procedure here
	

	declare @str varchar(max)
    select @str=masterdata.JSForInformation('PMPO')
    set @str=@str+masterdata.CSSForInformation('PMPO')
	set @str=@str+'<table class="altrowstable" id="alternatecolorPODrCrDetail">'
	set @str=@str+'<Caption><Font color="Maroon"><h3 align="Left"><b>Credit/Debit Detail</b></h3></Font></Caption>'
    set @str=@str+'<tr><th>Sl. No.</th><th>Doc Type(Cr/Dr)</th><th>Doc. No.</th><th>Doc. Date</th><th>Ref. No.</th><th>Ref. Date</th><th>Ded/Add Reason</th><th>Ded/Add Nature</th><th>Add/Ded Amount</th></tr>'
	set @str=@str
				 +( 
					replace( 
					replace(
					replace(
					replace( 
					replace(
					replace(
					replace(
					replace(
					replace(
							(
								Select  ROW_NUMBER() over ( Order by DCM.DocumentDate desc) as td1
								        ,tt.DrCrType  td2
										,DCM.DocumentNoYearly td3
										,Convert(varchar(20),DCM.DocumentDate,103) td4
										,ISNULL( DCM.VendorBillNo,'') td5
										,ISNULL( Convert(varchar(20),DCM.VendorBillDate,103),0) td6
										,CASE When DCM.ReasonNo = 5 Then 'Other :'+ DCM.OtherReason
												Else RSN.ReasonName
												End as  td7
										,Nature.Nature td8
										, CONVERT(Decimal(15,2),SUM(DCBD.Amount)) td9
								From [Finance].[DrCrNoteMain] DCM
								Inner Join (Select 1 DrCrTypeNo, 'Debit Note' [DrCrType]
											union 
											Select 2 DrCrTypeNo, 'Credit Note' [DrCrType]	
											) tt On DCM.DrCrTypeNo = tt.DrCrTypeNo
								Inner Join GlobalData.DrCrNoteReason RSN On DCM.ReasonNo = RSN.DrCrNoteReasonNo 
								Inner Join (select 1 Id ,'Percentage' [Nature]
											union all select 2 ,'Amount'
											union all select 3 ,'Rate Difference'
											union all select 4 ,'Fix Rate'
											union all select 5 ,'Manual'	
											)Nature On DCM.NatureNo = Nature.Id
								Inner Join Finance.DrCrNoteBillDetail DCBD On DCM.DrCrNoteNo = DCBD.DrCrNoteNo 
								Inner Join ( Select DCBD.DrCrNoteBillDetailNo
													,POAM.PONo PoNo
											From Finance.DrCrNoteMain DCM
											Inner Join Finance.DrCrNoteBillDetail DCBD On DCM.DrCrNoteNo = DCBD.DrCrNoteNo  
											Left Join Finance.PurchaseBillItemDetail PBIM On DCBD.PurchaseBillNo = PBIM.PurchaseBillNo
											LEFT Join Finance.FreightInvoiceMain FIM On DCBD.FreightInvoiceNo = FIM.FreightInvoiceNo
											Left Join Finance.FreightInvoiceDocumentDetail FID On FIM.FreightInvoiceNo = FID.FreightInvoiceNo
																									 
											Left Join GateEntry.InvGateEntryItemDetail GEI ON FID.RefDocNo=GEI.GateEntryNo
																									And FIM.FreightDocumentNo = 1
											Left Join Inventory.GRNItemDetail GRNI On FID.RefDocNo=GRNI.GRNNo
																									And FIM.FreightDocumentNo = 6
																									

											Left Join Finance.ServiceBillItemDetail SBIM On DCBD.ServiceBillNo = SBIM.ServiceBillNo
																					And DCM.SourceDocumentNo =4
											Inner Join  Purchase.POAmendmentMain POAM On GEI.POAmendmentNo = POAM.POAmendmentNo
																						Or GRNI.POAmendmentNo = POAM.POAmendmentNo
																						Or PBIM.POAmendmentNo = POAM.POAmendmentNo
																						Or SBIM.POAmendmentNo = POAM.POAmendmentNo
											Where POAM.PONo = @PONo
											Group By DCBD.DrCrNoteBillDetailNo
													,POAM.PONo
											) ttpo  On DCBD.DrCrNoteBillDetailNo = ttpo.DrCrNoteBillDetailNo
								
								Inner Join Purchase.POMain POM On ttpo.PONo = POM.PONo
								Where POM.PONo = @PONo
										And POM.DocumentDate = @DocumentDate
										And POM.CompanyNo = @CompanyNo
								Group By tt.DrCrType 
										,DCM.DocumentNoYearly
										,DCM.DocumentDate
										,DCM.VendorBillNo 
										,DCM.VendorBillDate 
										,CASE When DCM.ReasonNo = 5 Then 'Other :'+ DCM.OtherReason
												Else RSN.ReasonName
												End 
										,Nature.Nature 
								
							for xml path('tr')),'<td1>','<td style="text-align: Center;">')
							                   ,'<td2>','<td>')
		   									   ,'<td3>','<td>')				
											   ,'<td4>','<td>')
											   ,'<td5>','<td>')
		   									   ,'<td6>','<td>')
											   ,'<td7>','<td>')
											   ,'<td8>','<td>')
											   ,'<td9>','<td style="text-align: Right;">')	
									    	)
								
				   
	set @str=@str+'</table>'
	
	select @str
END
