CREATE PROCEDURE [Information].[PendingMRNforInvoiceForVendor]  
	-- Add the parameters for the stored procedure here
	@CompanyNo tinyint,
	@DocumentDate smalldatetime,
	@VendorLocationNo int
AS
BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;

    -- Insert statements for procedure here
	declare @str varchar(max)
    select @str=masterdata.JSForInformation('PMPO')
    set @str=@str+masterdata.CSSForInformation('PMPO')
	set @str=@str+'<table class="altrowstable" id="alternatecolorMRNInvoiceInfo">'
	set @str=@str+'<Caption><Font color="Maroon"><h3 align="Left"><b>Pending MRN for Invoice</b></h3></Font></Caption>'
    set @str=@str+'<tr><th>Sl. No.</th><th>Division</th><th>Department</th><th>Document No.</th><th>Document Date</th><th>Order Type</th><th>Order No.</th><th>MRN Amount</th><th>Created By</th><th>Authorized By</th></tr>'
	set @str=@str
				 +( replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					replace((
					
					Select ROW_NUMBER() over ( Order by td5 desc) as td1
					       ,td2
					       ,td3
					       ,td4
					       ,td5
					       ,td6
					       ,td7
					       ,td8
					       ,td9
					       ,td10 
					From (      -------For GRN 
						   Select  Isnull(DM.DivisionName,'') td2
									,Isnull(DP.DeptName,'') td3
									,GRNM.DocumentNoYearly  td4
									,convert(varchar(20),GRNM.DocumentDate,103) td5
									,SRC.SourceDocument  td6
									,Coalesce(GE.DocumentNoYearly,ST.DocumentNoYearly,DC.DocumentNoYearly,POA.DocumentNoYearly,'') td7
									,cast(Sum(GRNI.AcceptedQty*GRNI.Rate) as Decimal(15,2)) td8----Group by Clause For this
									,L.PrintingName td9
									,Isnull(L1.PrintingName,'') td10
							From  Inventory.GRNMain GRNM
							INNER JOIN GlobalData.GRNSourceDcoument SRC ON  GRNM.GRNSourceNo = SRC.GRNSourceNo
							LEFT JOIN payroll.tblDivisionMaster DM on GRNM.CompanyNo = DM.CompanyNo  
															And GRNM.DivisionNo=DM.DivisionNo
							LEFT JOIN GlobalData.FreightType FT ON GRNM.FreightTypeNo = FT.FreightTypeNo
							Left Join payroll.tblDeptMaster DP ON DP.DeptNo=GRNM.DeptNo
							Inner Join Inventory.GRNItemDetail GRNI ON GRNI.GRNNo=GRNM.GRNNo
							Left Join Finance.PurchaseBillItemDetail PBI ON PBI.GRNNo=GRNM.GRNNo
															And GRNI.ItemNo=PBI.ItemNo
															And ISNULL(GRNI.MakeNo,0)=ISNULL(PBI.MakeNo,0)
															And ISNULL(GRNI.TechnicalGradeNo,0)=ISNULL(PBI.TechnicalGradeNo,0)
							Left Join Purchase.POAmendmentMain POA ON POA.POAmendmentNo=GRNI.POAmendmentNo
							Left Join GateEntry.InvGateEntryMain GE ON GE.GateEntryNo=GRNI.GateEntryNo
							Left Join Inventory.InvStockTransferMain ST ON ST.InvStockTransferNo=GRNI.InvStockTransferNo
							Left Join Sales.DispatchChallan DC ON DC.DispatchChallanNo=GRNI.DispatchChallanNo
							Inner Join Security.Login L ON L.LoginNo=GRNM.CreatedBy
							Left Join Security.Login L1 ON L1.LoginNo=GRNM.AuthorizedBy
							Where GRNM.FormCode='IGRN' And GRNM.VendorLocationNo=@VendorLocationNo
									And GRNM.DocumentDate<=@DocumentDate
									And GRNM.CompanyNo=@CompanyNo
									And (PBI.PurchaseBillNo Is null)
									And GRNM.AuthorizedBy IS NOT NULL
							Group by DM.DivisionName
									,DP.DeptName
									,GRNM.DocumentNoYearly  
									,GRNM.DocumentDate 
									,GRNM.GRNSourceNo
									,SRC.SourceDocument  
									,GE.DocumentNoYearly,ST.DocumentNoYearly,DC.DocumentNoYearly,POA.DocumentNoYearly
									,L.PrintingName 
									,L1.PrintingName 
									
						Union All   ------- For SCMR
						
						Select  Isnull(DM.DivisionName,'') 
							   ,Isnull(DP.DeptName,'') 
							   ,GRNM.DocumentNoYearly  
							   ,convert(varchar(20),GRNM.DocumentDate,103) 
								,SRC.SourceDocument  
								,Coalesce(GE.DocumentNoYearly,POA.DocumentNoYearly,SCA.DocumentNoYearly,DC.DocumentNoYearly,'') 
								,Cast(SUM(GRSC.Quantity*GRSC.Rate) as Decimal(15,2))
								,L.PrintingName 
								,Isnull(L1.PrintingName,'')
						From  Inventory.GRNMain GRNM
							Inner JOIN GlobalData.GRNSourceDcoument SRC ON  GRNM.GRNSourceNo = SRC.GRNSourceNo
							LEFT JOIN payroll.tblDivisionMaster DM on GRNM.CompanyNo = DM.CompanyNo  
											And GRNM.DivisionNo=DM.DivisionNo
							LEFT JOIN GlobalData.FreightType FT ON GRNM.FreightTypeNo = FT.FreightTypeNo
							Left Join payroll.tblDeptMaster DP ON DP.DeptNo=GRNM.DeptNo
							Inner join Inventory.GRNRefDocDetail GRNR ON GRNR.GRNNo=GRNM.GRNNo
							Inner Join Inventory.GRNSCOrderDetail GRSC ON GRSC.GRNNo=GRNM.GRNNo
							Left Join Finance.ServiceBillItemDetail SBI ON SBI.GRNNo=GRNM.GRNNo	
																		And SBI.ItemNo=GRSC.ItemNo
							Left Join Purchase.POAmendmentMain POA ON POA.POAmendmentNo=GRNR.POAmendmentNo
							Left Join GateEntry.InvGateEntryMain GE ON GE.GateEntryNo=GRNR.GateEntryNo
							Left Join Sales.DispatchChallan DC ON DC.DispatchChallanNo=GRNR.DispatchChallanNo
							Left join Purchase.SCOrderAmendmentMain SCA ON SCA.SCOrderAmendmentNo=GRNR.SCOrderAmendmentNo
							Inner Join Security.Login L ON L.LoginNo=GRNM.CreatedBy
							Left Join Security.Login L1 ON L1.LoginNo=GRNM.AuthorizedBy
						Where FormCode='SCMR'
									And GRNM.VendorLocationNo=@VendorLocationNo
									And GRNM.DocumentDate<=@DocumentDate
									And GRNM.CompanyNo=@CompanyNo
									And SBI.ServiceBillNo Is null
									And GRNM.AuthorizedBy IS NOT NULL
						Group by DM.DivisionName
								,DP.DeptName
								,GRNM.DocumentNoYearly  
								,GRNM.DocumentDate 
								,GRNM.GRNSourceNo
								,SRC.SourceDocument 
								,GE.DocumentNoYearly,POA.DocumentNoYearly,SCA.DocumentNoYearly,DC.DocumentNoYearly
								,L.PrintingName 
								,L1.PrintingName 
						) tt
							for xml path('tr')),'<td1>','<td style="text-align: center;">')
										   ,'<td2>','<td>')
										   ,'<td3>','<td>')
										   ,'<td4>','<td>')
										   ,'<td5>','<td style="text-align: center;">')
										   ,'<td6>','<td>')
										   ,'<td7>','<td>')
										   ,'<td8>','<td style="text-align: right;">')
										   ,'<td9>','<td>')
										   ,'<td10>','<td>')
										   )
				   
	set @str=@str+'</table>'
	
	select @str
	
END
