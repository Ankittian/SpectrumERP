Reads: Finance.DrCrNoteMain, Finance.GSTAdvanceAdjustmentDetail, Finance.GSTAdvanceAdjustmentMain, Finance.GSTAdvanceReceiptMain, Finance.ReferenceMain, Finance.VoucherGSTDetail, GlobalData.FormMaster, Sales.CommercialInvoiceAdvanceDetail, Sales.InvoiceAdvanceDetail

CREATE PROCEDURE [Finance].[VoucherGSTDetailSave]
	@DocumentNo int, 
	@DocumentNoYearly varchar(30), 
	@DocumentDate smalldatetime, 
	@FormCode varchar(4), 
	@GSTTransactionTypeNo tinyint, 
	@VendorLocationNo int, 
	@AccountNo Int, 
	@HSNSACCodeNo smallint, 
	@InvoiceValue decimal(15,2), 
	@TaxableValue  decimal(15,2), 
	@IGSTInput  decimal(15,2), 
	@CGSTInput  decimal(15,2), 
	@SGSTInput  decimal(15,2), 
	@UTGSTInput decimal(15,2), 
	@IGSTOutput decimal(15,2), 
	@CGSTOutput decimal(15,2), 
	@SGSTOutput decimal(15,2), 
	@UTGSTOutput decimal(15,2), 
	@ITCCGSTInput decimal(15,2), 
	@ITCSGSTInput decimal(15,2), 
	@ITCIGSTInput decimal(15,2), 
	@ITCUTGSTInput decimal(15,2), 
	@IGSTPayable decimal(15,2),
	@CGSTPayable decimal(15,2), 
	@SGSTPayable decimal(15,2), 
	@UTGSTPayable decimal(15,2), 
	@IGSTExpense decimal(15,2), 
	@CGSTExpense decimal(15,2), 
	@SGSTExpense decimal(15,2), 
	@UTGSTExpense decimal(15,2), 
	@GSTRSectionNo smallint, 
	@VoucherNo int, 
	@GSTRegistrationTypeNo tinyint,
	
	-- 01 july 2017
	@IGSTAdvance decimal(15,2), 
	@CGSTAdvance decimal(15,2), 
	@SGSTAdvance decimal(15,2), 
	@UTGSTAdvance decimal(15,2),
	
	-- 11 july 2017
	@CessInput decimal(15,2), 
	@CessOutput decimal(15,2), 
	@CessExpense decimal(15,2), 
	@CessPayable decimal(15,2),
	@IsAdvanceAdjustment bit = 0,
	@ReferenceNo int = null,

	-- 22 December 2022
	@TCSAmount decimal(15,2) = NULL
AS
BEGIN
begin try
	
	SET NOCOUNT ON;
	
	declare @ParentVoucherGSTDetailNo int , @Multiple as float = 1
	 if @FormCode ='FIDC'
		 begin
		 
			if exists(select 1 from Finance.DrCrNoteMain  where DrCrTypeNo = 2 and SourceDocumentNo in(2,3,5,8 ) and DrCrNoteNo = @DocumentNo )
			begin
				set @Multiple = -1
			end
			if exists(select 1 from Finance.DrCrNoteMain  where DrCrTypeNo = 1 and SourceDocumentNo in(1,4,6,9) and DrCrNoteNo = @DocumentNo )
			begin
				set @Multiple = -1
			end
		 end
	else if @FormCode ='SDSR'
	  set @Multiple = -1	 
		 
	 
	 if @IsAdvanceAdjustment =1
	 begin
		 if @FormCode = 'SDEI'
		 begin
			 select @ParentVoucherGSTDetailNo = VoucherGSTDetailNo  from Sales.InvoiceAdvanceDetail iad
			 inner join Finance.ReferenceMain rm on rm.ReferenceNo = iad.ReferenceNo
			 inner join  finance.VoucherGSTDetail vd on vd.DocumentNo = rm.SourceDocumentno and vd.FormCode = 'GPIR'
			 and vd.HSNSACCodeNo = @HSNSACCodeNo and iad.InvoiceNo = @DocumentNo  and iad.ReferenceNo = @ReferenceNo
			 
			 if @ParentVoucherGSTDetailNo is null
				  select @ParentVoucherGSTDetailNo = VoucherGSTDetailNo   from Sales.InvoiceAdvanceDetail iad
				 inner join Finance.ReferenceMain rm on rm.ReferenceNo = iad.ReferenceNo
				 inner join finance.GSTAdvanceReceiptMain gsta on gsta.ReferenceNo=rm.ReferenceNo
				 inner join  finance.VoucherGSTDetail vd on vd.DocumentNo = gsta.GSTAdvanceReceiptNo  
				 where   iad.InvoiceNo = @DocumentNo  and iad.ReferenceNo = @ReferenceNo
		end	 
		 if @FormCode = 'SDCI' Or @FormCode ='ERCI'  --- ERCI is added on 28-09-2020 for eros commercial Invoice
		 begin
			 select @ParentVoucherGSTDetailNo = VoucherGSTDetailNo  from Sales.CommercialInvoiceAdvanceDetail iad
			 inner join Finance.ReferenceMain rm on rm.ReferenceNo = iad.ReferenceNo
			 inner join  finance.VoucherGSTDetail vd on vd.DocumentNo = rm.SourceDocumentno and vd.FormCode = 'GPIR'
			 and vd.HSNSACCodeNo = @HSNSACCodeNo and iad.CommercialInvoiceNo = @DocumentNo  and iad.ReferenceNo = @ReferenceNo
			 
			 if @ParentVoucherGSTDetailNo is null
				  select @ParentVoucherGSTDetailNo = VoucherGSTDetailNo   from Sales.InvoiceAdvanceDetail iad
				 inner join Finance.ReferenceMain rm on rm.ReferenceNo = iad.ReferenceNo
				 inner join finance.GSTAdvanceReceiptMain gsta on gsta.ReferenceNo=rm.ReferenceNo
				 inner join  finance.VoucherGSTDetail vd on vd.DocumentNo = gsta.GSTAdvanceReceiptNo  
				 where   iad.InvoiceNo = @DocumentNo  and iad.ReferenceNo = @ReferenceNo
			 
		 end 
	 	 if @FormCode = 'FGAA'
		 begin
		 --Comented by sandeep on 29/08/17
		 ---------------------------------
			 --select @ParentVoucherGSTDetailNo = VoucherGSTDetailNo  from  
				--Finance.GSTAdvanceAdjustmentMain gam
				--	inner join Finance.GSTAdvanceAdjustmentDetail gad on gad.GSTAdvanceAdjustmentNo = gam.GSTAdvanceAdjustmentNo
				--	inner join Finance.ReferenceMain rm on rm.ReferenceNo = gam.ReferenceNo
				--	inner join GlobalData.FormMaster fm on fm.FormNo = rm.SourceDocumentFormNo 
				--	inner join Finance.VoucherGSTDetail gst on gst.DocumentNo = rm.SourceDocumentNo and gst.FormCode = fm.FormCode 
				--where gam.GSTAdvanceAdjustmentNo = @DocumentNo and gad.AdvanceReferenceNo = @ReferenceNo
		---------------------------------
		--New Code By sandeep 		 on 29/08/17
			select @ParentVoucherGSTDetailNo = VoucherGSTDetailNo  from  
			Finance.GSTAdvanceAdjustmentMain gam
				inner join Finance.GSTAdvanceAdjustmentDetail gad on gad.GSTAdvanceAdjustmentNo = gam.GSTAdvanceAdjustmentNo
				inner join Finance.ReferenceMain rm on rm.ReferenceNo = gad.AdvanceReferenceNo
				inner join GlobalData.FormMaster fm on fm.FormNo = rm.SourceDocumentFormNo 
				inner join Finance.VoucherGSTDetail gst on gst.DocumentNo = rm.SourceDocumentNo and gst.FormCode = 'GPIR'
			where gam.GSTAdvanceAdjustmentNo = @DocumentNo and gad.AdvanceReferenceNo = @ReferenceNo
				
			  if @ParentVoucherGSTDetailNo is null
				select @ParentVoucherGSTDetailNo = VoucherGSTDetailNo  from  
				    Finance.GSTAdvanceAdjustmentMain gam
					inner join Finance.GSTAdvanceAdjustmentDetail gad on gad.GSTAdvanceAdjustmentNo = gam.GSTAdvanceAdjustmentNo
					inner join Finance.ReferenceMain rm on rm.ReferenceNo = gad.AdvanceReferenceNo
					inner join finance.GSTAdvanceReceiptMain gsta on gsta.ReferenceNo=rm.ReferenceNo
					inner join Finance.VoucherGSTDetail gst on gst.DocumentNo = gsta.GSTAdvanceReceiptNo 
				where gam.GSTAdvanceAdjustmentNo = @DocumentNo and gad.AdvanceReferenceNo = @ReferenceNo
			---------------------------------
		 end 
	end
	
	 
	 insert into Finance.VoucherGSTDetail(DocumentNo, DocumentNoYearly, DocumentDate, FormCode, GSTTransactionTypeNo, VendorLocationNo, 
			AccountNo, HSNSACCodeNo, InvoiceValue, TaxableValue, IGSTInput, CGSTInput, SGSTInput, UTGSTInput, IGSTOutput, CGSTOutput, 
			SGSTOutput, UTGSTOutput, ITCCGSTInput, ITCSGSTInput, ITCIGSTInput, ITCUTGSTInput, IGSTPayable, CGSTPayable, SGSTPayable, 
			UTGSTPayable, IGSTExpense, CGSTExpense, SGSTExpense, UTGSTExpense, GSTRSectionNo, VoucherNo, GSTRegistrationTypeNo,
			IGSTAdvance,CGSTAdvance,SGSTAdvance,UTGSTAdvance,CessInput,CessOutput,CessExpense,CessPayable ,IsAdvanceAdjustment,
			ParentVoucherGSTDetailNo,ReferenceNo,TCSAmount)
	 values (@DocumentNo, @DocumentNoYearly, @DocumentDate, @FormCode, @GSTTransactionTypeNo, @VendorLocationNo, 
			@AccountNo, @HSNSACCodeNo, @InvoiceValue * @Multiple, @TaxableValue * @Multiple, @IGSTInput * @Multiple, @CGSTInput  * @Multiple, @SGSTInput  * @Multiple, 
			@UTGSTInput  * @Multiple, @IGSTOutput  * @Multiple, @CGSTOutput  * @Multiple, @SGSTOutput  * @Multiple, @UTGSTOutput  * @Multiple, 
			@ITCCGSTInput  * @Multiple, @ITCSGSTInput  * @Multiple, @ITCIGSTInput  * @Multiple, @ITCUTGSTInput  * @Multiple, @IGSTPayable  * @Multiple, 
			@CGSTPayable  * @Multiple, @SGSTPayable  * @Multiple, @UTGSTPayable  * @Multiple, @IGSTExpense  * @Multiple, @CGSTExpense  * @Multiple, 
			@SGSTExpense  * @Multiple, @UTGSTExpense  * @Multiple, @GSTRSectionNo , @VoucherNo, @GSTRegistrationTypeNo,
			@IGSTAdvance, @CGSTAdvance , @SGSTAdvance, @UTGSTAdvance,@CessInput * @Multiple
			,@CessOutput * @Multiple,@CessExpense,@CessPayable ,@IsAdvanceAdjustment,
			@ParentVoucherGSTDetailNo , @ReferenceNo,@TCSAmount)
	 
	 select 1, IDENT_CURRENT ('Finance.VoucherGSTDetail') 
end try

begin catch
	select 0,ERROR_MESSAGE ()
	 
end catch

    
END
