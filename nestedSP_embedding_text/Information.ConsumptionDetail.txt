CREATE PROCEDURE [Information].[ConsumptionDetail] 
	-- Add the parameters for the stored procedure here
	@CompanyNo tinyint,
	@DocumentDate smalldatetime,
	@ItemNo int
AS
BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;

    -- Insert statements for procedure here
	Declare @LastConsumptionDays tinyint
	Select @LastConsumptionDays=isnull(LastConsumptionDays,0)
	From Inventory.Setting
	where CompanyNo=@CompanyNo    
    
	declare @str varchar(max)
    select @str=masterdata.JSForInformation('MMPR')
    set @str=@str+masterdata.CSSForInformation('MMPR')
	set @str=@str+'<table class="altrowstable" id="alternatecolorConsInfo">'
	set @str=@str+'<Caption><Font color="Maroon"><h3 align="Left"><b>Last '+convert(varchar(3),@LastConsumptionDays)+'Days Issue</b></h3></Font></Caption>'
    set @str=@str+'<tr><th>Sl. No.</th><th>Division</th><th>Warehouse</th><th>Document No.</th><th>Document Date</th><th>Issued Quantity</th></tr>'
	set @str=@str
				  +(replace(
					replace(
					replace(
					replace(
					replace(
					replace((    
							Select  Row_number() over (Order By ld.DocumentDate,ld.DocumentNoYearly) as [td1]
							       ,isnull(DV.DivisionName,'') AS [td2]
							       ,WH.WareHouseName AS [td3]
							       ,ld.DocumentNoYearly AS [td4]
							       ,convert(varchar,ld.DocumentDate,103) AS [td5]
							       ,Convert(Decimal(12,3),Sum(isnull(ld.IssuedQty,0))) AS [td6]
							From inventory.invlotDetail ld
								  Left Join payroll.tblDivisionMaster DV on ld.DivisionNo=DV.DivisionNo
								  Inner Join masterdata.mWarehouse WH on WH.WareHouseNo=ld.WarehouseNo
								  --Left Join masterdata.mMakeMaster MM on MM.MakeNo=ld.MakeNo
							Where ld.IssuedQty is not null	
								And ld.DocumentDate between @DocumentDate-@LastConsumptionDays and @DocumentDate
								And ld.ItemNo=@ItemNo
								And ld.CompanyNo=@CompanyNo	 
							Group By DV.DivisionName,WH.WareHouseName,ld.DocumentNoYearly,ld.DocumentDate	     
							Order By ld.DocumentDate,ld.DocumentNoYearly
							for xml path('tr')),'<td1>','<td style="text-align: center;">')
							                   ,'<td2>','<td>')
											   ,'<td3>','<td>')
											   ,'<td4>','<td>')
											   ,'<td5>','<td style="text-align: center;">')
											   ,'<td6>','<td style="text-align: right;">'))
				   
	set @str=@str+'</table>'
	
	select @str
END
