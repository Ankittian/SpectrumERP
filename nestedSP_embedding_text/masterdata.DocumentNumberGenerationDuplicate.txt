CREATE PROCEDURE [masterdata].[DocumentNumberGenerationDuplicate]
	-- Add the parameters for the stored procedure here
	 @DocumentDate smalldatetime
	,@DocumentSettingNo smallint
	,@DocumentNoYearly varchar(50)
	,@Status tinyint output
	,@Remarks nvarchar(4000) output
AS
BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;

    -- Insert statements for procedure here

  BEGIN TRY

	declare @Padding tinyint,@Frequency tinyint,@CompanyNo tinyint,@NumberStartsFrom smallint

	select @Padding = Padding 
			,@Frequency = Frequency 
			,@CompanyNo = CompanyNo
			,@NumberStartsFrom = NumberStartsFrom
	from masterdata.DocumentSetting 
	where DocumentSettingNo=@DocumentSettingNo

	declare @YearNo tinyint

	select @YearNo = YearNo
	from masterdata.mFYear
	where @DocumentDate between StartDate and EndDate

	declare @str nvarchar(max)

	--Alt 2 9 is ↔
	declare @strMultiTable nvarchar(max)
	select @strMultiTable = replace(substring((select distinct ' Union All '+N'select 1 [Col] from '+FormMainTableinDB+' mT 
																where documentsettingno = @DocumentSettingNo and CompanyNo = @CompanyNo 
																and @FrequencyQuery and DocumentNoYearly=''@DocumentNoYearly'''
										from masterdata.documentFormdetail fd
											inner join GlobalData.FormMaster fm on fm.formno=fd.formNo
										where documentsettingno = @documentsettingno 
										for xml path('')),12,8000),'&#x0D;','')

	declare @query nvarchar(max)
	select @query='select isnull(count(Col),0) from ('+@strMultiTable+')tt'

	declare @FrequencyQuery varchar(500)
	IF @Frequency = 1		--Date
	BEGIN
		set @FrequencyQuery = ' day(cast(convert(varchar(11),DocumentDate) as smalldatetime)) = day(''@DocumentDate'')
							 and month(cast(convert(varchar(11),DocumentDate) as smalldatetime)) = month(''@DocumentDate'')
							 and year(cast(convert(varchar(11),DocumentDate) as smalldatetime)) = year(''@DocumentDate'')'
	END
	ELSE IF @Frequency = 2	--Month
	BEGIN
		set @FrequencyQuery = ' month(cast(convert(varchar(11),DocumentDate) as smalldatetime)) = month(''@DocumentDate'')
							and year(cast(convert(varchar(11),DocumentDate) as smalldatetime)) = year(''@DocumentDate'')'
	END
	ELSE IF @Frequency = 3	--Year
	BEGIN
		set @FrequencyQuery = ' YearNo = @YearNo'
	END

	set @query=replace(@query,'@FrequencyQuery',@FrequencyQuery)

	set @query=replace(@query,'@Padding',@Padding)
	set @query=replace(@query,'@YearNo',isnull(@YearNo,0))
	set @query=replace(@query,'@CompanyNo',@CompanyNo)
	set @query=replace(@query,'@DocumentDate',@DocumentDate)
	set @query=replace(@query,'@DocumentSettingNo',@DocumentSettingNo)
	set @query=replace(@query,'@DocumentNoYearly',@DocumentNoYearly)

	declare @DocumentTable Table (NoOfRows tinyint)
	insert into @DocumentTable
	exec(@query)
				
	select @Status=1,@Remarks=NoOfRows
	from @DocumentTable
	
  End Try

  Begin Catch
	select @Status=0,@Remarks=ERROR_MESSAGE()
  End Catch

END
