CREATE PROCEDURE [Information].[AgreementDetail] 
	-- Add the parameters for the stored procedure here
	@CompanyNo tinyint,
	@DocumentDate smalldatetime,
	@ItemNo int
	
AS
BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;

    -- Insert statements for procedure here
	

	declare @str varchar(max)
    select @str=masterdata.JSForInformation('PRFQ')
    set @str=@str+masterdata.CSSForInformation('PRFQ')
	set @str=@str+'<table class="altrowstable" id="alternatecolorAgreementDetailInfo">'
	set @str=@str+'<Caption><Font color="Maroon"><h3 align="Left"><b>Agreement  Detail</b></h3></Font></Caption>'
    set @str=@str+'<tr><th>Sl. No.</th><th>Vendor Name</th><th>Document No.</th><th>Document Date</th><th>Make</th><th>Agreement Type</th><th>Agreement Qty.</th><th>Agreement Rate</th><th>Agreement Amount</th></tr>'
	set @str=@str
				 +( 
					replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					 (    
					      Select   ROW_NUMBER() over (order By  AM.DocumentDate desc,AM.DocumentNoyearly Desc)
                                       as td1,
                                   VD.Vendorname td2,
								   AM.DocumentNoyearly td3,
								   Convert(VArchar(20),AM.DocumentDate,103) td4,
								   Isnull(MM.MakeName,'') td5,
								   AT.AgreementType td6,
								   Isnull(AI.Quantity,0) td7,
								   Cast(Isnull(AI.Rate,0) as decimal(15,2)) td8,
								   Cast(isnull(AI.Amount,0)as decimal(15,2)) td9      
							From Purchase.AgreementMain as AM
							Inner Join Purchase.AgreementItemDetail AI ON AI.AgreementNo=AM.AgreementNo
							LEFT Join masterdata.mMakeMaster MM ON MM.MakeNo=AI.MakeNo
							Inner Join Globaldata.AgreementType AT on AT.AgreementTypeNo=AM.AgreementTypeNo
							Inner Join masterdata.mVendorLocationDetail as VD
							               On AM.VendorLocationNo=VD.VendorLocationNo
							Where AM.CompanyNo=@CompanyNo
							And @DocumentDate between AM.AgreementFrom And AM.AgreementTo
							And AI.ItemNo=@ItemNo
							order By  AM.DocumentDate desc,AM.DocumentNoyearly Desc
							
		               	For xml path('tr')
		               	),'<td1>','<td style="text-align: center;">')
		               	 ,'<td2>','<td>')
		   				 ,'<td3>','<td>')		
						 ,'<td4>','<td style="text-align: Center;">')
						 ,'<td5>','<td>')
						 ,'<td6>','<td>')
						 ,'<td7>','<td style="text-align: Right;">')
						 ,'<td8>','<td style="text-align: Right;">')
						 ,'<td9>','<td style="text-align: Right;">')
					)
								
				   
	set @str=@str+'</table>'
	
	select @str
END
