Stored procedure Inventory.IMSTInterUnitFinancePosting: stock tranfer Online FInance Posting. Called from Finance.FinancePostingFillValuesDocumentDetails.
Reads: Finance.ARDFinanceBookRule, Finance.ARDIntermediatePostingAccountRule, Finance.ARDItemRule, FixedAsset.AssetMain, FixedAsset.AssetTagMain, masterdata.mAccountMaster, masterdata.mFBMain, masterdata.mItemMaster, Purchase.POAccountAssignmentDetail, Purchase.POAmendmentMain, Purchase.POAmendmentTransportationRoute
Parameters: @ItemDetail xml, @CompanyNo tinyint, @VendorLocationNo int, @NetAmount money, @ItemNos varchar
Calls procedures: Inventory.IGRNFinancePosting

Create PROCEDURE [Inventory].[IMSTInterUnitFinancePosting] 
	@ItemDetail xml ,
	@CompanyNo tinyint ,
	@VendorLocationNo int  ,
	@NetAmount money ,
	@ItemNos varchar(max)
	
AS
BEGIN

	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;

begin try
declare @IsManagingExpenseAccount bit

Create table #ItemDetail (ReferenceNo int,ReferenceType varchar(4),ItemNo int,MakeNo smallint,
		TechnicalGradeNo tinyint,AcceptedQty decimal(15,3), Rate decimal(18,6),GateEntryNo int 
		,RouteIdentity decimal(10,8) , IncludedTax decimal(18,6) , ChallanQty decimal(15,3),ReceivedQty decimal(15,3)
		,TransporterNo int,MAXRouteIdentity decimal(10,8), MINRouteIdentity decimal(10,8)
		,FromLocationNo smallint,ToLocationNo smallint
		)


create table #Posting ([Add to Basic] bit,Particular varchar(500),[Debit Amount] decimal(15,2) , 
		[Credit Amount] decimal(15,2),[Account Name] int, [Ref. Detail] varchar(50),[Cost Center] varchar(50),
		[Amount] decimal(15,2), [Type] tinyint,SlNo tinyint ,MiscChargeNo int , TaxCodeMiscChargesDetailNo int,
			ItemNos varchar(200))

		declare @ItemTable table (ItemNo int,BasicAmount money,Id int,Qty decimal(12,3),Rate money)
		insert into @ItemTable
		select Col1, sum(convert(money,Col2)), Col3, sum(CONVERT(decimal(12,3),Col4)), sum(CONVERT(money,Col5))
		from masterdata.Split_RowColumn(@ItemNos,'`','~')
		group by Col1, Col3
		
			declare @IMST_Flag tinyint ,@IMST_CompanyNo int,@ProcessingTypeNo tinyint,@IsReceivingPosting bit, @FBNo smallint
			
			SELECT @IMST_Flag=v.value('Flag[1]','tinyint')
					,@IMST_CompanyNo=v.value('Id[1]','int')
					,@FBNo=v.value('FBNo[1]','smallint')
					,@ProcessingTypeNo=v.value('ProcessingTypeNo[1]','tinyint')
					,@IsReceivingPosting=v.value('IsReceivingPosting[1]','bit')
			FROM @ItemDetail.nodes('/DocumentElement/Table') x(v)
			
			if @IMST_Flag=3
			begin
			
			;with cte as 
	(
		SELECT v.value('ReferenceNo[1]','int') AS ReferenceNo,
		v.value('ReferenceType[1]','varchar(4)') ReferenceType,
		v.value('ItemNo[1]','int') ItemNo,
		v.value('MakeNo[1]','smallint') MakeNo,
		v.value('TechnicalGradeNo[1]','tinyint') TechnicalGradeNo,
		v.value('AcceptedQty[1]','decimal(15,3)') AcceptedQty,
		v.value('Rate[1]','decimal(18,6)') Rate,
		v.value('GateEntryNo[1]','int') GateEntryNo,
		v.value('RouteIdentity[1]','decimal(10,8)') RouteIdentity,
		v.value('IncludedTax[1]','decimal(18,6)') IncludedTax,
		v.value('ChallanQty[1]','decimal(15,3)') ChallanQty,
		v.value('ReceivedQty[1]','decimal(15,3)') ReceivedQty,
		v.value('TransporterNo[1]','int') TransporterNo,
		v.value('FromLocationNo[1]','int') FromLocationNo,
		v.value('ToLocationNo[1]','int') ToLocationNo
		FROM @ItemDetail.nodes('/DocumentElement/Table') x(v)
	)
	Insert into #ItemDetail (ReferenceNo ,ReferenceType ,ItemNo ,MakeNo ,TechnicalGradeNo ,AcceptedQty , 
	Rate ,GateEntryNo ,RouteIdentity , IncludedTax , ChallanQty,ReceivedQty	,TransporterNo ,
	FromLocationNo ,ToLocationNo ,MAXRouteIdentity , MINRouteIdentity )
	select ReferenceNo ,ReferenceType ,ItemNo ,MakeNo ,TechnicalGradeNo ,AcceptedQty , 
	Rate ,GateEntryNo ,RouteIdentity , IncludedTax , ChallanQty ,ReceivedQty ,TransporterNo ,FromLocationNo
	,ToLocationNo
	,(select max(ptd.RouteIdentity ) from Purchase.POAmendmentTransportationRoute ptd
				where ptd.POAmendmentNo =cte.ReferenceNo and ReferenceNo is not null and cte.ReferenceType ='PMPO'
				and ptd.RouteIdentity - cte.RouteIdentity between 
				CONVERT(decimal(18,8),'0.'+REPLICATE ('0', LEN(CONVERT(int,CONVERT(float,REVERSE (cte.RouteIdentity))))) +'1')
					and CONVERT(decimal(18,8),'0.'+REPLICATE ('0', LEN(CONVERT(int,CONVERT(float,REVERSE (cte.RouteIdentity))))) +'9'))
	,(select MIn(ptd.RouteIdentity ) from Purchase.POAmendmentTransportationRoute ptd
				where ptd.POAmendmentNo =cte.ReferenceNo and ReferenceNo is not null and cte.ReferenceType ='PMPO'
				and ptd.RouteIdentity - cte.RouteIdentity between 
				CONVERT(decimal(18,8),'0.'+REPLICATE ('0', LEN(CONVERT(int,CONVERT(float,REVERSE (cte.RouteIdentity))))) +'1')
					and CONVERT(decimal(18,8),'0.'+REPLICATE ('0', LEN(CONVERT(int,CONVERT(float,REVERSE (cte.RouteIdentity))))) +'9'))
	from cte where ItemNo is not null
	
			if @ProcessingTypeNo=1 and @IsReceivingPosting=1
			  begin
			    if exists(select 1 from #ItemDetail where ReferenceNo is not null) 
				 insert into #Posting 
					exec inventory.IGRNFinancePosting @ItemDetail,@CompanyNo ,@VendorLocationNo
			     
			     else
			     begin
			     
			     Insert into #Posting  select cast(0 as bit) [Add to Basic],'Stock Account-DIRECT' as Particular
				,convert(decimal(15,2),sum(Amount)) [Debit Amount]
				,convert(decimal(15,2),null) [Credit Amount]
				, ItemAC.AccountNo  [Account Name]
				,cast(null as varchar) [Ref. Detail]
				,cast(null as varchar) [Cost Center]
				,SUM(Amount) [Amount]
				,1 [Type]
				,ROW_NUMBER()over(order by ItemAC .AccountNo ) SlNo
				,null MiscChargeNo
				,null TaxCodeMiscChargesDetailNo
				,'' ItemNos
			from (select  SUM(AcceptedQty * Rate) Amount ,ItemNo ,MakeNo,TechnicalGradeNo  
							from #ItemDetail  where ReferenceNo is null
							group by ItemNo,MakeNo,TechnicalGradeNo)I
				inner join (select ItemNo,max(IR.AccountNo)AccountNo
							from Finance.ARDItemRule IR
							inner join masterdata.mAccountMaster AM on AM.AccountNo=IR.AccountNo
							where AM.AccountTypeNo=14 group by ItemNo ) ItemAC on ItemAC.ItemNo=I.ItemNo

			group by ItemAC.AccountNo 
							
				Insert into #Posting 
		select cast(0 as bit) [Add to Basic]
				,'Stock In Transit' as Particular
				,convert(decimal(15,2), null) [Debit Amount]
				,convert(decimal(15,2),sum(Amount)) [Credit Amount]
				,IR.AccountNo [Account Name]
				,cast(null as varchar) [Ref. Detail]
				,cast(null as varchar) [Cost Center]
				,SUM(Amount) [Amount]
				,2 [Type]
				,ROW_NUMBER()over(order by IR.AccountNo ) + (select isnull(max(SlNo),0) from #Posting) SlNo
				,null MiscChargeNo
				,null TaxCodeMiscChargesDetailNo
				,'' ItemNos
			from (select  
				sum((AcceptedQty * Rate)) Amount ,
						ItemNo ,MakeNo,TechnicalGradeNo  
							from #ItemDetail where 
							(RouteIdentity is null or RouteIdentity <> MINRouteIdentity )
							group by ItemNo,MakeNo,TechnicalGradeNo)I
				left join (select ItemNo,max(IR.AccountNo)AccountNo
							from 
							Finance.ARDIntermediatePostingAccountRule AR
							inner join Finance.ARDItemRule IR on IR.ARDNo=AR.ARDNo and IR.AccountNo = AR.AccountNo
							inner join masterdata.mAccountMaster AM on AM.AccountNo=IR.AccountNo
							where AM.AccountTypeNo=54 and AR.IntermediateAccountTypeNo =18
							group by ir.ItemNo ) IR on IR.ItemNo=I.ItemNo
			
		group by IR.AccountNo 
		
			     end 
			      
			   end
			else
				
			  begin   
			 
				Insert into #Posting 
		select cast(0 as bit) [Add to Basic]
				,'Stock Account-1' as Particular
				,convert(decimal(15,2),NUll) [Debit Amount]
				,convert(decimal(15,2),sum(Amount)) [Credit Amount]
				, ItemAC.AccountNo  [Account Name]
				,cast(null as varchar) [Ref. Detail]
				,cast(null as varchar) [Cost Center]
				,SUM(Amount) [Amount]
				,2 [Type]
				,ROW_NUMBER()over(order by ItemAC .AccountNo ) SlNo
				,null MiscChargeNo
				,null TaxCodeMiscChargesDetailNo
				,'' ItemNos
			from (select  SUM(AcceptedQty * Rate) Amount ,ItemNo ,MakeNo,TechnicalGradeNo  
							from #ItemDetail 
							group by ItemNo,MakeNo,TechnicalGradeNo)I
				outer apply (SELECT pad.AccountNo FROM Purchase.POAccountAssignmentDetail pad
						inner join Purchase.POAmendmentMain pam on pam.PONo = pad.PONo
						inner join #ItemDetail PID on PID.ReferenceNo = pam.POAmendmentNo 
						where PID.ReferenceType ='PMPO' and pad.Flag ='IM' and pad.Id = PID.ItemNo
						and pad.AccountTypeNo in (14 ,16)
						and PID.ItemNo = I.ItemNo 
						group by pad.AccountNo 
						union all
						SELECT pad.AccountNo  FROM Purchase.POAccountAssignmentDetail pad
						inner join Purchase.POAmendmentMain pam on pam.PONo = pad.PONo
						inner join #ItemDetail PID on PID.ReferenceNo = pam.POAmendmentNo 
						where PID.ReferenceType ='PMPO' and pad.Flag ='IM3' and pad.Id = PID.ItemNo
						and pad.AccountTypeNo in (54)
						and PID.ItemNo = I.ItemNo 
						group by pad.AccountNo  
						union all
						select top 1 (IR.AccountNo)AccountNo 
							from Finance.ARDItemRule IR
							inner join masterdata.mAccountMaster AM on AM.AccountNo=IR.AccountNo
							inner join  #ItemDetail ID on ID.ItemNo = IR.ItemNo 
							where AM.AccountTypeNo=14 and ID.ReferenceNo is null
							and ir.ItemNo = I.ItemNo 
							
						) ItemAC 
						
			group by ItemAC .AccountNo 
		 
				Insert into #Posting 
		select cast(0 as bit) [Add to Basic]
				,'Stock In Transit' as Particular
				,convert(decimal(15,2), sum(Amount)) [Debit Amount]
				,convert(decimal(15,2),null) [Credit Amount]
				,IR.AccountNo [Account Name]
				,cast(null as varchar) [Ref. Detail]
				,cast(null as varchar) [Cost Center]
				,SUM(Amount) [Amount]
				,1 [Type]
				,ROW_NUMBER()over(order by IR.AccountNo ) + (select isnull(max(SlNo),0) from #Posting) SlNo
				,null MiscChargeNo
				,null TaxCodeMiscChargesDetailNo
				,'' ItemNos
			from (select  
				sum((AcceptedQty * Rate)) Amount ,
						ItemNo ,MakeNo,TechnicalGradeNo  
							from #ItemDetail where 
							(RouteIdentity is null or RouteIdentity <> MINRouteIdentity )
							group by ItemNo,MakeNo,TechnicalGradeNo)I
				left join (select ItemNo,max(IR.AccountNo)AccountNo
							from 
							Finance.ARDIntermediatePostingAccountRule AR
							inner join Finance.ARDItemRule IR on IR.ARDNo=AR.ARDNo and IR.AccountNo = AR.AccountNo
							inner join masterdata.mAccountMaster AM on AM.AccountNo=IR.AccountNo
							where AM.AccountTypeNo=54 and AR.IntermediateAccountTypeNo =18
							group by ir.ItemNo ) IR on IR.ItemNo=I.ItemNo
			
		group by IR.AccountNo 
     
			  end
			end
			   
			else if @IMST_Flag=1 
			begin
				declare @AssetLotDetail table (AssetTagNo int,InvLotDetailNo int,Qty decimal(12,3))
				insert into @AssetLotDetail
				SELECT v.value('AssetTagNo[1]','int') AS AssetTagNo
					,v.value('InvLotDetailNo[1]','int') AS InvLotDetailNo
					,v.value('Qty.[1]','decimal') AS Qty
				FROM @ItemDetail.nodes('/DocumentElement/Table') x(v)
			
				Insert into #Posting
				select top 1 
					 cast(0 as bit) [Add to Basic],'Inter FB Account' Particular,
					convert(decimal(15,2),case when @IsReceivingPosting=1 then null else @NetAmount end) [Debit Amount]
					,convert(decimal(15,2),case when @IsReceivingPosting=1 then @NetAmount else null end) [Credit Amount]
					,am.AccountNo [Account Name]
					,cast(null as varchar) [Ref. Detail],cast(null as varchar)[Cost Center] ,@NetAmount [Amount],
					case when @IsReceivingPosting=1 then 2 else 1 end [Type]
					,1 SlNo
					,cast(null as smallint) MiscChargeNo,null TaxCodeMiscChargesDetailNo ,CAST(null as varchar(5000)) ItemNos
				from masterdata.mFBMain FBM 
					inner join Finance.ARDFinanceBookRule FBR on FBR.FBNo=FBM.FBNo
					inner join masterdata.mAccountMaster AM on AM.AccountNo=FBR.AccountNo
					--									and AM.AccountTypeNo=1
				where FBM.FBNo=@IMST_CompanyNo 
					and exists(select 1 from Finance.ARDFinanceBookRule  where ARDNo = FBR.ARDNo 
									and FBNo = @FBNo)--and FBM.IsDefault=1
					and FBM.FBTypeNo=1
					
				union all
				select cast(0 as bit) [Add to Basic] ,'Stock Account-2' Particular ,
				convert(decimal(15,2),case when @IsReceivingPosting=1 then sum(I.BasicAmount) else null end) [Debit Amount]
				,convert(decimal(15,2),case when @IsReceivingPosting=1 then null else sum(I.BasicAmount) end) [Credit Amount]
				,IR.AccountNo [Account Name] 
				,cast(null as varchar) [Ref. Detail] ,cast(null as varchar) [Cost Center] 
				,sum(I.BasicAmount) Amount ,
				case when @IsReceivingPosting=1 then 1 else 2 end [Type] 
				,ROW_NUMBER() over(order by isnull(cast(IR.AccountNo as int),I.ItemNo))+1 SlNo
				,CAST(null as smallint) MiscChargeNo,null TaxCodeMiscChargesDetailNo,CAST(null as varchar(5000))ItemNos
				from @ItemTable I
					inner join masterdata.mItemMaster IM on IM.ItemNo=I.ItemNo
					left join (select ItemNo,max(IR.AccountNo) AccountNo
								from Finance.ARDItemRule IR
								inner join masterdata.mAccountMaster AM on AM.AccountNo=IR.AccountNo
								and AM.AccountTypeNo=14 Group by ItemNo) IR on IR.ItemNo=I.ItemNo
				where not exists (select 1 from @AssetLotDetail where AssetTagNo is not null)
				group by IR.AccountNo,isnull(cast(IR.AccountNo as int),I.ItemNo)
				
				union all
					select cast(0 as bit) [Add to Basic] ,'Basic Amount' Particular 
						,convert(decimal(15,2),null) [Debit Amount]
						,sum(Amount+DA.DiffAmt) [Credit Amount]
						,tt.AccountNo [Account Name] 
						,cast(null as varchar) [Ref. Detail] ,cast(null as varchar) [Cost Center] 
						,sum(Amount+DA.DiffAmt) [Amount] ,2 [Type] 
						,2 SlNo
						,CAST(null as smallint) MiscChargeNo,null TaxCodeMiscChargesDetailNo,CAST(null as varchar(5000))ItemNos
					from (
							select sum(ALD.Qty*IT.BasicAmount/IT.Qty) Amount
								,AM.AccountNo
								,IT.ItemNo
							from @AssetLotDetail ALD
								inner join FixedAsset.AssetTagMain ATM on ATM.AssetTagNo=ALD.AssetTagNo
								inner join FixedAsset.AssetMain AM on AM.AssetNo=ATM.AssetNo
								inner join @ItemTable IT on IT.ItemNo=ATM.ItemNo
							where ALD.AssetTagNo is not null
							group by AM.AccountNo,IT.ItemNo
						) tt
						inner join @ItemTable IT on IT.ItemNo=tt.ItemNo
						cross apply (select IT.BasicAmount-tt.Amount DiffAmt) DA
					group by tt.AccountNo
				
			end
			else
			begin
				Insert into #Posting
				select cast(0 as bit) [Add to Basic] ,'Stock Account-3' Particular 
					,sum(v.value('Amount[1]','money')) [Debit Amount] ,CAST(null as money) [Credit Amount] 
					,IR.AccountNo [Account Name] 
					,cast(null as varchar) [Ref. Detail] ,cast(null as varchar) [Cost Center] ,sum(v.value('Amount[1]','money')) Amount ,1 [Type] 
					,ROW_NUMBER() over(order by isnull(cast(IR.AccountNo as int),IM.ItemNo)) SlNo
					,CAST(null as smallint) MiscChargeNo,null TaxCodeMiscChargesDetailNo,CAST(null as varchar(5000))ItemNos
				FROM @itemdetail.nodes('/DocumentElement/Table') x(v)
					inner join masterdata.mItemMaster IM on IM.ItemNo=v.value('Id[1]','int')
					left join (select ItemNo,IR.AccountNo
								from Finance.ARDItemRule IR
								inner join masterdata.mAccountMaster AM on AM.AccountNo=IR.AccountNo
														and AM.AccountTypeNo=14) IR on IR.ItemNo=IM.ItemNo
				where v.value('Flag[1]','tinyint')=2
				group by IR.AccountNo,isnull(cast(IR.AccountNo as int),IM.ItemNo)
				
				union all
				select cast(0 as bit) [Add to Basic] ,'Stock Account-4' Particular 
					,CAST(null as money) [Debit Amount] ,sum(I.BasicAmount) [Credit Amount] 
					,IR.AccountNo [Account Name] 
					,cast(null as varchar) [Ref. Detail] ,cast(null as varchar) [Cost Center] ,sum(I.BasicAmount) Amount ,2 [Type] 
					,ROW_NUMBER() over(order by isnull(cast(IR.AccountNo as int),I.ItemNo))
						+(select count(1) from @ItemDetail.nodes('/DocumentElement/Table') x(v)) SlNo
					,CAST(null as smallint) MiscChargeNo,null TaxCodeMiscChargesDetailNo,CAST(null as varchar(5000))ItemNos
				from @ItemTable I
					inner join masterdata.mItemMaster IM on IM.ItemNo=I.ItemNo
					left join (select ItemNo,IR.AccountNo
								from Finance.ARDItemRule IR
								inner join masterdata.mAccountMaster AM on AM.AccountNo=IR.AccountNo
													and AM.AccountTypeNo=14) IR on IR.ItemNo=I.ItemNo
				group by IR.AccountNo,isnull(cast(IR.AccountNo as int),I.ItemNo)
			end
	  

	 

 if exists(select 1 from #Posting having abs(SUM([Credit Amount])- SUM([Debit Amount]))<=.5)
 begin
	update PD set [Debit Amount] = [Debit Amount] + ISNULL(diff,0)
		from
		#Posting PD
		cross join (select SUM([Credit Amount])- SUM([Debit Amount]) Diff from #Posting) tt
		where PD.SlNo  = 1
	
	update PD set [Credit Amount] = [Credit Amount] + ISNULL(diff,0)
		from
		#Posting PD
		cross join (select SUM([Debit Amount])- SUM([Credit Amount]) Diff from #Posting) tt
		where PD.SlNo  = 1
			
 end
 
 
select * from #Posting
 where isnull([Debit Amount],0) >0 or isnull([Credit Amount],0) >0
   
   
   
END TRY
	
	BEGIN CATCH
		select ERROR_MESSAGE()
		INSERT INTO [masterdata].[mErrorLog]
			   ([ErrorDate]
			   ,[ErrorNo]
			   ,[ErrorSeverity]
			   ,[ErrorState]
			   ,[ErrorProcedure]
			   ,[ErrorLine]
			   ,[ErrorDescription]
			   ,[UserName]
			   ,[FormCode])
		VALUES
			   (getdate()
			   ,ERROR_NUMBER()
			   ,ERROR_SEVERITY()
			   ,ERROR_STATE()
			   ,ERROR_PROCEDURE()
			   ,ERROR_LINE()
			   ,ERROR_MESSAGE()
			   ,SUSER_SNAME()
			   ,'IGRN')
	END CATCH
END
