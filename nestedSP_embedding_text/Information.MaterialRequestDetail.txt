CREATE PROCEDURE [Information].[MaterialRequestDetail]  
	-- Add the parameters for the stored procedure here
	@CompanyNo tinyint,
	@DocumentDate smalldatetime,
	@ItemNo int
AS
BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;

    -- Insert statements for procedure here
	declare @str varchar(max)
    select @str=masterdata.JSForInformation('MMPR')
    set @str=@str+masterdata.CSSForInformation('MMPR')
	set @str=@str+'<table class="altrowstable" id="alternatecolorMRInfo">'
	set @str=@str+'<Caption><Font color="Maroon"><h3 align="Left"><b>Pending Material Request</b></h3></Font></Caption>'
    set @str=@str+'<tr><th>Sl. No.</th><th>Division</th><th>Department</th><th>Make</th><th>Document No.</th><th>Document Date</th><th>MR Quantity</th><th>PR Quantity</th><th>Issued Quantity</th><th>Balance Quantity</th><th>Created By</th><th>Authorized By</th></tr>'
	set @str=@str
				 +( replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					replace(
					replace((    
						      Select Row_number() over (Order By IR.DocumentDate,IR.DocumentNoYearly) as td1
						            ,isnull(DV.DivisionName,'') As [td2]
						            ,isnull(DM.DeptName,'') As [td3]
						            ,isnull(MM.MakeName,'') As [td4]
                                    ,IR.DocumentNoYearly As [td5]
                                    ,convert(varchar,IR.DocumentDate,103) As [td6]
                                    ,ID.Quantity As [td7]
                                    ,isnull(IDIT.IndentQty,0) As [td8]
                                    ,isnull(ID.IssueQty,0) As [td9]
                                    ,(ID.Quantity-isnull(ID.IssueQty,0)) As [td10]
                                    ,L.PrintingName [td11]
                                    ,Isnull(L1.PrintingName,'') [td12]
							From Inventory.IndentRequisitionMain IR
							Inner Join Inventory.IndentRequisitionItemDetail ID on ID.IndentRequisitionNo=IR.IndentRequisitionNo
							Inner Join masterdata.mItemMaster IM on IM.ItemNo=ID.ItemNo
							Inner Join payroll.tblDeptMaster DM on DM.DeptNo=IR.DeptNo
							Left Join payroll.tblDivisionMaster DV on DV.DivisionNo=IR.DivisionNo
							Left Join masterdata.mMakeMaster MM on MM.MakeNo=ID.MakeNo 
							Left Join Inventory.IndentMain IDM on IR.IndentRequisitionNo=IDM.RefDocumentNo
															   And IDM.IndentAgainstNo=3----For Matarial Request
							Left Join Inventory.IndentItemDetail IDIT ON IDIT.IndentNo=IDM.IndentNo
                                          And IDIT.ItemNo=ID.ItemNo
                                          And isnull(IDIT.MakeNo,0)=isnull(ID.MakeNo,0)
                                          And isnull(IDIT.TechnicalGradeNo,0)=isnull(ID.TechnicalGradeNo,0)
                            Left Join Security.Login L ON L.LoginNo=IR.CreatedBy
                            Left Join Security.Login L1 ON L1.LoginNo=IR.AuthorizedBy
                            Where ID.StatusNo in (1,5,6,8)
							And IR.CompanyNo=@CompanyNo
							And IR.DocumentDate<=@DocumentDate
							And ID.ItemNo=@ItemNo
						Order By IR.DocumentDate,IR.DocumentNoYearly
						for xml path('tr')),'<td1>','<td style="text-align: center;">')
						                   ,'<td2>','<td>')
										   ,'<td3>','<td>')
	            						   ,'<td4>','<td>')
										   ,'<td5>','<td>')
										   ,'<td6>','<td style="text-align: center;">')
										   ,'<td7>','<td style="text-align: right;">')
										   ,'<td8>','<td style="text-align: right;">')
										   ,'<td9>','<td style="text-align: right;">')
										   ,'<td10>','<td style="text-align: right;">')
										   ,'<td11>','<td>')
										   ,'<td12>','<td>'))
				   
	set @str=@str+'</table>'
	
	select @str
END
