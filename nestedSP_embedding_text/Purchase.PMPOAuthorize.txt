Stored procedure Purchase.PMPOAuthorize. Called from Notifications.TANRSaveUserApproval.
Writes: Notifications.TATAuthorizationDetail, Purchase.AgreementItemDetail, Purchase.AgreementMain, Purchase.POAmendmentItemLRate, Purchase.POAmendmentMain, Purchase.POAmendmentTransportationRoute, Purchase.POItemLRate, Purchase.POMain, Purchase.POTransportationRoute
Reads: Inventory.InvLotSummary, Inventory.InvLotTaxDetail, masterdata.mItemMaster, Notifications.MultiAuthorizationSetting, Notifications.TATAuthorizationDetail, Purchase.AgreementItemDetail, Purchase.AgreementMain, Purchase.POAmendmentExpenseHead, Purchase.POAmendmentItemDetail, Purchase.POAmendmentItemLRate, Purchase.POAmendmentMain, Purchase.POAmendmentTransportationRoute, Purchase.POItemDetail, Purchase.POMain, Purchase.PortalSetting, Purchase.POTransportationRoute, Purchase.RevisedQuotationMain, Security.LoginRoleDetail, Security.LoginTempRoleDetail, Security.mRoleMasterFormDetail
Parameters: @POAmendmentNo int, @PONo int, @OutputStatusNo tinyint = null OUTPUT, @OutputStatusRemarks varchar(4000) = null OUTPUT

CREATE PROCEDURE [Purchase].[PMPOAuthorize]

	@POAmendmentNo int,
	@PONo int,
    @OutputStatusNo tinyint = null output,
    @OutputStatusRemarks varchar(4000)  = null output
AS
BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;
begin try
   
	-- Quotation status change	
 	Declare @RFQNo int = 0,@RevisedQuotationNo int = 0,@CompanyNo tinyint = 0,@iRow int = 0 ,@iCount int = 0
 	IF Exists(select 1 from purchase.POMain  where PONo = @PONo  and 
						(PORefDocumentTypeNo = 1  or(PORefDocumentTypeNo=8 and IsCSAgainstAgreement =0)) and PODocumentTypeNo<>1)
	begin
		
		Select @RFQNo = RQM.RFQNo
		,@CompanyNo = PAM.CompanyNo
		,@RevisedQuotationNo =  RQM.RevisedQuotationNo
		From Purchase.POAmendmentMain PAM
		Inner Join Purchase.RevisedQuotationMain RQM on RQM.RevisedQuotationNo = PAM.RefDocumentNo
		Where  POAmendmentNo = @POAmendmentNo
			
		Declare @RevisedQuotationDetail as table(id int,RevisedQuotationNo int )	
		Insert into @RevisedQuotationDetail
		Select 1,@RevisedQuotationNo 
		Union 
		Select row_number() over (Order By RevisedQuotationNO) + 1,RevisedQuotationNO 
		From Purchase.RevisedQuotationMain Where RFQNo = @RFQNo and RevisedQuotationNO <> @RevisedQuotationNo
		
		set @iRow = @@rowcount
		set @iCount = 0
		while @iRow >@iCount
		Begin
		      Select @RevisedQuotationNo = RevisedQuotationNo
		      From @RevisedQuotationDetail Where id = @iCount + 1
		      
			  Exec [Purchase].[QuotationChangeStatus]  @RevisedQuotationNo,@POAmendmentNo,7,@CompanyNo,0,0
			  
			  set @iCOunt = @iCOunt + 1
		End
	end
 	

	
 	
 	Declare @DocumentStatusNo tinyint ,@DivisionNo int ,@AuthDetailNo int , @AmendMentNo tinyint 
		
	Select @DocumentStatusNo = DocumentStatusNo,@CompanyNo =CompanyNo  ,@DivisionNo = DivisionNo, @AmendMentNo = (AmendmentNo)
	from Purchase.POAmendmentMain sam 
	Where POAmendmentNo = @POAmendmentNo
 	
	If @AmendMentNo >0
	Begin
		If EXISTS(Select 1 
						From  (	select ls.ItemNo,ls.MakeNo,ls.TechnicalGradeNo,  td.ExpenseheadNo
								from Inventory.InvLotSummary ls 
								inner join Inventory.InvLotTaxDetail td on td.InvLotSummaryNo = ls.InvLotSummaryNo 
								inner join Purchase.POAmendmentMain pm on pm.POAmendmentNo = ls.POAmendmentNo 		
								where pm.PONo = @PONo And td.ExpenseHeadNo is not null --And pm.AmendmentNo= @AmendMentNo-1
								group by ls.ItemNo,ls.MakeNo,ls.TechnicalGradeNo,  td.ExpenseheadNo
								)tt
						Left Join (
									Select ped.ExpenseheadNo , poid.ItemNo,poid.MakeNo,poid.TechnicalGradeNo 
									From Purchase.POAmendmentExpenseHead ped  
									Cross Join Purchase.POAmendmentItemDetail poid
									where ped.POAmendmentNo= poid.POAmendmentNo And   ped.POAmendmentNo =@POAmendmentNo
								)tt1 On tt.ExpenseheadNo =tt1.ExpenseheadNo 
								And ISNULL(tt.ItemNo,0)=  ISNULL(tt1.ItemNo,0)
								And ISNULL(tt.MakeNo,0)=  ISNULL(tt1.MakeNo,0)
								And ISNULL(tt.TechnicalGradeNo,0)=  ISNULL(tt1.TechnicalGradeNo	,0)
						Where tt1.ExpenseheadNo Is  null
				 
						 )
			Begin
				select 0,'Expensehead used in GRN can not be deleted.'		
				return
			End 	
		End



 	Declare  @UserCount int 
	Declare @NextUsers table(SNo smallint,AuthDetailNo int,LevelNo int,LoginNo int) 
	Declare @UserNO int,@LevelNo smallint,@UserCounter int =1
	
	 ---------- Multi Level Authorization Query
 	   If Exists (Select 1  From Notifications.MultiAuthorizationSetting where CompanyNo =@CompanyNo and ISNULL(DivisionNo,0) = ISNULL(@DivisionNo,0)
 					and FormCode ='PMPA') and @DocumentStatusNo = 20
 		BEGIN
 		
 			 if Exists (Select 1 From Notifications.TATAuthorizationDetail where FormCode ='PMPA' and InternalNo = @POAmendmentNo 
 							and IsReady = 1 and IsEmailSent=0)
				BEGIN
						-- Send Email
					
					Insert into @NextUsers
					select ROW_NUMBER () over (order by LoginNo) SlNo,AuthDetailNo, LevelNo, LoginNo  from (
					Select LevelNo,LoginNo,AuthDetailNo
					 From Notifications.TATAuthorizationDetail where FormCode ='PMPA' and InternalNo = @POAmendmentNo 
						and IsReady = 1 and IsEmailSent =0	and LoginNo is not null
					union 
					Select LevelNo,lrd.LoginNo,AuthDetailNo
					 From Notifications.TATAuthorizationDetail tad
					 inner join Security.mRoleMasterFormDetail rfd on rfd.RoleMasterNo = tad.RoleNo and rfd.FormNo in( 188,199)
					 inner join Security.LoginRoleDetail lrd on rfd.RoleMasterNo = lrd.RoleMasterNo 
					 where FormCode ='PMPA' and InternalNo = @POAmendmentNo 
						and IsReady = 1 and IsEmailSent =0	and tad.RoleNo  is not null	
					union 
					Select LevelNo,lrd.LoginNo,AuthDetailNo
					 From Notifications.TATAuthorizationDetail tad
					 inner join Security.mRoleMasterFormDetail rfd on rfd.RoleMasterNo = tad.RoleNo and rfd.FormNo in( 188,199)
					 inner join Security.LoginTempRoleDetail lrd on rfd.RoleMasterNo = lrd.RoleMasterNo 
					 where FormCode ='PMPA' and InternalNo = @POAmendmentNo and CONVERT(smalldatetime,convert(varchar(15), GETDATE(),106))
						 between lrd.FromDate and ISNULL( lrd.ToDate,CONVERT(smalldatetime,convert(varchar(15), GETDATE(),106)))
						and IsReady = 1 and IsEmailSent =0	and tad.RoleNo  is not null	) tt
						group by LevelNo, LoginNo 	,AuthDetailNo							
					
					
					Set @UserCount = @@ROWCOUNT	
					
					While @UserCounter <= @UserCount
					 Begin
					
						 Select @UserNO=LoginNo , @LevelNo = LevelNo,@AuthDetailNo=AuthDetailNo From @NextUsers Where SNo = @UserCounter
								
						 Exec [Purchase].[PMPOSendAuthorizationMail] @POAmendmentNo,@UserNo,@LevelNo ,@AuthDetailNo
						 
						Update Notifications.TATAuthorizationDetail  set IsEmailSent =1 where FormCode = 'PMPA' and InternalNo = @POAmendmentNo 
								and LevelNo = @LevelNo and IsReady =1
						 Set @UserCounter=@UserCounter+1 
					 End
				END
 		 END
 	  else If Exists (Select 1  From Notifications.MultiAuthorizationSetting where CompanyNo =@CompanyNo and ISNULL(DivisionNo,0) = ISNULL(@DivisionNo,0)
 					and FormCode ='PMPA') and Exists (Select 1 From Notifications.TATAuthorizationDetail where FormCode ='PMPA' 
 						and InternalNo = @POAmendmentNo and IsRejectedMailSend =1) -- rejection mail
 		BEGIN
 			 
 			 if Exists (Select 1 From Notifications.TATAuthorizationDetail where FormCode ='PMPA' and InternalNo = @POAmendmentNo 
 							and IsRejectedMailSend =1)
				BEGIN
						-- Send Email
					 
					Insert into @NextUsers
					select ROW_NUMBER () over (order by LoginNo) SlNo,AuthDetailNo, LevelNo, LoginNo  from (
					Select LevelNo,LoginNo,AuthDetailNo
					 From Notifications.TATAuthorizationDetail where FormCode ='PMPA' and InternalNo = @POAmendmentNo 
						and IsRejectedMailSend =1 and LoginNo is not null
					union 
					Select LevelNo,lrd.LoginNo,AuthDetailNo
					 From Notifications.TATAuthorizationDetail tad
					 inner join Security.mRoleMasterFormDetail rfd on rfd.RoleMasterNo = tad.RoleNo and rfd.FormNo in( 188,199)
					 inner join Security.LoginRoleDetail lrd on rfd.RoleMasterNo = lrd.RoleMasterNo 
					 where FormCode ='PMPA' and InternalNo = @POAmendmentNo 
						and IsRejectedMailSend =1 and tad.RoleNo  is not null	
					union 
					Select LevelNo,lrd.LoginNo,AuthDetailNo
					 From Notifications.TATAuthorizationDetail tad
					 inner join Security.mRoleMasterFormDetail rfd on rfd.RoleMasterNo = tad.RoleNo and rfd.FormNo in( 188,199)
					 inner join Security.LoginTempRoleDetail lrd on rfd.RoleMasterNo = lrd.RoleMasterNo 
					 where FormCode ='PMPA' and InternalNo = @POAmendmentNo and CONVERT(smalldatetime,convert(varchar(15), GETDATE(),106))
						 between lrd.FromDate and ISNULL( lrd.ToDate,CONVERT(smalldatetime,convert(varchar(15), GETDATE(),106)))
						and IsRejectedMailSend =1 and tad.RoleNo  is not null	) tt
						group by LevelNo, LoginNo 	,AuthDetailNo							
					
					
					Set @UserCount = @@ROWCOUNT	
					
					
					While @UserCounter <= @UserCount
					 Begin
					
						 Select @UserNO=LoginNo , @LevelNo = LevelNo,@AuthDetailNo=AuthDetailNo From @NextUsers Where SNo = @UserCounter
								
						 Exec [Purchase].[PMPOSendAuthorizationMail] @POAmendmentNo,@UserNo,@LevelNo ,@AuthDetailNo ,1
						 
						Update Notifications.TATAuthorizationDetail  set IsRejectedMailSend =0 where FormCode = 'PMPA' and InternalNo = @POAmendmentNo 
								and LevelNo = @LevelNo and IsRejectedMailSend =1
						 Set @UserCounter=@UserCounter+1 
					 End
				END				  	
 		 END 		  
 		
 	  -- Sending mail if last level got approved
 	  If exists (	 Select 1 
 					 From Notifications.TATAuthorizationDetail TAT
 					 Inner join Purchase.POAmendmentMain POAM on POAM.POAmendmentNo = TAT.InternalNo
 					 where InternalNo = @POAmendmentNo and FormCode = 'PMPA' and POAM.DocumentStatusNo = 30 -- Authorized
						   And 
						   TAT.LevelNo = (
							Select MAX(LevelNo)
							From Notifications.TATAuthorizationDetail
							Where InternalNo=@POAmendmentNo	and FormCode= 'PMPA'
						)  and TAT.StatusNo = 13 -- Approve
				 )
	  BEGIN 
		Exec [Purchase].[PMPOSendAuthorizationMail]  @POAmendmentNo = @POAmendmentNo,@LoginNo = 0 ,@LevelNo = 0 ,@AuthDetailNo =0 ,@IsRejected=0
	  END			
 	  			  
 	   
 	   update Purchase.POMain set items=replace (stuff((select ',' + ItemDescription from masterdata.mItemMaster im  inner join Purchase.POItemDetail pid on pid.ItemNo  = im.ItemNo   where pid.PONo = @PONo  Group by ItemDescription  for xml path ('')),1,1,''),'&amp;','&')  
 		  where PONo = @PONo 
 	   
 	    update Purchase.POAmendmentMain set items=replace (stuff((select ',' + ItemDescription from masterdata.mItemMaster im  inner join Purchase.POAmendmentItemDetail pid on pid.ItemNo  = im.ItemNo   where pid.POAmendmentNo = @POAmendmentNo  Group by ItemDescription  for xml path ('')),1,1,''),'&amp;','&')  
 		  where POAmendmentNo = @POAmendmentNo 
 		  
 	If Exists (Select 1 from purchase.POAmendmentItemLRate   Where POAmendmentNo =@POAmendmentNo)
 	BEGIN
 		Delete From purchase.POAmendmentItemLRate where POAmendmentNo=@POAmendmentNo
 	END
 	   insert into  purchase.POAmendmentItemLRate(POAmendmentNo ,ItemNo ,MakeNo ,TechnicalGradeNo ,
 			RouteIdentity,LRate )
 	   select @POAmendmentNo ,ItemNo,MakeNo,TechnicalGradeNo ,RouteIdentity,ISNULL(LRate,0)
 	    from Purchase.funcPOItemLRate(@POAmendmentNo)
 	    
 	      
		update t1 set ParentRouteId  = (select RouteIdentity from Purchase.POAmendmentTransportationRoute 
								 tt where RouteIdentity = LEFT ( t1.RouteIdentity,len( CONVERT(float, t1.RouteIdentity))-1) 
								 and POAmendmentNo = t1.POAmendmentNo  
								   ) 
			 from Purchase.POAmendmentTransportationRoute  t1
			 where POAmendmentNo = @POAmendmentNo 
			 
		update pr set IsGateEntry = 0 
		from purchase.poamendmenttransportationroute pr ,Purchase.POAmendmentMain pm
		where pm.POAmendmentNo = pr.POAmendmentNo and pm.StatusNo in(1,8) and pr.IsGateEntry =1 and ParentRouteId is not null
		and pm.POAmendmentNo = @POAmendmentNo 
		
		
		 if exists(select 1 from purchase.poamendmenttransportationroute ptr where parentrouteid is null and isgateentry =0
				and not exists(select 1 from purchase.poamendmenttransportationroute  where poamendmentno = ptr.poamendmentno 
			and Parentrouteid = ptr.routeidentity ) and ptr.POAmendmentNo = @POAmendmentNo  )
		 and exists(select 1 from Purchase.POAmendmentMain where POAmendmentNo = @POAmendmentNo and AuthorizedBy is not null)	
		begin
				raiserror ('Some problem in route creation',16,-1)
		end


 	    IF EXISTS (SELECT 1 
						   FROM [Purchase].[POAmendmentMain] 
						   WHERE POAmendmentNo = @POAmendmentNo
						   AND ([AmendmentNo]=0 OR ISNULL([AuthorizedBy],0)<>0)
						 )
				  BEGIN
					
					update t1 set ParentRouteId  = (select RouteIdentity from Purchase.POTransportationRoute 
								 tt where RouteIdentity = LEFT ( t1.RouteIdentity,len( CONVERT(float, t1.RouteIdentity))-1) 
								 and PONo = t1.PONo 
								   ) 
						 from Purchase.POTransportationRoute  t1
						 where PONo = @PONo 
						 
				      update pr set IsGateEntry = 0 
						from purchase.POTransportationRoute pr ,Purchase.POMain  pm
						where pm.PONo = pr.PONo and pm.StatusNo in(1,8) and pr.IsGateEntry =1 and ParentRouteId is not null
						and pm.PONo = @PONo 
		
					 Delete from Purchase.POItemLRate where PONo = @PONo
					 
					 insert into  purchase.POItemLRate
						(PONo ,ItemNo ,MakeNo ,TechnicalGradeNo ,RouteIdentity,LRate )
 						select @PONo  ,ItemNo,MakeNo,TechnicalGradeNo ,RouteIdentity,ISNULL(LRate,0)
 						from Purchase.funcPOItemLRate(@POAmendmentNo)
				  End

	If Exists (Select 1 From Purchase.PortalSetting where IsInsertIntoPortal=1)
	BEGIN
		DECLARE	@return_value int,
				@PortalOutputStatusNo tinyint,
				@PortalOutputStatusRemarks varchar(4000)
				
		EXEC	@return_value = [Purchase].[PMPOInsertPOIntoPortal]
				@POAmendmentNo = @POAmendmentNo,
				@OutputStatusNo = @PortalOutputStatusNo OUTPUT,
				@OutputStatusRemarks = @PortalOutputStatusRemarks OUTPUT

		If @PortalOutputStatusNo=0
		BEGIN
			Raiserror(@PortalOutputStatusRemarks,16,-1)
		END
	End 
	
	---- Budget Validation
	BEGIN	
		Declare 	@OutputStatusNo1 tinyint = null ,@OutputStatusRemarks1 varchar(4000) = null 	
		exec [Project].[ProjectBudgetValidation] @POAmendmentNo,Null,null,@OutputStatusNo1 output,@OutputStatusRemarks1 output

		If @OutputStatusNo1=0
		BEGIN
			Raiserror(@OutputStatusRemarks1,16,-1) 	
		END		
	END

	Declare @AgreementQty decimal(10,3),@ToleranceQty decimal(10,3)
	Declare @SumOfPOQty decimal(10,3), @AgreementType tinyint,@ReferenceNo int,@POQty decimal(10,3)

	select @ReferenceNo = RefDocumentNo from purchase.POAmendmentMain where POAmendmentNo = @POAmendmentNo 
	And PORefDocumentTypeNo = 2

	if exists(Select 1
				from Purchase.AgreementMain
				Where AgreementNo = @ReferenceNo
				And IsAllowAlternateItemInPO = 1
			)
	begin
		select @SumOfPOQty =sum(Quantity) 
		from Purchase.POAmendmentItemdetail  pd 
		inner join purchase.POAmendmentMain pm on 
		pd.PoAmendmentNo = pm.POAmendmentNo
		and AmendmentNo = (
							select max(AmendmentNo) 
							from Purchase.POAmendmentMain 
							where PONo = pm.PONo
						)
		where PORefDocumentTypeNo=2 and RefDocumentNo=@ReferenceNo
		
		select @POQty =sum(Quantity) 
		from Purchase.POAmendmentItemdetail  pd 
		inner join purchase.POAmendmentMain pm on 
		pd.PoAmendmentNo = pm.POAmendmentNo
		and AmendmentNo = (
							select max(AmendmentNo) 
							from Purchase.POAmendmentMain 
							where PONo = pm.PONo
						)
		And PM.PONo = @PONo
		where PORefDocumentTypeNo=2 and RefDocumentNo=@ReferenceNo

		select @AgreementQty = SUM(Quantity) 
		,@ToleranceQty = SUM(Case When ToleranceBasisNo = 1 Then ad.TolerancePlus 
			When ToleranceBasisNo = 2 Then Cast(Quantity * (TolerancePlus/100) as decimal(12,3))
			Else 0 End)
		,  @AgreementType = AgreementTypeNo
		from Purchase.AgreementItemDetail ad 
		inner join Purchase.AgreementMain am on ad.AgreementNo = am.AgreementNo
		where ad.AgreementNo = @ReferenceNo
		Group by AgreementTypeNo
		
		if 	@AgreementType=2  -- Quantity Agreement
		begin
			if isnull(@SumOfPOQty,0) > @AgreementQty + ISNULL(@ToleranceQty,0)
			begin
				select 0,'PO Qty can''t be greater than Balance Agreement qty ' 
							+ convert(varchar,@AgreementQty - ISNULL(@SumOfPOQty - ISNULL(@POQty,0),0)) +'.'	
				return
			end
			else if isnull(@SumOfPOQty,0) >= @AgreementQty
			begin
				Update Purchase.AgreementItemDetail 
				set StatusNo = 2 
				where AgreementNo = @ReferenceNo 

				update Purchase.AgreementMain set Statusno = 2 where AgreementNo  = @ReferenceNo 
			end
			else if isnull(@SumOfPOQty,0) < @AgreementQty
			begin
				Update Purchase.AgreementItemDetail 
				set StatusNo = 8 
				where AgreementNo = @ReferenceNo 

				update Purchase.AgreementMain set Statusno = 8 where AgreementNo  = @ReferenceNo 
			end
		end		
			 
	end     

	   Select @OutputStatusNo = 1,@OutputStatusRemarks = 'Success'				  
 	   Select 1,'Success'				  
End Try

Begin Catch
		Select 0 ,Error_Message()
		select @OutputStatusNo = 0 , @OutputStatusRemarks = ERROR_MESSAGE ()		
End Catch
    
END
