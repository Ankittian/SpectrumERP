 , @Result bit output

		select @IsReturnThroughDC =IsReturnThroughDC , @CurrencyConversionRate = convert(decimal(18,6),DivisibilityRate* MultiplicativeRate)
			 from Inventory.GRNMain where GRNNo = @DocumentNo 
		if @CurrencyConversionRate is null
			set @CurrencyConversionRate=1
		declare @PurchaseFreightAmount TABLE 
						(
							-- Add the column definitions for the TABLE variable here
							GateEntryNo int,
							GRNNo int,
							FreightAmount money, 
							DeductionAmount money
						)
						
	 
 		--- Expenses In PO

Create table #ExpenseHeads (invlotsummaryNo int,ExpenseHeadNo smallint,ExpenseRate decimal(18,6))
Create table #FreightHeads (invlotsummaryNo int,FreightRate decimal(18,6))
Create table #SCDiffRate (InvlotsummaryNo int,DiffRate decimal(18,6))

 
if exists(select 1 from Inventory.GRNItemDetail gid inner join Purchase.POAmendmentExpenseHead peh on peh.POAmendmentNo = gid.POAmendmentNo 
				and gid.GRNNo = @Documentno and SCOrderAmendmentNo is null)
begin
		insert into #ExpenseHeads
		 select ls.InvLotSummaryNo,peh.ExpenseheadNo ,(((PEH.Rate * case PEH.BillingQtyNo when 1 then id.ChallanQty  
						  when 2 then Id.ChallanQty 
						  when 4 then ID.ReceivedQty else   
						  case when @IsReturnThroughDC =1 then id.AcceptedQty + ISNULL(id.RejectedQty,0)
							else id.AcceptedQty end end) 
								* (case when @IsReturnThroughDC =1 then ID.AcceptedQty + RejectedQty else ID.AcceptedQty end * isnull(ID.SecondCF,1) /isnull(ID.FirstCF,1))) /
					isnull((select case when @IsReturnThroughDC =1 then sum(pgd.AcceptedQty + ISNULL(pgd.RejectedQty,0))else sum(pgd.AcceptedQty) end
							from Inventory.GRNItemIndentDetail pgd 
							where pgd.GRNItemDetailNo= ID.GRNItemDetailNo),case when @IsReturnThroughDC =1 then ID.AcceptedQty + RejectedQty else ID.AcceptedQty end)) 
								/ case when @IsReturnThroughDC =1 then ID.AcceptedQty + RejectedQty else ID.AcceptedQty end

		from 
		Purchase.POAmendmentMain PAM
		inner join Purchase.POAmendmentExpenseHead PEH on PAM.POAmendmentNo= PEH.POAmendmentNo

					isnull(ID.gateEntryNo,0) = isnull(ls.gateEntryNo,0) and
					isnull(ID.invstocktransferNo,0) = isnull(ls.invstocktransferNo,0) and

		case when  @IsReturnThroughDC =1 then IsRejectedLot else 0 end= ls.IsRejectedLot and

						isnull(PEH.TechnicalGradeNo,0) = isnull(PID.TechnicalGradeNo,0) ))

							where pgd.GRNItemDetailNo= ID.GRNItemDetailNo)

							IsNULL(

							where GILD.GRNItemDetailNo = ID.GRNItemDetailNo

		Purchase.POAmendmentMain PAM
		inner join Purchase.POAmendmentExpenseHead PEH on PAM.POAmendmentNo= PEH.POAmendmentNo 
		outer apply (select APEOD.ItemNo from  Purchase.POAmendmentExpenseHeadOutputDetail APEOD
						inner join Purchase.POAmendmentExpenseHead APEH on APEH.POAmendmentExpenseDetailNo = APEOD.POAmendmentExpenseDetailNo

			from(
				select ls.InvLotSummaryNo , peh.ExpenseheadNo ,
				(sum( PEH.Rate * cnt *  case when @IsReturnThroughDC =1 then ID.AcceptedQty + ISNULL(ID.RejectedQty,0) else ID.AcceptedQty end * ID.Rate
					)/ TotAmount)  ExpenseAmount ,ls.ReceivedQty 
				, SUM(CID2.cnt2 * CID2.Rate) / SUM(FIT.TotReceivedQty) avgRate
				from 
				Purchase.POAmendmentMain PAM
				inner join (select count(distinct isnull(GateEntryNo,1)) cnt ,POAmendmentNo,GateEntryNo 
						from Inventory.GRNItemDetail id where grnno = @documentno
					   group by POAmendmentNo ,GateEntryNo )Cid
						  on Cid.POAmendmentNo = PAM.POAmendmentNo 
				inner join Purchase.POAmendmentExpenseHead PEH on PAM.POAmendmentNo= PEH.POAmendmentNo 
				outer apply (select APEOD.ItemNo from  Purchase.POAmendmentExpenseHeadOutputDetail APEOD
						inner join Purchase.POAmendmentExpenseHead APEH on APEH.POAmendmentExpenseDetailNo = APEOD.POAmendmentExpenseDetailNo 
						where  APEH.POAmendmentNo = PEH.POAmendmentNo  ) PEOD
				inner join masterdata.mExpenseHead eh on eh.ExpenseHeadNo = PEH.ExpenseheadNo and eh.NatureNo <>2
				inner join Purchase.POAmendmentItemDetail PID on PID.POAmendmentNo  = PEH.POAmendmentNo
				inner join Inventory.GRNItemDetail ID on (ID.ItemNo= PEOD.ItemNo or ID.ItemNo = PID.ItemNo) and 
								isnull(ID.MakeNo,0) = isnull(PID.MakeNo,0) and
								isnull(ID.GateEntryNo,0) = isnull(Cid.GateEntryNo ,0) and
								isnull(ID.TechnicalGradeNo,0) = isnull(PID.TechnicalGradeNo,0) and
								ID.POAmendmentNo = PAM.POAmendmentNo 
				--Left Join (
				--			Select GID.GRNItemDetailNo,Count(IID.IMSTInvLotSummaryNo) IMSTLotCnt -- When Having More then one lot in IMST then bifurcate it
				--			From Inventory.GRNItemDetail GID

				--			Where GID.GRNNo=@Documentno And IID.IMSTInvLotSummaryNo is Not Null
				--			Group By GID.GRNItemDetailNo
				--		  )STLot on STLot.GRNItemDetailNo=ID.GRNItemDetailNo						
			inner join (select sum(case when @IsReturnThroughDC =1 then AcceptedQty + ISNULL(RejectedQty,0) else gpd.AcceptedQty end * gpd.Rate) TotAmount
							,gpd.GRNNo  ,GateEntryNo, PPEH.ExpenseheadNo
							from  Inventory.GRNItemDetail gpd 
							inner join Purchase.POAmendmentExpenseHead PPEH on gpd.POAmendmentNo= PPEH.POAmendmentNo 
							inner join Purchase.POAmendmentItemDetail PPID on PPID.POAmendmentNo  = PPEH.POAmendmentNo
				
								where gpd.GRNNo=@DocumentNo and gpd.Rate<>0 and
								 gpd.ItemNo= PPID.ItemNo and 
								 isnull(gpd.RouteIdentity,0) = isnull(PPEH.RouteIdentity,0) and
								isnull(gpd.MakeNo,0) = isnull(PPID.MakeNo,0) and
								isnull(gpd.TechnicalGradeNo,0) = isnull(PPID.TechnicalGradeNo,0) and
								gpd.POAmendmentNo = PPID.POAmendmentNo and
								(PPEH.itemno is null or(PPEH.ItemNo = PPID.ItemNo and 
								isnull(PPEH.MakeNo,0) = isnull(PPID.MakeNo,0) and
								isnull(PPEH.TechnicalGradeNo,0) = isnull(PPID.TechnicalGradeNo,0)))
								
							group by gpd.GRNNo  ,GateEntryNo,PPEH.ExpenseheadNo) tot on tot.GRNNo= id.grnno and
									PEH.ExpenseheadNo = tot.ExpenseheadNo  and 
						      isnull(id.GateEntryNo,0) = isnull(tot.GateEntryNo,0)
				inner join Inventory.InvLotSummary ls on ls.DocumentNo = ID.GRNNo and ls.ItemNo = ID.ItemNo and
							isnull(ID.MakeNo,0) = isnull(ls.MakeNo,0) and
							isnull(ID.TechnicalGradeNo,0) = isnull(ls.TechnicalGradeNo,0) and
							isnull(ID.invstocktransferNo,0) = isnull(ls.invstocktransferNo,0) and
							ID.POAmendmentNo = ls.POAmendmentNo
							and isnull(ID.GateEntryNo,0) = isnull(Cid.GateEntryNo ,0) 
				---
				Outer apply	(Select SUM(Ils.ReceivedQty) TotReceivedQty from Inventory.InvLotSummary ILS
				  inner join  Purchase.POAmendmentExpenseHead POEH1 on POEH1.POAmendmentNo = Ils.POAmendmentNo 
				  inner join Purchase.POAmendmentMain POAM on POAM.POAmendmentNo = ILS.POAmendmentNo and POAM.POAmendmentNo = POEH1.POAmendmentNo  
				  where POAM.POAmendmentNo = PAM.POAmendmentNo 
				  and POEH1.ExpenseheadNo = PEh.ExpenseheadNo 
				  and ILS.DocumentNo = Id.GRNNo and ILs.TransactionFlag like 'IGRN'
				  ) FIT
				---
				---
				outer apply (select count(distinct IsNULL(GateEntryNo,1)) cnt2,POEH2.POAmendmentNo,POEH2.rate  
						from Inventory.GRNItemDetail id2
						inner join Purchase.POAmendmentExpenseHead POEH2 on POEH2.POAmendmentNo = id2.POAmendmentNo and POEH2.POAmendmentNo = PAM.POAmendmentNo 
						where grnno = @documentno and POEH2.ExpenseheadNo = peh.ExpenseheadNo 
					   group by POEH2.POAmendmentNo,POEH2.Rate)CID2 
				---
				where
				case when  @IsReturnThroughDC =1 then IsRejectedLot else 0 end= ls.IsRejectedLot and
				PEH.IncludeInValuation =1 and ls.ReceivedQty >0 and
				PEH.FreightRateTypeNo  =3 and -- per trip
				(PEH.RouteIdentity = id.RouteIdentity or peh.BillingLocationNo = pam.ToLocationNo) and
				(PEH.itemno is null or(PEH.ItemNo = PID.ItemNo and 
								isnull(PEH.MakeNo,0) = isnull(PID.MakeNo,0) and
								isnull(PEH.TechnicalGradeNo,0) = isnull(PID.TechnicalGradeNo,0)))
					group by ls.InvLotSummaryNo,peh.ExpenseheadNo ,Cid.GateEntryNo,ls.ReceivedQty ,TotAmount 
				)tt  group by InvLotSummaryNo, ExpenseheadNo,avgRate  
	
end

					from Inventory.GRNItemDetail pgd

		  + IsNULL((Select  (SUM(IsNULL(FITD.TotalValue,0)) * gid.Rate * LS.ReceivedQty) 
		                    
		       from Finance.FreightInvoiceTaxDetail FITD 
			  inner join Finance.FreightInvoiceDocumentDetail fidd on fidd.FreightInvoiceNo = fITD.FreightInvoiceNo 
	          inner join Finance.FreightInvoiceDocumentDetails fidds on fidds.FreightInvoiceNo = fidd.FreightInvoiceNo

				 / (select sum((case when @IsReturnThroughDC =1 then AcceptedQty + ISNULL(RejectedQty,0) else AcceptedQty  end* isnull(FirstCF,1) /isnull(SecondCF,1) )* Rate) 
					from Inventory.GRNItemDetail pgd 
						 where  pgd.GRNNo = @DocumentNo and Rate >0 and isnull(GateEntryNo,0) = isnull(ls.gateEntryNo,0))
				 ,0) 		
			
	    )
						 / LS.ReceivedQty

		and isnull(gid.MakeNo,0) = isnull(ls.MakeNo,0) and 
		isnull(gid.TechnicalGradeNo,0) = isnull(LS.TechnicalGradeNo,0) 
		and isnull(gid.POAmendmentNo,0) = isnull(ls.POAmendmentNo,0)and 
		isnull(gid.InvStockTransferNo ,0)= isnull(ls.invStockTransferNo ,0)
		and isnull(gid.GateEntryNo ,0) = isnull(LS.GateEntryNo ,0)
	where  ls.receivedQty>0 And
		gid.GRNNo = @documentno and
	case when  @IsReturnThroughDC =1 then IsRejectedLot else 0 end= ls.IsRejectedLot 
	and fm.AuthorizedBy is not null and fm.AuthorizedDate < @AuthorizedDate
	group by LS.InvLotSummaryNo , ls.gateEntryNo,gid.Rate,LS.ReceivedQty

				--		else 0 end)/sad.Quantity ,0))
				from

						and gid.ItemNo = ls.ItemNo and gid.SCOrderAmendmentNo = ls.SCOrderAmendmentNo

						and std.MiscChargeNo = td.MiscChargeNo and std.IncludeInBasic =1

			and gid.GRNNo = @documentno

			 Group by ls.invlotsummaryno,SCValuationMethodNo,said.Rate ,said.ActualRate ,BifercationPercentage,gsd.Amount ,gid.AcceptedQty,sad.Quantity
			 
			 
			if exists(select 1 from Inventory.GRNItemDetail where POAmendmentNo is null and SCOrderAmendmentNo is not null and GRNNo =@documentno )

			    INSERT INTO Inventory.InvLotTaxDetail(InvLotSummaryNo, InvLotDetailNo, InvLotBillDetailNo,

					,1, td.ChargeType ,satd.IncludeInBasic

				Inner join Inventory.InvLotSummary ls  on ls.DocumentNo = gid.GRNNo and ls.TransactionFlag ='IGRN'
						and gid.ItemNo = ls.ItemNo and gid.SCOrderAmendmentNo = ls.SCOrderAmendmentNo

							when td.IncludeInBasic =1 and mc.CalculationNature =2 then 0 -td.RatePerUnit
							else 0 end),0) DiffVrate,
							 isnull(sum(case when isnull(mc.CalculationNature,1) =1  then td.RatePerUnit 
							when mc.CalculationNature =2 then 0 -(td.RatePerUnit  )
							else 0 end),0) DiffinLRate

					left join masterdata.mMiscCharges mc on mc.MiscChargeNo = td.MiscChargeNo
