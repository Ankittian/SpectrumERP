Stored procedure Inventory.MGPOFinancePosting: Material Issue Posting. Called from Finance.FinancePostingFillValuesDocumentDetails.
Reads: Finance.ARDItemRule, Finance.ARDVendorLocationRule, Inventory.GatePassInItemDetail
Parameters: @OtherDataInXml xml, @CompanyNo tinyint, @VendorLocationNo int, @ItemNos varchar

					-- =============================================

-- Create date:  04 03 2020
-- Description:	Material Issue Posting
-- =============================================
--- Inventory].[IMSTFinancePosting] '<DocumentElement />' 
CREATE PROCEDURE [Inventory].[MGPOFinancePosting] 
	@OtherDataInXml xml ,
	@CompanyNo tinyint ,
	@VendorLocationNo int,
	@ItemNos varchar(max)
as

BEGIN
begin try
declare @IsManagingExpenseAccount bit
 

create table #Posting ([Add to Basic] bit,Particular varchar(500),[Debit Amount] decimal(15,2) , 
		[Credit Amount] decimal(15,2),[Account Name] int, [Ref. Detail] varchar(50),[Cost Center] varchar(50),
		[Amount] decimal(15,2), [Type] tinyint,SlNo tinyint identity(1,1) ,MiscChargeNo int , TaxCodeMiscChargesDetailNo int,
			ItemNos varchar(200),GroupNo smallint, GroupTypeNo tinyint, NatureNo  tinyint, AccountTypeNo tinyint)

		declare @ItemTable table (GatePassInNo int,ItemNo int,MakeNo int,Qty decimal(12,3),Rate money,BasicAmount money,GatePassInRate Money)
		insert into @ItemTable (GatePassInNo,ItemNo,MakeNo,Qty,Rate,BasicAmount)
		select Col1,Col2,Col3,sum(convert(money,Col4)), Col5, sum(CONVERT(decimal(12,3),Col6))
		from masterdata.Split_RowColumn(@ItemNos,'`','~')
		group by Col1, Col2,Col3,Col5
 
		Update IT
		Set IT.GatePassInRate = GPIID.Rate 
		from @ItemTable IT
		inner join Inventory.GatePassInItemDetail GPIID on GPIID.GatePassInNo = IT.GatePassInNo 
														and GPIID.ItemNo = IT.ItemNo 
														and IsNULL(GPIID.MakeNo,0) = ISNULL(IT.MakeNo,0)   
		


declare @MGPO_GatePassTypeNo tinyint

			SELECT @MGPO_GatePassTypeNo=v.value('GatePassTypeNo[1]','TinyInt')
			FROM @OtherDataInXml.nodes('/DocumentElement/Table') x(v)
			
			if ISNULL(@MGPO_GatePassTypeNo,0) = 7 
			Begin 
				insert into #Posting ([Add to Basic] ,Particular ,[Debit Amount] ,
								[Credit Amount] ,[Account Name] ,Amount ,[Type] ,
								MiscChargeNo ,TaxCodeMiscChargesDetailNo ,ItemNos)
				select cast(0 as bit) [Add to Basic] ,'Stock A/C' Particular 
					,CAST(null as money) [Debit Amount] ,sum(I.BasicAmount) [Credit Amount] 
					,IR.AccountNo [Account Name] 
					,sum(I.BasicAmount) Amount ,2 [Type] 
					,CAST(null as smallint) MiscChargeNo,null TaxCodeMiscChargesDetailNo,CAST(null as varchar(5000))ItemNos
				from @ItemTable I
					left join (select ItemNo,IR.AccountNo--,AM.AccountName
								from Finance.ARDItemRule IR
								--inner join masterdata.mAccountMaster AM on AM.AccountNo=IR.AccountNo
								where IR.AccountTypeNo=14) IR on IR.ItemNo=I.ItemNo 
				group by IR.AccountNo,isnull(IR.AccountNo,I.ItemNo)
				Union
				select cast(0 as bit) [Add to Basic] ,'Customer Loan A/C' Particular 
				,sum(I.BasicAmount)[Debit Amount],CAST(null as money) [Credit Amount] 
				,IR.AccountNo [Account Name] 
				,sum(I.BasicAmount) Amount ,1 [Type] 
				,CAST(null as smallint) MiscChargeNo,null TaxCodeMiscChargesDetailNo,CAST(null as varchar(5000))ItemNos
				from @ItemTable I
					Outer Apply (select IR.AccountNo
								from Finance.ARDVendorLocationRule IR
								where IR.AccountTypeNo=71 
								and IR.VendorLocationNo = @VendorLocationNo/*Supplier Loan Account*/) IR
				group by IR.AccountNo
			End 
			Else If ISNULL(@MGPO_GatePassTypeNo,0) = 6 
			Begin 
				insert into #Posting ([Add to Basic] ,Particular ,[Debit Amount] ,
								[Credit Amount] ,[Account Name] ,Amount ,[Type] ,
								MiscChargeNo ,TaxCodeMiscChargesDetailNo ,ItemNos)
				select cast(0 as bit) [Add to Basic] ,'Stock A/C' Particular 
					,CAST(null as money) [Debit Amount] ,sum(I.BasicAmount) [Credit Amount] 
					,IR.AccountNo [Account Name] 
					,sum(I.BasicAmount) Amount ,2 [Type] 
					,CAST(null as smallint) MiscChargeNo,null TaxCodeMiscChargesDetailNo,CAST(null as varchar(5000))ItemNos
				from @ItemTable I
					left join (select ItemNo,IR.AccountNo--,AM.AccountName
								from Finance.ARDItemRule IR
								--inner join masterdata.mAccountMaster AM on AM.AccountNo=IR.AccountNo
								where IR.AccountTypeNo=14) IR on IR.ItemNo=I.ItemNo 
				group by IR.AccountNo,isnull(IR.AccountNo,I.ItemNo)
				Union
				select cast(0 as bit) [Add to Basic] ,'Supplier Loan A/C' Particular 
				,sum(I.GatePassInRate*I.Qty)[Debit Amount],CAST(null as money) [Credit Amount] 
				,IR.AccountNo [Account Name] 
				,sum(I.GatePassInRate*I.Qty) Amount ,1 [Type] 
				,CAST(null as smallint) MiscChargeNo,null TaxCodeMiscChargesDetailNo,CAST(null as varchar(5000))ItemNos
				from @ItemTable I			
					Outer Apply (select IR.AccountNo
								from Finance.ARDVendorLocationRule IR
								where IR.AccountTypeNo=72 
								and IR.VendorLocationNo = @VendorLocationNo/*Supplier Loan Account*/) IR
				Where IsNULL(I.GatePassInRate,0) <> 0 
				group by IR.AccountNo
				Union 
				select cast(0 as bit) [Add to Basic] ,'Purchase Price Variance A/C' Particular 

					,Case When SUM(I.BasicAmount - (I.GatePassInRate * I.Qty)) > 0 then
					           SUM(I.BasicAmount - (I.GatePassInRate * I.Qty))
						  Else 
							   CAST(null as money)	
						  End  [Debit Amount] 

					,Case When SUM(I.BasicAmount - (I.GatePassInRate * I.Qty)) < 0 then
					           -1 * SUM(I.BasicAmount - (I.GatePassInRate * I.Qty))
						  Else 
							   CAST(null as money)	
						  End [Credit Amount] 

					,IR.AccountNo [Account Name] 
					,Case When SUM(I.BasicAmount - (I.GatePassInRate * I.Qty)) < 0 then
					           -1 * SUM(I.BasicAmount - (I.GatePassInRate * I.Qty))
						  Else 
							   SUM(I.BasicAmount - (I.GatePassInRate * I.Qty))	
						  End Amount 
					,Case When SUM(I.BasicAmount - (I.GatePassInRate * I.Qty)) > 0 then 1
						  When SUM(I.BasicAmount - (I.GatePassInRate * I.Qty)) < 0 then 2	
					      End [Type] 
					,CAST(null as smallint) MiscChargeNo,null TaxCodeMiscChargesDetailNo,CAST(null as varchar(5000))ItemNos
				from @ItemTable I
					left join (select ItemNo,IR.AccountNo--,AM.AccountName
								from Finance.ARDItemRule IR
								--inner join masterdata.mAccountMaster AM on AM.AccountNo=IR.AccountNo
								where IR.AccountTypeNo=18) IR on IR.ItemNo=I.ItemNo 
				group by IR.AccountNo
				Having SUM(I.BasicAmount - (I.GatePassInRate * I.Qty)) <> 0
			End 

 if exists(select 1 from #Posting having abs(SUM([Credit Amount])- SUM([Debit Amount]))<=.5)
 begin
	update PD set [Debit Amount] = [Debit Amount] + ISNULL(diff,0)
		from
		#Posting PD
		cross join (select SUM([Credit Amount])- SUM([Debit Amount]) Diff from #Posting) tt
		where PD.SlNo  = 1
	
	update PD set [Credit Amount] = [Credit Amount] + ISNULL(diff,0)
		from
		#Posting PD
		cross join (select SUM([Debit Amount])- SUM([Credit Amount]) Diff from #Posting) tt
		where PD.SlNo  = 1
			
 end
 
 
select * from #Posting
 where isnull([Debit Amount],0) >0 or isnull([Credit Amount],0) >0
   
  
  drop table #Posting 
   
END TRY
	
	BEGIN CATCH
		select ERROR_MESSAGE()
		INSERT INTO [masterdata].[mErrorLog]
			   ([ErrorDate]
			   ,[ErrorNo]
			   ,[ErrorSeverity]
			   ,[ErrorState]
			   ,[ErrorProcedure]
			   ,[ErrorLine]
			   ,[ErrorDescription]
			   ,[UserName]
			   ,[FormCode])
		VALUES
			   (getdate()
			   ,ERROR_NUMBER()
			   ,ERROR_SEVERITY()
			   ,ERROR_STATE()
			   ,ERROR_PROCEDURE()
			   ,ERROR_LINE()
			   ,ERROR_MESSAGE()
			   ,SUSER_SNAME()
			   ,'MGPO')
	END CATCH
END
